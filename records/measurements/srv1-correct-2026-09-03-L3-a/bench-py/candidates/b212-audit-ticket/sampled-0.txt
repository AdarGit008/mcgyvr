def audit_ticket(ticket: str) -> str:
    if not isinstance(ticket, str) or not ticket.isupper():
        raise ValueError("Invalid ticket format. Ticket must be a string of two capital letters followed by a hyphen, four digits, a hyphen, and one check digit.")
    
    letters = ticket[0:2]
    digits = ticket[3:7]
    check_digit = int(ticket[7])
    
    total = sum(ord(letter) - 64 + int(digit) * (i + 1) for i, (letter, digit) in enumerate(zip(letters, digits)))
    
    if check_digit != total % 11:
        return "check"
    elif total % 11 == 10:
        return "shape"
    else:
        return "ok"
