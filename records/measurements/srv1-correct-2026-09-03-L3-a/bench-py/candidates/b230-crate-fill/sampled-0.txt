```python
from typing import List

def crate_fill(items: int, crates: int) -> List[int]:
    if crates == 0:
        return []
    
    avg_items_per_crate = items // crates
    extra_items = items % crates
    
    return [avg_items_per_crate] * crates + [extra_items] * extra_items

def crate_total(sizes: List[int]) -> int:
    return sum(sizes)
```
