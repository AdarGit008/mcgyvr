def value_keys(store: dict) -> dict:
    """
    Given a store, return the values it holds paired with how many keys hold each one.

    Args:
    store (dict): The store to process.

    Returns:
    dict: A mapping from value to count.
    """
    if not store:
        return {}

    value_counts = {}
    for key, value in store.items():
        if value in value_counts:
            value_counts[value] += 1
        else:
            value_counts[value] = 1

    return value_counts
