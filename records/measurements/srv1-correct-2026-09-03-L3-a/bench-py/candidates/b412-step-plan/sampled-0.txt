def step_allowed(start: str, end: str, allowed: list) -> bool:
    if start == end:
        return False
    if start not in allowed:
        return False
    if end not in allowed:
        return False
    for move in allowed:
        if move == (start, end):
            return True
    return False

def step_plan(states: list, allowed: list) -> int:
    if not states:
        raise ValueError("Run holding no states")
    
    for i in range(len(states) - 1):
        if not step_allowed(states[i], states[i + 1], allowed):
            return i
    return -1
