# solution.py
from typing import *
from string import *

def slug_word(word: str) -> str:
    """Lowercase the input word and drop non-letter/digit characters."""
    return ''.join(char for char in word if char.isalnum())

def slug_make(phrase: str) -> str:
    """Join slugged words of a phrase with a single hyphen."""
    slugged_words = [slug_word(word) for word in phrase.split()]
    return '-'.join(word for word in slugged_words if word)
