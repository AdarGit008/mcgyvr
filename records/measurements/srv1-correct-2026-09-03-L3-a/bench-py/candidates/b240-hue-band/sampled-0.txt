def hue_band(degrees: int) -> str:
    """Return the band the hue falls into: red below 60, green below 180, blue below 300, and red again from 300 up."""
    degrees = degrees % 360  # Wrap around the circle if necessary
    if degrees < 60:
        return "red"
    elif degrees < 180:
        return "green"
    elif degrees < 300:
        return "blue"
    else:
        return "red"
