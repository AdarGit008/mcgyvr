def field_mean(records: list, field: str) -> int:
    """
    Given records, return the mean of a specified numeric field across them,
    rounded down. Records without the field are passed over, and a set without
    the field averages to nothing.
    
    :param records: List of records, each a dictionary with numeric fields.
    :param field: Name of the numeric field to average.
    :return: Integer mean of the field, rounded down.
    """
    total = 0
    count = 0
    for record in records:
        if field in record:
            try:
                value = float(record[field])
                total += value
                count += 1
            except ValueError:
                pass
    if count == 0:
        return 0
    return int(total // count)
