def plural_of(noun: str) -> str:
    """
    Return the plural of an English noun. A noun ending in s, x, ch or sh takes es.
    A noun ending in y after a consonant loses the y and takes ies instead. Every other noun simply takes s.
    """
    if not noun or noun[-1] not in 'sxchsh':
        return noun + 's'
    elif noun[-1] == 'y' and noun[-2] not in 'aeiouAEIOU':
        return noun[:-1] + 'ies'
    else:
        return noun + 's'
