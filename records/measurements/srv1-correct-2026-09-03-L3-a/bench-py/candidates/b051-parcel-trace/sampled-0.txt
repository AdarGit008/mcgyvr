```python
from typing import List

def trace_parcel(events: List[str]) -> List[str]:
    if not isinstance(events, list):
        raise ValueError("Input must be a list")
    
    states = ["created"]
    
    for event in events:
        if event not in ["created", "packed", "shipped", "delivered", "returned"]:
            raise ValueError(f"Invalid event: {event}")
        
        if states[-1] == "delivered" or states[-1] == "returned":
            raise ValueError("No further events allowed after delivered or returned")
        
        if event == "created":
            states.append("packed")
        elif event == "packed":
            states.append("shipped")
        elif event == "shipped":
            states.append("delivered")
        elif event == "delivered":
            states.append("returned")
    
    return states
```
