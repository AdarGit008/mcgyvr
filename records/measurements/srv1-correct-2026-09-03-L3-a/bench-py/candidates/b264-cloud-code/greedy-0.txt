```python
def cloud_code(table: dict, code: str) -> str:
    """
    Turn a short weather code into words using the table you are handed,
    which maps codes to descriptions. Codes may arrive in any case and are
    matched in upper case. A code the table does not carry becomes the
    single word unknown.
    """
    if not table:
        raise ValueError("Table is empty")
    
    code = code.upper()
    if code not in table:
        return "unknown"
    
    return table[code]
```
