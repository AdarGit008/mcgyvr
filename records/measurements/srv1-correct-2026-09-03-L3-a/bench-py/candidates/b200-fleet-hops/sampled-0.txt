def fleet_hops(count: int, lts: list, builds: list, hop: int) -> int:
    if not (0 <= hop <= count):
        raise ValueError("Hop must be between 0 and count inclusive")
    if not all(0 <= build <= count for build in builds):
        raise ValueError("All builds must be between 0 and count inclusive")

    total_hops = 0
    for build in builds:
        if build == count:
            continue
        while build < lts[-1] and build + hop <= count:
            build += hop
            total_hops += 1

    return total_hops
