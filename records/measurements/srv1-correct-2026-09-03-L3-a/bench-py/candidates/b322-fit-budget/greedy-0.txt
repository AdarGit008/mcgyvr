```python
def fit_budget(prices: list, budget: int) -> int:
    if budget < 0:
        raise ValueError("Budget cannot be negative.")
    
    prices.sort()
    total_bought = 0
    remaining_budget = budget
    
    for price in prices:
        if price > remaining_budget:
            break
        total_bought += 1
        remaining_budget -= price
    
    return total_bought
```
