def run_decode(coded: str) -> str:
    """
    Decode a run of characters where each character is followed by a number indicating how many times it repeats.
    
    Args:
    coded (str): A string representing the run of characters to decode.
    
    Returns:
    str: The decoded string.
    """
    decoded = ""
    i = 0
    while i < len(coded):
        char = coded[i]
        if char.isdigit():
            count = int(coded[i:i+2])
            decoded += char * count
            i += 2
        else:
            decoded += char
            i += 1
    return decoded
