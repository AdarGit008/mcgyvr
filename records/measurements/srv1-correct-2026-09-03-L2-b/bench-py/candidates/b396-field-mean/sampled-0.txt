```python
from typing import List, Union

def field_mean(records: List[dict], field: str) -> int:
    if not records or not field:
        raise ValueError("Input records and field must not be empty")
    
    total_sum = 0
    count = 0
    
    for record in records:
        if field in record:
            try:
                value = int(record[field])
                total_sum += value
                count += 1
            except ValueError:
                continue
    
    if count == 0:
        return 0
    
    return total_sum // count
```
