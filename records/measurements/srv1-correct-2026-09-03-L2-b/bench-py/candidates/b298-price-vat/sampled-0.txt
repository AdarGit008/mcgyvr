def gross_price(net: int, rate: int) -> int:
    """A net price with a whole-percent tax added, in whole pence."""
    if rate == 0:
        return net
    elif rate > 0:
        tax_pence = (rate * net) // 100
        return net + tax_pence
    else:  # rate is negative
        raise ValueError("Rate must be non-negative")
