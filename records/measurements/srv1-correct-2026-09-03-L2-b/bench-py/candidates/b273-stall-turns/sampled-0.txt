def stall_turns(counts: list, limit: int) -> int:
    if limit <= 0:
        raise ValueError("Limit must be positive.")
    
    total_turns = 0
    for count in counts:
        if count > limit:
            total_turns += count // limit
            if count % limit > 0:
                total_turns += 1
        else:
            total_turns += 1
    
    return total_turns
