def step_back(moves: str) -> int:
    """
    Walks a line of moves ('F' for forward and 'B' for backward) and returns the furthest forward position reached.
    Non-letter moves ('f', 'b', etc.) are ignored.
    """
    current_position = 0
    furthest_position = 0
    for move in moves:
        if move == 'F':
            current_position += 1
            furthest_position = max(furthest_position, current_position)
        elif move == 'B':
            current_position -= 1
    return furthest_position
