```python
def step_value(step: str) -> int:
    """Returns what one instruction adds to a running total: a plus adds one, a minus takes one away and anything else adds nothing."""
    if step == "+":
        return 1
    if step == "-":
        return -1
    return 0


def scan_tally(steps: list) -> list:
    """The running total after each instruction."""
    total = 0
    for step in steps:
        total += step_value(step)
    return total
```
