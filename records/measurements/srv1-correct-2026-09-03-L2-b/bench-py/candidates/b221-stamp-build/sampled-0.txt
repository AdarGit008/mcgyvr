def stamp_build(major: int, minor: int, patch: int) -> int:
    if not isinstance(major, int) or not isinstance(minor, int) or not isinstance(patch, int):
        raise ValueError("All components must be integers.")
    if major < 0 or minor < 0 or patch < 0:
        raise ValueError("Components cannot be negative.")
    if major == 0 and minor == 0 and patch == 0:
        raise ValueError("No component can be zero.")
    if major == 0 and minor == 1000:
        raise ValueError("Major cannot be zero and minor cannot be 1000.")
    if major == 1000 and minor == 0:
        raise ValueError("Major cannot be 1000 and minor cannot be 0.")
    if patch >= 1000:
        raise ValueError("Patch cannot be 1000 or more.")
    
    return patch + 1000 * minor + 1000000 * major
