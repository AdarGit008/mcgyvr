from typing import List

def mean_window(readings: List[int], size: int) -> List[int]:
    if size == 0:
        return []
    
    window_means = []
    for i in range(len(readings) - size + 1):
        window = readings[i:i + size]
        if not window:
            window_means.append(0)
        else:
            mean = sum(window) // size
            window_means.append(mean)
    
    return window_means
