def split_at(entries: list, marker: str) -> list:
    """
    Break a list in two at the first entry equal to a marker, returning what stood before it and what stood after.
    A list not holding the marker gives everything as the first part and nothing as the second.
    """
    try:
        index = entries.index(marker)
        return entries[:index], entries[index + 1:]
    except ValueError:
        return entries, []
