def wire_text(text: str) -> str:
    if not isinstance(text, str) or '\n' in text:
        raise ValueError("Text must be a non-empty string without newline characters.")
    return f"s{len(text)}:{text}"

def wire_value(value) -> str:
    if isinstance(value, str):
        return wire_text(value)
    elif isinstance(value, int):
        if value < 0:
            return f"-{abs(value)};"
        else:
            return f"n{value};"
    elif isinstance(value, list):
        return f"[{', '.join(wire_value(item) for item in value)}]"
    else:
        raise ValueError("Unsupported value type.")
