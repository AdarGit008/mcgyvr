def batch_ends(count: int, size: int) -> list:
    """The position of the first entry of each full batch."""
    if size <= 0:
        raise ValueError("Batch size must be a positive integer.")
    ends = []
    start = 0
    while start + size <= count:
        ends.append(start + size - 1)  # Adjusting to include the first entry of the batch
        start += size
    return ends
