def camel_break(name: str) -> list:
    """
    Break a camel-cased identifier into its individual words, convert them to lowercase,
    and return them as a list.

    :param name: The camel-cased identifier to break.
    :return: A list of words in lowercase.
    """
    if not name:
        return []

    words = []
    current_word = ''
    for char in name:
        if char.isupper():
            if current_word:
                words.append(current_word.lower())
                current_word = ''
            words.append(char.lower())
        else:
            current_word += char
    if current_word:
        words.append(current_word.lower())

    return words
