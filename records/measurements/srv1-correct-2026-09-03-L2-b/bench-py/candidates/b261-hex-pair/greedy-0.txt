def hex_split(colour: str) -> list:
    if not colour or len(colour) != 7 or not all(c in '0123456789abcdef' for c in colour):
        raise ValueError("Invalid colour format")
    return [colour[i:i+2].lower() for i in range(0, 7, 2)]

def hex_join(parts: list) -> str:
    if len(parts) != 3:
        raise ValueError("Invalid number of parts")
    return ''.join(parts)
