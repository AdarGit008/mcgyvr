def step_cost(kind: str) -> int:
    """Return the cost of a step of the given kind."""
    if kind == "walk":
        return 1
    elif kind == "run":
        return 2
    elif kind == "bike":
        return 3
    else:
        raise ValueError("Invalid step kind")

def path_toll(steps: list[str]) -> int:
    """Total the path, except that every third step along it is free."""
    total_cost = 0
    for i, step in enumerate(steps):
        cost = step_cost(step)
        if (i + 1) % 3 == 0:
            cost = 0
        total_cost += cost
    return total_cost
