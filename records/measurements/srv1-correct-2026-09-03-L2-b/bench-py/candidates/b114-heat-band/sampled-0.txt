from typing import List, Tuple, Dict

def run_thermostat(start: int, low: int, high: int, power: int, drifts: List[int]) -> Dict[str, int]:
    if not isinstance(start, int) or not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("temperatures must be integers")
    if low >= high:
        raise ValueError("low must lie strictly below high")
    if isinstance(power, bool) or not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("drifts must be an integer")

    current_temp = start
    heated_ticks = 0
    switches = 0
    coldest = start

    for drift in drifts:
        current_temp += drift
        if current_temp < low:
            heated_ticks += 1
            switches += 1
            current_temp = low
        elif current_temp > high:
            heated_ticks += 1
            switches += 1
            current_temp = high
        elif current_temp < coldest:
            coldest = current_temp

    return {"temp": current_temp, "heated": heated_ticks, "switches": switches, "coldest": coldest}
