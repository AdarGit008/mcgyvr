def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not isinstance(start, int) or not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("temperatures must be integers")
    if low >= high:
        raise ValueError("low must lie strictly below high")
    if isinstance(power, bool) or not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("drifts must be integers")
    # TODO: walk the drifts, switching the heater by the two-point rule,
    # and track the heated ticks, the switches and the coldest reading.
    temp = start
    heated = 0
    switches = 0
    coldest = start
    for drift in drifts:
        if temp + drift < low:
            temp = low
            heated += 1
            switches += 1
        elif temp + drift > high:
            temp = high
            heated += 1
            switches += 1
        else:
            temp += drift
        if temp < coldest:
            coldest = temp
    return {"temp": temp, "heated": heated, "switches": switches, "coldest": coldest}
