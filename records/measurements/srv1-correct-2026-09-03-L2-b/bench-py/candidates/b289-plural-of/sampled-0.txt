def plural_of(noun: str) -> str:
    """
    Return the plural of an English noun.
    """
    if noun.endswith(('s', 'x', 'ch', 'sh')):
        return noun + 'es'
    elif noun.endswith('y') and noun[-2] not in 'aeiou':
        return noun[:-1] + 'ies'
    else:
        return noun + 's'
