def vend_credit(coins: list, price: int) -> int:
    if not coins or price <= 0 or price % 5 != 0:
        raise ValueError("Invalid input")

    credit = 0
    items_dropped = 0

    for coin in coins:
        credit += coin
        if credit >= price:
            items_dropped += 1
            credit -= price

    if credit < 0:
        raise ValueError("Insufficient funds")

    return credit
