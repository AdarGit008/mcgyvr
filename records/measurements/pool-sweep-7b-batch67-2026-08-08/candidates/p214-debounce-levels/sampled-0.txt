/**
 * Debounces levels in a binary sample sequence based on a hold threshold.
 *
 * @param samples - A non-empty list of 0s and 1s.
 * @param hold - A positive integer indicating the number of consecutive differing samples required to change the level.
 * @returns An array representing the debounced levels at each step.
 */
export function debounceLevels(samples: unknown, hold: unknown): number[] {
    if (!Array.isArray(samples) || samples.length === 0) {
        throw new Error('Samples must be a non-empty list.');
    }

    for (const sample of samples) {
        if (sample !== 0 && sample !== 1) {
            throw new Error('All samples must be either 0 or 1.');
        }
    }

    if (!Number.isInteger(hold) || hold <= 0) {
        throw new Error('Hold must be a positive whole number.');
    }

    const debouncedLevels: number[] = [];
    let currentLevel = samples[0];
    let tally = 0;

    for (const sample of samples) {
        if (sample === currentLevel) {
            tally = 0;
        } else {
            tally++;
            if (tally >= hold) {
                currentLevel = sample;
                tally = 0;
            }
        }
        debouncedLevels.push(currentLevel);
    }

    return debouncedLevels;
}
