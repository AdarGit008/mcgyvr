/** How far apart two people stand in a register of forebears. */
export function kinshipDegree(
  register: Record<string, string[]>,
  one: string,
  other: string,
): Record<string, unknown> {
  if (typeof register !== 'object' || register === null) {
    throw new Error('Register must be an object.');
  }

  const names = [one, other];
  for (const name of names) {
    if (typeof name !== 'string' || name.trim() === '') {
      throw new Error('Names must be non-empty strings.');
    }
    if (!(name in register)) {
      throw new Error(`Name '${name}' is not a key in the register.`);
    }
  }

  const validateForebears = (forebears: string[]): void => {
    if (!Array.isArray(forebears) || forebears.length > 2) {
      throw new Error('Forebear list must be an array of at most two strings.');
    }
    for (const forebear of forebears) {
      if (typeof forebear !== 'string' || forebear.trim() === '') {
        throw new Error('Forebears must be non-empty strings.');
      }
      if (!(forebear in register)) {
        throw new Error(`Forebear '${forebear}' is not a key in the register.`);
      }
      if (forebears.includes(forebear, forebears.indexOf(forebear) + 1)) {
        throw new Error('Forebear list must not contain duplicate names.');
      }
      if (forebear === name) {
        throw new Error(`Person cannot be their own forebear.`);
      }
    }
  };

  for (const name of names) {
    validateForebears(register[name]);
  }

  const climb = (start: string): Map<string, number> => {
    const seen = new Map<string, number>();
    let frontier = register[start] ?? [];
    let step = 1;
    while (frontier.length > 0) {
      const next: string[] = [];
      for (const name of frontier) {
        if (!seen.has(name)) {
          seen.set(name, step);
          next.push(...(register[name] ?? []));
        }
      }
      frontier = next;
      step += 1;
    }
    return seen;
  };

  const mine = climb(one);
  const theirs = climb(other);

  for (const [name, up] of mine) {
    if (name === other && up === 0 || name === one && theirs.get(name) === 0) {
      return { steps: 0, line: "direct", meet: name };
    }
    const down = theirs.get(name);
    if (down !== undefined) {
      return { steps: up + down, line: "collateral", meet: name };
    }
  }

  return { steps: 0, line: "apart", meet: "" };
}
