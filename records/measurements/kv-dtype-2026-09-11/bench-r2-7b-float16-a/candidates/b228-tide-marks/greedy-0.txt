def tide_marks(levels: list) -> list:
    """Return the positions of every local peak in the given list of water levels."""
    if len(levels) < 3:
        return []
    
    peaks = [i for i in range(1, len(levels) - 1) if levels[i] > levels[i - 1] and levels[i] > levels[i + 1]]
    return peaks
