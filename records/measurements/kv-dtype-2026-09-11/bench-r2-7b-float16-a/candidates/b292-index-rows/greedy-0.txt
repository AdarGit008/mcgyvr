def row_key(name: str) -> str:
    """Return the first letter of a name in upper case."""
    if not name:
        raise ValueError("Name cannot be empty")
    return name[0].upper()

def index_rows(names: list) -> dict:
    """File names under their row_key, maintaining the order they arrived."""
    index = {}
    for name in names:
        key = row_key(name)
        if key not in index:
            index[key] = []
        index[key].append(name)
    return index
