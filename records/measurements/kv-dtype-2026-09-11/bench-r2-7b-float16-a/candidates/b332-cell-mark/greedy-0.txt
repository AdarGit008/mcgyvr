def live_count(neighbours: list) -> int:
    """Return how many of a cell's given neighbors are alive."""
    return sum(1 for n in neighbours if n)

def live_next(alive: bool, neighbours: list) -> bool:
    """Say whether the cell is alive in the next round."""
    count = live_count(neighbours)
    return alive and (count == 2 or count == 3) or count == 3
