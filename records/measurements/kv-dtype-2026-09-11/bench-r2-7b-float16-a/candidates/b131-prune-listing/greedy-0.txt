"""Prune a file listing with ordered drop and keep rules."""

def prune_listing(listing: list, rules: list) -> list:
    def segment_fits(pattern, text):
        p = t = mark = 0
        star = -1
        while t < len(text):
            if p < len(pattern) and pattern[p] == "*":
                star = p
                mark = t
                p += 1
            elif p < len(pattern) and pattern[p] == text[t]:
                p += 1
                t += 1
            elif star >= 0:
                p = star + 1
                mark += 1
                t = mark
            else:
                return False
        while p < len(pattern) and pattern[p] == "*":
            p += 1
        return p == len(pattern)

    def split_strict(text, what):
        if not isinstance(text, str) or text == "":
            raise ValueError(f"{what} must be a non-empty string")
        segments = text.split("/")
        if any(segment == "" for segment in segments):
            raise ValueError(f"{what} has an empty segment: {text}")
        return segments

    # Validate every rule and path with split_strict
    for rule in rules:
        split_strict(rule, "rule")

    for path in listing:
        split_strict(path, "path")

    # Keep each path the last matching rule allows; a path no rule matches survives.
    surviving_paths = []
    for path in listing:
        match_found = False
        for rule in reversed(rules):
            if rule.startswith("!"):
                if segment_fits(rule[1:], "/".join(split_strict(path, "path"))):
                    surviving_paths.append(path)
                    match_found = True
                    break
            elif segment_fits(rule, "/".join(split_strict(path, "path"))):
                match_found = True
                break
        if not match_found:
            surviving_paths.append(path)

    return surviving_paths
