def fill_order(sources: list, needed: int) -> dict:
    if not isinstance(needed, int) or needed <= 0:
        raise ValueError("Needed must be a positive integer")
    
    for source in sources:
        if not isinstance(source, tuple) or len(source) != 2:
            raise ValueError("Each source must be a pair (cost, stock)")
        unit_cost, stock = source
        if not isinstance(unit_cost, int) or unit_cost <= 0:
            raise ValueError("Cost must be a positive integer")
        if not isinstance(stock, int) or stock <= 0:
            raise ValueError("Stock must be a positive integer")
    
    total_stock = sum(stock for _, stock in sources)
    if total_stock < needed:
        raise ValueError("Not enough stock to fulfill the order")
    
    order = sorted(range(len(sources)), key=lambda index: sources[index][0])
    taken = [0] * len(sources)
    cost = 0
    remaining = needed
    
    for index in order:
        if remaining <= 0:
            break
        unit_cost, stock = sources[index]
        take = min(stock, remaining)
        taken[index] = take
        cost += take * unit_cost
        remaining -= take
    
    leftover = [stock - taken[i] for i, (_, stock) in enumerate(sources)]
    
    return {"cost": cost, "taken": taken, "leftover": leftover}
