def steer_forklift(aisles: int, bays: int, moves: list) -> list:
    """
    Steers a forklift on a warehouse floor grid.

    Parameters:
    aisles (int): The number of aisles on the floor.
    bays (int): The number of bays on the floor.
    moves (list): A list of strings representing the moves ('north', 'south', 'east', 'west').

    Returns:
    list: The final position of the forklift as [aisle, bay].

    Raises:
    ValueError: If aisles or bays are not positive integers, or if a move is not valid.
    """
    if not isinstance(aisles, int) or not isinstance(bays, int) or aisles <= 0 or bays <= 0:
        raise ValueError("Aisles and bays must be positive integers.")
    
    if not all(move in ['north', 'south', 'east', 'west'] for move in moves):
        raise ValueError("Moves must be one of 'north', 'south', 'east', 'west'.")
    
    position = [0, 0]
    
    for move in moves:
        if move == 'north':
            if position[1] + 1 > bays:
                raise ValueError("Cannot move north beyond the floor boundaries.")
            position[1] += 1
        elif move == 'south':
            if position[1] - 1 < 0:
                raise ValueError("Cannot move south beyond the floor boundaries.")
            position[1] -= 1
        elif move == 'east':
            if position[0] + 1 > aisles:
                raise ValueError("Cannot move east beyond the floor boundaries.")
            position[0] += 1
        elif move == 'west':
            if position[0] - 1 < 0:
                raise ValueError("Cannot move west beyond the floor boundaries.")
            position[0] -= 1
    
    return position
