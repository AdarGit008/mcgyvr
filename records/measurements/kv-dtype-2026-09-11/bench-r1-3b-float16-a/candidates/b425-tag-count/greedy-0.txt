def tag_count(line: str) -> int:
    """
    Returns the number of markers in a line, where a marker is an opening angle bracket
    followed by everything up to the next closing angle bracket. An unmatched closing bracket
    raises a ValueError.
    """
    if not line:
        return 0

    stack = []
    marker_count = 0

    for char in line:
        if char == '<':
            stack.append(char)
        elif char == '>':
            if not stack:
                raise ValueError("Unmatched closing bracket found.")
            stack.pop()
            marker_count += 1

    if stack:
        raise ValueError("Unmatched opening bracket found.")

    return marker_count
