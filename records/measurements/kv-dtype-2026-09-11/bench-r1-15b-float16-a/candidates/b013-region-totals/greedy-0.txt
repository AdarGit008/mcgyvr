from typing import List

def region_totals(grid: List[List[int]], queries: List[List[int]]) -> List[int]:
    if not isinstance(grid, list) or not grid:
        raise ValueError("grid must be a non-empty list of rows")
    width = len(grid[0]) if isinstance(grid[0], list) else 0
    if width == 0:
        raise ValueError("rows must be non-empty lists")
    for row in grid:
        if not isinstance(row, list) or len(row) != width:
            raise ValueError("rows must all share one length")
        for cell in row:
            if isinstance(cell, bool) or not isinstance(cell, int):
                raise ValueError("cells must be integers")
    if not isinstance(queries, list) or not queries:
        raise ValueError("queries must be a non-empty list of queries")
    for query in queries:
        if not isinstance(query, list) or len(query) != 4:
            raise ValueError("queries must be lists of exactly four integers")
        if query[0] >= query[2] or query[1] >= query[3]:
            raise ValueError("query must be within the grid")
        if query[0] < 0 or query[1] < 0 or query[2] < 0 or query[3] < 0:
            raise ValueError("query indices must be non-negative")
    # Build a prefix-sum table over the grid
    prefix_sum = [[0] * (width + 1) for _ in range(len(grid) + 1)]
    for i in range(1, len(grid) + 1):
        for j in range(1, width + 1):
            prefix_sum[i][j] = prefix_sum[i - 1][j] + prefix_sum[i][j - 1] - prefix_sum[i - 1][j - 1] + grid[i - 1][j - 1]
    # Validate each query and answer it from the table
    results = []
    for query in queries:
        top, left, bottom, right = query
        total = prefix_sum[bottom + 1][right + 1] - prefix_sum[bottom + 1][left] - prefix_sum[top][right + 1] + prefix_sum[top][left]
        results.append(total)
    return results
