def step_cost(kind: str) -> int:
    """
    Returns the cost of a single step of a given kind.
    """
    if kind == "walk":
        return 1
    elif kind == "run":
        return 2
    elif kind == "jump":
        return 3
    else:
        raise ValueError("Invalid step kind")

def path_toll(steps: list[str]) -> int:
    """
    Totals a path, except that every third step along it is free, counting the steps from one.
    """
    if not steps:
        return 0
    
    total_cost = 0
    for i, step in enumerate(steps):
        if (i + 1) % 3 == 0:
            total_cost += 0
        else:
            total_cost += step_cost(step)
    
    return total_cost
