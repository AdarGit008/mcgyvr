def turn_tail(entries: list[str], count: int) -> list[str]:
    if count == 0:
        return entries
    if count < 0:
        raise ValueError("Count cannot be negative")
    if count > len(entries):
        count = len(entries)
    out = []
    for i in range(count - 1, -1, -1):
        out.append(entries[i])
    for i in range(count, len(entries)):
        out.append(entries[i])
    return out
