def hex_split(colour: str) -> list:
    if not colour or len(colour) != 7 or colour[0] != '#':
        raise ValueError("Invalid colour format")
    return [colour[i:i+2].lower() for i in range(1, 7, 2)]

def hex_join(parts: list) -> str:
    if len(parts) != 3:
        raise ValueError("Invalid number of parts")
    return '#' + ''.join(parts)
