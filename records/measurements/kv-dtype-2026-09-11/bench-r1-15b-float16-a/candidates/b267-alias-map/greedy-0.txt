def alias_resolve(aliases: dict, name: str) -> str:
    """
    Resolves a name to its alias, following one hop and no further.
    If the name stands for nothing, returns the name itself.
    """
    if name in aliases:
        return aliases[name]
    return name


def alias_names(aliases: dict) -> list:
    """
    Lists all names that stand for something, in alphabetical order.
    """
    return sorted(aliases.keys())
