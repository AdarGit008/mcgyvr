```python
from typing import List

def cross_up(readings: List[int], level: int) -> int:
    """
    Return how many times a run of readings crosses a level going upward.
    A crossing is a reading that reaches the level when the reading immediately before it stood below.
    """
    if not readings:
        return 0
    
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > level and readings[i-1] <= level:
            count += 1
    
    return count
```
