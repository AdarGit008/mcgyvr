```python
from typing import List

def sift_marks(entries: List[str], mark: str) -> List[List[str]]:
    """
    Sort entries into two runs: those opening with a given mark, and those that do not.
    Each run holds its entries in the order they arrived. Give back the two runs together, the marked run first.
    Entries holding nothing at all give back two empty runs.
    """
    marked = []
    unmarked = []
    
    for entry in entries:
        if entry.startswith(mark):
            marked.append(entry)
        else:
            unmarked.append(entry)
    
    return [marked, unmarked]
```
