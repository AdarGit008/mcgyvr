def kettle_hold(readings: list, target: int) -> int:
    """
    Count the number of readings at or above the target without interruption.
    
    Parameters:
    readings (list): A list of temperature readings.
    target (int): The target temperature.
    
    Returns:
    int: The number of readings that sat at or above the target.
    """
    if not readings:
        return 0
    
    count = 0
    for reading in readings:
        if reading >= target:
            count += 1
        else:
            break
    
    return count
