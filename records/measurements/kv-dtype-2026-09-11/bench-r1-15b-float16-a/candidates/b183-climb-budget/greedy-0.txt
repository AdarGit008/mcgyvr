def climb_budget(tolls: list) -> int:
    if not tolls:
        return 0

    total_cost = 0
    for i, toll in enumerate(tolls):
        if not toll.isdigit():
            raise ValueError(f"Invalid toll at position {i}: {toll}")
        if toll[0] == '0':
            raise ValueError(f"Leading zero found at position {i}: {toll}")

        if toll == '0':
            continue

        total_cost += int(toll)

    return total_cost
