from typing import *
from collections import *

def cut_roll(length: int, board: list[tuple[int, int]]) -> dict:
    if isinstance(length, bool) or not isinstance(length, int) or length < 0:
        raise ValueError("length must be a non-negative whole number")
    
    # Sort the board by price in descending order
    order = sorted(board, key=lambda entry: -entry[1])
    
    # Initialize the DP table
    best = [0] * (length + 1)
    taken = [0] * (length + 1)
    
    # Fill the DP table
    for metres in range(1, length + 1):
        best[metres] = -1
        for piece, price in order:
            if piece <= metres and price + best[metres - piece] > best[metres]:
                best[metres] = price + best[metres - piece]
                taken[metres] = piece
    
    # Walk back from the full roll to collect the pieces cut at each step
    pieces: list[int] = []
    current_metres = length
    while current_metres > 0:
        if taken[current_metres] > 0:
            pieces.append(taken[current_metres])
            current_metres -= taken[current_metres]
        else:
            break
    
    # Reverse the list of pieces to get them in the correct order
    pieces.reverse()
    
    return {"takings": best[length], "pieces": pieces}
