def step_back(moves: str) -> int:
    """
    Calculate the furthest forward position reached during a series of moves.
    
    Parameters:
    moves (str): A string containing 'F' for forward and 'B' for backward steps.

    Returns:
    int: The furthest forward position reached, or 0 if no forward steps were taken.
    """
    max_position = 0
    current_position = 0

    for move in moves:
        if move == 'F':
            current_position += 1
            max_position = max(max_position, current_position)
        elif move == 'B':
            current_position -= 1

    return max_position
