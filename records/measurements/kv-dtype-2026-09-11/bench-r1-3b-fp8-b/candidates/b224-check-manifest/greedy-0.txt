def check_manifest(line: dict) -> dict:
    """
    Vets a shipping manifest line and returns a fresh mapping with sku, count, and note.
    Raises ValueError for invalid inputs.
    """
    if not isinstance(line, dict):
        raise ValueError("Input must be a dictionary.")
    
    sku = line.get('sku', '')
    count = line.get('count', 0)
    note = line.get('note', '')

    if not isinstance(sku, str) or not sku.strip():
        raise ValueError("Invalid sku: must be a non-empty string.")
    if not isinstance(count, int) or count <= 0:
        raise ValueError("Invalid count: must be a positive whole number.")
    if not isinstance(note, str):
        raise ValueError("Invalid note: must be a string.")

    # Trim spaces from the sku
    sku = sku.strip()

    return {'sku': sku.upper(), 'count': count, 'note': note}
