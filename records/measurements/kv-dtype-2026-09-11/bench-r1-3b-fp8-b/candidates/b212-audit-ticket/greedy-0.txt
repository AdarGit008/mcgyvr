def audit_ticket(ticket: str) -> str:
    """
    Audit a depot ticket to determine if it is valid.
    
    Args:
        ticket (str): The depot ticket to audit.
    
    Returns:
        str: "shape" if the ticket does not read exactly as described,
             "check" if it reads correctly but its check digit is not the last digit of the total,
             "ok" if neither fault is present.
    """
    # Check if the ticket is a string
    if not isinstance(ticket, str):
        raise ValueError("Input must be a string.")
    
    # Trim leading and trailing spaces
    ticket = ticket.strip()
    
    # Check if the ticket matches the required format
    if len(ticket) != 10 or ticket[2] != '-' or ticket[6] != '-':
        return "shape"
    
    # Extract components of the ticket
    letters = ticket[:2]
    digits = ticket[3:7]
    check_digit = ticket[9]
    
    # Calculate the total from the letters
    letter_total = sum((ord(letter.upper()) - ord('A') + 1) * (i + 1) for i, letter in enumerate(letters))
    
    # Calculate the total from the digits
    digit_total = sum(int(digit) * (i + 1) for i, digit in enumerate(digits))
    
    # Calculate the total
    total = letter_total + digit_total
    
    # Check the check digit
    if check_digit != str(total % 10):
        return "check"
    
    # If neither fault is present
    return "ok"
