```python
from typing import Dict

def value_keys(store: Dict[str, str]) -> Dict[str, int]:
    """
    Returns a dictionary mapping each unique value in the store to its count.
    
    Args:
        store (Dict[str, str]): The store containing key-value pairs.

    Returns:
        Dict[str, int]: A dictionary mapping each unique value to its count.
    """
    if not store:
        return {}

    value_counts = {}
    for value in store.values():
        if value in value_counts:
            value_counts[value] += 1
        else:
            value_counts[value] = 1

    return value_counts
```
