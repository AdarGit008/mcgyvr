def mid_pair(values: list) -> list:
    """
    Return the two values standing closest to the middle of a list once it is put in order,
    the lower one first. A list of an odd length gives its middle value twice, and a list holding nothing gives two nothings.
    """
    if not values:
        return [None, None]
    
    sorted_values = sorted(values)
    mid_index = len(sorted_values) // 2
    
    if len(sorted_values) % 2 == 0:
        return [sorted_values[mid_index - 1], sorted_values[mid_index]]
    else:
        return [sorted_values[mid_index], sorted_values[mid_index]]
