def line_cost(price: int, quantity: int) -> int:
    """Calculate the cost of one line."""
    if price < 0 or quantity < 0:
        raise ValueError("Price and quantity must be non-negative.")
    return price * quantity

def basket_cost(lines: list, discount: int) -> int:
    """Calculate the total cost of a basket after applying a discount."""
    if discount < 0 or discount > 100:
        raise ValueError("Discount must be between 0 and 100.")
    
    if not lines:
        return 0
    
    total_cost = sum(line_cost(price, quantity) for price, quantity in lines)
    
    # Apply the discount
    discounted_cost = total_cost * (1 - discount / 100)
    
    # Round down to the nearest whole pence
    return int(discounted_cost)
