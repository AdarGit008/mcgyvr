```python
def slug_word(word: str) -> str:
    """
    Lowercase a word and remove all non-letter and non-digit characters.
    """
    return ''.join(c.lower() for c in word if c.isalnum())

def slug_make(phrase: str) -> str:
    """
    Join slugged words of a phrase with single hyphens, ignoring empty slugs.
    """
    return '-'.join(slug_word(word) for word in phrase.split() if slug_word(word))
```
