```python
def camel_break(name: str) -> list:
    if not name:
        return []
    
    words = []
    current_word = []
    
    for char in name:
        if char.isupper():
            if current_word:
                words.append(''.join(current_word))
                current_word = []
        current_word.append(char.lower())
    
    if current_word:
        words.append(''.join(current_word))
    
    return words
```
