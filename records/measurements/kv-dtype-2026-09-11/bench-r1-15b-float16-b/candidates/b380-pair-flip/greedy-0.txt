```python
from typing import List

def flip_one(pair: List) -> List:
    """
    Returns a pair with its two members the other way round.
    """
    if len(pair) != 2:
        raise ValueError("A pair must hold exactly two members.")
    return [pair[1], pair[0]]

def flip_all(pairs: List) -> List:
    """
    Returns a list with each pair in the original list, but with its two members the other way round.
    """
    return [flip_one(pair) for pair in pairs]
```
