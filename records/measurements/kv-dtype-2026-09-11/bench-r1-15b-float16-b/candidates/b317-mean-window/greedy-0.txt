from typing import List, Union

def mean_window(readings: List[int], size: int) -> List[int]:
    if size <= 0:
        raise ValueError("Window size must be positive")
    
    if size > len(readings):
        return []
    
    result = []
    for i in range(len(readings) - size + 1):
        window = readings[i:i + size]
        mean = sum(window) // size
        result.append(mean)
    
    return result
