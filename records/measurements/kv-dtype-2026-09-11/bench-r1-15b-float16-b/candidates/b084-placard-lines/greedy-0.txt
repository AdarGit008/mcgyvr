```python
"""Lay a notice's text into placard lines."""


def placard_lines(text: str, width: int) -> list:
    if not isinstance(text, str) or not text:
        raise ValueError("Text must be a non-empty string")
    if not isinstance(width, int) or width <= 0:
        raise ValueError("Width must be a positive integer")
    if text.startswith(" "):
        raise ValueError("Text must not start with a space")
    if text.endswith("  ") or text.endswith(" "):
        raise ValueError("Text must not end with two or more spaces")
    if any(word > width for word in text.split(" ")):
        raise ValueError("Text must not contain a word wider than the placard")
    lines = []
    line = ""
    for word in text.split(" "):
        if line == "":
            line = word
        elif len(line) + len(word) <= width:
            line = line + " " + word
        else:
            lines.append(line)
            line = word
    lines.append(line)
    return lines
```
