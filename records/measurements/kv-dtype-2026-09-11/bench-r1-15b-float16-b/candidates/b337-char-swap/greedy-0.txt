def swap_pair(first: str, second: str) -> str:
    """Swap two characters in the other order."""
    return second + first

def swap_all(text: str) -> str:
    """Walk a text in twos and swaps each pair it finds, leaving a final odd character exactly where it is."""
    result = []
    for i in range(0, len(text), 2):
        if i + 1 < len(text):
            result.append(text[i + 1])
            result.append(text[i])
        else:
            result.append(text[i])
    return ''.join(result)
