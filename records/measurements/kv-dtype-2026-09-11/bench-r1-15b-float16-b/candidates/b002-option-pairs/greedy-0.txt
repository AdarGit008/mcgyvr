import re


def parse_option(segment: str) -> list:
    key, eq, raw = segment.partition("=")
    if not eq or re.fullmatch(r"[A-Za-z][A-Za-z0-9_]*", key) is None:
        raise ValueError("malformed option segment: " + segment)
    if len(raw) >= 2 and raw.startswith('"') and raw.endswith('"'):
        raw = raw[1:-1]
    return [key, raw]


def scan_pairs(text: str) -> list:
    if not isinstance(text, str):
        raise ValueError("input must be a string")
    if text == "":
        return []
    if text.isspace():
        return []
    if text.count('"') % 2 != 0:
        raise ValueError("unterminated quote")
    if text.count(';') == 0:
        return parse_option(text)
    segments = text.split(';')
    pairs = []
    for segment in segments:
        if segment.startswith('"') and segment.endswith('"'):
            segment = segment[1:-1]
        if segment.count('=') == 0:
            raise ValueError("segment must contain an equals sign")
        key, eq, raw = segment.partition("=")
        if not eq or re.fullmatch(r"[A-Za-z][A-Za-z0-9_]*", key) is None:
            raise ValueError("malformed option segment: " + segment)
        if len(raw) >= 2 and raw.startswith('"') and raw.endswith('"'):
            raw = raw[1:-1]
        pairs.append([key, raw])
    return pairs
