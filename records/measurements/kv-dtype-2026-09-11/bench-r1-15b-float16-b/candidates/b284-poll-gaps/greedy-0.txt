from typing import List

def poll_gaps(minutes: List[int]) -> List[int]:
    """
    Given a list of minutes on which a monitor was meant to run, in increasing order,
    return the minutes between the first and the last on which it did not run, in increasing order.
    Fewer than two runs means there is nothing to report.
    """
    if not minutes:
        return []

    first_run = minutes[0]
    last_run = minutes[-1]

    if first_run == last_run:
        return []

    return [last_run - first_run - 1]
