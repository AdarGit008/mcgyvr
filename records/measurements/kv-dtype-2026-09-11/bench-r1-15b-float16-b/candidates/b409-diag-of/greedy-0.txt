from typing import List, Optional

def cell_at(grid: List[List[int]], row: int, column: int) -> Optional[int]:
    """
    Returns the value at a row and column of a grid, or None if either falls outside it.
    """
    if not (0 <= row < len(grid) and 0 <= column < len(grid[0])):
        return None
    return grid[row][column]

def diag_of(grid: List[List[int]]) -> List[int]:
    """
    Returns the values running down the grid's main diagonal from the top left, one for each row the grid holds.
    """
    if len(grid) != len(grid[0]):
        raise ValueError("Grid must be square to have a main diagonal.")
    return [grid[i][i] for i in range(len(grid))]
