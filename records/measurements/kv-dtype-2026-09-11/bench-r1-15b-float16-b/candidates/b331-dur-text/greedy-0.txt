def dur_hours(minutes: int) -> int:
    """
    Returns the whole hours in a count of minutes, discarding what is left over.
    """
    if minutes < 0:
        raise ValueError("Count of minutes cannot be negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    """
    Writes the count as hours and minutes joined, like 1h30m; a whole number of hours drops the minutes, a count under an hour drops the hours, and zero is 0m.
    """
    if minutes < 0:
        return "Invalid count of minutes"
    hours = minutes // 60
    minutes = minutes % 60
    return f"{hours}h{minutes:02d}m" if hours > 0 else f"{minutes}m"
