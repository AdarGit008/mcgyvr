def fleet_hops(count: int, lts: list, builds: list, hop: int) -> int:
    if hop < 0:
        raise ValueError("Hop must be non-negative")
    if not all(0 <= build <= count - 1 for build in builds):
        raise ValueError("All builds must be between 0 and count-1")
    if not all(0 <= lt <= count - 1 for lt in lts):
        raise ValueError("All LTS must be between 0 and count-1")
    if hop > count:
        raise ValueError("Hop cannot exceed count")

    hops = 0
    for build in builds:
        if build == count:
            continue
        if build in lts:
            continue
        hops += 1
        if build + hop > count:
            break

    return hops
