```python
from typing import List

def route_stops(route: str) -> List[str]:
    """
    Breaks a route written as stop names joined by greater-than signs into the stops themselves, trimming any spaces around each one.
    
    :param route: A string representing a route with stop names separated by greater-than signs.
    :return: A list of stop names.
    """
    if not route:
        return []
    
    stops = route.split('>')
    stops = [stop.strip() for stop in stops]
    return stops

def route_hops(route: str) -> int:
    """
    Counts the journeys between stops, one fewer than the stops, and none at all for a blank route.
    
    :param route: A string representing a route with stop names separated by greater-than signs.
    :return: The number of journeys between stops.
    """
    if not route:
        return 0
    
    stops = route.split('>')
    stops = [stop.strip() for stop in stops]
    return len(stops) - 1
```
