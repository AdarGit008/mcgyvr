def pair_off(marks: list[str]) -> list[str]:
    if not marks:
        return []

    result = []
    i = 0
    while i < len(marks):
        if i + 1 < len(marks) and marks[i] == marks[i + 1]:
            # Remove the two matching marks
            i += 2
        else:
            # Add the current mark to the result
            result.append(marks[i])
            i += 1

    return result
