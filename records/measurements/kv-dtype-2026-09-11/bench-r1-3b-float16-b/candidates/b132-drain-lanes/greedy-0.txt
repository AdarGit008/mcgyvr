"""Drain labelled items across planned lanes, round by round."""

from typing import Dict, List, Tuple

def drain_lanes(plan: List[Tuple[str, int]], items: List[Tuple[str, str]]) -> Dict[str, Union[List[str], int]]:
    # Validate the plan
    if not plan:
        raise ValueError("Plan cannot be empty.")
    
    for lane, quota in plan:
        if not isinstance(lane, str) or not isinstance(quota, int) or quota <= 0:
            raise ValueError("Invalid plan entry: lane must be a non-empty string and quota must be a positive integer.")
        
        if lane in queues:
            raise ValueError("Lane declared twice in the plan.")
    
    for label, lane in items:
        if not isinstance(label, str) or not isinstance(lane, str):
            raise ValueError("Invalid item: label and lane must be strings.")
    
    queues = {}
    for lane, _quota in plan:
        queues[lane] = []
    
    for label, lane in items:
        if lane in queues:
            queues[lane].append(label)
    
    order = []
    rounds = 0
    while True:
        rounds += 1
        for lane, quota in plan:
            queue = queues[lane]
            taken = 0
            while taken < quota and queue:
                order.append(queue.pop(0))
                taken += 1
            if taken < quota:
                return {"order": order, "rounds": rounds}
