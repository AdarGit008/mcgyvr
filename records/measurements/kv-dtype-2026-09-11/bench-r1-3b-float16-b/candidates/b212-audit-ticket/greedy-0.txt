def audit_ticket(ticket: str) -> str:
    if not isinstance(ticket, str):
        raise ValueError("Input must be a string.")
    
    # Trim leading and trailing spaces
    ticket = ticket.strip()
    
    # Check if the ticket matches the required format
    if len(ticket) != 11 or ticket[2] != '-' or ticket[7] != '-':
        return "shape"
    
    # Extract components
    letters = ticket[:2]
    digits = ticket[3:7]
    check_digit = ticket[8]
    
    # Calculate the total for the letters
    letter_total = sum((ord(char) - ord('A') + 1) * (i + 1) for i, char in enumerate(letters))
    
    # Calculate the total for the digits
    digit_total = sum(int(digit) * (i + 1) for i, digit in enumerate(digits))
    
    # Calculate the total
    total = letter_total + digit_total
    
    # Check the check digit
    if check_digit != str(total % 10):
        return "check"
    
    return "ok"
