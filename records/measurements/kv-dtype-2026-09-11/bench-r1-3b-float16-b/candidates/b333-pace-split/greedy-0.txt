def pace_of(seconds: int, kilometres: int) -> int:
    if kilometres <= 0:
        raise ValueError("Distance must be greater than zero.")
    return seconds // kilometres

def pace_list(legs: list) -> list:
    if not legs:
        return []
    
    pace_list = []
    for leg in legs:
        if leg['distance'] <= 0:
            raise ValueError("Distance must be greater than zero.")
        pace = pace_of(leg['seconds'], leg['distance'])
        pace_list.append(pace)
    
    return pace_list
