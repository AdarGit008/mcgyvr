def in_quote(ch: str) -> bool:
    """Return True if ch is a quotation mark."""
    return ch == '"' or ch == "'"

def quote_split(line: str) -> list:
    """Break a line on commas except for those inside quotation marks."""
    pieces = []
    current_piece = ""
    in_quotation = False
    for char in line:
        if in_quote(char):
            in_quotation = not in_quotation
        elif char == "," and not in_quotation:
            pieces.append(current_piece + char)
            current_piece = ""
        else:
            current_piece += char
    pieces.append(current_piece)
    return pieces
