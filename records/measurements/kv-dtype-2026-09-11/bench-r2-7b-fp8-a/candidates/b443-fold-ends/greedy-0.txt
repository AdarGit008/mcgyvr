def fold_ends(values: list) -> list:
    if not values:
        return []
    
    n =           len(values)
    folded = [0] * n
    for i in range(n // 2):
        folded[i[i] =            values[i]
        folded[n - 1 - i]:
            values[n - 1 - i]
    if n % 2 == 0:
        folded[n // 2]:
            values[n // 2]
    return folded
