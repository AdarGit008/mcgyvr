def served_by(queue: list, turn: int) -> str:
    if not queue:
        raise ValueError("a queue needs somebody in it")
    return queue[turn % len(queue)]


def queue_round(queue: list, turns: int) -> list:
    """Who is served on each turn up to a stated count, stopping at the end of the queue."""
    if not queue:
        raise ValueError("a queue needs somebody in it")
    served = []
    for turn in range(turns):
        if turn >= len(queue):
            break
        served.append(queue[turn])
    return served
