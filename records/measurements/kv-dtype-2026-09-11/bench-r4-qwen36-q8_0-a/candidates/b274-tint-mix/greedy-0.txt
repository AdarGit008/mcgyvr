def tint_mix(volume_a: int, strength_a: int, volume_b: int, strength_b: int) -> int:
    """Calculate the strength of a mixture of two tins of paint, rounded down to a whole percent."""
    total_volume = volume_a + volume_b
    if total_volume == 0:
        return 0
    total_paint = volume_a * strength_a + volume_b * strength_b
    return total_paint // total_volume
