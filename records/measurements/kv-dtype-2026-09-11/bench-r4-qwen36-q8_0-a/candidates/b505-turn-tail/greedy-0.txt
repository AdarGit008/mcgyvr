def turn_tail(entries: list[str], count: int) -> list[str]:
    """Turn round the closing stretch of a run, leaving everything before it standing as it was."""
    if count <= 0:
        return list(entries)
    
    n = len(entries)
    if count >= n:
        return list(reversed(entries))
    
    # The 'stop' index marks where the tail starts (exclusive)
    # We want to reverse the last 'count' elements
    # So the tail is entries[n - count : n]
    # The head is entries[: n - count]
    
    head = entries[:n - count]
    tail = entries[n - count:]
    
    return head + list(reversed(tail))
