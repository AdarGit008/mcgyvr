def step_back(moves: str) -> int:
    """Return the furthest forward position reached during the walk."""
    if not isinstance(moves, str):
        raise ValueError("moves must be a string")
    
    current_position = 0
    max_position = 0
    
    for char in moves:
        if char == 'F':
            current_position += 1
            if current_position > max_position:
                max_position = current_position
        elif char == 'B':
            current_position -= 1
        # Other characters are ignored
    
    return max_position
