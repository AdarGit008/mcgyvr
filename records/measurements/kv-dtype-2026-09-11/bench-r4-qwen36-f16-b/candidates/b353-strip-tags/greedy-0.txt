def strip_tags(line: str) -> str:
    """Remove every HTML-like tag marker from a line of text.
    
    A marker is defined as an opening angle bracket '<', followed by 
    everything up to the next closing angle bracket '>', including that bracket.
    If an opening bracket is never closed, everything from there to the end 
    of the line is removed.
    """
    result = []
    i = 0
    length = len(line)
    
    while i < length:
        char = line[i]
        if char == '<':
            # Find the next closing bracket
            close_index = line.find('>', i + 1)
            if close_index == -1:
                # No closing bracket found; remove everything from here to end
                break
            else:
                # Skip the entire tag including < and >
                i = close_index + 1
        else:
            result.append(char)
            i += 1
            
    return ''.join(result)
