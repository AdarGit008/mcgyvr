def driest_window(rain: list, width: int) -> int:
    """Return the starting index of the width-day stretch with the smallest rainfall sum."""
    if not isinstance(rain, list):
        raise ValueError("rain must be a list")
    if not isinstance(width, int) or isinstance(width, bool):
        raise ValueError("width must be an integer")
    if width <= 0:
        raise ValueError("width must be positive")
    if width > len(rain):
        raise ValueError("width cannot exceed the number of days")
    
    for i, reading in enumerate(rain):
        if not isinstance(reading, int) or isinstance(reading, bool):
            raise ValueError(f"Reading at index {i} must be an integer")
        if reading < 0:
            raise ValueError(f"Reading at index {i} must be non-negative")

    n = len(rain)
    if n == 0:
        return 0

    # Calculate the sum of the first window
    current_sum = sum(rain[:width])
    min_sum = current_sum
    min_index = 0

    # Slide the window
    for i in range(1, n - width + 1):
        # Update sum: subtract the element leaving the window, add the element entering
        current_sum = current_sum - rain[i - 1] + rain[i + width - 1]
        if current_sum < min_sum:
            min_sum = current_sum
            min_index = i

    return min_index
