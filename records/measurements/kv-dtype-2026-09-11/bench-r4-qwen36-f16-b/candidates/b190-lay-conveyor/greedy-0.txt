"""Lay a straight eastward conveyor run across a factory floor plan."""


def lay_conveyor(floor: list, row: int, col: int, length: int) -> list:
    """Return a new floor plan with a straight eastward conveyor run laid."""
    if row < 0 or row >= len(floor):
        raise ValueError("the run starts off the plan")
    
    cells = list(floor[row])
    end_col = col + length
    
    if end_col > len(cells):
        raise ValueError("the run passes the last column")
    
    for i in range(col, end_col):
        if cells[i] != ".":
            raise ValueError("the run covers a cell that is not open")
    
    for i in range(col, end_col):
        cells[i] = "="
    
    new_floor = list(floor)
    new_floor[row] = "".join(cells)
    return new_floor
