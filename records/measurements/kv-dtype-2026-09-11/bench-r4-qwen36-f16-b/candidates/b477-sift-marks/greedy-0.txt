def sift_marks(entries: list[str], mark: str) -> list[list[str]]:
    """Sort entries into two runs: those opening with a given mark, and those that do not."""
    if not entries:
        return [[], []]
    
    marked_run = []
    unmarked_run = []
    
    for entry in entries:
        if entry.startswith(mark):
            marked_run.append(entry)
        else:
            unmarked_run.append(entry)
            
    return [marked_run, unmarked_run]
