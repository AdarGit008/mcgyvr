def field_mean(records: list, field: str) -> int:
    """Return the mean of the named numeric field across records, rounded down."""
    values = []
    for record in records:
        if not isinstance(record, dict):
            continue
        if field not in record:
            continue
        value = record[field]
        if isinstance(value, bool):
            continue
        if not isinstance(value, (int, float)):
            continue
        values.append(value)
    if not values:
        return 0
    total = sum(values)
    count = len(values)
    # Use math.floor for rounding down, but since we need int return,
    # and floor of positive/negative works as expected for "rounded down" (towards -inf).
    import math
    return math.floor(total / count)
