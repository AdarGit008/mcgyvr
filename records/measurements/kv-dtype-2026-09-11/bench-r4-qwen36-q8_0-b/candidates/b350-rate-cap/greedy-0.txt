def rate_cap(price: int, rate: int, most: int) -> int:
    """Raise a price by a whole-percent rate, capped at a maximum increase."""
    if price < 0 or rate < 0 or most < 0:
        raise ValueError("price, rate, and most must be non-negative")
    
    # Calculate the increase amount based on the rate (whole percent)
    # Increase = price * rate / 100, rounded down (integer division)
    increase = (price * rate) // 100
    
    # Cap the increase at 'most'
    if increase > most:
        increase = most
    
    return price + increase
