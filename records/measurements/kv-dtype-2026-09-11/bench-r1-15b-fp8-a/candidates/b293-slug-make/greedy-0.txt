def slug_word(word: str) -> str:
    """Lowercase a word and drop non-letter, non-digit characters."""
    return ''.join(c.lower() for c in word if c.isalnum())

def slug_make(phrase: str) -> str:
    """Join slugged words of a phrase with hyphens,    excluding words that slugs away to nothing."""
    slugs = [slug_word(word) for word in phrase.split()]
    return '-'.join(slug for slug in slugs if slug)
