def dur_hours(minutes: int) -> int:
    if minutes < 0:
        raise ValueError("Number of minutes cannot be negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    if minutes < 0:
        raise ValueError("Number of minutes cannot be negative")
    hours = minutes // 60
    minutes = minutes % 60
    return f"{ hours}h{minutes}m"
