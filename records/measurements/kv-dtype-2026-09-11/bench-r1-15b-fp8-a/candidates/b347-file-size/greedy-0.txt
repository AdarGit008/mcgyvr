def size_text(bytes_config: int) -> str:
    if bytes_config < 0:
        raise ValueError("a byte count cannot be negative")
    unit = size_unit(bytes_config)
    if unit == "B":
        return str(bytes_config) + " B"
    if unit == "KB":
        return str(bytes_config // 1024) + " KB"
    return str(bytes_config // (1024 * 1024)) + " MB"
