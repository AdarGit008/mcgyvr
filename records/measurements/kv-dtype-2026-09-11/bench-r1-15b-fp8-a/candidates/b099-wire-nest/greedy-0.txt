def wire_value(value) -> str:
    if not isinstance(value, str):
        raise ValueError("Value must be a string")
    if value == "":
        return "n"
  # Check for non-ASCII characters
    if any(ord(c) > 127 for c in value):
        raise ValueError("Text must contain non-ASCII characters")
    return wire_text(value)

def wire_text(text: str) -> str:
    if not isinstance(text, str):
        raise ValueError("Text must be a string")
    if text == "":
        return "s0"
    if "\n" in text:
        raise ValueError("Text cannot contain a newline")
    if text.isascii():
        return "s" + str(len(text)) + ":" + text
    else:
        return "n" + str(len(text)) + ":" + text
 # Check for non non-ASCII characters
    if any(ord(c) > 127 for c in text):
        raise ValueError("Text must contain non-ASCII characters")
    return wire_text(text)
