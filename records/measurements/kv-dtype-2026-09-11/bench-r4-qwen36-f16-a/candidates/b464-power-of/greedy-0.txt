def power_of(base: int, power: int) -> int:
    """Raise a base to a whole power by multiplying it by itself that many times.
    
    A power of zero gives one, whatever the base. Rejects negative powers.
    """
    if power < 0:
        raise ValueError("Power must be non-negative")
    
    result = 1
    for _ in range(power):
        result *= base
    return result
