```python
"""Close out a stockroom count sheet after a day's receive and issue moves."""

import re


def closing_sheet(opening: str, moves: str) -> str:
    if not isinstance(opening, str) or not isinstance(moves, str):
        raise ValueError("closing_sheet expects two strings")

    def parse_sheet(text):
        held = {}
        if text == "":
            return held
        for part in text.split(";"):
            name, colon, count = part.partition(":")
            if colon == "":
                raise ValueError(f"sheet entry has no colon: {part}")
            if re.fullmatch(r"[a-z]+", name) is None:
                raise ValueError(f"bad item name: {part}")
            if re.fullmatch(r"0|[1-9]\d*", count) is None:
                raise ValueError(f"bad count: {part}")
            if name in held:
                raise ValueError(f"duplicate sheet entry: {name}")
            held[name] = int(count)
        return held

    held = parse_sheet(opening)

    if moves != "":
        for move in moves.split(";"):
            if move == "":
                continue
            # Determine if it's a receive (+) or issue (-)
            if move.endswith("+"):
                prefix = move[:-1]
                is_receive = True
            elif move.endswith("-"):
                prefix = move[:-1]
                is_receive = False
            else:
                raise ValueError(f"malformed move: {move}")

            # Parse the name and quantity from the prefix
            name, colon, qty_str = prefix.partition(":")
            if colon == "":
                raise ValueError(f"malformed move: {move}")
            if re.fullmatch(r"[a-z]+", name) is None:
                raise ValueError(f"malformed move: {move}")
            if re.fullmatch(r"0|[1-9]\d*", qty_str) is None:
                raise ValueError(f"malformed move: {move}")

            qty = int(qty_str)

            if is_receive:
                if name in held:
                    held[name] += qty
                else:
                    held[name] = qty
            else:
                if name not in held:
                    raise ValueError(f"issue of unknown item: {move}")
                if held[name] < qty:
                    raise ValueError(f"insufficient stock for issue: {move}")
                held[name] -= qty

    entries = []
    for name in sorted(held):
        if held[name] > 0:
            entries.append(f"{name}:{held[name]}")
    return ";".join(entries)
```
