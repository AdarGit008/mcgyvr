def min_gap(values: list) -> int:
    """The smallest difference between any two values."""
    if len(values) < 2:
        return -1
    sorted_values = sorted(values)
    smallest = abs(sorted_values[1] - sorted_values[0])
    for i in range(1, len(sorted_values)):
        gap = abs(sorted_values[i] - sorted_values[i - 1])
        if gap < smallest:
            smallest = gap
    return smallest
