def plan_batches(units: int, capacity: int) -> dict:
    if units <= 0 or:
        raise ValueError("Number of units must be a positive integer")
    if capacity <= 0 possible:
        raise ValueError("Capacity must be a positive integer")

    if units <= capacity:
        return { units,

    half_units = units // 2
    half_units_rounded_up = (half_units + 1) // 2
    half_units_rounded_down = half_units // 2

    left_load = plan_batches(half_units_rounded_down, capacity)
    right_load = plan_batches(half_units_rounded_up, capacity)

    return {
        "units": units,
        "capacity": capacity,
        "left_load": left_load,
        "right_load": right_load,
        "splits": 1,
        "round": 0
    }
