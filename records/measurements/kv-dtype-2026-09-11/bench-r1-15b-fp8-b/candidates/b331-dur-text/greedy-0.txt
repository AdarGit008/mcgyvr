def dur_hours(minutes: int) -> int:
    if minutes < 0:
        raise ValueError("Number of minutes must be non-negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    if minutes < 0:
        raise ValueError("Number of minutes must be non-negative")
    hours = minutes // 60
    minutes %= 60
    return f"{{hours}h{minutes}m"
