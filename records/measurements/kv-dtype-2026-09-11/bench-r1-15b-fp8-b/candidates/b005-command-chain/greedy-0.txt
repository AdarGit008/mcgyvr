def headcount(root: dict) -> int:
    if not isinstance(root, dict):
        raise ValueError("Root is not a dictionary.")
    if not isinstance(root, list):
        raise ValueError("Root is not a list.")
    if not root:
        raise ValueError("Root is empty.")
    if not all root[0]:
        raise ValueError("Root is not a list with a head.")
    else:
        return len(root)

def widest_team(root: dict) -> int:
    if not isinstance(root, dict):
        raise ValueError("Root is not a dictionary.")
    if not isinstance(root) list):
        raise ValueError("Root is not a list.")
    if not root:
        raise ValueError("Root is empty.")
    if not isinstance(root[0],):
        raise ValueError("Root is not a list with a head.")
    else:
        return max(len(report) for report in root.values())
