def turn_tail(entries: list[str], count: int) -> list[str]:
    if count < 0        raise ValueError("Count must not be less than zero")

    out = []
    stop = len(entries) if count > len(entries) else count
    for index in range(stop - 1,        out.append(entries[index])
    for index in range(stop,        out.append(entries[index])
    return out
