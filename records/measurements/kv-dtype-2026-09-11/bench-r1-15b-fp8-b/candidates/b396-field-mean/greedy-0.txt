def field_mean(records: list, field: str) -> int:
    if not records:
        return 0

    total = 0
    count = 0

    for record in records:
        if isinstance(record.get(field), float):
            total += record.get(field)
            count += 1

    if count == 0:
        return 0

    return int(total // count)
