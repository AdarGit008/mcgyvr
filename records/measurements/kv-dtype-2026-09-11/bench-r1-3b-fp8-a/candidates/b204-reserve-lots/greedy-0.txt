from typing import List, Tuple, Dict, Optional

def reserve_lots(lots: List[Tuple[int, str, int]], orders: List[Tuple[int, int]]) -> Dict[str, List[Tuple[int, int, int]]]:
    """
    Reserves lots for orders based on earliest use-by date and lot id.
    
    :param l: List of tuples (lot_id, use_by_date, units_on_hand)
    :param orders: List of tuples (order_id, units_wanted)
    :return: Dictionary with 'picks' and 'short' lists
    """
    # Sort lots by use-by date and then by lot_id
    l.sort(key=lambda x: (x[1], x[0]))
    
    # Initialize picks and short lists
    picks = []
    short = []
    
    # Iterate over orders
    for order_id, units_wanted in orders:
        for lot in l:
            if lot[2] > 0:
                units_taken = min(units_wanted, lot[2])
                picks.append((order_id, lot[0], units_taken))
                lot[2] -= units_taken
                units_wanted -= units_taken
                if units_wanted == 0:
                    break
    
    # If there are remaining units wanted, record them as shortfalls
    if units_wanted > 0:
        short.append((order_id, units_wanted))
    
    return {'picks': picks, 'short': short}
