def split_at(entries: list, marker: str) -> list:
    """
    Splits a list into two parts at the first occurrence of a marker.
    
    Parameters:
    entries (list): The list to be split.
    marker (str): The marker to split the list at.
    
    Returns:
    list: A list containing two parts: the part before the marker and the part after the marker.
    """
    try:
        index = entries.index(marker)
        return [entries[:index], entries[index+1:]]
    except ValueError:
        return [entries, []]
