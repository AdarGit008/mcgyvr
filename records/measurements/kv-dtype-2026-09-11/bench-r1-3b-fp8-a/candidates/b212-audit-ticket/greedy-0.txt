def audit_ticket(ticket: str) -> str:
    """
    Audit a depot ticket to determine if it is valid.
    
    Args:
        ticket (str): The depot ticket to audit.
    
    Returns:
        str: "shape" if the ticket does not read exactly as described,
             "check" if it reads correctly but its check digit is not the last digit of the total,
             "ok" if neither fault is present.
    """
    if not isinstance(ticket, str):
        raise ValueError("Input must be a string.")
    
    # Trim leading and trailing spaces
    ticket = ticket.strip()
    
    # Check if the ticket matches the required format
    if len(ticket) != 10 or ticket[2] != '-' or ticket[6] != '-':
        return "shape"
    
    # Extract components
    letters = ticket[:2]
    digits = ticket[3:7]
    check_digit = ticket[7]
    
    # Calculate the total from letters and digits
    total = sum((ord(letter.upper()) - ord('A') + 1) * (i + 1) for i, letter in enumerate(letters))
    total += sum(int(digit) * (i + 1) for i, digit in enumerate(digits))
    
    # Check the check digit
    if total % 10 != int(check_digit):
        return "check"
    
    
    return "ok"
