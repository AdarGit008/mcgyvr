```python
from typing import List, Dict

def count_by(records: List[Dict], field: str) -> Dict:
    """
    Count occurrences of each value in a specified field across records.
    
    Args:
        records: A list of dictionaries, where each dictionary represents a record.
        field: The name of the field to count occurrences of.

    Returns:
        A dictionary where keys are unique values from the specified field,
        and values are the counts of those values in the records.

    Raises:
        ValueError: If the field name is empty text.
    """
    if not field:
        raise ValueError("Field name cannot be empty text.")
    
    count_dict = {}
    for record in records:
        if field in record:
            value = record[field]
            if value in count_dict:
                count_dict[value] += 1
            else:
                count_dict[value] = 1
    
    return count_dict
```
