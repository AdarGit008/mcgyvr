def slot_pack(items: list, capacity: int) -> list:
    """Group items into slots of a fixed capacity, in order."""
    if capacity <= 0:
        raise ValueError("Capacity must be greater than zero.")
    
    slots = []
    for i in range(0, len(items), capacity):
        # Use min(capacity, len(items) - i) to ensure no slot exceeds the capacity
        slots.append(items[i : i + min(capacity, len(items) - i)])
    
    return slots
