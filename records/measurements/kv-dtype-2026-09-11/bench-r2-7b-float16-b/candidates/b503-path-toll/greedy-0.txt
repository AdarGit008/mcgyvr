def path_toll(steps: list[str]) -> int:
    """Total the cost of a path, skipping every third step starting from the first step."""
    if not steps:
        return 0
    
    total_cost = 0
    for i, step in enumerate(steps):
        if (i + 1) % 3 != 0:
            total_cost += step_cost(step)
    
    return total_cost
