def wave_count(readings: list) -> int:
    if len(readings) < 2:
        return 0
    
    count = 0
    prev_slope = 0
    
    for i in range(1, len(readings)):
        current_slope = readings[i] - readings[i - 1]
        
        if prev_slope * current_slope < 0:
            count += 1
        
        prev_slope = current_slope
    
    return count
