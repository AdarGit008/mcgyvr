def kettle_hold(readings: list, target: int) -> int:
    """Report how many readings at the very end of the run sat at or above the target without interruption."""
    if not readings:
        return 0
    
    hold_count = 0
    for reading in reversed(readings):
        if reading >= target:
            hold_count += 1
        else:
            break
    
    return hold_count
