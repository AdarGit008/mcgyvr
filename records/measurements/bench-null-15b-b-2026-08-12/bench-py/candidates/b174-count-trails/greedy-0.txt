from typing import List

def count_trails(rows: int, cols: int, blocked: List[List[int]]) -> int:
    if rows < 1 or cols < 1:
        raise ValueError("rows and cols must be at least one")
    
    if not all(isinstance(cell, list) and len(cell) == 2 for cell in blocked):
        raise ValueError("blocked must be a list of two-item lists")
    
    if any(not (0 <= cell[0] < rows and 0 <= cell[1] < cols) for cell in blocked):
        raise ValueError("blocked cells must be within the grid bounds")
    
    if any(cell == [0, 0] or cell == [rows - 1, cols - 1] for cell in blocked):
        raise ValueError("rope cannot cross the entrance or exit cell")
    
    def is_valid_move(row: int, col: int) -> bool:
        return 0 <= row < rows and 0 <= col < cols
    
    def dfs(row: int, col: int, visited: set) -> int:
        if (row, col) in visited:
            return 0
        visited.add((row, col))
        
        count = 1
        
        for dr, dc in [(0, 1), (1, 0)]:
            new_row, new_col = row + dr, col + dc
            if is_valid_move(new_row, new_col) and [new_row, new_col] not in blocked:
                count += dfs(new_row, new_col, visited)
        
        return count
    
    return dfs(0, 0, set())
