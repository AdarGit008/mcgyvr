def hue_band(degrees: int) -> str:
    """
    Returns the band name based on the hue in degrees. Hue ranges from 0 to 359.
    Bands are defined as follows:
    - Red below 60 degrees
    - Green below 180 degrees
    - Blue below 300 degrees
    - Red again from 300 degrees onwards
    """
    if degrees < 0:
        degrees += 360
    elif degrees >= 360:
        degrees %= 360

    if degrees < 60:
        return "Red"
    elif degrees < 180:
        return "Green"
    elif degrees < 300:
        return "Blue"
    else:
        return "Red"
