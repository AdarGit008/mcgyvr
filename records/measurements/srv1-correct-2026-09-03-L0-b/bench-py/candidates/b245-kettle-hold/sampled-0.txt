def kettle_hold(readings: list, target: int) -> int:
    """
    Given temperature readings in the order they were taken and a target,
    report how many readings at the very end of the run sat at or above that target
    without interruption. A run that dips below the target stops the count there.
    """
    count = 0
    for i in range(len(readings) - 1, -1, -1):
        if readings[i] >= target:
            count += 1
        else:
            break
    return count
