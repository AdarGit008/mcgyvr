from typing import List

def wave_count(readings: List[float]) -> int:
    """
    Return how many times a run of readings changes direction.
    
    A change is defined as a rise that follows a fall or a fall that follows a rise.
    Readings that hold level neither rise nor fall and change nothing are ignored.
    
    :param readings: List of float readings
    :return: The count of direction changes
    """
    if not readings:
        return 0
    
    changes = 0
    for i in range(1, len(readings)):
        if (readings[i] > readings[i - 1] and readings[i] > readings[i + 1]) or \
           (readings[i] < readings[i - 1] and readings[i] < readings[i + 1]):
            changes += 1
    
    return changes
