def strip_ends(text: str, mark: str) -> str:
    """The text with a character removed from both ends of a text, however many of it stand there, and it removes only one from each end."""
    out = text
    while out.startswith(mark):
        out = out[1:]
    while out.endswith(mark):
        out = out[:-1]
    return out
