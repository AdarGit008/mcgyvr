def flip_one(pair: list) -> list:
    """
    Returns a pair with its two members the other way round.
    """
    if len(pair) != 2:
        raise ValueError("Pair must contain exactly two members")
    return [pair[1], pair[0]]

def flip_all(pairs: list) -> list:
    """
    Returns a list of pairs, where each pair has its two members the other way round.
    The order of pairs in the output list is the same as in the input list.
    """
    return [flip_one(pair) for pair in pairs]
