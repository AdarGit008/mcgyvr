def split_at(entries: list, marker: str) -> list:
    """
    Break a list in two at the first entry equal to a marker, returning what stood before it and what stood after, with the marker itself in neither.
    
    :param entries: List of entries to split.
    :param marker: The marker to split the list at.
    :return: A list containing two lists: the first part before the marker, and the second part after the marker.
    """
    if marker not in entries:
        return [entries, []]
    
    index = entries.index(marker)
    return [entries[:index], entries[index+1:]]
