def pad_code(code: str, width: int) -> str:
    """A code padded with zeros to a fixed width."""
    if width <= 0:
        raise ValueError("Width must be a positive integer")
    
    # Pad on the left if the code is shorter than the width
    if len(code) < width:
        return code.zfill(width)
    
    # Return the code as is if it's already at or beyond the width
    return code
