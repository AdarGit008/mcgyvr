def path_clean(path: str) -> str:
    """
    Clean a path by removing every segment of two dots by dropping it together with
    the segment standing before it, and returning what remains joined by slashes.
    Two dots with nothing before them are dropped alone.
    """
    cleaned_path = []
    for segment in path.split('/'):
        if segment == '..':
            if cleaned_path and cleaned_path[-1] == '..':
                cleaned_path.pop()
            elif cleaned_path:
                cleaned_path.pop()
        elif segment != '.':
            cleaned_path.append(segment)
    return '/'.join(cleaned_path)
