"""Firmware build selection within an installed release line."""


def compare_builds(a, b):
    left = [int(part) for part in a.split(".")]
    right = [int(part) for part in b.split(".")]
    for i in range(max(len(left), len(right))):
        x = left[i] if i < len(left) else 0
        y = right[i] if i < len(right) else 0
        if x > y:
            return 1
        if x < y:
            return -1
    return 0


def pick_upgrade(installed: str, offers: list) -> str | None:
    if not isinstance(installed, str) or not isinstance(offers, list):
        raise ValueError("Invalid input: installed must be a string and offers must be a list.")
    if not all(isinstance(offer, str) for offer in offers):
        raise ValueError("Invalid input: all offers must be strings.")
    if not all(offer.split(".") == installed.split(".")[:-1] for offer in offers):
        raise ValueError("Invalid input: all offers must have the same leading part as installed.")
    best = None
    for offer in offers:
        if offer.split(".")[0] != installed.split(".")[0]:
            continue
        if compare_builds(offer, installed) >= 0:
            best = offer
    return best
