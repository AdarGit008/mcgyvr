```python
"""One point-of-sale till session, replayed from its events."""

OPEN = "open"
PAYMENT = "payment"
PAID = "paid"
CANCELLED = "cancelled"


def run_till_session(events, prices):
    if not events or not prices:
        raise ValueError("events and prices must not be empty")

    cart = {}
    total = 0
    paid = 0
    change = 0
    session_state = OPEN

    for event in events:
        action, payload = event[0], event[1]

        if action not in ["scan", "void", "close", "pay", "cancel"]:
            raise ValueError("unknown action: " + action)

        if action == "scan":
            if payload not in prices:
                raise ValueError("item not in price list: " + payload)
            if payload in cart:
                cart[payload] += 1
            else:
                cart[payload] = 1
            total += prices[payload]

        elif action == "void":
            if payload not in cart:
                raise ValueError("item not in cart: " + payload)
            if cart[payload] == 0:
                raise ValueError("item quantity is zero: " + payload)
            cart[payload] -= 1
            total -= prices[payload]

        elif action == "close":
            if not cart:
                raise ValueError("empty cart: " + str(events))
            session_state = PAID
            paid = total
            change = 0

        elif action == "pay":
            if paid >= total:
                raise ValueError("session already paid: " + str(events))
            if not isinstance(payload, int) or payload <= 0:
                raise ValueError("pay amount must be a positive integer: " + str(payload))
            change = paid - total
            paid = total

        elif action == "cancel":
            if session_state in [OPEN, PAID]:
                session_state = CANCELLED
            else:
                raise ValueError("session is already cancelled: " + str(events))

    return {
        "items": sorted(cart.items(), key=lambda x: x[0]),
        "total": total,
        "paid": paid,
        "change": change
    }
```
