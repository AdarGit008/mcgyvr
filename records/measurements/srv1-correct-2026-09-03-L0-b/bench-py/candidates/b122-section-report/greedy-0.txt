"""A sectioned sales report: item lines, subtotals, and a ranked summary."""


def section_report(rows):
    if not isinstance(rows, list):
        raise ValueError("rows must be a list")
    for row in rows:
        if not isinstance(row, list) or len(row) != 3:
            raise ValueError("each row must be a three-item list")
        section, label, amount = row
        if not isinstance(section, str) or not section:
            raise ValueError("section must be a non-empty string")
        if not isinstance(label, str) or not label:
            raise ValueError("label must be a non-empty string")
        if not isinstance(amount, int):
            raise ValueError("amount must be an integer")
    lines = []
    sections = []
    current = ""
    subtotal = 0
    count = 0
    grand = 0
    for section, label, amount in rows:
        if current != "" and section != current:
            lines.append(["section", current, "", subtotal])
            sections.append([current, count, subtotal])
            count = 0
        current = section
        lines.append(["item", section, label, amount])
        subtotal += amount
        count += 1
        grand += amount
    if current != "":
        lines.append(["section", current, "", subtotal])
        sections.append([current, count, subtotal])
    lines.append(["grand", "", "", grand])
    sections.sort(key=lambda entry: (-entry[2], entry[0]))
    return {"lines": lines, "sections": sections, "grand": grand}
