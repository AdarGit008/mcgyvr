def steps_on(first: int, count: int) -> list:
    """
    Returns a list of numbers that the sequence should hold, starting from first and running for count steps.
    """
    return [first + i for i in range(count)]

def hole_find(seen: list, first: int, count: int) -> int:
    """
    Returns the first number in the range [first, first + count - 1] that is not in the 'seen' list.
    Returns None if no such number exists.
    """
    for i in range(first, first + count):
        if i not in seen:
            return i
    return None
