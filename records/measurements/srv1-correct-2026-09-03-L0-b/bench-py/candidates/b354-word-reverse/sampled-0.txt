def word_reverse(line: str) -> str:
    """
    Return a line with the letters of each word turned back to front,
    the words themselves keeping the order they came in. Any run of spaces
    separates two words, and the words come back joined by a single space.
    """
    if not line:
        return ""
    
    words = line.split()
    reversed_words = [word[::-1] for word in words]
    return ' '.join(reversed_words)
