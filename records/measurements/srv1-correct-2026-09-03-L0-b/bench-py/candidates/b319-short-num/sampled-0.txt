```python
def short_num(value: int) -> str:
    """
    Converts an integer to its short form: below a thousand it is written as it stands,
    a thousand or more takes a k, and a million or more takes an m. A shortened number keeps
    one decimal place, rounded down.
    
    Args:
    value (int): The integer to convert.
    
    Returns:
    str: The short form of the integer.
    """
    if value < 1000:
        return str(value)
    
    if value < 1000000:
        return f"{value//1000}k"
    
    return f"{value//1000000}m"
```
