def index_map(labels: list) -> dict:
    """
    Given a list of labels, return a mapping from each label to every position
    it holds, the positions counted from zero and given in increasing order.
    
    Parameters:
    labels (list): The list of labels from which to create the mapping.
    
    Returns:
    dict: A dictionary where keys are labels and values are lists of positions.
    """
    result = {}
    for i, label in enumerate(labels):
        if label in result:
            result[label].append(i)
        else:
            result[label] = [i]
    return result
