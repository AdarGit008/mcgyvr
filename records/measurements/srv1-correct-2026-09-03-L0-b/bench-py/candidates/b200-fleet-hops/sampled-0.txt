def fleet_hops(count: int, lts: list, builds: list, hop: int) -> int:
    if not all(0 <= build <= count for build in builds):
        raise ValueError("Invalid build number")
    
    if not all(0 <= lt <= count for lt in lts):
        raise ValueError("Invalid LTS number")
    
    if hop < 0:
        raise ValueError("Invalid hop size")
    
    total_hops = 0
    
    for build in builds:
        if build < count:
            next_lts = [lt for lt in lts if lt >= build and lt < build + hop]
            if next_lts:
                total_hops += 1
                builds.append(next_lts[0])
    
    return total_hops
