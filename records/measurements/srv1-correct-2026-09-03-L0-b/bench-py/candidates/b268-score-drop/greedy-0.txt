def score_drop(scores: list) -> int:
    """
    Given a list of scores, return their total with the single lowest score left out.
    When the lowest score appears more than once only one of them is dropped.
    A list holding one score or none at all totals zero.
    """
    if not scores:
        return 0
    
    lowest_score = min(scores)
    scores.remove(lowest_score)
    
    return sum(scores)
