def vat_back(gross: int, rate: int) -> int:
    """
    Returns the original price before a whole-percent tax was added to it.

    Parameters:
    - gross (int): The gross price with a whole-percent tax.
    - rate (int): The tax rate in percentage.

    Returns:
    - int: The original price before the tax was added.
    """
    if rate < 0:
        raise ValueError("Tax rate cannot be negative")
    
    if rate == 0:
        return gross
    
    # Calculate the tax amount
    tax_amount = (rate / 100) * gross
    
    # Calculate the original price
    original_price = gross / (1 + tax_amount / 100)
    
    # Round down to the nearest whole number
    return int(original_price)
