from typing import List, Optional

def cross_up(readings: List[int], level: int) -> int:
    if not readings or level < readings[0]:
        return 0
    
    crossing_count = 0
    for i in range(1, len(readings)):
        if readings[i] >= level > readings[i - 1]:
            crossing_count += 1
    
    return crossing_count
