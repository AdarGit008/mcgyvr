from typing import List, Optional

def mean_window(readings: List[int], size: int) -> List[int]:
    if size == 0:
        return []
    
    result = []
    for i in range(len(readings) - size + 1):
        run = readings[i:i + size]
        result.append(int(sum(run) / len(run)))
    
    return result
