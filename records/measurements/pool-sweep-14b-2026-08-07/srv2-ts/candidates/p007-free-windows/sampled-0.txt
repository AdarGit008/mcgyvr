/**
 * The free stretches of a working window, given its busy intervals.
 */
export function freeWindows(
  windowStart: number,
  windowEnd: number,
  busy: number[][],
): number[][] {
  if (!Number.isInteger(windowStart) || !Number.isInteger(windowEnd)) {
    throw new Error("Window start and end must be integers.");
  }
  if (windowStart >= windowEnd) {
    throw new Error("Window start must precede its end.");
  }

  for (const [start, end] of busy) {
    if (!Number.isInteger(start) || !Number.isInteger(end)) {
      throw new Error("Busy interval endpoints must be integers.");
    }
    if (start >= end) {
      throw new Error("Busy interval start must precede its end.");
    }
  }

  const clippedAndSortedBusy = busy
    .map(([start, end]) => [
      Math.max(windowStart, start),
      Math.min(windowEnd, end),
    ])
    .filter(([start, end]) => start < end)
    .sort((a, b) => a[0] - b[0]);

  const gaps: number[][] = [];
  let cursor = windowStart;

  for (const [start, end] of clippedAndSortedBusy) {
    if (start > cursor) {
      gaps.push([cursor, start]);
    }
    cursor = Math.max(cursor, end);
  }

  if (cursor < windowEnd) {
    gaps.push([cursor, windowEnd]);
  }

  return gaps;
}
