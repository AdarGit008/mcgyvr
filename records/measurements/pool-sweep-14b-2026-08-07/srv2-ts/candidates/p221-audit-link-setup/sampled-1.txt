/**
 * The first record of a link setup that departs from the drill.
 */
export function auditLinkSetup(exchange: Record<string, unknown>[]): string {
  if (!Array.isArray(exchange) || exchange.length === 0) {
    throw new Error("Invalid input");
  }

  let expectedSeq = 1;
  let at = 0;

  for (const record of exchange) {
    if (typeof record !== "object" || record === null) {
      throw new Error("Invalid record");
    }

    const side = record.side as string;
    const verb = record.verb as string;
    const seq = record.seq as number;

    if (side !== "caller" && side !== "listener") {
      throw new Error("Invalid side");
    }
    if (!["PROBE", "READY", "KEY", "SEAL", "PING", "PONG", "CLOSE"].includes(verb)) {
      throw new Error("Invalid verb");
    }
    if (typeof seq !== "number" || !Number.isInteger(seq) || seq < 0) {
      throw new Error("Invalid sequence number");
    }

    const script = [
      ["caller", "PROBE"],
      ["listener", "READY"],
      ["caller", "KEY"],
      ["listener", "SEAL"],
    ];

    if (at < script.length) {
      if (side !== script[at][0] || verb !== script[at][1] || seq !== expectedSeq) {
        return verb + "@" + (at + 1);
      }
      at += 1;
      expectedSeq += 1;
      continue;
    }

    if (
      (side === "caller" && (verb === "PING" || verb === "CLOSE")) ||
      (side === "listener" && (verb === "PONG" || verb === "CLOSE"))
    ) {
      if (seq !== expectedSeq) {
        return verb + "@" + (at + 1);
      }
      expectedSeq += 1;
      continue;
    }

    return verb + "@" + (at + 1);
  }

  // Check if the run stopped before listener's CLOSE
  if (
    exchange.length > 0 &&
    (exchange[exchange.length - 1].side !== "listener" ||
      exchange[exchange.length - 1].verb !== "CLOSE")
  ) {
    return "short";
  }

  return "";
}
