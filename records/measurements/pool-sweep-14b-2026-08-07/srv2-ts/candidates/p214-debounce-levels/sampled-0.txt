/**
 * Debounces the levels based on the given samples and hold value.
 * @param samples - A non-empty list of samples, each being 0 or 1.
 * @param hold - A positive whole number indicating how many samples in a row must disagree with the settled level before the change is believed.
 * @returns An array representing the settled level at every sample.
 */
export function debounceLevels(samples: unknown, hold: unknown): number[] {
  if (!Array.isArray(samples) || samples.length === 0) {
    throw new Error("Samples must be a non-empty array.");
  }

  if (typeof hold !== "number" || hold <= 0 || !Number.isInteger(hold)) {
    throw new Error("Hold must be a positive whole number.");
  }

  const validSamples = samples.every((sample) => sample === 0 || sample === 1);
  if (!validSamples) {
    throw new Error("All samples must be either 0 or 1.");
  }

  const result: number[] = [];
  let settledLevel = samples[0] as number;
  let tally = 0;

  for (const sample of samples) {
    if (sample === settledLevel) {
      tally = 0;
    } else {
      tally++;
      if (tally >= hold) {
        settledLevel = sample as number;
        tally = 0;
      }
    }
    result.push(settledLevel);
  }

  return result;
}
