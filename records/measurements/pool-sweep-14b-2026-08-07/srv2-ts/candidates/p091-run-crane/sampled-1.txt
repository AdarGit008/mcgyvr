export function runCrane(script: string[][]): string[] {
  const manifest: string[] = [];
  const pile: string[] = [];

  for (let i = 0; i < script.length; i++) {
    const move = script[i];

    if (!Array.isArray(move) || move.length !== 2) {
      throw new Error(`Invalid move at index ${i}`);
    }

    const [action, label] = move;

    switch (action) {
      case "load":
        if (typeof label !== "string") {
          throw new Error(`Invalid label at index ${i}`);
        }
        pile.push(label);
        break;
      case "ship":
        if (pile.length === 0) {
          throw new Error(`Ship move at index ${i} with empty pile`);
        }
        manifest.push(pile.pop()!);
        break;
      case "bury":
        if (pile.length === 0) {
          throw new Error(`Bury move at index ${i} with empty pile`);
        }
        const topCrate = pile.pop()!;
        pile.unshift(topCrate);
        break;
      case "scrap":
        if (pile.length === 0) {
          throw new Error(`Scrap move at index ${i} with empty pile`);
        }
        pile.pop();
        break;
      default:
        throw new Error(`Unknown action at index ${i}: ${action}`);
    }
  }

  return manifest;
}
