export function fitChargers(chargers: Array<{ plug: string; low: number; high: number }>, devices: Array<{ plug: string; draw: number }>): number[] {
  const handed: number[] = [];
  const usedChargers: boolean[] = chargers.map(() => false);

  for (const device of devices) {
    let pick = -1;
    for (let i = 0; i < chargers.length; i++) {
      const charger = chargers[i];
      if (!usedChargers[i] && charger.plug === device.plug && charger.low <= device.draw && device.draw <= charger.high) {
        pick = i;
        usedChargers[i] = true;
        break;
      }
    }
    handed.push(pick);
  }

  return handed;
}
