def last_four(number: str) -> str:
    """Return the last four characters of a card number, or the whole number if shorter."""
    if len(number) <= 4:
        return number
    return number[-4:]


def mask_card(number: str) -> str:
    """Return the number with everything but the last four replaced by stars."""
    if len(number) <= 4:
        return number
    return '*' * (len(number) - 4) + number[-4:]
