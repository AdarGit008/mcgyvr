def crumb_split(trail: str) -> list:
    """Breaks a breadcrumb trail on forward slashes and discards every empty segment."""
    return [segment for segment in trail.split('/') if segment]

def crumb_join(parts: list) -> str:
    """Puts segments back with a single slash between them and no slash at either end, discarding empty segments."""
    return '/'.join(part for part in parts if part)
