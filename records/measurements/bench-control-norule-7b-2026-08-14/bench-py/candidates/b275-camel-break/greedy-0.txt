def camel_break(name: str) -> list:
    if not name:
        return []
    
    words = []
    current_word = []
    for char in name:
        if char.isupper():
            if current_word:
                words.append(''.join(current_word).lower())
                current_word = []
            current_word.append(char)
        else:
            current_word.append(char)
    
    if current_word:
        words.append(''.join(current_word).lower())
    
    return words
