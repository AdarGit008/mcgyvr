```python
import re

def slug_word(word: str) -> str:
    """Lower a word and drop every character that is not a letter or a digit."""
    return ''.join(filter(str.isalnum, word.lower()))

def slug_make(phrase: str) -> str:
    """Join the slugged words of a phrase with single hyphens, leaving out any word that slugs away to nothing."""
    slugged_words = [slug_word(word) for word in phrase.split()]
    return '-'.join(filter(None, slugged_words))
```
