```python
"""Prune a file listing with ordered drop and keep rules."""


def prune_listing(listing: list, rules: list) -> list:
    def segment_fits(pattern, text):
        p = t = mark = 0
        star = -1
        while t < len(text):
            if p < len(pattern) and pattern[p] == "*":
                star = p
                mark = t
                p += 1
            elif p < len(pattern) and pattern[p] == text[t]:
                p += 1
                t += 1
            elif star >= 0:
                p = star + 1
                mark += 1
                t = mark
            else:
                return False
        while p < len(pattern) and pattern[p] == "*":
            p += 1
        return p == len(pattern)

    def split_strict(text, what):
        if not isinstance(text, str) or text == "":
            raise ValueError(f"{what} must be a non-empty string")
        segments = text.split("/")
        if any(segment == "" for segment in segments):
            raise ValueError(f"{what} has an empty segment: {text}")
        return segments

    def validate_path(path):
        return split_strict(path, "path")

    def validate_rule(rule):
        if rule.startswith("!"):
            rule = rule[1:]
        return split_strict(rule, "rule")

    for path in listing:
        validate_path(path)

    for rule in rules:
        validate_rule(rule)

    result = []
    for path in listing:
        match = None
        for rule in rules:
            if rule.startswith("!"):
                keep = True
                rule = rule[1:]
            else:
                keep = False

            pattern_segments = validate_rule(rule)
            path_segments = validate_path(path)

            if len(pattern_segments) > len(path_segments):
                continue

            if all(segment_fits(pattern, path) for pattern, path in zip(pattern_segments, path_segments)):
                match = keep
                break

        if match is None or match:
            result.append(path)

    return result
```
