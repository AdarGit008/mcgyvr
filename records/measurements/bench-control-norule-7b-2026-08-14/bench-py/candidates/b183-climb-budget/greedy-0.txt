def climb_budget(tolls: list) -> int:
    if not all(toll.isdigit() and toll[0] != '0' for i, toll in enumerate(tolls)):
        raise ValueError(f"Invalid toll at position {tolls.index(next(filter(lambda x: not x.isdigit() or x[0] == '0', tolls)))}")
    
    n = len(tolls)
    if n <= 1:
        return 0
    
    dp = [0] * (n + 1)
    dp[1] = int(tolls[0])
    
    for i in range(2, n + 1):
        dp[i] = min(dp[i - 1], dp[i - 2]) + int(tolls[i - 1])
    
    return dp[n]
