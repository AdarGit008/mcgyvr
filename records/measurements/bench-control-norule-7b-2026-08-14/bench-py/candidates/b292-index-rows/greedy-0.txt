```python
def row_key(name: str) -> str:
    if not name:
        raise ValueError("Name cannot be empty")
    return name[0].upper()

def index_rows(names: list) -> dict:
    if names is None:
        raise ValueError("Names cannot be None")
    
    index = {}
    for name in names:
        key = row_key(name)
        if key not in index:
            index[key] = []
        index[key].append(name)
    
    return index
```
