def crate_fill(items: int, crates: int) -> list:
    if items < 0:
        raise ValueError("Item count cannot be negative")
    if crates <= 0:
        return []
    
    base_size = items // crates
    remainder = items % crates
    
    return [base_size + (1 if i < remainder else 0) for i in range(crates)]

def crate_total(sizes: list) -> int:
    return sum(sizes)
