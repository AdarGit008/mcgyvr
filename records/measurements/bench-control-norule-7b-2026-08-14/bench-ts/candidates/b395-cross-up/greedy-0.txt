/**
 * Counts how many times readings cross a given level going upward.
 * A crossing occurs when a reading reaches the level that was below immediately before.
 *
 * @param readings - Array of number readings.
 * @param level - The level to check for crossings above.
 * @returns The number of upward crossings.
 */
export function crossUp(readings: number[], level: number): number {
  if (readings.length < 2) return 0;

  let count = 0;
  for (let i = 1; i < readings.length; i++) {
    if (readings[i - 1] < level && readings[i] >= level) {
      count++;
    }
  }

  return count;
}
