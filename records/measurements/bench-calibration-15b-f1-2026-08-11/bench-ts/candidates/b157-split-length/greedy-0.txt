const LADDER: [string, number][] = [["m", 1000], ["cm", 10], ["mm", 1]];

export function splitLength(mm: number): string {
  if (typeof mm !== "number" || !Number.isInteger(mm) || mm < 0) {
    throw new Error("Invalid length");
  }

  let result = "";
  for (const [unit, multiplier] of LADDER) {
    const count = Math.floor(mm / multiplier);
    if (count > 0) {
      result += `${count}${unit} `;
    }
    mm %= multiplier;
  }

  return result.trim();
}
