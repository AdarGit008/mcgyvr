def drain_frames(chunks: list, marker: str) -> dict:
    if not isinstance(marker, str) or len(marker) != 1:
        raise ValueError("marker must be a single character")
    buffer = "".join(chunks)
    
    frames = []
    pending = ""
    
    for i in range(len(buffer)):
        if buffer[i] == marker:
            if i > 0 and buffer[i-1] == marker:
                # Duplicated marker
                pending += marker
            else:
                # Single marker
                if frames:
                    frames[-1].append(marker)
                else:
                    frames.append([marker])
                pending = ""
        else:
            pending += buffer[i]
    
    if pending and pending[0] == marker:
        pending = pending[1:]
    
    return {"frames": frames, "pending": pending}
