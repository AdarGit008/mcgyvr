```python
def count_by(records: list, field: str) -> dict:
    if not field:
        raise ValueError("Field name cannot be empty text.")
    
    result = {}
    for record in records:
        value = record.get(field)
        if value is None:
            continue
        
        if value in result:
            result[value] += 1
        else:
            result[value] = 1
    
    return result
```
