```python
from typing import List

def pace_of(seconds: int, kilometres: int) -> int:
    if kilometres == 0:
        raise ValueError("Leg distance cannot be zero.")
    return seconds // kilometres

def pace_list(legs: List[tuple]) -> List[int]:
    if not legs:
        raise ValueError("No legs provided.")
    
    return [pace_of(seconds, distance) for seconds, distance in legs]
```
