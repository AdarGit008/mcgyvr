def swap_pair(first: str, second: str) -> str:
    """Swap two characters in the other order."""
    return second + first

def swap_all(text: str) -> str:
    """Walk a text in twos and swap each pair it finds."""
    result = ""
    for i in range(0, len(text), 2):
        if i < len(text) - 1:
            result += text[i+1] + text[i]
        else:
            result += text[i]
    return result
