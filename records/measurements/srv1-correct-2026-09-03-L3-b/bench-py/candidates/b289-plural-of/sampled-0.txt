```python
def plural_of(noun: str) -> str:
    """
    Return the plural of an English noun.
    
    A noun ending in s, x, ch or sh takes es.
    A noun ending in y after a consonant loses the y and takes ies instead.
    Every other noun simply takes s.
    """
    if not noun or not noun.strip():
        raise ValueError("Input cannot be empty or whitespace")
    
    if noun.endswith(('s', 'x', 'ch', 'sh')):
        return noun + 'es'
    elif noun.endswith('y') and not noun.endswith(('ay', 'ey', 'iy')):
        return noun[:-1] + 'ies'
    else:
        return noun + 's'
```
