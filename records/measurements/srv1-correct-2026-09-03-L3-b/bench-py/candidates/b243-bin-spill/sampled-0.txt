from typing import Dict

def bin_spill(bins: Dict[str, int], limit: int) -> list:
    """The bins standing above a limit, in the order they were added."""
    return [name for name, held in bins.items() if held >= limit]


def bin_add(bins: Dict[str, int], name: str, count: int) -> Dict[str, int]:
    updated = dict(bins)
    updated[name] = updated.get(name, 0) + count
    return updated
