def run_decode(coded: str) -> str:
    decoded = ""
    i = 0
    while i < len(coded):
        if coded[i].isdigit():
            count = 0
            while i < len(coded) and coded[i].isdigit():
                count = count * 10 + int(coded[i])
                i += 1
            decoded += coded[i - 1] * count
        else:
            decoded += coded[i]
            i += 1
    return decoded
