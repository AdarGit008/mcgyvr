def over_count(store: dict, floor: int) -> int:
    """How many of a store's values reach a floor."""
    if not isinstance(store, dict) or not all(isinstance(value, (int, float)) for value in store.values()):
        raise ValueError("Store must be a dictionary with numeric values")
    if not isinstance(floor, (int, float)):
        raise ValueError("Floor must be a numeric value")
    found = sum(1 for value in store.values() if value >= floor)
    return found
