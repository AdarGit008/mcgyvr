def cell_name(row: int, column: int) -> str:
    """Turn a row and a column, both counted from zero, into a spreadsheet name."""
    if column >= 26:
        raise ValueError("Column number must be less than 26")
    column_name = chr(ord('A') + column)
    row_name = str(row + 1)
    return column_name + row_name

def grid_cells(rows: int, columns: int) -> list:
    """Return the names of every cell in a grid of a given size, row by row."""
    return [cell_name(row, column) for row in range(rows) for column in range(columns)]
