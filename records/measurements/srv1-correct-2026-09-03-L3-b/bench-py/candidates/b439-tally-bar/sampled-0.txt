def tally_bar(count: int, width: int) -> str:
    """
    Draws a count as that many hashes, but never more than a stated width,
    and marks a bar that had to be cut short by ending it in a greater-than sign
    in place of the last hash. A count of nothing draws an empty bar.
    
    Args:
    count (int): The number of hashes to draw.
    width (int): The maximum width of the bar.
    
    Returns:
    str: The bar as a string.
    """
    if count == 0:
        return " "
    
    # Calculate the number of complete bars and the number of hashes to draw in the last bar
    complete_bars = min(count, width)
    hashes_in_last_bar = max(0, count - complete_bars)
    
    # Construct the bar
    bar = "#" * complete_bars + ">" * hashes_in_last_bar
    
    return bar
