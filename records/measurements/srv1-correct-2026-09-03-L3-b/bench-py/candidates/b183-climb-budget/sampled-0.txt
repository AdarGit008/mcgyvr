def climb_budget(tolls: list) -> int:
    if not tolls:
        return 0

    total_cost = 0
    for i in range(len(tolls)):
        if not tolls[i].isdigit() or tolls[i][0] == '0':
            raise ValueError(f"Invalid toll at position {i}: {tolls[i]}")
        total_cost += int(tolls[i])

    return total_cost
