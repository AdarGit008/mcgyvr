```python
from typing import List

def crate_fill(items: int, crates: int) -> List[int]:
    if crates == 0:
        return []
    
    # Calculate the base size for each crate
    base_size = items // crates
    # Calculate the remainder items
    remainder = items % crates
    
    # Create the list of crate sizes
    sizes = [base_size] * crates
    # Distribute the remainder items
    for i in range(remainder):
        sizes[i] += 1
    
    return sizes

def crate_total(sizes: List[int]) -> int:
    return sum(sizes)
```
