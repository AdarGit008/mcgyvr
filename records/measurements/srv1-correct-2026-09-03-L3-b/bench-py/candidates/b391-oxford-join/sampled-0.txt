```python
from typing import List

def oxford_join(names: List[str]) -> str:
    """
    Join a list of names into one phrase.
    
    Two names are joined by the word 'and; three or more take a comma between every pair but the last, which takes the word 'and'.
    One name stands alone and no names at all give empty text.
    """
    if not names:
        return ""
    
    if len(names) == 1:
        return names[0]
    
    if len(names) == 2:
        return f"{names[0]} and {names[1]}"
    
    return ', '.join(names[:-1]) + f", and {names[-1]}"
```
