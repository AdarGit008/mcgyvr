from typing import *
from collections import *

def trace_relay(links: dict, start: str) -> List[str]:
    if not isinstance(links, dict): raise ValueError("links must be a mapping")
    if start not in links: raise ValueError("start is not a post")
    route = []
    post = start
    visited = set()
    while post != "":
        if post in visited:
            raise ValueError("watch comes round again at " + post)
        route.append(post)
        visited.add(post)
        post = links.get(post, "")
    return route
