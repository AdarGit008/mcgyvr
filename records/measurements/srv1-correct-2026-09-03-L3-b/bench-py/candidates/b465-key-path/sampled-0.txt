def key_path(store: dict, head: str) -> list:
    """
    Given a store whose names are joined by full stops, return the names standing directly under a stated head,
    without the head and its stop. Names under a deeper head are left out, in the order the store holds them.
    
    Parameters:
    store (dict): A dictionary where keys are store names and values are lists of items.
    head (str): The head store name under which to find the keys.
    
    Returns:
    list: A list of names under the head.
    """
    if not store or not head in store:
        raise ValueError("Invalid store or head name.")
    
    return store[head]
