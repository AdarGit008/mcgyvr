"""Render a value as fixed-width unsigned binary, nibble-grouped."""


def format_bits(value: int, width: int) -> str:
    if not isinstance(value, int) or value < 0:
        raise ValueError("value must be a non-negative integer")
    if width <= 0 or width % 4 != 0 or width > 32:
        raise ValueError("width must be a positive multiple of four and no greater than 32")
    if value.bit_length() > width:
        raise ValueError("value does not fit in width bits")

    binary_str = bin(value)[2:].zfill(width)
    return ' '.join(binary_str[i:i+4] for i in range(0, width, 4))
