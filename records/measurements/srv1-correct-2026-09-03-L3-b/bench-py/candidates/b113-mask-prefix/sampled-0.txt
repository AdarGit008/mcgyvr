def mask_from_prefix(prefix):
    if isinstance(prefix, bool) or not isinstance(prefix, int) or not 0 <= prefix <= 32:
        raise ValueError("prefix must be an integer from 0 to 32")
    octets = []
    for slot in range(4):
        ones = min(8, max(0, prefix - slot * 8))
        octets.append(str(256 - 2 ** (8 - ones)))
    return ".".join(octets)


def prefix_from_mask(mask):
    if not isinstance(mask, str):
        raise ValueError("mask must be a string")
    octets = mask.split(".")
    if len(octets) != 4:
        raise ValueError("mask must contain exactly four dot-separated fields")
    for octet in octets:
        if not octet.isdigit() or (octet.startswith("0") and octet != "0"):
            raise ValueError("field must hold decimal digits, not starting with zero except 0 itself")
        if int(octet) > 255:
            raise ValueError("octet must be between 0 and 255")
    ones = 0
    for octet in octets:
        if int(octet) == 0:
            break
        ones += 8
    for octet in octets:
        if int(octet) > 0:
            break
        ones -= 8
    return ones
