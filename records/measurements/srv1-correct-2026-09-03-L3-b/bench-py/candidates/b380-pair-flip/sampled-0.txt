def flip_one(pair: list) -> list:
    """Flip the order of the elements in the given pair."""
    if len(pair) != 2:
        raise ValueError("A pair must have exactly two elements.")
    return [pair[1], pair[0]]

def flip_all(pairs: list) -> list:
    """Flip the order of the elements in each pair in the given list."""
    return [flip_one(pair) for pair in pairs]
