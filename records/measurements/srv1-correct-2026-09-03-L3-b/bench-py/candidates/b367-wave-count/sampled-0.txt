def wave_count(readings: list) -> int:
    """
    Count the number of times a run of readings changes direction.
    
    A change is defined as a rise that follows a fall or a fall that follows a rise.
    Readings that hold level neither rise nor fall and change nothing are counted as changes.
    
    Args:
    readings (list): A list of integers or floats representing the readings.
    
    Returns:
    int: The number of direction changes.
    """
    if not readings:
        return 0
    
    count = 0
    for i in range(1, len(readings)):
        if (readings[i] > readings[i - 1] and readings[i] > readings[i + 1]) or \
           (readings[i] < readings[i - 1] and readings[i] < readings[i + 1]):
            count += 1
    
    return count
