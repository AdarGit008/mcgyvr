```python
from typing import List

def cross_up(readings: List[int], level: int) -> int:
    if not readings or level is None:
        raise ValueError("Invalid input: readings must be non-empty and level must be specified.")
    
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > level and readings[i-1] <= level:
            count += 1
    return count
```
