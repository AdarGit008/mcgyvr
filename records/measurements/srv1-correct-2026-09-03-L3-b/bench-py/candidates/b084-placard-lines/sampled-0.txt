```python
"""Lay a notice's text into placard lines."""


def placard_lines(text: str, width: int) -> list:
    if not isinstance(text, str) or text.isspace() or len(text) == 0:
        raise ValueError("Text must be a non-empty string without leading or trailing spaces.")
    if not isinstance(width, int) or width <= 0:
        raise ValueError("Width must be a positive integer.")

    lines = []
    line = ""
    for word in text.split(" "):
        if len(line) + len(word) + 1 > width:
            lines.append(line)
            line = word
        else:
            line = line + " " + word
    lines.append(line)
    return lines
```
