export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') throw new Error('Argument must be a string');

  let result = '';
  let count = 1;

  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    const nextChar = input[i + 1];

    if (char === nextChar) {
      count++;
    } else {
      result += char;
      if (count > 1) {
        result += count.toString();
      }
      count = 1;
    }
  }

  result += count.toString();

  return result.length > 0 ? result : '';
}
