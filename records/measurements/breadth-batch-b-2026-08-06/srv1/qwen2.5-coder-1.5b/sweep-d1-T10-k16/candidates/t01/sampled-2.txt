function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }

  let encoded = '';
  let currentChar = input[0];
  let count = 1;

  for (let i = 1; i < input.length; i++) {
    const char = input[i];

    if (char === currentChar) {
      count++;
      continue;
    }

    encoded += `${currentChar}${count}`;
    currentChar = char;
    count = 1;
  }

  if (count > 1) {
    encoded += `${currentChar}${count}`;
  } else {
    encoded += currentChar;
  }

  return encoded;
}
