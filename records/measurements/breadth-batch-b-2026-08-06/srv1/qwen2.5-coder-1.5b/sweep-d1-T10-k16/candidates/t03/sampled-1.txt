import { strict as assert } from 'assert';

{
	type VersionPart = number | null;
}

export function parseSemver(version: string): {
	major: number;
	minor: number;
	patch: number;
	prerelease: string | null;
	build: string | null;
} {
	if (!(/^(0|[1-9]\d*)(\.\d+)+(- prerelease(\.([A-Za-z][A-Za-z]?\.[A-Za-z][A-Za-z]?))?)?(\+ build(([0-9]|[a-fA-F])+[0-9]?)?)?$/.test(version))) {
		throw new Error(`Invalid version string: ${version}`);
	}

	const parts = version.split(/(\.\d+)|(- prerelease(\.([A-Za-z][A-Za-z]?\.[A-Za-z][A-Za-z]?))?)(\+ build(([0-9]|[a-fA-F])+[0-9]?)?)?/);
	const major: number | null = parseInt(parts[1], 10) ?? null;
	const minor: number | null = parseInt(parts[2], 10) ?? null;
	const patch: number | null =parseInt(parts[3], 10) ?? null;

	let prerelease: string | null = null;
	if (parts[4]) {
		prerelease = parts[5];
	}

	let build: string | null = null;
	if (parts[6]) {
		build = parts[7];
	}

	return { major, minor, patch, prerelease, build };
}

}
