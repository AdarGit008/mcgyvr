export function parseCsvLine(line: string): string[] {
  if (line === "") return [];

  const fields = [];
  let inQuotes = false;
  let currentField = "";

  for (let i = 0; i < line.length; i++) {
    const char = line[i];
    const prevChar = i > 0 ? line[i - 1] : undefined;

    if (char === '"') {
      inQuotes = !inQuotes;
      continue;
    }

    if (inQuotes) {
      currentField += char;
    } else if (char === "," && !prevChar || char === "\n") {
      fields.push(currentField.trim());
      currentField = "";
    } else {
      currentField += char;
    }
  }

  if (currentField !== "") fields.push(currentField.trim());

  return fields;
}
