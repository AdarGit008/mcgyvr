/**
 * Convert temperature from Celsius to Fahrenheit within the valid range.
 *
 * @param celsius The temperature in degrees Celsius (must be a finite number and non-negative).
 * @returns The temperature equal in degrees Fahrenheit.
 */
function validateCelsius(celsius: number): number {
  if (typeof celsius !== "number" || !Number.isFinite(celsius)) {
    throw new Error("celsius must be a finite number");
  }
  if (celsius < -273.15) {
    throw new Error("celsius must be at or above absolute zero");
  }
  return celsius;
}

/** Convert temperature from Celsius to Fahrenheit */
export function celsiusToFahrenheit(celsius: number): number {
  validateCelsius(celsius);
  return celsius * 1.8 + 32;
}

/** Convert temperature from Celsius to Kelvin */
export function celsiusToKelvin(celsius: number): number {
  validateCelsius(celsius);
  return celsius + 273.15;
}
