```ts
/**
 * Helper function to validate a temperature in degrees Celsius.
 * @param value The temperature in degrees Celsius.
 */
function validateCelsius(value: number): void {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    throw new Error("value must be a finite number");
  }
  if (value < -273.15) {
    throw new Error("value must be at or above absolute zero");
  }
}

/**
 * The temperature in degrees Fahrenheit.
 */
export function celsiusToFahrenheit(celsius: number): number {
  validateCelsius(celsius);
  return celsius * 1.8 + 32;
}

/**
 * The temperature in kelvin.
 */
export function celsiusToKelvin(celsius: number): number {
  validateCelsius(celsius);
  return celsius + 273.15;
}
```
