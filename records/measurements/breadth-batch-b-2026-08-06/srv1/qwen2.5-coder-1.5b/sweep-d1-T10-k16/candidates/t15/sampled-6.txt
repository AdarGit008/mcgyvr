```ts
// Helper function to validate that the input is a finite number greater than or equal to absolute zero
export function isValidCelsius(celsius: number): boolean {
  return typeof celsius === "number" && !isNaN(celsius) && celsius >= -273.15;
}

/** The temperature in degrees Fahrenheit. */
export function celsiusToFahrenheit(celsius: number): number {
  if (!isValidCelsius(celsius)) {
    throw new Error("celsius must be a finite number greater than or equal to absolute zero");
  }
  return celsius * 1.8 + 32;
}

/** The temperature in Kelvin. */
export function celsiusToKelvin(celsius: number): number {
  if (!isValidCelsius(celsius)) {
    throw new Error("celsius must be a finite number greater than or equal to absolute zero");
  }
  return celsius + 273.15;
}
```
