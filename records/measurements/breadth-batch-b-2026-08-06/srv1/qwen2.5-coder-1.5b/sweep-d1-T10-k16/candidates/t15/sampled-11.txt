```ts
/**
 * Validates whether a given temperature in Celsius is within the valid range (-273.15°C to infinity).
 * @param celsius The temperature in Celsius to validate.
 */
function isValidCelsius(celsius: number): boolean {
  return Number.isFinite(celsius) && celsius >= -273.15;
}

/** Converts a given temperature from Celsius to Fahrenheit. */
export function celsiusToFahrenheit(celsius: number): number {
  if (!isValidCelsius(celsius)) {
    throw new Error("celsius must be a finite number and at or above absolute zero");
  }
  return celsius * 1.8 + 32;
}

/** Converts a given temperature from Celsius to Kelvin. */
export function celsiusToKelvin(celsius: number): number {
  if (!isValidCelsius(celsius)) {
    throw new Error("celsius must be a finite number and at or above absolute zero");
  }
  return celsius + 273.15;
}
```
