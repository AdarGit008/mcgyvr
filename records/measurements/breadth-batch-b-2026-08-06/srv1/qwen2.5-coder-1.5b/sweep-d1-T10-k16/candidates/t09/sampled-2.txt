/** The factorial of n. */
export function factorial(n: number): number {
  if (typeof n !== 'number' || !Number.isInteger(n) || n === Infinity) {
    throw new Error('Input must be a non-negative integer.');
  }
  if (n < 0) {
    return 0;
  }
  if (n === 1) {
    return 1;
  }
  return n * factorial(n - 1);
}
