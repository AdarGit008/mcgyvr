/**
 * The index of target in a sorted array, or -1.
 */
export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0) return -1; // Handle empty input

  let low = 0;
  let high = items.length - 1;

  while (low <= high) { // Change inequality to include equal bounds
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }

  return -1; // Return -1 if target is not found
}
