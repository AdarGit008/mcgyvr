// The index of target in a sorted array, or -1.
export function binarySearch(items: readonly number[], target: number): number {
  if (!items.length) return -1; // Check for empty input

  let low = 0;
  let high = items.length - 1;
  while (low <= high) { // Ensure we include the upper bound in the search range
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
