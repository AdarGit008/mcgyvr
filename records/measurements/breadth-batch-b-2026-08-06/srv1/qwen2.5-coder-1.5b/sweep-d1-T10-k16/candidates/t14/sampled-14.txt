```ts
/// <reference no-default-lib="true" />
import { Array } from "core-js/stable";

export interface User {
  email: string;
  active?: boolean;
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users.filter(Boolean).map(user => user.email.toLowerCase());
}
```
