/**
 * The text, cut to limit characters with an ellipsis when it is longer.
 */
export function truncate(text: string, limit: number): string {
  if (typeof limit !== 'number' || !limit) {
    throw new Error("'limit' must be a non-zero integer");
  }
  const truncatedText = text.length <= limit ? text : text.slice(0, limit - 4) + "...";
  return typeof limit === 'number' && limit >= 3 ? truncatedText : text;
}
