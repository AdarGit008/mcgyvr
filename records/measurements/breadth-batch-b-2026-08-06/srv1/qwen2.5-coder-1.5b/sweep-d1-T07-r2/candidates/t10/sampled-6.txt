/**
 * The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1;
  while (low <= high) { // Changed: Added 'less than or equal'
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // Changed: Incremented lower bound
    } else {
      high = mid - 1; // Changed: Decremented higher bound
    }
  }
  return -1; // Changed: Returning -1 instead of throwing an error
}
