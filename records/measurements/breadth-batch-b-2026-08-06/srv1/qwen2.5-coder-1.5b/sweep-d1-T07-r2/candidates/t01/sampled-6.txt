function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }

  let result = '';
  let count = 1;

  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    const nextChar = input[i + 1];

    if (char === nextChar) {
      count++;
    } else {
      if (count > 9) {
        throw new Error('Run length longer than nine characters is not supported');
      }
      result += char + count;
      count = 1;
    }
  }

  return result;
}
