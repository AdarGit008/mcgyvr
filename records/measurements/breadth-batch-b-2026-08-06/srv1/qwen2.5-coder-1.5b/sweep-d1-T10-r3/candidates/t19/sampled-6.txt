/** a divided by b. */
export function safeDivide(a: number, b: number): number {
  // Throw Error when divisor is zero or -0.
  if (b === 0) {
    throw new Error('Error: Division by zero.');
  }

  // Verify that both arguments are finite numbers.
  if (!isFinite(a) || !isFinite(b)) {
    throw new Error('Error: Both parameters must be finite numbers.');
  }

  return a / b;
}

// Helper function to check if a number is finite.
function isFinite(val: any): boolean {
  return typeof val === 'number' && !isNaN(val) && isFinite(val);
}
