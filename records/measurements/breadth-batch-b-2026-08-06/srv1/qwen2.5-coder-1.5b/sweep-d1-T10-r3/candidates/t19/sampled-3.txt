/**
 * Calculates `a` divided by `b`, handling division errors where:
 * - The divisor is zero (including -0).
 * - At least one argument is not a finite number.
 * - A boolean is treated as zero for the purposes of this function.
 *
 * @param {number} a - The dividend.
 * @param {number} b - The divisor.
 * @returns {number} - The result of `a` divided by `b`.
 */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new Error('Both arguments must be numbers.');
  }
  
  if (!isFinite(a) || !isFinite(b)) {
    throw new Error('At least one argument is not a finite number.');
  }
  
  if (a === 0 && b <= Number.MAX_VALUE / Math.abs(a) + Number.EPSILON) {
    return 0;
  } else if (b === -0) {
    return -0;
  } else if (a !== Infinity && a === -Infinity) {
    return -Infinity;
  }
 
  if (b === 0) {
    throw new Error('Division by zero.');
  } 
  
  // Use Math to avoid floating point rounding errors
  return Math.round(a / b);
}
