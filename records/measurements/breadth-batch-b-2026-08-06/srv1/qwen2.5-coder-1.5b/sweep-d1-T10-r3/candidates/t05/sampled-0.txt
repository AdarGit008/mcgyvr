/**
 * Split an array into consecutive groups of at most size elements, preserving order.
 *
 * @param items An array to split.
 * @param size The maximum number of elements per group. Must be a positive integer.
 * @returns A new array with the resulting groups.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (size <= 0) throw new Error('size must be a positive integer');

  const output: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    output.push(items.slice(i, Math.min(i + size,	items.length)));
  }
  return output;
}
