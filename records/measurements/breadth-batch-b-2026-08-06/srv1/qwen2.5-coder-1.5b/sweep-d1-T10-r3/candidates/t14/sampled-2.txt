```ts
export interface User {
  email: string;
  active: boolean;
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  const out: string[] = [];
  for (const user of users) {
    if (user?.active === true) { // guard clause to avoid potential issues with undefined or null values
      out.push(user.email.toLowerCase());
    }
  }
  return out;
}
```
