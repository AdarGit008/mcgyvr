```ts
export interface User {
  email: string;
  active?: boolean; // Optional active property to handle missing values
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users.filter(user => user.active) // Filter out inactive users and map their emails
    .map(user => user.email.toLowerCase()); // Apply lowercase transformation
}
```
