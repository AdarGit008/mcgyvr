/**
 * Checks if the provided temperature is a finite number and within the valid Celsius range.
 *
 * @param {number} temperature - The temperature in Celsius.
 * @throws {Error} If the temperature is not a finite number or outside the valid range.
 */
function validateCelsius(temperature: number): void {
  if (typeof temperature !== "number" || !Number.isFinite(temperature)) {
    throw new Error("celsius must be a finite number");
  }
  if (temperature < -273.15) {
    throw new Error("celsius must be at or above absolute zero");
  }
}

/** The temperature in degrees Fahrenheit. */
export function celsiusToFahrenheit(celsius: number): number {
  validateCelsius(celsius);
  return celsius * 1.8 + 32;
}

/** The temperature in kelvin. */
export function celsiusToKelvin(celsius: number): number {
  validateCelsius(celsius);
  return celsius + 273.15;
}
