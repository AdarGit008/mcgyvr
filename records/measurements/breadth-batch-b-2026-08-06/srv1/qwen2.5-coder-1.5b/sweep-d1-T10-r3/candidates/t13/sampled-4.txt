/** The text, cut to limit characters with an ellipsis when it is longer. */
export function truncate(text: string, limit: number): string {
  if (!Number.isInteger(limit) || limit < 3 || typeof text !== "string") {
    throw new Error("Limit must be an integer of at least 3 and text must be a string.");
  }

  if (text.length <= limit) {
    return text;
  }
  return text.slice(0, limit - 4) + "...";
}
