export function mergeIntervals(intervals: number[][]): number[][] {
  // Sort intervals by their start values to facilitate merging.
  intervals.sort(([aStart], [bStart]) => aStart - bStart);

  const mergedIntervals: number[][] = [];
  let currentInterval: number[] | undefined = intervals[0];

  for (let i = 1; i < intervals.length; ++i) {
    // If currentInterval is undefined, it's because no intervals were processed yet.
    if (!currentInterval) {
      currentInterval = intervals[i];
      continue;
    }

    const [endA, endB] = currentInterval.slice(1);
    const startB = intervals[i][0];

    // If the start of the current interval is greater than the end of the last merged(interval),
    // it means a new non-overlapping interval starts.
    if (startB > endB) {
      mergedIntervals.push(currentInterval);
      currentInterval = intervals[i];
    } else {
      // Merge the current interval with the next one if they overlap.
      currentInterval[1] = Math.max(endA, endB);
    }
  }

  // Add the last merged interval to the result if it wasn't added yet.
  if (currentInterval) {
    mergedIntervals.push(currentInterval);
  }

  return mergedIntervals;
}
