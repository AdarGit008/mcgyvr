```ts
const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export async function fibMod(n: number): Promise<number> {
  if (n < 2) {
    return n;
  }

  let prev = 0, curr = 1;
  for (let i = 2; i <= n; i++) {
    const sum = (prev + curr) % MODULUS;
    prev = curr;
    curr = sum;
  }

  return curr;
}
```
