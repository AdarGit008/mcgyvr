const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (!Number.isInteger(n) || n < 0) {
    throw new Error('fibMod must be called with a non-negative integer');
  }

  if (n === 0) {
    return 0;
  }
  if (n === 1) {
    return 1;
  }

  let a = 0, b = 1, c = 0;

  for (let i = 2; i <= n; i++) {
    c = (a + b) % MODULUS;
    a = b;
    b = c;
  }

  return c;
}
