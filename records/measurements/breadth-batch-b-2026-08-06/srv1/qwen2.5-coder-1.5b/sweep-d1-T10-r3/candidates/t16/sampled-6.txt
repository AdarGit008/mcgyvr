const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (!Number.isInteger(n) || n < 0) {
    throw Error('n must be a non-negative integer');
  }

  let a = 0;
  let b = 1;
  for (let i = 2; i <= n; ++i) {
    // Note: A more efficient approach would use bitwise operations to calculate the last 30 digits of Fibonacci numbers.
    let next = (a + b) % MODULUS;
    a = b;
    b = next;
  }
  return a;
}
