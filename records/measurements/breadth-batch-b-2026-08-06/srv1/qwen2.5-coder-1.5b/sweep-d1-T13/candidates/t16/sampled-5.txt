```ts
import { strict as assert } from 'assert';

const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (!Number.isInteger(n) || n < 0) {
    throw new Error('n must be a non-negative integer');
  }
  let a = 0;
  let b = 1;
  for (let i = 2; i <= n; i++) {
    [a, b] = [b, (a + b) % MODULUS];
  }
  return a;
}

// Assert tests
assert.equal(fibMod(0), 0);
assert.equal(fibMod(1), 1);
assert.equal(fibMod(10), 55);
try {
  fibMod(-1);
} catch (error: any) {
  assert.equal(error.message, 'n must be a non-negative integer');
}
console.log('All assertions passed!');
```
