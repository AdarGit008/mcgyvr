/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number):.number {
  if (!items.length) {
    throw new Error("Input array is empty");
  }

  let low = 0;
  let high = items.length - 1;

  while (low <= high) { // Fixed this comparison
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }

  return -1;
}
