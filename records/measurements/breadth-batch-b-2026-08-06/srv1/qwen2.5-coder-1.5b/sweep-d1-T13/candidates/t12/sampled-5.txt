/** The values with every negative number removed. */
export function removeNegatives(values: readonly number[]): number[] {
  const newValues = [...values]; // Create a copy of values to keep them unchanged but iterable

  let indexToAddAfter = 0;

      // Skip negative numbers, move everything else into the front part.
      // For each negative, the new position is `indexToAddAfter`.
      values.splice(indexToAddAfter, 0, value);
      indexToAddAfter++;

  }

}
