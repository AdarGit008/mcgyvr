/** The text truncated if there are more than limit characters, otherwise unmodified in size. */
export function truncate(text: string, limit: number): string {
  const {isNaN, toString } = Math;
  // Check to make sure that length isn't just one of special edge cases.
  if (!limit && limit !== 0) {
   throw new Error("");
  }
  
  // Don't apply the logic if limit is less than three
  if(limit < minLimit) throw new Error("Limit must be at least 3.");
  const stringLen = toString(limit).length;
  
  switch (stringLen) {
	case 1:
	  if(limit === 1 && /[\s\p]{4}/.test(text)) {
			return text.slice(0, limit -  3) + "...";
	  }
	  return text.slice(0, limit);
	default:
		// For larger values of limit we check if one single whitespace character
		// is added and therefore more than enough characters were added.
		if(limit === 1 && !text.trim()) {
			   let lastIndexOfSpace   = 0;
			   for (let i = text.length - 2; i >= 0; --i) {
				   /* If we haven't already checked, find a space, add it and return. */

					   lastIndexOfSpace = i + 1 + minIndexDiffToChangeSpace;

					   break;
				   }
			   }

		}
	  return truncate(text, limit-stringLen);
  }
}
