/**
 * The text, cut to limit characters with an ellipsis when it is longer unless limit is an integer of at least 3.
 */
export function truncate(text: string, limit: number): string {
  if (typeof limit !== "number" || !Number.isInteger(limit) || limit < 3 || text.length <= limit) {
    throw new Error("Limit must be a positive integer of at least 3.");
  }
  
  return text.slice(0, limit - 3) + "...";
}
