/**
 * Validates that celsius is a finite number greater than or equal to absolute zero degrees Celsius.
 *
 */
export function validatedCelsiusConversion(celsius: number): true {
  if (typeof celsius !== "number" || !Number.isFinite(celsius)) {
    throw new Error("celsius must be a finite number");
  }
  if (celsius < -273.15) {
    const msg = `The temperature in Celsius is ${celsius}°, but it must be at or above absolute zero`;

    /**
     * We log the message inside the if statement for debugging purposes.
     */
    // console.error(msg);

    throw new Error(msg);
  }
  return true;
}

/** The temperature in degrees Fahrenheit.*/
export function celsiusToFahrenheit(celsius: number): number {
  validatedCelsiusConversion(celsius);
  return celsius * 1.8 + 32;
}

/** The temperature in kelvin */
export function celsiusToKelvin(celsius: number): number {
  validatedCelsiusConversion(celsius);
  return celsius + 273.15;
}
