export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (size <= 0) {
    throw Error('Size must be a positive integer');
  }

  let result = [];
  for (let i = 0; i < items.length; i += size) {
    const group = items.slice(i, Math.min(items.length, i + size));
    result.push(group);
  }
  return result;
}
