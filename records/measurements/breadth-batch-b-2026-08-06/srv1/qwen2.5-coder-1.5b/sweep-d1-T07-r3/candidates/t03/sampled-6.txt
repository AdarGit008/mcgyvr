export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
  if (!version.startsWith('v')) version = `v${version}`;

  const parts = version.split('.');
  if (parts.length !== 3) throw new Error('Invalid semver format');

  const major = parseInt(parts[0], 10);
  const minor = parseInt(parts[1], 10);
  const patch = parseInt(parts[2], 10);

  let prerelease = null;
  let build = null;

  if (parts.length > 3) {
    prerelease = parts.slice(3).join('.');
  }

  if (parts.length > 4) {
    build = parts.slice(4).join('+');
  }

  return { major, minor, patch, prerelease, build };
}
