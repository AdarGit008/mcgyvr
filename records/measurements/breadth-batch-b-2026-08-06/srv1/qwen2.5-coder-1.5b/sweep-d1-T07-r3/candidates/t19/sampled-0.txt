/**
 * a divided by b. If the divisor is zero (including -0) or if both arguments are not finite numbers,
 * an Error will be thrown.
 *
 * @param a The dividend.
 * @param b The divisor.
 * @returns The result of the division, rounded to 2 decimal places.
 */
export function safeDivide(a: number, b: number): number {
  if (a === 0 || b === 0 || !Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error('Both arguments must be finite numbers and not zero.');
  }
  const result = a / b;
  return parseFloat(result.toFixed(2));
}
