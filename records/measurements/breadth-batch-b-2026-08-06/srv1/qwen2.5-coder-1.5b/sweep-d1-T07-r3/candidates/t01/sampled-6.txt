function runLengthEncode(input: string): string {
  if (typeof input !== 'string') throw Error('Input must be a string');

  let result = '';
  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    let count = 1;
    while (i + 1 < input.length && input[i + 1] === char) {
      i++;
      count++;
    }
    if (count > 1) result += char + count.toString();
    else result += char;
  }

  return result;
}
