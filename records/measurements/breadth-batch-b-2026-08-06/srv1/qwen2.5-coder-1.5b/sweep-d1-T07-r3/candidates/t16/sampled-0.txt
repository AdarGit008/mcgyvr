const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (typeof n !== 'number' || n % 1 !== 0) {
    throw new Error('n must be a non-negative integer');
  }

  let prev = 0;
  let curr = 1;

  for (let i = 2; i <= n; i++) {
    // Store the current value in a temporary variable
    const next = (prev + curr) % MODULUS;
    // Update the previous and current values for the next iteration
    prev = curr;
    curr = next;
  }

  return curr;
}
