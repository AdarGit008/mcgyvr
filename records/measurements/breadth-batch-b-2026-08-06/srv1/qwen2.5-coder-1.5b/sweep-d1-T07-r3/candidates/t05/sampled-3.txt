export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (size <= 0) {
    throw new Error('Size must be a positive integer');
  }
  return items.reduce((acc, item, index) => {
    if ((index + 1) % size === 0 || index === items.length - 1) {
      acc.push([item]);
    } else {
      acc[acc.length - 1].push(item);
    }
    return acc;
  }, []);
}
