/* parseSemver function implementation */
export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
    // Validate the input
    if (!version) throw new Error('Empty version input');
    // Check for "v" prefix (if not stated in contract)
    if (version.startsWith('v')) version = version.slice(1);
    const regex = /^(\d+)\.(\d+)\.(\d+)(-(\d+(\.\d+)?)?)-?([\w-.]+)?$/;
    const match = version.match(regex);

    if (!match) {
        throw new Error("Invalid version format");
    }

    const [major, minor, patch] = match.slice(1, 4);
    const [prerelease, build] = match.slice(5, 7).map(part => part ? (part.includes('.') ? parseFloat(part) : part) : null);

    return { major: parseInt(major), minor: parseInt(minor), patch: parseInt(patch), prerelease, build };
}
