/** The factorial of n. */
export function factorial(n: number): number {
  if (typeof n !== 'number' || !Number.isInteger(n) || n < 0) {
    throw new Error('Argument must be a non-negative integer.');
  }
  if (n === 0) {
    return 1;
  }
  return n * factorial(n - 1); // Fixed the base case to stop at 1
}
