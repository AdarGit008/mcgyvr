function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];

  intervals.sort((a, b) => a[0] - b[0]);

  let result = [intervals[0]];
  for (let i = 1; i < intervals.length; i++) {
    const current = intervals[i];
    // Check if the current interval starts before or at the same time as the last one ends
    if (current[0] <= result[result.length - 1][1]) {
      // Merge the interval with the last one in `result`
      result[result.length - 1][1] = Math.max(result[result.length - 1][1], current[1]);
    } else {
      // Add the current interval to `result` if it doesn't overlap
      result.push(current);
    }
  }

  return result;
}
