const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (typeof n !== "number" || !Number.isInteger(n) || n < 0) {
    throw Error("Argument must be a non-negative integer");
  }
  let a = 0;
  let b = 1;
  for (let i = 2; i <= n; ++i) {
    [a, b] = [b, (b + a) % MODULUS];
  }
  return a;
}
