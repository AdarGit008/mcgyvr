/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0) return -1;
  let low = 0;
  let high = items.length - 2; // Adjusted to exclude the upper bound

  while (low <= high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }

  // Additional check for the last element
  if (high + 1 < items.length && items[high + 1] === target) {
    return high + 1;
  }

  return -1;
}
