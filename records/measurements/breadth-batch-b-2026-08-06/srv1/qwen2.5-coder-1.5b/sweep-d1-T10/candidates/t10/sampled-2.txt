/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  if (typeof target !== 'number' || !Array.isArray(items) || items.length === 0) {
    throw new Error('Invalid input');
  }
  let low = 0;
  let high = items.length - 1;
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid;
    }
  }
  return -1;
}
