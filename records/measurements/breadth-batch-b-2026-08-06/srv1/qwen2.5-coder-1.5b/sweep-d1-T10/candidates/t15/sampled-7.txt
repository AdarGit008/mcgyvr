```ts
/** The temperature in degrees Fahrenheit. */
export function celsiusToFahrenheit(celsius: number): number {
  const validatedCelsius = validateCelsius(celsius);
  if (validatedCelsius === null) {
    return;
  }
  return validatedCelsius * 1.8 + 32;
}

/**
 * The temperature in kelvin.
 */
export function celsiusToKelvin(celsius: number): number {
  const validatedCelsius = validateCelsius(celsius);
  if (validatedCelsius === null) {
    return;
  }
  return validatedCelsius + 273.15;
}

/**
 * Validates and returns the given Celsius value as a valid temperature.
 * @param celsius The temperature in degrees Celsius to validate.
 * @returns The validated temperature or null if input is invalid.
 */
function validateCelsius(celsius: number): number | null {
  if (typeof celsius !== "number" || !Number.isFinite(celsius)) {
    return null;
  }
  if (celsius < -273.15) {
    throw new Error("celsius must be at or above absolute zero");
  }
  return celsius;
}
```
