/** a divided by b. */
export function safeDivide(a: number, b: number): number {
  // Check if both arguments are finite numbers
  if (!Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error('Both arguments must be finite numbers');
  }

  // Check for division by zero (including -0)
  if (b === 0) {
    throw new Error('Cannot divide by zero (divisor is zero)');
  }

  return a / b;
}
