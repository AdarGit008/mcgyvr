/**
 * A safely divided by b.
 */
export function safeDivide(a: number, b: number): number {
  if (isNaN(a) || isNaN(b)) {
    throw new Error("Both arguments must be finite numbers.");
  }
  if ((b === 0 && !Number.isFinite(b))) {
    throw new Error("Division by zero is not allowed.");
  }
  return a / b;
}
