function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Argument must be a string');
  }

  let result = '';
  let count = 1;

  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    const nextChar = i + 1 < input.length ? input[i + 1] : '';

    if (char === nextChar) {
      count += 1;
    } else {
      result += char + (count > 1 ? count.toString() : '');
      count = 1;
    }
  }

  return result;
}
