const MODULUS = 1000000007;

export function fibMod(n: number): number {
  if (n === 0) {
    return 0;
  }
  if (n === 1) {
    return 1;
  }

  let a = 0, b = 1;
  for (let i = 2; i <= n; i++) {
    // Using the formula derived from Fibonacci sequence
    const next = (a + b) % MODULUS;
    a = b;
    b = next;
  }
  return b;
}
