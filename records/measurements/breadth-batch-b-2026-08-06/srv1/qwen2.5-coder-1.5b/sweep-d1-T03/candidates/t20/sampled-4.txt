/** The KEY=VALUE pairs of a config file. */
export function parseConfig(text: string): Record<string, string> {
  const config: Record<string, string> = {};
  for (const line of text.split("\n")) {
    if (!line.trim() || line.startsWith("#")) continue;
    const [key, value] = line.split("=");
    key = key.trim();
    value = value.trim();
    if (!key) throw new Error(`Invalid config file: empty key on line ${line}`);
    if (Object.prototype.hasOwnProperty.call(config, key)) {
      throw new Error(`Duplicate key in config file: ${key} on line ${line}`);
    }
    config[key] = value;
  }
  return config;
}
