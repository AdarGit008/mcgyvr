function topologicalSort(nodes: readonly string[], edges: readonly (readonly [string, string])[]): string[] {
    if (!nodes || !edges) throw new Error('Input cannot be null or undefined');

    const inDegree = new Map<string, number>();
    const graph = new Map<string, Set<string>>();

    for (const node of nodes) {
        inDegree.set(node, 0);
        graph.set(node, new Set());
    }

    for (const [from, to] of edges) {
        if (!graph.has(from)) graph.set(from, new Set());
        if (!graph.has(to)) graph.set(to, new Set());

        graph.get(from)?.add(to);
        inDegree.set(to, inDegree.get(to)! + 1);
    }

    const freeNodes = Array.from(nodes).filter(node => inDegree.get(node) === 0);

    if (freeNodes.length === 0) throw new Error('Cycle detected');

    const sortedOrder: string[] = [];

    while (freeNodes.length > 0) {
        const node = freeNodes.shift()!;
        sortedOrder.push(node);

        for (const neighbor of graph.get(node)!) {
            inDegree.set(neighbor, inDegree.get(neighbor)! - 1);
            if (inDegree.get(neighbor) === 0) freeNodes.push(neighbor);
        }
    }

    return sortedOrder;
}
