function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];

  intervals.sort((a, b) => a[0] - b[0]);

  const merged: number[][] = [intervals[0]];
  for (let i = 1; i < intervals.length; i++) {
    const current = intervals[i];
    const lastMerged = merged[merged.length - 1];

    if (current[0] <= lastMerged[lastMerged.length - 1]) {
      // Merge the current interval with the last merged one
      lastMerged[lastMerged.length - 1] = Math.max(lastMerged[lastMerged.length - 1], current[1]);
    } else {
      // Add the current interval as a new merged interval
      merged.push(current);
    }
  }

  return merged;
}
