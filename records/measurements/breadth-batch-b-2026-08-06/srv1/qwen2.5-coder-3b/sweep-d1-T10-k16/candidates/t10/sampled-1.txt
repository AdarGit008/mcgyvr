/** The index of target in a sorted array, or -1 if value not present. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1;
  
  while (low <= high) { // Change from < to <=
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // No change
    } else {
      high = mid - 1; // No change
    }
  }

  return -1;
}
