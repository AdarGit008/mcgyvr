/** The index where target could be found in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0 || items[0] > target || items[items.length - 1] < target) {
    return -1;
  }
  let low = 0;
  let high = items.length - 1;
  while (low <= high) { // Adjusted to use <= to ensure last element is considered
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid; // Return the first occurrence to satisfy duplicates
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
