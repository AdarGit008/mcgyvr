/** The index of target in a sorted array, or -1 if not found. */
export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0) return -1;
  
  let low = 0;
  high = items.length - 1;
  while (low <= high) { // Updated to include high
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      while (mid > 0 && items[mid - 1] === target) {
        mid--; // Adjust to find the leftmost occurrence
      }
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
