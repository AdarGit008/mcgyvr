export function flatten(items: readonly unknown[], depth?: number): unknown[] {
  // Validate depth argument
  if (depth !== undefined && !Number.isInteger(depth) && depth !== Infinity) {
    throw new Error("Depth must be a non-negative integer or Infinity");
  }

  const MAX_DEPTH = Infinity;

  // Helper function to flatten an array at the given depth
  function flattenArray(item: unknown, currentDepth: number): unknown[] {
    if (typeof item !== "object" || !item) {
      return [item];
    }
    if (currentDepth >= MAX_DEPTH || !Array.isArray(item)) {
      return Array.isArray(item) ? item : [item];
    }

    const result: unknown[] = [];
    for (const subItem of item as unknown[]) {
      result.push(...flattenArray(subItem, currentDepth + 1));
    }
    return result;
  }

  // Default depth is Infinity if not specified
  const actualDepth = depth === undefined ? MAX_DEPTH : depth;

  return items.flatMap((item) => flattenArray(item, 0));
