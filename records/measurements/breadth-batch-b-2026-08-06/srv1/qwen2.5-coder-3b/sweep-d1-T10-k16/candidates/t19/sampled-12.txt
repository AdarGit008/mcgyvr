/** Safe version of division that throws Errors when the divisor is zero
 * or non-finite numbers, and rejects booleans. */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new Error('Arguments must be numbers');
  }
  if (!Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error('Both arguments must be finite numbers');
  }
  if (isNaN(a) || isNaN(b)) {
    throw new Error('Arguments cannot be NaN');
  }
  if (!(Number.isSafeInteger(a) && Number.isSafeInteger(b))) {
    throw new Error('Arguments must be safe integers');
  }
  if (b === 0 || b === -0) {
    throw new Error('Divisor cannot be zero');
  }
  return a / b;
}
