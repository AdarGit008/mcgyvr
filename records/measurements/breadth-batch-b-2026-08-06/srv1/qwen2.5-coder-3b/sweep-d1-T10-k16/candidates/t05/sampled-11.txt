export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (size <= 0) {
    throw new Error("Size must be a positive integer");
  }

  return items.reduce(
    (acc, item) => (!acc.length || acc[acc.length - 1].length < size)
      ? [...acc, [item]]
      : [...acc.slice(0, -1), [...acc[acc.length - 1], item]],
    [] as T[][]
  );
}
