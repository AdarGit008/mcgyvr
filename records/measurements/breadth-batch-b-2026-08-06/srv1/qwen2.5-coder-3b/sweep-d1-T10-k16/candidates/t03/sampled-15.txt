export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
  const regex = /^v?(\d+)(?:\.(\d+))?(?:\.(\d+))?$|-|$/;
  const parts = version.match(regex);

  if (!parts) {
    throw new Error(`Invalid semver version format: ${version}`);
  }

  return {
    major: parseInt(parts[1], 10),
    minor: parts[2] ? parseInt(parts[2], 10) : 0,
    patch: parts[3] ? parseInt(parts[3], 10) : 0,
    prerelease: parts[4] || null,
    build: parts[5] || null
  };
}
