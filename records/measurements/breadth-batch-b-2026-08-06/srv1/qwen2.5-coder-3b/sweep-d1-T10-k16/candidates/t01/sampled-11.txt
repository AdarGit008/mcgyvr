export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') throw new Error('Argument must be a string');

  if (!input) return '';

  let result = '';
  let count = 1;

  for (let i = 1; i < input.length; i++) {
    if (input[i] === input[i - 1]) {
      count++;
    } else {
      result += `${input[i - 1]}${count !== 1 ? count : ''}`;
      count = 1;
    }
  }

  // Add the last character and its count
  result += `${input[input.length - 1]}${count !== 1 ? count : ''}`;

  return result;
}
