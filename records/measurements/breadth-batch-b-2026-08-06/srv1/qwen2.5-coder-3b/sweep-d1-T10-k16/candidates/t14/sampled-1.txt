```ts
export interface User {
  email: string;
  active?: boolean; // Optional property for handling absent 'active'
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter(user => user.active !== false) // Filter out inactive users
    .map(user => user.email.toLowerCase()); // Map to lowercase and return
}
```
