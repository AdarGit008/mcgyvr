```typescript
export interface User {
  email: string;
  active?: boolean; // Optional property to handle users without 'active' set
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users.filter(user => user.active || typeof user.active === 'undefined').map(user => user.email.toLowerCase());
}
```
