```typescript
export interface User {
  email: string;
  active: boolean | null | undefined; // Allow for null or undefined values
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter(user => user.active !== false && user.email) // Filter out non-boolean active and emails that are truthy
    .map(user => user.email.toLowerCase());
}
```
