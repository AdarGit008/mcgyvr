/* The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Changed from `items.length - 1` to `items.length`

  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // No change needed
    } else {
      high = mid;
    }
  }

  // Adjust for case where target is the last element
  if (low === high && items[low] === target) {
    return low;
  }

  return -1;
}
