/** The index of target in a sorted array,
 * or -1 if no such element exists. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length;

  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // Move lower bound up to mid
    } else {
      high = mid;      // Move upper boundary down to mid
    }
  }

  return -1;
}
