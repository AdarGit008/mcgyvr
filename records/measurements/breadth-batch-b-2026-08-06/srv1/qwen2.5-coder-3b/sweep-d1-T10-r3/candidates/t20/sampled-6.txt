export function parseConfig(text: string): Record<string, string> {
  const config: Record<string> = {};
  for (const line of text.split("\n").filter(line => line.trim().charAt(0) !== '#')) {
    if (!line.includes('=')) {
      throw new Error(`Line without '=' at ${line}`);
    }
    const [key, ...valueParts] = line.split('=', 2);
    const value = valueParts.join('=');
    config[key.trim()] = value.trim();
  }
  return config;
}
