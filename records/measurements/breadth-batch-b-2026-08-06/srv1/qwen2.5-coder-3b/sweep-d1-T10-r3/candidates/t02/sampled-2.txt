export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.some(pair => pair[0] > pair[1])) {
    throw new Error("Each interval must have a start no greater than an end.");
  }

  // Create a map to store the indices of intervals we are considering
  const indexLookup = new Map<number, { index: number; start: number }>();
  for (let i = 0; i < intervals.length; i++) {
    indexLookup.set(i, { index: i, start: intervals[i][0] });
  }

  // Sort intervals by start time
  intervals.sort((a, b) => a[0] - b[0]);

  let result: number[][] = [];
  let currentInterval = intervals[0];

  for (let i = 1; i < intervals.length; i++) {
    const start = indexLookup.get(i).start;
    const end = intervals[i][1];
    if (end >= currentInterval[0]) { // Merge
      currentInterval = [currentInterval[0], Math.max(currentInterval[1], end)];
    } else { // New interval
      result.push([...currentInterval]);
      currentInterval = intervals[i];
    }
  }

  result.push([...currentInterval]);

  return result;
}
