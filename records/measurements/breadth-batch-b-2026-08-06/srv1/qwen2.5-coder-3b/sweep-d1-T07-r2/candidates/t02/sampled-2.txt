export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.some(([start, end]) => start > end)) {
    throw new Error("Invalid interval: start cannot be greater than end.");
  }

  intervals.sort((a, b) => a[0] - b[0]);

  const mergedIntervals: number[][] = [];
  let currentInterval = intervals[0];

  for (let i = 1; i < intervals.length; i++) {
    const [nextStart, nextEnd] = intervals[i];
    if (currentInterval[1] >= nextStart) {
      // Merge the current interval with the next one
      currentInterval[1] = Math.max(currentInterval[1], nextEnd);
    } else {
      // Add the current interval to the merged list and start a new one
      mergedIntervals.push([...currentInterval]);
      currentInterval = [nextStart, nextEnd];
    }
  }

  // Add the last merged interval
  mergedIntervals.push([...currentInterval]);

  return mergedIntervals;
}
