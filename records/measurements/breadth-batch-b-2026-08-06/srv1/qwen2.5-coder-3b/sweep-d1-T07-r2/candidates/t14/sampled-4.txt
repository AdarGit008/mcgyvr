```typescript
export interface User {
  email: string;
  active?: boolean; // Updated to allow optional 'active' property
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter(user => user.active !== false) // Filter out non-active and undefined active properties
    .map(user => user.email.toLowerCase());
}
```
