export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1;
  while (low <= high) { // Change from < to <=
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid; // Return the index of any duplicate found
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
