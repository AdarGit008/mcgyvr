export function parseCsvLine(line: string): string[] {
  if (line.trim() === "") return [];

  const result: string[] = [];
  let currentField = "";
  let inQuote = false;
  let escapedQuote = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (char === '"') {
      if (!inQuote) {
        inQuote = true;
        escapedQuote = false;
      } else if (escapedQuote) {
        currentField += '"';
        escapedQuote = false;
      } else {
        // If not in a quote and we encounter a second quote, it's a closing quote
        if (i + 1 < line.length && line[i + 1] === '"') {
          currentField += '"';
          i++; // Skip the next character as it's part of the field value
        } else {
          escapedQuote = true;
        }
      }
    } else if (char === "," && !inQuote) {
      result.push(currentField);
      currentField = "";
    } else {
      if (!escapedQuote && char === " ") continue; // Ignore leading spaces
      currentField += char;
    }
  }

  // Add the last field
  result.push(currentField);

  return result;
}
