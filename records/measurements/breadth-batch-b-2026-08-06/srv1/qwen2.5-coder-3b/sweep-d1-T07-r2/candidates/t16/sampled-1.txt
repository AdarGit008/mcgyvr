const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (n < 0 || !Number.isInteger(n)) {
    throw new Error("Invalid input: n must be a non-negative integer.");
  }

  let a = 0;
  let b = 1;

  for (let i = 0; i < n; i++) {
    const temp = (a + b) % MODULUS;
    a = b;
    b = temp;
  }

  return a;
}
