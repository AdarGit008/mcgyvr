/** Safely divides two numbers. Throws an error if the divisor is zero or not finite,
 *  and rejects booleans as division arguments. */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error("Both arguments must be finite numbers.");
  }
  if (b === 0 || Number.isNaN(b)) {
    throw new Error("Division by zero or NaN is not allowed.");
  }
  return a / b;
}
