/** The temperature in degrees Fahrenheit. */
export function celsiusToFahrenheit(celsius: number): number {
  if (!validateCelsiusInput(celsius)) throw new Error("celsius must be a finite number and at or above absolute zero");
  return celsius * 1.8 + 32;
}

/** The temperature in kelvin. */
export function celsiusToKelvin(celsius: number): number {
  if (!validateCelsiusInput(celsius)) throw new Error("celsius must be a finite number and at or above absolute zero");
  return celsius + 273.15;
}

function validateCelsiusInput(celsius: number): boolean {
  return typeof celsius === "number" && !isNaN(celsius) && Number.isFinite(celsius) && celsius >= -273.15;
}
