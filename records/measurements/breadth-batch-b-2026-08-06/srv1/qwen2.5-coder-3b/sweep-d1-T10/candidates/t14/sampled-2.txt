```typescript
export interface User {
  email: string;
  active?: boolean; // Optional active property with a default of false
}

/**
 * The lowercased addresses of the active users, in order.
 */
export function activeEmails(users: readonly User[]): string[] {
  return (
    users
      .filter(({ active }) => Boolean(active)) // Filter for active users
      .map(user => user.email.toLowerCase())           // Map to lowercased email
  );
}
```
