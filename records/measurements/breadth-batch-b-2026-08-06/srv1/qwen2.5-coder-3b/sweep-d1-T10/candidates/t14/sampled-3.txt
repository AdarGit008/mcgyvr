```ts
export interface User {
  email: string;
  active?: boolean; // Marking 'active' as optional
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter(user => user.active) // Use filter to include only active users
    .map(user => user.email.toLowerCase()); // Map to lowercase and collect results
}
```
