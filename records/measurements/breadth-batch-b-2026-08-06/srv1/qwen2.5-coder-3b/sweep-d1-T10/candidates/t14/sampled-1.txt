```ts
export interface User {
  email: string;
  active: boolean | null | undefined;
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return (
    users.filter(user => user.active !== false) // Filter out non-active users
      .map(user => user.email.toLowerCase())     // Convert all email addresses to lowercase
  );
}
```
