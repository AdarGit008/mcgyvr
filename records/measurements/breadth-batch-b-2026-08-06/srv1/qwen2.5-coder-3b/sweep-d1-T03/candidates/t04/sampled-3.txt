export class LruCache<K, V> {
  private capacity: number;
  private cache: Map<K, { value: V; timestamp: number }>;
  private keys: K[] = [];

  constructor(capacity: number) {
    if (capacity <= 0 || !Number.isInteger(capacity)) {
      throw new Error("Capacity must be a positive integer.");
    }
    this.capacity = capacity;
    this.cache = new Map();
  }

  get(key: K): V | undefined {
    const entry = this.cache.get(key);
    if (entry) {
      // Update the timestamp to reflect recent use
      entry.timestamp = Date.now();
      this.keys.splice(this.keys.indexOf(key), 1); // Remove from keys list
      this.keys.push(key); // Add to end of keys list
      return entry.value;
    }
    return undefined;
  }

  set(key: K, value: V): void {
    if (this.cache.has(key)) {
      // Update the existing key's value and timestamp
      const entry = this.cache.get(key)!; // Safe because we just checked has
      entry.value = value;
      entry.timestamp = Date.now();
      this.keys.splice(this.keys.indexOf(key), 1); // Remove from keys list
      this.keys.push(key); // Add to end of keys list
    } else {
      if (this.cache.size >= this.capacity) {
        // Evict the least recently used item
        const lruKey = this.keys.shift()!;
        this.cache.delete(lruKey);
      }
      // Add the new key-value pair
      this.cache.set(key, { value, timestamp: Date.now() });
      this.keys.push(key); // Add to end of keys list
    }
  }

  get size(): number {
    return this.cache.size;
  }
}
