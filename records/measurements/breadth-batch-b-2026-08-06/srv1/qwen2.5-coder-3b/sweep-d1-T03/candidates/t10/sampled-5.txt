/** The index of target in a sorted array, or -1 if not found. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Changed to high = items.length for inclusive search
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // No change needed, remains the same
    } else {
      high = mid; // Changed to high = mid for inclusive search
    }
  }
  return -1;
}
