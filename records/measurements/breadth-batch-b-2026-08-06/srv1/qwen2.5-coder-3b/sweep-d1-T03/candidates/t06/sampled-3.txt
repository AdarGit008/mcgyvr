export function parseCsvLine(line: string): string[] {
  if (line.trim() === "") return [];

  const result: string[] = [];
  let currentField = "";
  let inQuotes = false;
  let escapedQuote = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (char === '"') {
      if (!inQuotes) {
        inQuotes = true;
        continue;
      } else if (escapedQuote) {
        currentField += '"';
        escapedQuote = false;
      } else {
        throw new Error("Quoted field is never closed");
      }
    }

    if (char === "," && !inQuotes) {
      result.push(currentField.trim());
      currentField = "";
      continue;
    }

    if (char === " ") {
      if (!inQuotes) {
        if (currentField.length > 0) {
          result.push(currentField.trim());
          currentField = "";
        }
        continue;
      }
    }

    if (char === "\n") {
      throw new Error("Unexpected newline in CSV line");
    }

    currentField += char;

    escapedQuote = char === "\\";
  }

  if (inQuotes) {
    throw new Error("Quoted field is never closed");
  }

  result.push(currentField.trim());
  return result;
}
