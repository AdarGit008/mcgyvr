```ts
/** A safe division that handles edge cases and throws errors where necessary. */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new Error('Arguments must be numbers.');
  }
  // Check for NaN
  if (isNaN(a) || isNaN(b)) {
    throw new Error('Non-finite inputs are not allowed.');
  }
  // Check for zero division, except -0
  if (b === 0) {
    throw new Error('Zero division error. Cannot divide by zero.');
  }
  return a / b;
}
```
