/** Safely divides two numbers, throwing an Error if the divisor is zero or infinite,
 * and rejecting non-number types. */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new Error('Both inputs must be finite numbers.');
  }
  if (!Number.isFinite(b) || b === -0) {
    throw new Error('Cannot divide by zero or infinity.');
  }
  return a / b;
}
