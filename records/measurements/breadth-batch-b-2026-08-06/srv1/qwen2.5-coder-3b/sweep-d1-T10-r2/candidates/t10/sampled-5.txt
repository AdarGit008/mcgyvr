/** The index of target in a sorted array or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length;
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid; // Adjusted to include the middle element
    }
  }

  if (low === items.length || items[low] !== target) {
    return -1; // Return -1 for absent target
  }

  return low;
}
