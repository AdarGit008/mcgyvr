/** The index of target in a sorted array, or -1 if not found. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Change to items.length for non-strict check on last element

  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid; // Change to mid for correct lower bound check
    }
  }

  return -1;
}
