/** The index of target in a sorted array (including -1 if absent). */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length;
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid; // Ensure high is mid inclusive
    }
  }
  return -1;
}
