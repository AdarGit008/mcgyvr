/** Calculate the factorial n */
export function factorial(n: number): number {
  if (n < 0) throw new Error("Negative argument.");
  if (!Number.isInteger(n)) throw new Error("Non-integer argument");
  if (n > BigInt(Number.MAX_SAFE_INTEGER)) throw new Error("Overflow.");
  if (n === 0 || n === 1) return 1;
  let result = 2;
  for (let i = 3; i <= n; i++) {
    result *= i;
  }
  return result;
}
