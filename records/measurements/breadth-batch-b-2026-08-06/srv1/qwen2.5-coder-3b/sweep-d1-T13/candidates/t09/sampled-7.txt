/** The factorial of n. */
export function factorial(n: number): number {
  if (
    !Number.isInteger(n) ||
    n < 0
  ) {
    throw new Error("factorial is undefined for non-integer or negative arguments");
  }
  if (n === 1 || n === 0) {
    return 1;
  }
  return n * factorial(n - 1);
}
