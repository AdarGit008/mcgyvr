/** The factorial of n. Throws an error if input is non-integer or below zero. */
export function factorial(n: number): number {
  // ThrowError if n is a non-integer or negative;
  if (!Number.isInteger(n) || n < 0) {
    throw new Error("Invalid argument: 'n' must be a non-negative integer.");
  }
  // Base case for factorial of 0 is 1
  return (n === 0 ? 1 : n * factorial(n - 1));
}
