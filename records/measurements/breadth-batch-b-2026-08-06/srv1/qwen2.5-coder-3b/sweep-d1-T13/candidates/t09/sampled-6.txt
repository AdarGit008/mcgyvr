/** The factorial of n. Handles non-integer negative numbers and returns 1 for n = 0,
 * Otherwise throws an Error if n is < 0. */
export function factorial(n: number): number {
  if (!Number.isInteger(n)) {
    throw new Error('Factorial is not defined for non-integers.');
  }
  if (n < 0) {
    throw new Error('Factorial is not defined for negative numbers.');
  }
  if (n === 0) {
    return 1;
  }
  return n * factorial(n - 1);
}
