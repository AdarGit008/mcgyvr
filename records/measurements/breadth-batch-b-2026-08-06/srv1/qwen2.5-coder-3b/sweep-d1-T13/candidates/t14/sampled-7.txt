```typescript
export interface User {
  email: string;
  active?: boolean; // Note the optional nature for unspecified 'active' property
}

/**
 * The lowercased addresses of the active users, in order, or all elements if undefined active is provided.
 */
export function activeEmails(users: readonly User[]): string[] {
  return users.filter(user => user.active || user.active === undefined).map(user => user.email.toLowerCase());
}
```
