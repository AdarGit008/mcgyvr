export function truncate(text: string, limit: number): string {
  if (limit < 3 || !Number.isInteger(limit)) {
    throw new Error("limit must be an integer of at least 3.");
  }

  const ellipsis = "...";

  if (text.length <= limit) {
    return text;
  }

  const lengthToCutOff = Math.min(4, limit - ellipsis.length);
  return text.slice(0, limit - lengthToCutOff) + ellipsis;
}
