export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Argument must be a string');
  }
  
  const result: string[] = [];
  let count = 1;
  for (let i = 1; i <= input.length; i++) {
    if (input[i] === input[i - 1]) {
      count++;
    } else {
      result.push(input[i - 1], count.toString());
      count = 1;
    }
  }
  
  return result.join('');
}
