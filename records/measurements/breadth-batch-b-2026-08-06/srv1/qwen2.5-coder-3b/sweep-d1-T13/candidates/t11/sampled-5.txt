/** The numbers, ascending. */
export function sortAscending(values: readonly number[]): number[] {
  // Assuming numbers are correctly represented for sorting (valid for typical use cases)
  if (!Array.isArray(values)) {
    throw new Error("Input must be an array of numbers");
  }

  return [...values].slice().sort((a, b) => {
    if (Number.isNaN(a) || Number.isNaN(b)) {
      // NaN should appear last, following JavaScript's numeric sort behavior for undefined order
      return NaN - NaN;
    }
    
    return a - b;
  });
}
