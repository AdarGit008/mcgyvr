```typescript
/** The factorial of n. Throws Error for negative or non-integer arguments. */
export function factorial(n: number): number {
  if (n < 0 || !Number.isInteger(n)) {
    throw new Error("Negative or non-integer argument");
  }
  if (n === 0) {
    return 1;
  }
  return n * factorial(n - 1);
}
```
