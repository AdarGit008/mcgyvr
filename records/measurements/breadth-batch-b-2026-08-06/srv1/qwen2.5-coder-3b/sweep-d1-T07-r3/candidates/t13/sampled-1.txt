/** The text, cut to limit characters with an ellipsis when it is longer. */
export function truncate(text: string, limit: number): string {
  if (limit < 3 || !Number.isInteger(limit)) {
    throw new Error("Limit must be an integer of at least 3.");
  }
  
  if (text.length <= limit) {
    return text;
  }
  
  const ellipsis = "...";
  const truncatedText = text.slice(0, limit - ellipsis.length) + ellipsis;
  
  // Ensure the last three characters are not within the limit
  if (truncatedText.length > limit) {
    const overlappingChars = limit - text.length;
    return text.slice(0, limit - overlappingChars) + ellipsis;
  }
  
  return truncatedText;
}
