```ts
export interface User {
  email: string;
  active?: boolean; // Marking 'active' as optional since it's not explicitly stated if absent is considered inactive
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users.filter(user => user.active === true).map(user => user.email.toLowerCase());
}
```
