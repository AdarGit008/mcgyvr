```ts
export interface User {
  email: string;
  active?: boolean; // Added optional property for inactive users
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter(user => user.active === true) // Filter active users
    .map(user => user.email.toLowerCase()); // Lowercase and map to email addresses
}
```
