export function flatten(items: readonly unknown[], depth?: number): unknown[] {
  if (typeof depth !== 'number' || !Number.isFinite(depth) || depth < 0) {
    throw new Error('Depth must be a non-negative integer or Infinity');
  }

  const result: unknown[] = [];
  const stack: { item: unknown, level: number }[] = items.map(item => ({
    item,
    level: depth ?? 1
  }));

  while (stack.length > 0) {
    const { item, level } = stack.pop()!;
    if (!Array.isArray(item)) {
      result.push(item);
    } else {
      for (let i = 0; i < item.length; i++) {
        if (level > 1) {
          stack.push({ item: item[i], level: level - 1 });
        } else {
          result.push(item[i]);
        }
      }
    }
  }

  return result;
}
