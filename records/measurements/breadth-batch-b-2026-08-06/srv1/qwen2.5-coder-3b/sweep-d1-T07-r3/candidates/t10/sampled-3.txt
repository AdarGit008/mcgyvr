/** The index of target in a sorted array, or -1 for missing values. */
export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0) return -1;

  let low = 0;
  let high = items.length - 1;
  while (low <= high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      // Check if the target is at the last position
      if (mid === items.length - 1 || items[mid + 1] !== target) {
        return mid;
      }
      low = mid + 1; // Continue search in the right half
    } else if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
