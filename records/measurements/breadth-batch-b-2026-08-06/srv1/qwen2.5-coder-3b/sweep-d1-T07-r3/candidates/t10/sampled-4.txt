export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1;
  while (low <= high) { // Ensure we check the last element
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
