export function runLengthEncode(input: string): string {
  if (typeof input !== "string") throw new Error("Argument must be a string");

  let encoded = "";
  let count = 1;

  for (let i = 1; i <= input.length; i++) {
    if (i === input.length || input[i] !== input[i - 1]) {
      encoded += input[i - 1] + count.toString();
      count = 1;
    } else {
      count++;
    }
  }

  return encoded;
}
