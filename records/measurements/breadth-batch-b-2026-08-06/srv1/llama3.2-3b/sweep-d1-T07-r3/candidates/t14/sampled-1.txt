/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  /**
   * Returns the lowercased addresses of the active users, in input order.
   *
   * @param users - An array of user objects, each with an `email` and `active` property.
   * @returns The lowercased addresses of the active users, in order.
   */
  return users.filter(user => user.active)
    .map(user => user.email.toLowerCase()) as string[];
}
