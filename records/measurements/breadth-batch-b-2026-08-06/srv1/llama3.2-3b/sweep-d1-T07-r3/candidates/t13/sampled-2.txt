/** The text, cut to limit characters with an ellipsis when it is longer. */
export function truncate(text: string, limit: number): string {
  if (typeof limit !== 'number' || limit < 3) {
    throw new Error('Limit must be a positive integer of at least 3');
  }
  const tailLength = Math.max(0, limit - text.length);
  return text.slice(0, limit - tailLength) + '...'.padStart(tailLength, ' ');
}
