```typescript
/** An append-only log of timestamped events. */
export interface Event {
  name: string;
  timestamp: number;
}

/**
 * @param event - The event to record.
 */
export class EventLog {
  private #events: Event[] = [];

  /**
   * Append one event.
   */
  public record(event: Event): void {
    this.#events.push(event);
  }

  /**
   * Every event at or after the given timestamp, in insertion order.
   * @param timestamp - The timestamp to filter by.
   */
  public since(timestamp: number): Event[] {
    return [...this.#events.filter((event) => event.timestamp >= timestamp)];
  }
}
```
