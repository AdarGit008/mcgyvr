```typescript
/**
 * The factorial of n.
 */
export function factorial(n: number): number {
  if (typeof n !== 'number' || !Number.isInteger(n) || n < 0) {
    throw new Error('n must be a non-negative integer');
  }
  if (n === 0) {
    return 1;
  } else {
    return n * factorial(n - 1);
  }
}
```
