/** The factorial of n. */
export function factorial(n: number): number {
  if (typeof n !== 'number' || n < 0) {
    throw new Error('n must be a non-negative integer');
  }
  if (!Number.isInteger(n)) {
    throw new Error('n must be an integer');
  }
  if (n === 0 || n === 1) {
    return 1;
  }
  return n * factorial(n - 1);
}
