/** An append-only log of timestamped events. */
export class EventLog<T extends recordable<Event>> {
  #events: Event[] = [];

  /** Append one event. */
  record(event: T) {
    this.#events.push(event);
  }

  /** Every event at or after the given timestamp, in insertion order. */
  since(timestamp: number): T[] {
    return this.#events.filter((event) => event.timestamp >= timestamp);
  }
}

export interface Event<T extends recordable<Record<Path, value>> = Path> {
  name: string;
  timestamp: number;
}
