/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  // Ensure high is the last element's potential index
  let low = 0;
  let high = items.length - 1;

  while (low <= high) { // changed 'while' to 'while'
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1; // also changed to prevent mid-1 errors
    }
  }

  return -1;
}
