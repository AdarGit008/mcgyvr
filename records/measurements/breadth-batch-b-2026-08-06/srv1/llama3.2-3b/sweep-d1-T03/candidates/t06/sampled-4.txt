/**
 * Splits one line of CSV into its fields on commas.
 * A field wrapped in double quotes may contain commas, and a doubled quote inside such a field means one literal quote character;
 * the surrounding quotes are not part of the value. Fields that are not quoted are taken literally, including any surrounding spaces.
 * An empty line is one empty field.
 */
export function parseCsvLine(line: string): string[] {
    const result: string[] = [];
    let currentField: string | null = null;
    let inQuote: boolean = false;

    for (const char of line) {
        if (char === '"') {
            inQuote = !inQuote;
        } else if (!inQuote && char === ',') {
            if (currentField !== null) {
                result.push(currentField);
                currentField = null;
            }
        } else if (!inQuote && char === '\r' || char === '\n') {
            // ignore newline characters
        } else if (char === ',' && inQuote) {
            throw new Error('Unbalanced quote in CSV line');
        } else {
            if (currentField === null) {
                currentField = char;
            } else {
                currentField += char;
            }
        }
    }

    if (currentField !== null) {
        result.push(currentField);
    }

    return result.filter(field => field !== '');
}
