```typescript
/**
 * Divides a by b, throwing an Error if either argument is invalid.
 */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new Error('Both arguments must be numbers');
  }
  
  const isFiniteA = !isNaN(a) && a === a; // check for NaN
  const isFiniteB = !isNaN(b) && b === b;
  
  if (!isFiniteA || !isFiniteB) {
    throw new Error('Both arguments must be finite numbers');
  }
  
  if (b === 0) {
    throw new Error('Cannot divide by zero');
  }
  
  return a / b;
}
```
