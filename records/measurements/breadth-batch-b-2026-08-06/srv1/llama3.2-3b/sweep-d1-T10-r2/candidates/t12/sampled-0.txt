/** Returns a copy of values with every negative number removed. */
export function removeNegatives(values: readonly number[]): readonly number[] {
  const result = [...values];
  for (let i = result.length - 1; i >= 0; i--) {
    if (result[i] < 0) {
      result.splice(i, 1);
    } else {
      break;
    }
  }
  return result;
}
