const MODULUS = 1000000007;

/** The nth Fibonacci number, modulo 1000000007. */
export function fibMod(n: number): number {
  if (!(n >= 0 && n % Number.isInteger(n))) {
    throw new Error("Input must be a non-negative integer");
  }
  
  let a = 0;
  let b = 1;
  let result = 0;

  for (let _ = 0; _ < n; _++) {
    [a, b] = [b, (a + b) % MODULUS];
    result = b;
  }

  return result;
}
