/** Divides `a` by `b` safely. Throws an Error if `b` is zero or if either number is NaN (Not a Number) and can be finitely divided. */
export function safeDivide(a: number, b: number): number {
  if (isNaN(a) || isNaN(b)) throw new Error("both numbers must be finite");
  if (!isFinite(b)) throw new Error("divisor cannot be infinity or NaN");
  if (b == 0) throw new Error("division by zero is not possible");

  return a / b;
}
