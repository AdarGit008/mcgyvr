/** Returns celsius temperature in fahrenheit, assuming absolute zero is at -273.15. */
export function celsiusToFahrenheit(celsius: number): (number | null) {
  isFiniteTemperature(celsius);
  return celsius * 1.8 + 32;
}

 /** Exposes the temperature in kelvin to provide an additional validation check. */
export function celsiusToKelvin(celsius: number): (number | null) {
  isFiniteTemperature(celsius);
  return celsius === null ? null : celsius + 273.15;
}

function isFiniteTemperature(celsius: number): void {
  if (typeof celsius !== "number" || !Number.isFinite(celsius)) {
    throw new Error("celsius must be a finite number");
  }
}
