```typescript
/**
 * The factorial of n.
 */
export function factorial(n: number): number {
  if (n < 0) {
    throw new Error('Input must be a non-negative integer');
  }
  else if (!Number.isInteger(n)) {
    throw new Error('Input must be an integer');
  } 
  else if (n === 0 || n === 1) {
    return 1;
  }
  else {
    return n * factorial(n - 1);
  }
}
```
