/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1; // Adjusted the upper bound to include the last element
  while (low <= high) { // Changed to inclusive search
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1; // Adjusted to include the search window
    }
  }
  return -1; // Return -1 for absent value
}
