function validateTemperature(t: number): void {
  if (typeof t !== "number" || !Number.isFinite(t)) {
    throw new Error("must be a finite number");
  }
  if (t < -273.15) {
    throw new Error(" must be at or above absolute zero");
  }
}

// Implementation
function celsiusToFahrenheit(celsius: number): number {
  validateTemperature(celsius);
  return celsius * 1.8 + 32;
}

function celsiusToKelvin(celsius: number): number {
  validateTemperature(celsius);
  return celsius + 273.15;
}
