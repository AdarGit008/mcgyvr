/** The words grouped by their length, in first-seen order. */
export function groupByLength(words: readonly string[]): Map<number, string[]> {
  const groups = new Map();
  for (const word of words) {
    const bucketLength = Number(word.length);
    if (isNaN(bucketLength)) {
      throw new Error('Invalid argument: word length must be a number');
    }
    const bucket = groups.get(bucketLength);
    if (bucket === undefined) {
      groups.set(bucketLength, [word]);
    } else {
      bucket.push(word);
    }
  }
  return groups;
}
