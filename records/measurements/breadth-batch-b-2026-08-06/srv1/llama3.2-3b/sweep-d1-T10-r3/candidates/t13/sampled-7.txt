```typescript
/**
 * Returns text truncated to exactly limit characters with the last three replaced by "...".
 */
export function truncate(text: string, limit: number): string {
  if (typeof limit !== 'number' || isNaN(limit) || limit < 3) {
    throw new Error('Limit must be an integer of at least 3.');
  }
  return text.length <= limit
    ? text
    : text.slice(0, limit - 2) + '...';
}
```
