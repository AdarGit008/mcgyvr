export function runLengthEncode(input: string): string {
  /**
   * Encode a string by replacing every maximal run of one repeated character with that character followed by its count.
   *
   * @param input The string to be encoded.
   * @returns The run-length encoded string.
   */
  if (typeof input !== 'string') {
    throw new Error('Input must be a non-empty string.');
  }

  const result = [];
  let currentChar = input[0];
  let count = 1;

  for (let i = 1; i <= input.length; i++) {
    const char = input[i];

    if (char === currentChar) {
      count++;
    } else {
      result.push(currentChar + count.toString());
      currentChar = char;
      count = 1;
    }
  }

  return result.join('');
}
