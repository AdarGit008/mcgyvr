/**
 * The KEY=VALUE pairs of a config file.
 */
export function parseConfig(text: string): Record<string, string> {
  const config: Record<string, string> = {};
  for (const line of text.split("\n")) {
    if (line.trim() === "" || line.startsWith("#")) continue;
    const index = line.indexOf("=");
    if (index === -1) throw new Error(`Invalid line with no '=': ${line}`);
    const key = line.slice(0, index).trim();
    if (key === "") throw new Error(`Empty key found`);
    config[key] = line.slice(index + 1).trim();
  }
  // Prevent prototype pollution
  delete config["__proto__"];
  return config;
}
