/**
 * Replace every maximal run of one repeated character with that character followed by the run length.
 * A run of length one still carries its count, so "abc" encodes to "a1b1c1".
 * The empty string encodes to the empty string.
 * @param input - The string to encode using run-length encoding.
 * @returns The encoded string.
 */
export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }

  let encoded = '';
  let count = 1;

  for (let i = 1; i <= input.length; i++) {
    if (input[i] === input[i - 1]) {
      count++;
    } else {
      encoded += `${input[i - 1]}${count}`;
      count = 1;
    }
  }

  return encoded;
}
