type CacheEntry<K, V> = { key: K; value: V; timestamp: number };

export class LruCache<K, V> {
    private readonly capacity: number;
    private readonly cache: Map<K, CacheEntry<K, V>>;
    private timestamp: number;

    constructor(capacity: number) {
        if (isNaN(capacity) || capacity <= 0 || !Number.isInteger(capacity)) {
            throw new Error('Capacity must be a positive integer');
        }
        this.capacity = capacity;
        this.cache = new Map();
        this.timestamp = 0;
    }

    get(key: K): V | undefined {
        const entry = this.cache.get(key);
        if (entry) {
            entry.timestamp = ++this.timestamp;
            return entry.value;
        }
        return undefined;
    }

    set(key: K, value: V): void {
        if (this.cache.has(key)) {
            this.cache.get(key)!.timestamp = ++this.timestamp;
        } else {
            if (this.cache.size >= this.capacity) {
                const oldestEntry = [...this.cache.entries()].reduce((oldest, current) =>
                    current[1].timestamp < oldest[1].timestamp ? current : oldest
                );
                this.cache.delete(oldestEntry[0]);
            }
            this.cache.set(key, { key, value, timestamp: ++this.timestamp });
        }
    }

    get size(): number {
        return this.cache.size;
    }
}
