// flatten.ts

/**
 * Flattens an array up to the specified depth.
 * @param items - The array to flatten.
 * @param [depth=Infinity] - The maximum depth to flatten. Defaults to Infinity.
 * @returns A new flattened array.
 */
export function flatten(items: readonly unknown[], depth?: number): unknown[] {
    if (depth === undefined) {
        depth = Infinity;
    } else if (!Number.isInteger(depth) || depth < 0) {
        throw new Error('Depth must be a non-negative integer or Infinity.');
    }

    const result: unknown[] = [];

    items.forEach(item => {
        if (Array.isArray(item)) {
            if (depth > 1) {
                result.push(...flatten(item, depth - 1));
            } else {
                result.push(...item);
            }
        } else if (typeof item !== 'string') {
            result.push(item);
        }
    });

    return result;
}
