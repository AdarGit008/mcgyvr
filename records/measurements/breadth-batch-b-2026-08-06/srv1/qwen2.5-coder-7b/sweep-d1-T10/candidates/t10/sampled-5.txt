/**
 * The index of target in a sorted array, or -1.
 */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Adjusted the upper bound to be inclusive
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid; // Adjusted to include the current mid value in the next iteration
    }
  }
  return -1;
}
