/**
 * The index of target in a sorted array, or -1.
 */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Include the last element by setting upper bound to length
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid; // Adjust to include the middle element in the next iteration
    }
  }
  return -1;
}
