/**
 * Safely divides `a` by `b`, throwing an error if either argument is not a finite number or if the divisor is zero.
 *
 * @param {number} a - The dividend.
 * @param {number} b - The divisor.
 * @returns {number} - The result of the division if valid.
 */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number' || !isFinite(a) || !isFinite(b)) {
    throw new Error('Both arguments must be finite numbers.');
  }
  if (b === 0 || b === -0) {
    throw new Error('Divisor cannot be zero.');
  }
  return a / b;
}
