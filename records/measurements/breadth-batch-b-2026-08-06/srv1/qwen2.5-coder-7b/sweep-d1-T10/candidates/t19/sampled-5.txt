/**
 * a divided by b, ensuring safety by checking for valid numbers and non-zero divisors.
 * @param {number} a - The dividend.
 * @param {number} b - The divisor.
 * @returns {number} - The result of the division if safe, otherwise throws an error.
 */
export function safeDivide(a: number, b: number): number {
  if (isNaN(a) || isNaN(b)) {
    throw new Error('Both arguments must be finite numbers.');
  }
  if (b === 0 || b === -0) {
    throw new Error('Divisor cannot be zero.');
  }
  return a / b;
}
