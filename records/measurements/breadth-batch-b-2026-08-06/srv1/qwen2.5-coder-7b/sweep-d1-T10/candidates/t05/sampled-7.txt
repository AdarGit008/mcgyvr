/**
 * Splits an array into consecutive groups of at most size elements, preserving order.
 * The final group is shorter when the length does not divide evenly. An empty array yields 
 * an empty array. The input array must not be mutated. Throws Error unless size is a positive integer.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
    if (size <= 0 || !Number.isInteger(size)) {
        throw new Error('Size must be a positive integer');
    }
    return items.length === 0 ? [] : Array.from({ length: Math.ceil(items.length / size) }, (_, i) => 
        items.slice(i * size, i * size + size)
    );
}
