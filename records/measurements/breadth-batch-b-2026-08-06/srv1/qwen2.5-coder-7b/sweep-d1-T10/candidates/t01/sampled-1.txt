/**
 * Run-length encodes a string.
 * 
 * @param input The input string to encode.
 * @returns The run-length encoded string.
 */
export function runLengthEncode(input: string): string {
    if (typeof input !== 'string') {
        throw new Error('Invalid argument: input must be a string');
    }

    let encoded = '';
    let count = 1;

    for (let i = 1; i <= input.length; i++) {
        if (input[i] === input[i - 1]) {
            count++;
        } else {
            encoded += `${input[i - 1]}${count}`;
            count = 1;
        }
    }

    return encoded;
}
