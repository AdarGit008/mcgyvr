export class LruCache<K, V> {
    private readonly capacity: number;
    private cache: Map<K, { value: V; timestamp: number }>;
    private time: number;

    constructor(capacity: number) {
        if (!Number.isInteger(capacity) || capacity <= 0) {
            throw new Error('Capacity must be a positive integer');
        }
        this.capacity = capacity;
        this.cache = new Map();
        this.time = 0;
    }

    get(key: K): V | undefined {
        const entry = this.cache.get(key);
        if (entry) {
            this.updateTimestamp(entry);
            return entry.value;
        }
        return undefined;
    }

    set(key: K, value: V): void {
        if (this.cache.has(key)) {
            const entry = this.cache.get(key)!;
            entry.value = value;
            this.updateTimestamp(entry);
        } else {
            if (this.cache.size >= this.capacity) {
                const oldestEntry = Array.from(this.cache.entries())
                    .reduce((a, b) => a[1].timestamp < b[1].timestamp ? a : b);
                this.cache.delete(oldestEntry[0]);
            }
            this.cache.set(key, { value, timestamp: this.time });
        }
        this.time++;
    }

    get size(): number {
        return this.cache.size;
    }

    private updateTimestamp(entry: { timestamp: number }): void {
        entry.timestamp = this.time;
    }
}
