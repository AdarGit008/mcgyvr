export function topologicalSort(nodes: readonly string[], edges: readonly (readonly [string, string])[]): string[] {
    if (!nodes.some(node => edges.some(([from, to]) => from === node || to === node)) && edges.length > 0) {
        throw new Error("There are no nodes that match any edge.");
    }

    const graph: { [key: string]: string[] } = {};
    const inDegree: { [key: string]: number } = {};

    for (const node of nodes) {
        graph[node] = [];
        inDegree[node] = 0;
    }

    for (const [from, to] of edges) {
        if (!nodes.includes(from) || !nodes.includes(to)) {
            throw new Error(`Edge includes a node not in the list: ${from} or ${to}`);
        }
        graph[from].push(to);
        inDegree[to]++;
    }

    const queue = nodes.filter(node => inDegree[node] === 0);

    const sortedNodes: string[] = [];

    while (queue.length > 0) {
        const node = queue.shift()!;
        sortedNodes.push(node);

        for (const neighbor of graph[node]) {
            inDegree[neighbor]--;
            if (inDegree[neighbor] === 0) {
                queue.push(neighbor);
            }
        }
    }

    if (sortedNodes.length !== nodes.length) {
        throw new Error("Cycle detected");
    }

    return sortedNodes;
}
