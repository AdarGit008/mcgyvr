```typescript
/**
 * The text, cut to limit characters with an ellipsis when it is longer.
 * @param text - The input string to truncate.
 * @param limit - The maximum length of the returned string including the ellipsis.
 * @returns A new string that is no longer than the specified limit, ending with "..." if truncated.
 */
export function truncate(text: string, limit: number): string {
  if (!Number.isInteger(limit) || limit < 3) {
    throw new Error("Limit must be an integer of at least 3");
  }
  if (text.length <= limit) {
    return text;
  }
  return text.slice(0, limit - 3) + "...";
}
```
