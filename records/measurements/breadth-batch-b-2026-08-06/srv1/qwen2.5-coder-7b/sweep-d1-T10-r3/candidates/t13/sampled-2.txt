/**
 * The text, cut to limit characters, with an ellipsis at the end if trimmed. 
 * Throws an error if limit is less than 3.
 */
export function truncate(text: string, limit: number): string {
  if (typeof limit !== 'number' || !Number.isInteger(limit) || limit < 3) {
    throw new Error('Limit must be an integer of at least 3.');
  }
  if (text.length <= limit) {
    return text;
  }
  return text.slice(0, limit - 3) + "...";
}
