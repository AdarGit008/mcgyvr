/**
 * a divided by b. Throws an error if b is zero or either argument is not a finite number.
 */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || !isFinite(a)) throw new Error('First argument must be a finite number');
  if (typeof b !== 'number' || !isFinite(b)) throw new Error('Second argument must be a finite number');
  if (b === 0 || b === -0) throw new Error('Cannot divide by zero');
  return a / b;
}
