class LruCache<K, V> {
  private readonly capacity: number;
  private readonly cache = new Map<K, { value: V; timestamp: number }>();
  private timestamp = 0;

  constructor(capacity: number) {
    if (!Number.isInteger(capacity) || capacity <= 0) {
      throw new Error("Capacity must be a positive integer");
    }
    this.capacity = capacity;
  }

  get(key: K): V | undefined {
    const entry = this.cache.get(key);
    if (entry !== undefined) {
      entry.timestamp = ++this.timestamp;
      return entry.value;
    }
    return undefined;
  }

  set(key: K, value: V): void {
    if (this.cache.has(key)) {
      this.cache.get(key)!.timestamp = ++this.timestamp;
    } else {
      while (this.cache.size >= this.capacity) {
        const oldestEntry = this.cache.entries().next();
        if (!oldestEntry.done) {
          this.cache.delete(oldestEntry.value[0]);
        }
      }
    }
    this.cache.set(key, { value, timestamp: ++this.timestamp });
  }

  get size(): number {
    return this.cache.size;
  }
}
