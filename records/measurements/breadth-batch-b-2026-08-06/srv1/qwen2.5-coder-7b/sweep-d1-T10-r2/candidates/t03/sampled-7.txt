/**
 * Parses a semantic version string into its components.
 * @param version - The semantic version string to parse.
 * @returns An object containing the major, minor, patch, prerelease, and build parts.
 */
export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
    const parts = version.match(/^v?(\d+)\.(\d+)\.(\d+)(?:-([0-9a-zA-Z-.]+))?(?:\+([0-9a-zA-Z-.]+))?$/);
    if (!parts) {
        throw new Error("Invalid semantic version format");
    }
    const [, major, minor, patch, prerelease = null, build = null] = parts;
    return {
        major: parseInt(major),
        minor: parseInt(minor),
        patch: parseInt(patch),
        prerelease,
        build
    };
}
