/**
 * a divided by b, ensuring the divisor is not zero and both arguments are finite numbers.
 */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error('Both arguments must be finite numbers.');
  }
  if (b === 0 || b === -0) {
    throw new Error('Divisor cannot be zero or -zero.');
  }
  return a / b;
}
