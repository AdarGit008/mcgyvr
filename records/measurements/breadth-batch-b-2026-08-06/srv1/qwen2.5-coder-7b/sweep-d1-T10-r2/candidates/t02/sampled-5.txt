/**
 * Merges overlapping intervals into the smallest set of non-overlapping intervals.
 *
 * @param intervals - An array of [start, end] pairs representing intervals.
 * @returns A new array of merged and sorted intervals.
 * @throws Error if any interval has a start greater than its end.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
    if (intervals.some(([start, end]) => start > end)) {
        throw new Error('Invalid interval: start cannot be greater than end');
    }

    const sortedIntervals = intervals.slice().sort((a, b) => a[0] - b[0]);
    let merged: number[][] = [];

    for (const current of sortedIntervals) {
        if (!merged.length || merged[merged.length - 1][1] < current[0]) {
            merged.push(current);
        } else {
            merged[merged.length - 1][1] = Math.max(merged[merged.length - 1][1], current[1]);
        }
    }

    return merged;
}
