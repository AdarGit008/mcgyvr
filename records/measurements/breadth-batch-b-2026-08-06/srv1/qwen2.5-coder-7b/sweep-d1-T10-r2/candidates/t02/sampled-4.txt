import { isEqual } from 'lodash';

/**
 * Merges overlapping intervals into the smallest list of non-overlapping intervals covering the same points.
 * @param intervals - List of [start, end] pairs, sorted by start.
 * @returns A new list of merged intervals, sorted ascending by start.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];

  constmerged: number[][] = [];
  letcurrent: number[] = intervals[0];

  for (let i = 1; i < intervals.length; i++) {
    const next = intervals[i];
    if (current[1] >= next[0]) {
      current[1] = Math.max(current[1], next[1]);
    } else {
      merged.push([...current]);
      current = [...next];
    }
  }

  merged.push([...current]);

  return merged;
}
