export function flatten(items: readonly unknown[], depth?: number): unknown[] {
  if (depth === undefined) depth = Infinity;
  if (!Number.isInteger(depth) || depth < 0) {
    throw new Error("Depth must be a non-negative integer or Infinity");
  }

  const result: unknown[] = [];
  const stack: [unknown[], number][] = [[items, depth]];

  while (stack.length > 0) {
    const [currentItems, currentDepth] = stack.pop()!;
    for (const item of currentItems) {
      if (Array.isArray(item) && currentDepth > 0) {
        stack.push([item, currentDepth - 1]);
      } else {
        result.push(item);
      }
    }
  }

  return result;
}
