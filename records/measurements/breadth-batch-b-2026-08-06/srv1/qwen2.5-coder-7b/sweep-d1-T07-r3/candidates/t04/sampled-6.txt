export class LruCache<K, V> {
  private readonly capacity: number;
  private readonly cache: Map<K, V>;
  private readonly queue: K[];

  constructor(capacity: number) {
    if (typeof capacity !== 'number' || capacity <= 0 || !Number.isInteger(capacity)) {
      throw new Error('Capacity must be a positive integer');
    }

    this.capacity = capacity;
    this.cache = new Map<K, V>();
    this.queue = [];
  }

  get(key: K): V | undefined {
    if (!this.cache.has(key)) {
      return undefined;
    }

    // Move the key to the end of the queue (most recently used)
    const index = this.queue.indexOf(key);
    if (index !== -1) {
      this.queue.splice(index, 1);
    }
    this.queue.push(key);

    return this.cache.get(key)!;
  }

  set(key: K, value: V): void {
    if (this.cache.has(key)) {
      // Update the value and move to most recently used
      this.cache.set(key, value);
      this.get(key); // This will move the key to the end of the queue
    } else {
      if (this.queue.length === this.capacity) {
        // Evict the least recently used item
        const leastRecentlyUsed = this.queue.shift();
        if (leastRecentlyUsed !== undefined) {
          this.cache.delete(leastRecentlyUsed);
        }
      }
      // Add new key-value pair and mark as most recently used
      this.cache.set(key, value);
      this.queue.push(key);
    }
  }

  get size(): number {
    return this.cache.size;
  }
}
