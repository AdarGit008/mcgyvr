/**
 * Splits an array into consecutive groups of at most size elements, preserving order;
 * final group is shorter when length does not divide evenly. An empty array yields an empty array.
 * The input array must not be mutated. Throws an Error unless size is a positive integer.
 *
 * @param items - Array to split.
 * @param size - Maximum size of each chunk.
 * @returns Array of chunks.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (!Number.isInteger(size) || size <= 0) {
    throw new Error('Size must be a positive integer');
  }

  const result: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    result.push(items.slice(i, i + size));
  }
  return result;
}
