/**
 * Flattens an array up to a specified depth. A string is considered a value
 * and not iterable. A depth of 0 returns a copy of the input unchanged. The input
 * must not be mutated.
 *
 * @param items - The array to flatten.
 * @param depth - The maximum depth to flatten, defaults to Infinity.
 * @returns A new flattened array.
 */
export function flatten(items: readonly unknown[], depth: number = Infinity): unknown[] {
  if (depth < 0) {
    throw new Error('Depth must be a non-negative integer or Infinity');
  }

  return items.reduce((acc, item) => {
    if (Array.isArray(item) && depth > 0) {
      acc.push(...flatten(item, depth - 1));
    } else if (typeof item !== 'string') {
      acc.push(item);
    }
    return acc;
  }, [] as unknown[]);
}
