/**
 * Flattens a nested array up to a specified depth.
 * @param items - The array to flatten.
 * @param depth - The maximum depth of flattening. Defaults to Infinity.
 * @returns A new flattened array.
 */
export function flatten(items: readonly unknown[], depth?: number): unknown[] {
  if (depth === undefined) {
    depth = Infinity;
  }

  if (typeof depth !== 'number' || depth < 0 || !Number.isInteger(depth)) {
    throw new Error('Depth must be a non-negative integer or Infinity.');
  }

  return items.reduce((acc, item) => {
    if (Array.isArray(item) && depth > 0) {
      acc.push(...flatten(item, depth - 1));
    } else {
      acc.push(item);
    }
    return acc;
  }, [] as unknown[]);
}
