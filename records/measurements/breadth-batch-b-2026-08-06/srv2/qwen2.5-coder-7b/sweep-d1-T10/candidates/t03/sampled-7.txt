/**
 * Parses a semantic version string into its components.
 * @param version Semantic version string to parse.
 * @returns Object containing major, minor, patch numbers and prerelease/build strings.
 */
export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
    const regex = /^(\d{1,2})\.(\d{1,2})\.(\d{1,2})(-([0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*))?(\+([0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*))?$/;
    const match = version.match(regex);
    if (!match) {
        throw new Error(`Invalid semver: ${version}`);
    }
    return {
        major: parseInt(match[1], 10),
        minor: parseInt(match[2], 10),
        patch: parseInt(match[3], 10),
        prerelease: match[5] || null,
        build: match[7] || null
    };
}
