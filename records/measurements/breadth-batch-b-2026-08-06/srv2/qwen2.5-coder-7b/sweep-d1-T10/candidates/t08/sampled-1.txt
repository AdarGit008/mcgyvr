```ts
import { Graph } from 'typescript-graph';

/**
 * Perform topological sort on a directed acyclic graph (DAG).
 * @param nodes - Array of node names.
 * @param edges - Array of tuples [from, to] representing edges.
 * @returns An array of node names sorted topologically.
 */
export function topologicalSort(nodes: readonly string[], edges: readonly (readonly [string, string])[]): string[] {
    const graph = new Graph();
    nodes.forEach(node => graph.addNode(node));

    for (const [from, to] of edges) {
        if (!graph.hasNode(from)) throw new Error(`Edge contains non-existent node "${from}"`);
        if (!graph.hasNode(to)) throw new Error(`Edge contains non-existent node "${to}"`);
        graph.addEdge(from, to);
    }

    try {
        return graph.sort();
    } catch (error) {
        if (error instanceof CycleError) {
            throw new Error('Graph contains a cycle');
        } else {
            throw error;
        }
    }
}
```

```ts
import { Graph as TSG } from 'typescript-graph';

class Graph extends TSG<string> {
    addNode(node: string): this {
        if (this.hasNode(node)) throw new Error(`Duplicate node "${node}"`);
        return super.addNode(node);
    }

    sort(): string[] {
        const result: string[] = [];
        const inDegree = new Map<string, number>(Array.from(this.nodes()).map(n => [n, 0]));

        for (const edge of this.edges()) {
            inDegree.set(edge.to, inDegree.get(edge.to)! + 1);
        }

        const queue: string[] = Array.from(inDegree).filter(([_, count]) => count === 0).map(([node]) => node);

        while (queue.length > 0) {
            const node = queue.shift()!;
            result.push(node);

            for (const neighbor of this.successors(edge, node)) {
                inDegree.set(neighbor, inDegree.get(neighbor)! - 1);
                if (inDegree.get(neighbor)! === 0) queue.push(neighbor);
            }
        }

        if (result.length !== this.nodes().length) throw new CycleError();

        return result;
    }
}

class CycleError extends Error {}
```
