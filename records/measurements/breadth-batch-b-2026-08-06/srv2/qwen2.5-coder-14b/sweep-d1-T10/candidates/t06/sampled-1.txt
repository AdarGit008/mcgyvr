export function parseCsvLine(line: string): string[] {
  // Handle empty line case
  if (line.trim() === "") {
    return [""];
  }

  const fields: string[] = [];
  let currentField = "";
  let inQuotes = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (char === '"') {
      // Toggle quote state
      inQuotes = !inQuotes;
      continue;
    }

    if (char === "," && !inQuotes) {
      // End of a field
      fields.push(currentField);
      currentField = "";
      continue;
    }

    currentField += char;
  }

  // Handle unclosed quotes
  if (inQuotes) {
    throw new Error("Quoted field is never closed");
  }

  // Add the last field
  fields.push(currentField);

  return fields;
}
