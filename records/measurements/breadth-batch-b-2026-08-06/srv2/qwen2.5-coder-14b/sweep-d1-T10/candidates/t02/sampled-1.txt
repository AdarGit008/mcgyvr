/**
 * Merges overlapping intervals into the smallest list of non-overlapping pairs.
 * @param intervals - An array of [start, end] pairs.
 * @returns A new array of non-overlapping [start, end] pairs.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (!Array.isArray(intervals) || intervals.length === 0) return [];

  const sortedIntervals = [...intervals].sort((a, b) => a[0] - b[0]);

  const merged: number[][] = [sortedIntervals[0]];

  for (let i = 1; i < sortedIntervals.length; i++) {
    const currentStart = sortedIntervals[i][0];
    const currentEnd = sortedIntervals[i][1];
    const lastMergedStart = merged[merged.length - 1][0];
    const lastMergedEnd = merged[merged.length - 1][1];

    if (currentStart > currentEnd) {
      throw new Error("Interval start must not be greater than its end");
    }

    if (currentStart <= lastMergedEnd) {
      // Intervals overlap, merge them
      merged[merged.length - 1][1] = Math.max(lastMergedEnd, currentEnd);
    } else {
      // No overlap, add the current interval
      merged.push([currentStart, currentEnd]);
    }
  }

  return merged;
}
