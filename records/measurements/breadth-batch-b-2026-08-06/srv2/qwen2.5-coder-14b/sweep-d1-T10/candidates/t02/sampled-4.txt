/**
 * Merges overlapping intervals into non-overlapping ones.
 * @param intervals - An array of [start, end] pairs.
 * @returns A sorted array of non-overlapping intervals.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];

  // Validate and sort intervals
  const sortedIntervals = intervals.slice().sort((a, b) => a[0] - b[0]);
  for (const [start, end] of sortedIntervals) {
    if (start > end) {
      throw new Error("Invalid interval: start is greater than end.");
    }
  }

  const mergedIntervals: number[][] = [];
  let currentInterval = [...sortedIntervals[0]];

  for (let i = 1; i < sortedIntervals.length; i++) {
    const [start, end] = sortedIntervals[i];
    if (start <= currentInterval[1]) {
      // Intervals overlap or touch, merge them
      currentInterval[1] = Math.max(currentInterval[1], end);
    } else {
      // No overlap, add the current interval to the result and move on
      mergedIntervals.push([...currentInterval]);
      currentInterval = [start, end];
    }
  }

  // Add the last interval
  mergedIntervals.push([...currentInterval]);

  return mergedIntervals;
}
