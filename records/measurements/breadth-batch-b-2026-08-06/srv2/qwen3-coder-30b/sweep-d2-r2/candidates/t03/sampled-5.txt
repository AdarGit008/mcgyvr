export function topoSort(n: number, edges: [number, number][]): number[] {
  // Validate inputs
  if (n < 0) {
    throw new Error("n must be non-negative");
  }
  
  // Build adjacency list and in-degree count
  const adjList: Set<number>[] = Array(n).fill(null).map(() => new Set());
  const inDegree: number[] = Array(n).fill(0);
  
  for (const [u, v] of edges) {
    if (u < 0 || u >= n || v < 0 || v >= n) {
      throw new Error("Edge endpoints must be in [0, n)");
    }
    if (u === v) {
      throw new Error("Self-loop detected");
    }
    if (!adjList[u].has(v)) {
      adjList[u].add(v);
      inDegree[v]++;
    }
  }
  
  // Initialize queue with nodes of in-degree 0
  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) {
      queue.push(i);
    }
  }
  
  // Sort queue to ensure lexicographically smallest order
  queue.sort((a, b) => a - b);
  
  const result: number[] = [];
  
  while (queue.length > 0) {
    // Take the smallest node
    const u = queue.shift()!;
    result.push(u);
    
    // Process neighbors
    for (const v of adjList[u]) {
      inDegree[v]--;
      if (inDegree[v] === 0) {
        queue.push(v);
      }
    }
    
    // Re-sort queue to maintain lexicographic order
    queue.sort((a, b) => a - b);
  }
  
  // Check for cycles
  if (result.length !== n) {
    throw new Error("Cycle detected");
  }
  
  return result;
}
