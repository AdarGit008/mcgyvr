export function maxSlidingWindow(nums: number[], k: number): number[] {
  if (!Number.isInteger(k) || k < 1 || k > nums.length) {
    throw new Error("Invalid window size");
  }

  if (nums.length === 0) {
    return [];
  }

  const result: number[] = [];
  const deque: number[] = []; // stores indices

  // Initialize the deque for the first window
  for (let i = 0; i < k; i++) {
    // Remove indices whose values are less than current element
    while (deque.length > 0 && nums[deque[deque.length - 1]] <= nums[i]) {
      deque.pop();
    }
    deque.push(i);
  }

  // The front of the deque is the index of the maximum element in the first window
  result.push(nums[deque[0]]);

  // Process the remaining windows
  for (let i = k; i < nums.length; i++) {
    // Remove indices that are out of the current window
    while (deque.length > 0 && deque[0] <= i - k) {
      deque.shift();
    }

    // Remove indices whose values are less than current element
    while (deque.length > 0 && nums[deque[deque.length - 1]] <= nums[i]) {
      deque.pop();
    }

    deque.push(i);
    result.push(nums[deque[0]]);
  }

  return result;
}
