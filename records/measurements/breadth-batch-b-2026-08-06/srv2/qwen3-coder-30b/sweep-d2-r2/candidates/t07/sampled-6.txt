export function firstUnbalanced(input: string): number {
  const stack: { char: string; index: number }[] = [];
  const bracketPairs = new Map<string, string>([
    [')', '('],
    [']', '['],
    ['}', '{']
  ]);
  
  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    
    if (['(', '[', '{'].includes(char)) {
      stack.push({ char, index: i });
    } else if ([')', ']', '}'].includes(char)) {
      if (stack.length === 0) {
        return i;
      }
      
      const lastOpener = stack.pop();
      if (lastOpener.char !== bracketPairs.get(char)) {
        return i;
      }
    }
  }
  
  if (stack.length > 0) {
    return stack[0].index;
  }
  
  return -1;
}
