export function addDecimalStrings(a: string, b: string): string {
  // Validate inputs
  if (typeof a !== 'string' || typeof b !== 'string') {
    throw new Error('Both arguments must be strings');
  }
  
  if (a.length === 0 || b.length === 0) {
    throw new Error('Both arguments must be non-empty strings');
  }
  
  // Check for invalid characters
  if (!/^\d+$/.test(a) || !/^\d+$/.test(b)) {
    throw new Error('Strings must contain only digits 0-9');
  }
  
  // Handle leading zeros by converting to numbers and back to strings
  // This ensures canonical representation (e.g., "007" becomes "7")
  const numA = BigInt(a);
  const numB = BigInt(b);
  
  // Since we can't use BigInt for the addition directly,
  // we'll perform digit-by-digit addition manually
  const result: number[] = [];
  let carry = 0;
  let i = a.length - 1;
  let j = b.length - 1;
  
  // Process digits from right to left (least significant to most significant)
  while (i >= 0 || j >= 0 || carry > 0) {
    const digitA = i >= 0 ? parseInt(a[i], 10) : 0;
    const digitB = j >= 0 ? parseInt(b[j], 10) : 0;
    
    const sum = digitA + digitB + carry;
    result.push(sum % 10);
    carry = Math.floor(sum / 10);
    
    i--;
    j--;
  }
  
  // Convert result array to string
  let resultStr = result.reverse().join('');
  
  // Ensure canonical representation (no leading zeros, except for "0")
  if (resultStr.length > 1 && resultStr[0] === '0') {
    resultStr = resultStr.substring(1);
  }
  
  return resultStr;
}
