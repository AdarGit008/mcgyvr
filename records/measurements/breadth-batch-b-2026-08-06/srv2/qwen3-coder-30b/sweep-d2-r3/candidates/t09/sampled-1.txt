export function romanToInt(input: string): number {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }
  
  if (input === '') {
    throw new Error('Input cannot be empty');
  }
  
  const validChars = new Set(['I', 'V', 'X', 'L', 'C', 'D', 'M']);
  for (const char of input) {
    if (!validChars.has(char)) {
      throw new Error('Invalid character in input');
    }
  }
  
  const romanValues: { [key: string]: number } = {
    'I': 1,
    'V': 5,
    'X': 10,
    'L': 50,
    'C': 100,
    'D': 500,
    'M': 1000
  };
  
  let candidateValue = 0;
  let i = 0;
  
  while (i < input.length) {
    const currentChar = input[i];
    const currentValue = romanValues[currentChar];
    
    if (i + 1 < input.length) {
      const nextChar = input[i + 1];
      const nextValue = romanValues[nextChar];
      
      if (currentValue < nextValue) {
        candidateValue += nextValue - currentValue;
        i += 2;
        continue;
      }
    }
    
    candidateValue += currentValue;
    i++;
  }
  
  if (candidateValue < 1 || candidateValue > 3999) {
    throw new Error('Number out of range');
  }
  
  const canonicalForm = numberToRoman(candidateValue);
  
  if (canonicalForm !== input) {
    throw new Error('Input is not canonical');
  }
  
  return candidateValue;
}

function numberToRoman(num: number): string {
  const romanNumerals: [number, string][] = [
    [1000, 'M'],
    [900, 'CM'],
    [500, 'D'],
    [400, 'CD'],
    [100, 'C'],
    [90, 'XC'],
    [50, 'L'],
    [40, 'XL'],
    [10, 'X'],
    [9, 'IX'],
    [5, 'V'],
    [4, 'IV'],
    [1, 'I']
  ];
  
  let result = '';
  
  for (const [value, numeral] of romanNumerals) {
    while (num >= value) {
      result += numeral;
      num -= value;
    }
  }
  
  return result;
}
