/** a divided by b if both are finite numbers. */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error('Both arguments must be finite numbers');
  }
  
  // Reject boolean as number for the divisor
  if (typeof b === 'boolean') {
      throw new Error('Divisor cannot be a boolean value');
  }
  
  if(b == 0){    // including -0 cases
    throw new Error('Cannot divide by zero');
  }

  return a / b;
}
