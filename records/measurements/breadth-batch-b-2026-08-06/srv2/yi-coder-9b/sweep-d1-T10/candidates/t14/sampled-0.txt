Sure! Here is how you could refactor the `activeEmails` function to utilize array methods like `map()` and `filter()`:

```ts
export interface User {
  email: string;
  active: boolean;
}

/** The lowercased addresses of the active users, in order. */
export function activeEmails(users: readonly User[]): string[] {
  return users
    .filter((user) => user.active) 
    .map(({email}) => email.toLowerCase());
}
```
