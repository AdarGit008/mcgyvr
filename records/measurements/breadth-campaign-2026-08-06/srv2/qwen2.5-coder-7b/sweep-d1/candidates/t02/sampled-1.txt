/**
 * Merges overlapping intervals into the smallest list of non-overlapping intervals.
 * @param intervals - List of [start, end] pairs.
 * @returns Sorted list of merged [start, end] pairs.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (!Array.isArray(intervals) || intervals.length === 0) {
    throw new Error('Input must be a non-empty array of intervals');
  }

  const sortedIntervals = intervals.slice().sort((a, b) => a[0] - b[0]);
  const merged: number[][] = [];

  for (const interval of sortedIntervals) {
    if (
      typeof interval[0] !== 'number' ||
      typeof interval[1] !== 'number' ||
      interval[0] > interval[1]
    ) {
      throw new Error('Invalid interval: start must be less than or equal to end');
    }

    if (merged.length === 0 || merged[merged.length - 1][1] < interval[0]) {
      merged.push(interval);
    } else {
      merged[merged.length - 1][1] = Math.max(merged[merged.length - 1][1], interval[1]);
    }
  }

  return merged;
}
