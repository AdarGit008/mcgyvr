export function firstUnbalanced(input: string): number {
  const stack: { index: number; char: string }[] = [];
  const bracketPairs: { [key: string]: string } = {
    ')': '(',
    ']': '[',
    '}': '{'
  };
  
  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    
    if (char === '(' || char === '[' || char === '{') {
      stack.push({ index: i, char });
    } else if (char === ')' || char === ']' || char === '}') {
      if (stack.length === 0) {
        return i;
      }
      
      const last = stack.pop();
      if (last.char !== bracketPairs[char]) {
        return i;
      }
    }
  }
  
  if (stack.length > 0) {
    return stack[0].index;
  }
  
  return -1;
}
