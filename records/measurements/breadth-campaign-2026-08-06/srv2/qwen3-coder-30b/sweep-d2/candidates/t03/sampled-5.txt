export function topoSort(n: number, edges: [number, number][]): number[] {
  // Validate input
  if (n < 0) {
    throw new Error("n must be non-negative");
  }
  
  // Build adjacency list and in-degree count
  const adjList: Set<number>[] = Array(n).fill(null).map(() => new Set());
  const inDegree: number[] = Array(n).fill(0);
  
  for (const [u, v] of edges) {
    if (u < 0 || u >= n || v < 0 || v >= n) {
      throw new Error("Edge endpoints must be in range [0, n)");
    }
    if (u === v) {
      throw new Error("Self-loop detected");
    }
    if (!adjList[u].has(v)) {
      adjList[u].add(v);
      inDegree[v]++;
    }
  }
  
  // Initialize priority queue with nodes having zero in-degree
  const pq: number[] = [];
  for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) {
      pq.push(i);
    }
  }
  
  // Sort to maintain lexicographic order
  pq.sort((a, b) => a - b);
  
  const result: number[] = [];
  
  while (pq.length > 0) {
    // Take the smallest node
    const u = pq.shift()!;
    result.push(u);
    
    // For each neighbor, reduce in-degree
    for (const v of adjList[u]) {
      inDegree[v]--;
      if (inDegree[v] === 0) {
        pq.push(v);
        pq.sort((a, b) => a - b); // Keep sorted for lexicographic order
      }
    }
  }
  
  // Check for cycles
  if (result.length !== n) {
    throw new Error("Cycle detected");
  }
  
  return result;
}
