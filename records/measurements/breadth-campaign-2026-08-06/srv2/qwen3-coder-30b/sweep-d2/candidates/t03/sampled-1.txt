export function topoSort(n: number, edges: [number, number][]): number[] {
  // Validate input
  if (n < 0) {
    throw new Error("n must be non-negative");
  }
  
  // Build adjacency list and in-degree array
  const graph: number[][] = Array(n).fill(null).map(() => []);
  const inDegree = Array(n).fill(0);
  
  for (const [u, v] of edges) {
    if (u < 0 || u >= n || v < 0 || v >= n) {
      throw new Error("Edge endpoints must be in range [0, n)");
    }
    
    // Check for self-loop
    if (u === v) {
      throw new Error("Graph contains a cycle");
    }
    
    graph[u].push(v);
    inDegree[v]++;
  }
  
  // Initialize queue with nodes having zero in-degree
  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) {
      queue.push(i);
    }
  }
  
  // Sort queue to ensure lexicographical order
  queue.sort((a, b) => a - b);
  
  const result: number[] = [];
  
  while (queue.length > 0) {
    // Take the smallest node
    const u = queue.shift()!;
    result.push(u);
    
    // Process neighbors
    for (const v of graph[u]) {
      inDegree[v]--;
      if (inDegree[v] === 0) {
        queue.push(v);
      }
    }
    
    // Re-sort queue to maintain lexicographical order
    queue.sort((a, b) => a - b);
  }
  
  // Check for cycles (if result length is less than n)
  if (result.length !== n) {
    throw new Error("Graph contains a cycle");
  }
  
  return result;
}
