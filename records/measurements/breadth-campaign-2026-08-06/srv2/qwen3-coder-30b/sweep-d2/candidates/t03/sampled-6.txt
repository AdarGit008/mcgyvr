export function topoSort(n: number, edges: [number, number][]): number[] {
  if (n < 0) {
    throw new Error("n must be non-negative");
  }

  // Initialize in-degrees and adjacency list
  const inDegree = new Array(n).fill(0);
  const adjList: number[][] = Array(n)
    .fill(null)
    .map(() => []);

  // Process edges
  for (const [u, v] of edges) {
    if (u < 0 || u >= n || v < 0 || v >= n) {
      throw new Error("Edge endpoints must be in [0, n)");
    }
    if (u === v) {
      throw new Error("Self-loop detected");
    }
    inDegree[v]++;
    adjList[u].push(v);
  }

  // Initialize queue with nodes having zero in-degree
  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) {
      queue.push(i);
    }
  }

  // Sort queue to ensure lexicographically smallest order
  queue.sort((a, b) => a - b);

  const result: number[] = [];

  while (queue.length > 0) {
    // Take the smallest node from the queue
    const u = queue.shift()!;
    result.push(u);

    // Process neighbors
    for (const v of adjList[u]) {
      inDegree[v]--;
      if (inDegree[v] === 0) {
        queue.push(v);
      }
    }

    // Keep queue sorted to maintain lexicographical order
    queue.sort((a, b) => a - b);
  }

  // Check for cycles
  if (result.length !== n) {
    throw new Error("Cycle detected");
  }

  return result;
}
