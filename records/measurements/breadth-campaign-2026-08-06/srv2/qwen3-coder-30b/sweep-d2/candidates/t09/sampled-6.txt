export function romanToInt(input: string): number {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }
  
  if (input.length === 0) {
    throw new Error('Input cannot be empty');
  }
  
  const validChars = new Set(['I', 'V', 'X', 'L', 'C', 'D', 'M']);
  for (const char of input) {
    if (!validChars.has(char)) {
      throw new Error('Input contains invalid characters');
    }
  }
  
  const romanValues: { [key: string]: number } = {
    'I': 1,
    'V': 5,
    'X': 10,
    'L': 50,
    'C': 100,
    'D': 500,
    'M': 1000
  };
  
  let result = 0;
  let prevValue = 0;
  
  for (let i = input.length - 1; i >= 0; i--) {
    const char = input[i];
    const value = romanValues[char];
    
    if (value < prevValue) {
      result -= value;
    } else {
      result += value;
    }
    
    prevValue = value;
  }
  
  if (result <= 0 || result > 3999) {
    throw new Error('Number out of range 1-3999');
  }
  
  // Re-encode the number canonically and compare with input
  const canonical = intToRoman(result);
  if (canonical !== input) {
    throw new Error('Input is not in canonical form');
  }
  
  return result;
}

function intToRoman(num: number): string {
  if (num <= 0 || num > 3999) {
    throw new Error('Number out of range 1-3999');
  }
  
  const values = [1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1];
  const symbols = ['M', 'CM', 'D', 'CD', 'C', 'XC', 'L', 'XL', 'X', 'IX', 'V', 'IV', 'I'];
  
  let result = '';
  
  for (let i = 0; i < values.length; i++) {
    while (num >= values[i]) {
      result += symbols[i];
      num -= values[i];
    }
  }
  
  return result;
}
