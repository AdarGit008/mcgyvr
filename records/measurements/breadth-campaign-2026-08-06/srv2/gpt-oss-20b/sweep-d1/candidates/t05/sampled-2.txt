/**
 * Split an array into consecutive groups of at most {@link size} elements.
 *
 * @param items - The input array to be chunked.
 * @param size - The maximum size of each chunk; must be a positive integer.
 * @returns An array of chunks, preserving the original order. Does not mutate {@link items}.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (!Number.isInteger(size) || size <= 0) {
    throw new Error('size must be a positive integer');
  }

  const result: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    result.push(items.slice(i, i + size));
  }
  return result;
}
