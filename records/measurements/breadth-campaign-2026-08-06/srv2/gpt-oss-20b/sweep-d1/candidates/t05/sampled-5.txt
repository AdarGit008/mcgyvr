/**
 * Split an array into consecutive groups of at most size elements, preserving order.
 *
 * @template T - The type of the items in the array.
 * @param items - Readonly array of items to be chunked.
 * @param size - Positive integer defining maximum group size.
 * @returns An array of chunks (arrays), each containing up to `size` elements.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (!Number.isInteger(size) || size <= 0) {
    throw new Error('size must be a positive integer');
  }

  const result: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    result.push(items.slice(i, i + size));
  }
  return result;
}
