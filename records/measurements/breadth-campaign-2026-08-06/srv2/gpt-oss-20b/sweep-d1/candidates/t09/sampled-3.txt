/** The factorial of n. */
export function factorial(n: number): number {
  if (!Number.isFinite(n) || !Number.isInteger(n)) {
    throw new Error('Invalid argument');
  }
  if (n < 0) {
    throw new Error('Negative argument not allowed');
  }
  if (n <= 1) {
    return 1;
  }
  return n * factorial(n - 1);
}
