```ts
/** The factorial of n. */
export function factorial(n: number): number {
  if (!Number.isInteger(n)) {
    throw new Error('factorial only accepts integer arguments');
  }
  if (n < 0) {
    throw new Error('factorial does not support negative numbers');
  }
  if (n === 0 || n === 1) {
    return 1;
  }
  return n * factorial(n - 1);
}
```
