/**
 * Encodes a string using run-length encoding.
 * @param input The string to encode.
 * @returns A run-length encoded string.
 */
export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }

  const length = input.length;
  if (length === 0) return '';

  const parts: string[] = [];
  let currentChar = input[0];
  let count = 1;

  for (let i = 1; i < length; i++) {
    const char = input[i];
    if (char === currentChar) {
      count++;
    } else {
      parts.push(`${currentChar}${count}`);
      currentChar = char;
      count = 1;
    }
  }

  parts.push(`${currentChar}${count}`);
  return parts.join('');
}
