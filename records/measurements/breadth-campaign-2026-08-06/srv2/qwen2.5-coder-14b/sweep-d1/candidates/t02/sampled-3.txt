/**
 * Merges overlapping intervals and returns the smallest list of non-overlapping pairs.
 * @param intervals - An array of [start, end] pairs.
 * @returns The merged list of non-overlapping intervals.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (!Array.isArray(intervals) || intervals.length === 0) {
    return [];
  }

  const sortedIntervals = [...intervals].sort((a, b) => a[0] - b[0]);

  const merged: number[][] = [sortedIntervals[0]];

  for (let i = 1; i < sortedIntervals.length; i++) {
    const current = sortedIntervals[i];
    const lastMerged = merged[merged.length - 1];

    if (current[0] > current[1]) {
      throw new Error("Invalid interval: start is greater than end");
    }

    if (current[0] <= lastMerged[1]) {
      lastMerged[1] = Math.max(lastMerged[1], current[1]);
    } else {
      merged.push([...current]);
    }
  }

  return merged;
}
