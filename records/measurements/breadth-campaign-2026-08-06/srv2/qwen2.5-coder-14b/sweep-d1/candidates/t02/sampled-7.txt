/**
 * Merges overlapping intervals into a list of non-overlapping intervals.
 * @param intervals - An array of [start, end] pairs.
 * @returns A new array of non-overlapping [start, end] pairs sorted by start.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];

  // Validate intervals and create a copy
  const validIntervals = intervals.map(([start, end]) => {
    if (start > end) throw new Error("Invalid interval: start is greater than end");
    return [start, end];
  });

  // Sort intervals by start
  validIntervals.sort((a, b) => a[0] - b[0]);

  const merged: number[][] = [];
  let currentInterval = [...validIntervals[0]];

  for (let i = 1; i < validIntervals.length; i++) {
    const [start, end] = validIntervals[i];
    if (start <= currentInterval[1]) {
      // Merge intervals
      currentInterval[1] = Math.max(currentInterval[1], end);
    } else {
      // No overlap, add the current interval to merged and move to next
      merged.push([...currentInterval]);
      currentInterval = [start, end];
    }
  }

  // Add the last interval
  merged.push([...currentInterval]);

  return merged;
}
