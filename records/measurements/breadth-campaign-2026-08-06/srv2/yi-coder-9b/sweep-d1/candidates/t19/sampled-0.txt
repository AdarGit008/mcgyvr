/** a divided by b. */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(a) || !Number.isFinite(b) || typeof b === 'boolean') {
    throw new Error('Both arguments must be finite numbers and the divisor cannot be a boolean');
  } else if (b === 0 || b === -0) {
    throw new Error('Cannot divide by zero');
  }
  return a / b;
}
