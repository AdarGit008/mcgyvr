export function parseCsvLine(line: string): string[] {
    const result: string[] = [];
    let currentField = "";
    let insideQuotes = false;
    for (let i = 0; i < line.length; i++) {
        if (insideQuotes) {
            if (line[i] === '"') {
                if (i + 1 < line.length && line[i+1] === '"') {
                    currentField += '"';
                    i++;
                } else {
                    insideQuotes = false;
                }
            } else if (line[i] === ",") {
                result.push(currentField);
                currentField = "";
            } else {
                currentField += line[i];
            }
        } else {
            if (line[i] === '"') {
                insideQuotes = true;
            } else if (line[i] === ",") {
                result.push(currentField);
                currentField = "";
            } else {
                currentField += line[i];
            }
        }
    }
    if (insideQuotes) {
        throw new Error('Unexpected end of line in quoted field');
    }
    result.push(currentField);
    return result;
}
