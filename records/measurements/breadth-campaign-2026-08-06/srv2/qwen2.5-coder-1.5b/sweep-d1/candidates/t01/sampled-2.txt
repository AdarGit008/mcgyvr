export function runLengthEncode(input: string): string {
  if (!typeof input === 'string') {
    throw new Error("Input must be a string");
  }

  let result = '';
  let count = 1;
  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    if (char === input[i + 1]) {
      count++;
    } else {
      result += char + (count > 1 ? count.toString() : '');
      count = 1;
    }
  }

  return result;
}
