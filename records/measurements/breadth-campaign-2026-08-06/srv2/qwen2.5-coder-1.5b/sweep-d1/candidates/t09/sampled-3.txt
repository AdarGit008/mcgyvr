/** The factorial of n. */
export function factorial(n: number): number {
  if (n === 0) {
    return 1;
  }
  if (!Number.isInteger(n)) {
    throw new Error('Factorial is only defined for non-negative integers.');
  }
  if (n < 0) {
    throw new Error('Factorial is not defined for negative numbers.');
  }
  let result = 1;
  for (let i = 2; i <= n; i++) {
    result *= i;
  }
  return result;
}
