/**
 * The temperature in degrees Fahrenheit.
 */
export function celsiusToFahrenheit(celsius: number): number {
  validateTemperature(celsius);
  return celsius * 1.8 + 32;
}

/**
 * The temperature in kelvin.
 */
export function celsiusToKelvin(celsius: number): number {
  validateTemperature(celsius);
  return celsius + 273.15;
}

// Helper to validate the input temperature
function validateTemperature(temp: number) {
  if (typeof temp !== "number" || !Number.isFinite(temp)) {
    throw new Error("temp must be a finite number");
  }
  if (temp < -273.15) {
    throw new Error("temp must be at or above absolute zero");
  }
}
