/** a divided by b. */
export function safeDivide(a: number, b: number): number {
  if (b === 0) {
    throw new Error("Division by zero is not allowed.");
  }
  if (!Number.isFinite(a)) {
    throw new Error("The first argument must be a finite number.");
  }
  if (!Number.isFinite(b)) {
    throw new Error("The second argument must be a finite number.");
  }
  return a / b;
}
