/**
 * Safely divides two numbers, handling division by zero and non-finite numbers.
 *
 * @param a The numerator.
 * @param b The denominator.
 * @returns The result of the division, rounded to 10 decimal places if necessary.
 */
export function safeDivide(a: number, b: number): number {
  if (b === 0 || !Number.isFinite(b)) {
    throw new Error('Division by zero or non-finite number.');
  }
  const result = a / b;
  return parseFloat(result.toFixed(10));
}
