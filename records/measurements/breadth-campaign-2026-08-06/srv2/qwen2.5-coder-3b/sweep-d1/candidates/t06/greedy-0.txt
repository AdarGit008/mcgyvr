export function parseCsvLine(line: string): string[] {
  if (line.trim() === "") return [];

  const result: string[] = [];
  let currentField = "";
  let inQuotes = false;
  let escapedQuote = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (char === '"') {
      if (!inQuotes) {
        inQuotes = true;
      } else if (escapedQuote) {
        currentField += '"';
        escapedQuote = false;
      } else {
        throw new Error("Quoted field is never closed");
      }
    } else if (char === "," && !inQuotes) {
      result.push(currentField.trim());
      currentField = "";
    } else if (char === " ") {
      if (!inQuotes) continue; // Ignore spaces in quoted fields
    }

    currentField += char;
  }

  if (currentField.trim() !== "") result.push(currentField.trim());

  return result;
}
