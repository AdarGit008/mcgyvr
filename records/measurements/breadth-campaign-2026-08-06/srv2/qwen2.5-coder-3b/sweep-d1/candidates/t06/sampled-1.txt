export function parseCsvLine(line: string): string[] {
  if (line.trim() === "") return [];

  const fields: string[] = [];
  let currentField = "";
  let quoted = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (char === '"') {
      if (!quoted) {
        quoted = true;
      } else if (i + 1 < line.length && line[i + 1] === '"') {
        // Double quote inside a quoted field
        currentField += '"';
        i++; // Skip the next double quote
      } else {
        // End of quoted field
        quoted = false;
      }
    } else if (!quoted) {
      if (char === ",") {
        fields.push(currentField.trim());
        currentField = "";
      } else if (char !== " ") {
        currentField += char;
      }
    } else {
      // Append character to the current field
      currentField += char;
    }
  }

  // Add the last field if there is one
  if (currentField.length > 0) fields.push(currentField.trim());

  return fields;
}
