export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') throw new Error('Argument must be a string');

  let encoded = '';
  for (let i = 0; i < input.length; ) {
    const char = input[i];
    let count = 1;
    while (i + 1 < input.length && input[i + 1] === char) {
      count++;
      i++;
    }
    encoded += `${char}${count}`;
    i++;
  }
  return encoded;
}
