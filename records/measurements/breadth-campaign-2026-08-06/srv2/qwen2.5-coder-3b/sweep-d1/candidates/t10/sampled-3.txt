/** The index of target in a sorted array, or -1 if target is not found. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length; // Updated to include the last element
  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1; // Updated to include mid in the search window
    } else {
      high = mid; // Updated to exclude mid from the search window
    }
  }
  return -1;
}
