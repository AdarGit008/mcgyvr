```ts
export interface Event { name: string; timestamp: number }

class EventLog {
  #events: Event[] = [];

  record(event: Event): void {
    if (!event.name || !Number.isFinite(event.timestamp)) {
      throw new Error('Invalid event data');
    }
    this.#events.push(event);
  }

  since(timestamp: number): Event[] {
    return this.#events.filter((event) => event.timestamp >= timestamp);
  }
}
```
