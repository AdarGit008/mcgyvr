export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
  const regex = /^v?(\d+)(?:\.(\d+))?(?:\.(\d+))?(?:-(\w+(?:[-.]\w+)*)?)?(?:\+(\w+(?:[-.]\w+)*)?)?$/;
  const match = version.match(regex);

  if (!match) {
    throw new Error(`Invalid semantic version format: "${version}"`);
  }

  return {
    major: parseInt(match[1], 10),
    minor: parseInt(match[2] || '0', 10),
    patch: parseInt(match[3] || '0', 10),
    prerelease: match[4] || null,
    build: match[5] || null
  };
}
