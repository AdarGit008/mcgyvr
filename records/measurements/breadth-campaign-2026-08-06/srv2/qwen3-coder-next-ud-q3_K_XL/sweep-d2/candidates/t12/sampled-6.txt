/**
 * Computes the Levenshtein (edit) distance between two strings.
 * @param a The first string.
 * @param b The second string.
 * @returns The minimum number of single-character insertions, deletions, and substitutions needed to transform a into b.
 */
export function editDistance(a: string, b: string): number {
  if (typeof a !== 'string' || typeof b !== 'string') {
    throw new Error('Both arguments must be strings.');
  }

  const m = a.length;
  const n = b.length;

  // Create DP table with dimensions (m+1) x (n+1)
  const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));

  // Base cases: transforming empty string to/from another string
  for (let i = 0; i <= m; i++) {
    dp[i][0] = i; // i deletions needed to convert first i chars of a to empty string
  }
  for (let j = 0; j <= n; j++) {
    dp[0][j] = j; // j insertions needed to convert empty string to first j chars of b
  }

  // Fill the DP table
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (a[i - 1] === b[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1]; // No operation needed
      } else {
        const insertCost = dp[i][j - 1] + 1;    // Insert character into a
        const deleteCost = dp[i - 1][j] + 1;    // Delete character from a
        const replaceCost = dp[i - 1][j - 1] + 1; // Replace character in a
        dp[i][j] = Math.min(insertCost, deleteCost, replaceCost);
      }
    }
  }

  return dp[m][n];
}
