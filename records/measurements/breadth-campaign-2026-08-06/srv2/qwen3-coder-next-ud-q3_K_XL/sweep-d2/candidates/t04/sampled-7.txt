/**
 * Tokenizes the input string by splitting on spaces and tabs, respecting quoted sections,
 * escape sequences, and combining adjacent quoted/unquoted text into single tokens.
 * 
 * @param input The string to tokenize.
 * @returns An array of tokens.
 */
export function tokenize(input: string): string[] {
  if (input === "") {
    return [];
  }

  const tokens: string[] = [];
  let currentToken = "";
  let inQuotes = false;
  let i = 0;

  while (i < input.length) {
    const char = input[i];

    if (char === '\\') {
      // Escape sequence: include the next character literally, if it exists
      if (i + 1 >= input.length) {
        throw new Error("Trailing backslash");
      }
      currentToken += input[i + 1];
      i += 2;
      continue;
    }

    if (char === '"') {
      // Toggle quoted mode
      inQuotes = !inQuotes;
      i++;
      continue;
    }

    if ((char === ' ' || char === '\t') && !inQuotes) {
      // Separator outside quotes: end current token if non-empty, skip separators
      if (currentToken !== "") {
        tokens.push(currentToken);
        currentToken = "";
      }
      i++;
      // Skip runs of separators
      while (i < input.length && (input[i] === ' ' || input[i] === '\t')) {
        i++;
      }
      continue;
    }

    // Append current character to token (including spaces and tabs inside quotes)
    currentToken += char;
    i++;
  }

  // After loop, check for unterminated quote
  if (inQuotes) {
    throw new Error("Unterminated quote");
  }

  // Add final token if any
  if (currentToken !== "") {
    tokens.push(currentToken);
  }

  return tokens;
}
