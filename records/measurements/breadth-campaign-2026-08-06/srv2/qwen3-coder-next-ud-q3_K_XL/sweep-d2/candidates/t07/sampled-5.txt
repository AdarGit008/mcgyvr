/**
 * Scans a string for balanced brackets among (), [], {}.
 * Returns -1 if balanced. Otherwise returns the index of the first offending character.
 * An offending character is either:
 * - a closing bracket that doesn't match the most recent unclosed opener, or
 * - the earliest unclosed opener if the scan ends with openers remaining.
 */
export function firstUnbalanced(input: string): number {
  const stack: { char: string; index: number }[] = [];
  const openingBrackets = "([{";
  const closingBrackets = ")]}";
  const bracketMap: Record<string, string> = {
    ")": "(",
    "]": "[",
    "}": "{"
  };

  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    
    if (openingBrackets.includes(char)) {
      stack.push({ char, index: i });
    } else if (closingBrackets.includes(char)) {
      if (stack.length === 0 || stack[stack.length - 1].char !== bracketMap[char]) {
        return i;
      }
      stack.pop();
    }
  }

  if (stack.length > 0) {
    return stack[0].index;
  }

  return -1;
}
