/**
 * Merges overlapping or touching intervals into non-overlapping intervals sorted by start.
 * @param intervals An array of [start, end] pairs where start <= end.
 * @returns A new array of merged intervals sorted by start.
 * @throws Error if any interval is invalid (not a two-number pair, or start > end).
 */
export function mergeIntervals(intervals: [number, number][]): [number, number][] {
  // Validate input
  if (!Array.isArray(intervals)) {
    throw new Error('Intervals must be an array');
  }

  for (let i = 0; i < intervals.length; i++) {
    const interval = intervals[i];
    if (!Array.isArray(interval) || interval.length !== 2) {
      throw new Error(`Interval at index ${i} must be a two-number pair`);
    }
    
    const [start, end] = interval;
    if (typeof start !== 'number' || typeof end !== 'number') {
      throw new Error(`Interval at index ${i} must contain numbers`);
    }
    if (!Number.isFinite(start) || !Number.isFinite(end)) {
      throw new Error(`Interval at index ${i} must contain finite numbers`);
    }
    if (start > end) {
      throw new Error(`Interval at index ${i} has start > end`);
    }
  }

  // Handle empty array
  if (intervals.length === 0) {
    return [];
  }

  // Create a deep copy and sort by start
  const sorted = [...intervals].sort((a, b) => a[0] - b[0]);

  // Merge intervals
  const result: [number, number][] = [];
  let current = sorted[0];

  for (let i = 1; i < sorted.length; i++) {
    const [start, end] = sorted[i];
    
    // Check if intervals overlap or touch
    if (current[1] >= start) {
      // Merge: update current interval's end if needed
      current = [current[0], Math.max(current[1], end)];
    } else {
      // No overlap/touch: push current and move to next
      result.push(current);
      current = sorted[i];
    }
  }

  // Push the last interval
  result.push(current);

  return result;
}
