 ```ts
/** Divides two numbers safely, throwing an Error if the divisor is zero or either argument is not a finite number. */
export function safeDivide(a: number, b: number): number {
  if (b === 0 || !Number.isFinite(a) || !Number.isFinite(b)) {
    throw new Error("Invalid arguments for division");
  }
  return a / b;
}
```
