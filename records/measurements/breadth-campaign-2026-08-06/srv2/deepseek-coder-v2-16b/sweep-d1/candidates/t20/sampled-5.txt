 ```ts
/** The KEY=VALUE pairs of a config file. */
export function parseConfig(text: string): Record<string, string> {
  const config: Record<string, string> = {};
  for (const line of text.split("\n")) {
    // Ignore blank lines and comments
    if (!line.trim() || line.trim().startsWith("#")) continue;

    // Trim whitespace around the key and value
    let [key, value] = line.split("=");
    key = key.trim();
    value = value ? value.trim() : "";

    if (key === "") {
      throw new Error(`Invalid config line: ${line}`);
    }

    // Prevent keys from reaching Object.prototype
    if (Object.keys(config).includes(key)) {
      throw new Error(`Key "${key}" would reach Object.prototype`);
    }

    // Assign the key-value pair to the config object
    config[key] = value;
  }
  return config;
}
```
