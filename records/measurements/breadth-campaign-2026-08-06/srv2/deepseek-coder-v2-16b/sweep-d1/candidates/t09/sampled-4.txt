 ```ts
/** The factorial of n. */
export function factorial(n: number): number {
  if (typeof n !== 'number' || !Number.isInteger(n)) {
    throw new Error('Argument must be a non-negative integer');
  }
  if (n < 0) {
    throw new Error('Argument must be a non-negative integer');
  }
  if (n === 1) {
    return 1;
  }
  return n * factorial(n - 1);
}
```
