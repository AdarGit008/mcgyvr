export function parseSemver(version: string): { major: number; minor: number; patch: number; prerelease: string | null; build: string | null } {
    const parts = version.match(/^(\d+)\.(\d+)\.(\d+)(?:-([0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*))?(?:\+([0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*))?$/);
    if (!parts) {
        throw new Error(`Invalid semantic version string: ${version}`);
    }

    const major = parseInt(parts[1], 10);
    const minor = parseInt(parts[2], 10);
    const patch = parseInt(parts[3], 10);
    const prerelease = parts[4] || null;
    const build = parts[5] || null;

    if (isNaN(major) || isNaN(minor) || isNaN(patch)) {
        throw new Error(`Invalid semantic version number: ${version}`);
    }

    return { major, minor, patch, prerelease, build };
}
