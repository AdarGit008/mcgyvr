/**
 * Converts a temperature in degrees Celsius to Fahrenheit.
 *
 * @param {number} celsius - The temperature in degrees Celsius.
 * @returns {number} The equivalent temperature in degrees Fahrenheit.
 */
function validateCelsius(celsius: number): void {
  if (typeof celsius !== "number" || !Number.isFinite(celsius)) {
    throw new Error("celsius must be a finite number");
  }
  if (celsius < -273.15) {
    throw new Error("celsius must be at or above absolute zero");
  }
}

export function celsiusToFahrenheit(celsius: number): number {
  validateCelsius(celsius);
  return celsius * 1.8 + 32;
}

/**
 * Converts a temperature in degrees Celsius to Kelvin.
 *
 * @param {number} celsius - The temperature in degrees Celsius.
 * @returns {number} The equivalent temperature in Kelvin.
 */
export function celsiusToKelvin(celsius: number): number {
  validateCelsius(celsius);
  return celsius + 273.15;
}
