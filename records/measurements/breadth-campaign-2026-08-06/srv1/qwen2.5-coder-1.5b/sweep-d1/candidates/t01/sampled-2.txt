/**
 * Encodes runs of repeated characters in a string into a compressed form.
 * @param input The string to encode.
 * @returns The encoded string with run lengths.
 */
export function runLengthEncode(input: string): string {
  if (typeof input !== 'string') {
    throw new Error('Input must be a string');
  }

  let encoded = '';
  let currentChar = input[0];
  let count = 1;

  for (let i = 1; i < input.length; i++) {
    const char = input[i];

    if (char === currentChar) {
      count++;
    } else {
      encoded += `${currentChar}${count}`;
      currentChar = char;
      count = 1;
    }
  }

  // Add the last run
  encoded += `${currentChar}${count}`;

  return encoded;
}
