/**
 * Splits an array into consecutive groups of at most size elements, preserving order.
 * @param items The array to split.
 * @param size The maximum size of each group.
 * @returns An array of arrays, where each inner array contains up to `size` elements from the input array.
 */
export function chunk<T>(items: readonly T[], size: number): T[][] {
  if (typeof size !== 'number' || !Number.isInteger(size) || size <= 0) {
    throw new Error('Size must be a positive integer.');
  }
  const result: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    result.push(items.slice(i, Math.min(i + size, items.length)));
  }
  return result;
}
