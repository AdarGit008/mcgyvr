export function parseCsvLine(line: string): string[] {
  const result: string[] = [];
  let currentField = "";
  let inQuotes = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];
    if (char === '"') {
      inQuotes = !inQuotes;
      continue;
    }
    if (char === ',') {
      if (!inQuotes) {
        result.push(currentField.trim());
        currentField = "";
      }
      continue;
    }
    currentField += char;
  }

  // Handle the last field
  if (!inQuotes && currentField.length > 0) {
    result.push(currentField.trim());
  }

  return result;
}
