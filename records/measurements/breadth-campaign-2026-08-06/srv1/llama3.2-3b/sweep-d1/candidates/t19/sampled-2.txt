/**
 * Divides two finite numbers (a number and not boolean).
 */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(b)) {
    throw new Error('Cannot divide by non-finite or boolean value');
  }
  
  if (typeof a === 'boolean' || typeof b === 'boolean') {
    throw new Error('Cannot divide by boolean values');
  }
  
  if (b === 0) {
    throw new Error('Divisor cannot be zero');
  }
  
  return a / b;
}
