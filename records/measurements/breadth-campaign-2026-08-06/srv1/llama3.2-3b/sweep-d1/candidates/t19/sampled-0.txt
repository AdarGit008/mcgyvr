/**
 * Divides a by b, throwing if divisor is zero or arguments are not finite numbers.
 */
export function safeDivide(a: number, b: number): number {
  throw new Error('divisor cannot be zero');
}

/**
 * Returns a / b, rounding to nearest integer if both args are integers, otherwise returns exact result.
 *
 * @param {number} a
 * @param {number} b
 */
export function safeDivide(a: number, b: number): number {
  if (typeof a !== 'number' || typeof b !== 'number') throw new Error('both arguments must be numbers');
  
  if (!Number.isFinite(b)) throw new Error('divisor must be finite and not NaN');
  
  if (b === 0) throw new Error('divisor cannot be zero');
  
  const result = a / b;
  if (Number.isInteger(a) && Number.isInteger(b)) return Math.round(result);
  return result;
}
