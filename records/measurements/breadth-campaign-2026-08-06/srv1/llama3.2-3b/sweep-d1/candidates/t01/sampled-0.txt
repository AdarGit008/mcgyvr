export function runLengthEncode(input: string): string {
    /**
     * Replaces every maximal run of one repeated character with that character 
     * followed by the run length in order.
     *
     * @param input - The input string to be encoded.
     */
    
    if (typeof input !== 'string') {
        throw new Error('Invalid argument. Input must be a string.');
    }

    const result = [];
    let currentChar = input[0];
    let currentCount = 1;

    for (let i = 1; i <= input.length; i++) {
        if (i === input.length || input[i] !== currentChar) {
            result.push(currentChar + (currentCount > 9 ? currentCount.toString() : currentCount));
            if (input[i] !== currentChar) {
                currentChar = input[i];
                currentCount = 1;
            }
        } else {
            currentCount++;
        }
    }

    return result.join('');
}
