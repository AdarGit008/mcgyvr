/**
 * Replaces every maximal run of one repeated character in the input string with that character followed by its length,
 * in order: "aaabbc" encodes to "a3b2c1". A run of length one still carries its count, so "abc" encodes to "a1b1c1".
 *
 * @param input The input string
 * @throws {Error} If the argument is not a string
 */
export function runLengthEncode(input: string): string {
    if (typeof input !== 'string') {
        throw new Error('Input must be a string');
    }

    const result = [];
    let currentChar = input[0];
    let currentCount = 1;

    for (let i = 1; i <= input.length; i++) {
        if (i === input.length || input[i] !== currentChar) {
            result.push(currentChar + (currentCount > 9 ? '...' : `${currentCount}`));
            if (input[i] !== currentChar) {
                currentChar = input[i];
                currentCount = 1;
            }
        } else {
            currentCount++;
        }
    }

    return result.join('');
}
