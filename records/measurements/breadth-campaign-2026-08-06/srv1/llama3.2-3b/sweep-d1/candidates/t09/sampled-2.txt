```typescript
/**
 * The factorial of n.
 */
export function factorial(n: number): number {
  if (typeof n !== 'number' || n % 1 !== 0) {
    throw new Error('Input must be a non-negative integer');
  }
  if (n === 0 || n < 0) {
    throw new Error('Input must be a non-negative integer');
  }
  if (n === 1) {
    return 1;
  } else if (n > 1) {
    return n * factorial(n - 1);
  }
}
```
