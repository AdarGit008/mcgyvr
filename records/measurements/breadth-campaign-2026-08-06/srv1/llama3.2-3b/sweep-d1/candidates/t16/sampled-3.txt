const MODULUS = 1000000007;
let prevPrev = 0;
let prev = 1;

export function fibMod(n: number): number {
  if (n === 0) return 0;
  for (let i = 2; i <= n; i++) {
    let temp = (prev + prevPrev) % MODULUS;
    prevPrev = prev;
    prev = temp;
  }
  return prev;
}
