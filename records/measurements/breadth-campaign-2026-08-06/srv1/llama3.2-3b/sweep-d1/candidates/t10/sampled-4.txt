/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1; // adjusted upper bound to include the last element
  while (low <= high) { // changed condition to allow for equal bounds
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1; // adjusted lower bound to include the last element
    }
  }
  return -1;
}
