/** The index of target in a sorted array, or -1. */
export function binarySearch(items: readonly number[], target: number): number {
  let low = 0;
  let high = items.length - 1; // <--- CHANGED FROM high = items.length - 1 TO HIGH = Items LENGTH - 1
  while (low <= high) { // <--- CHANGED FROM low < high TO low <= high
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      return mid;
    }
    if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
