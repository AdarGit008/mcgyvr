export function flatten(items: readonly unknown[], depth = Infinity): unknown[] {
  if (!Number.isFinite(depth) || depth < 0) {
    throw new Error("Depth must be a non-negative integer or Infinity.");
  }

  const result: unknown[] = [];

  items.forEach(item => {
    if (Array.isArray(item)) {
      if (depth > 1) {
        result.push(...flatten(item, depth - 1));
      } else {
        result.push(...item);
      }
    } else {
      result.push(item);
    }
  });

  return result;
}
