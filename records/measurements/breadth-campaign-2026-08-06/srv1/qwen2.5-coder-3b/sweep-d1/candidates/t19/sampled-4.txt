/** Safe division that throws an error if the divisor is zero or non-finite. */
export function safeDivide(a: number, b: number): number {
  if (!Number.isFinite(b)) {
    throw new Error("Cannot divide by non-finite number.");
  }
  if (b === 0) {
    throw new Error("Cannot divide by zero.");
  }
  return a / b;
}
