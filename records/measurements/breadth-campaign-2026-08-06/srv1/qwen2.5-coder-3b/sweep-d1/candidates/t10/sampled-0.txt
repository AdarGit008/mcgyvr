export function binarySearch(items: readonly number[], target: number): number {
  if (items.length === 0) return -1;

  let low = 0;
  let high = items.length - 1;
  while (low <= high) {
    const mid = Math.floor((low + high) / 2);
    if (items[mid] === target) {
      // Check if there are duplicates and return the first occurrence
      if (mid > 0 && items[mid - 1] === target) {
        high = mid - 1;
      } else {
        return mid;
      }
    } else if (items[mid] < target) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return -1;
}
