/**
 * Performs topological sorting on a directed acyclic graph (DAG).
 * @param nodes - Array of node names.
 * @param edges - Array of [from, to] pairs representing directed edges.
 * @returns An array of node names sorted topologically.
 */
export function topologicalSort(nodes: readonly string[], edges: readonly (readonly [string, string])[]): string[] {
    const graph: { [key: string]: string[] } = {};
    const inDegree: { [key: string]: number } = {};

    // Initialize graph and in-degree count
    for (const node of nodes) {
        graph[node] = [];
        inDegree[node] = 0;
    }

    // Build the graph and calculate in-degrees
    for (const [from, to] of edges) {
        if (!nodes.includes(from)) throw new Error(`Node ${from} not found`);
        if (!nodes.includes(to)) throw new Error(`Node ${to} not found`);

        graph[from].push(to);
        inDegree[to]++;
    }

    // Queue for nodes with zero in-degree
    const queue: string[] = [];
    for (const node of nodes) {
        if (inDegree[node] === 0) {
            queue.push(node);
        }
    }

    const result: string[] = [];

    while (queue.length > 0) {
        const currentNode = queue.shift()!;
        result.push(currentNode);

        for (const neighbor of graph[currentNode]) {
            inDegree[neighbor]--;
            if (inDegree[neighbor] === 0) {
                queue.push(neighbor);
            }
        }
    }

    // Check if all nodes are included in the result
    if (result.length !== nodes.length) {
        throw new Error('Cycle detected');
    }

    return result;
}
