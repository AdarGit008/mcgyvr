/**
 * Merges overlapping intervals into the smallest list of non-overlapping pairs,
 * sorted ascending by start. Intervals that merely touch (end === next start) do merge.
 * Throws an Error for a pair whose start is greater than its end.
 *
 * @param intervals - Array of [start, end] pairs to be merged.
 * @returns A new array of merged non-overlapping intervals.
 */
export function mergeIntervals(intervals: number[][]): number[][] {
  if (intervals.length === 0) return [];
  
  // Sort intervals by start time
  const sortedIntervals = intervals.slice().sort((a, b) => a[0] - b[0]);

  let merged: number[][] = [sortedIntervals[0]];
  for (let i = 1; i < sortedIntervals.length; i++) {
    const current = sortedIntervals[i];
    const lastMerged = merged[merged.length - 1];

    if (current[0] > lastMerged[1]) {
      // No overlap, add new interval
      merged.push(current);
    } else if (current[1] > lastMerged[1]) {
      // Overlap, update end time of the last merged interval
      merged[merged.length - 1][1] = current[1];
    }
  }

  return merged;
}
