from typing import List

def span_merge(spans: List[List[int]]) -> List[List[int]]:
    if not spans:
        return []

    # Sort the spans by their start time
    sorted_spans = sorted(spans, key=lambda x: x[0])

    merged_spans = []
    current_span = sorted_spans[0]

    for span in sorted_spans[1:]:
        if span[0] <= current_span[1]:
            # Merge the current and new spans
            current_span[1] = max(current_span[1], span[1])
        else:
            # Add the current span to the merged list
            merged_spans.append(current_span)
            current_span = span

    # Add the last span to the merged list
    merged_spans.append(current_span)

    return merged_spans
