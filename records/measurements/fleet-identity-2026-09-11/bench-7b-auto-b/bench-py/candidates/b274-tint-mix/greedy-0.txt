def tint_mix(volume_a: int, strength_a: int, volume_b: int, strength_b: int) -> int:
    if volume_a < 0 or volume_b < 0:
        raise ValueError("Volume cannot be negative")
    total_volume = volume_a + volume_b
    if total_volume == 0:
        return 0
    total_strength = (volume_a * strength_a + volume_b * strength_b) // total_volume
    return total_strength
