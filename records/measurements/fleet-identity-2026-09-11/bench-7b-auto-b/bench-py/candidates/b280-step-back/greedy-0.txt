def step_back(moves: str) -> int:
    """Return the furthest forward position reached during a walk."""
    max_position = current_position = 0
    for move in moves:
        if move == 'F':
            current_position += 1
            max_position = max(max_position, current_position)
        elif move == 'B':
            current_position -= 1
    return max_position
