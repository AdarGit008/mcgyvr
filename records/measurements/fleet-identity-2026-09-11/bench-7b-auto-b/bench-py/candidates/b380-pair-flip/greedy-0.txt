def flip_one(pair: list) -> list:
    """Return a pair with its two members the other way round."""
    if len(pair) != 2:
        raise ValueError("Input must be a pair with exactly two members.")
    return [pair[1], pair[0]]

def flip_all(pairs: list) -> list:
    """Do that to every pair in a list, keeping the pairs in the order they came."""
    return [flip_one(pair) for pair in pairs]
