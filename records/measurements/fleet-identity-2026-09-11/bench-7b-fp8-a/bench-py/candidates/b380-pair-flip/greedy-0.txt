def flip_one(pair: list) -> list:
    if len(pair) != 2:
        raise ValueError("Pair must contain exactly two elements")
    return [pair[1], pair[0]]

def flip_all(pairs: list) -> list:
    if not all(isinstance(pair, list and len(pair) == 2 for pair in pairs):
        raise ValueError("All elements in the list must be pairs containing exactly two elements")
    return [flip_one(pair) for pair in pairs]
