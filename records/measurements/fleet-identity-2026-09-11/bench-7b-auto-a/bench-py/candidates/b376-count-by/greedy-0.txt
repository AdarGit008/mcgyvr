```python
from typing import List, Dict

def count_by(records: List[Dict], field: str) -> Dict[str, int]:
    if not field:
        raise ValueError("Field name cannot be empty")
    
    counts = {}
    for record in records:
        if field in record:
            value = record[field]
            counts[value] = counts.get(value, 0) + 1
    
    return counts
```
