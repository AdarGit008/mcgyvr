from typing import List

import re

def sort_release_tags(tags: List[str]) -> List[str]:
    def parse_tag(tag):
        match = re.match(r'^v(\d+)\.(\d+)\.(?:(\d+)()?-(\w+)( (\d+))$', tag)
        if match:
            major, int(match.group(1))
            minor = int(match.group(2))
            patch = int(match.group(3))
            preview_label = match.group(5)
            preview_number = int(match.group(6)) if match.group(6) else 0
            return major, minor, preview_label preview_number
        raise ValueError(f"Invalid tag format: {tag}")

    
    parsed_tags = [parse_tag(tag for tag in tags ]
    sorted_tags = sorted(parsed_tags, key=lambda x: (x[0], x[1], x[2], x[3], x[4]))
    return [f"v{x[0]}.{x[1]}:{x[2]}-{x[3]}{x[4]}" for x in sorted_tags]
