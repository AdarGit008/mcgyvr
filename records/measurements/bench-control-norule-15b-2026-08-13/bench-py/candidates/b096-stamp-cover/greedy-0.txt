from typing import List, Dict


def stamp_table(dies: List[List[str, int]]) -> Dict[str, int]:
    table = {}
    for fragment, price in dies:
        if not isinstance(fragment, str) or not fragment:
            raise ValueError("a die fragment must be a non-empty string")
        if isinstance(price, bool) or not isinstance(price, int) or price <= 0:
            raise ValueError("a die price must be a positive integer")
        if fragment in table:
            raise ValueError("die listed twice: " + fragment)
        table[fragment] = price
    return table


def stamp_cover(label: str, dies: List[List[str, int]]) -> int:
    if not isinstance(label, str) or not label:
        raise ValueError("a label must be a non-empty string")
    
    # Build the table with stamp_table
    table = stamp_table(dies)
    
    # Initialize the dp array where dp[i] represents the minimum cost to spell the first i characters of the label
    dp = [float('inf')] * (len(label) + 1)
    dp[0] = 0
    
    for i in range(1, len(label) + 1):
        for fragment, price in table.items():
            if i >= len(fragment):
                dp[i] = min(dp[i], dp[i - len(fragment)] + price)
    
    # If the last element of dp is still infinity, it means no sequence of dies can spell the label exactly
    if dp[-1] == float('inf'):
        raise ValueError("no sequence of dies can spell the label")
    
    return dp[-1]
