from typing import List

def batch_ends(count: int, size: int) -> List[int]:
    """The position of the last entry of each full batch."""
    if size <= 0:
        raise ValueError("Batch size must be positive.")
    
    ends = []
    start = 0
    while start + size <= count:
        ends.append(start)
        start += size
    
    # If there's a final part-batch, it doesn't have a last entry to report.
    if start < count:
        ends.pop()
    
    return ends
