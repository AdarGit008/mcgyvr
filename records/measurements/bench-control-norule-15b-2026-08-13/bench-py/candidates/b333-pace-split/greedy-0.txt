from typing import List, Tuple

def pace_of(seconds: int, kilometres: int) -> int:
    if kilometres <= 0:
        raise ValueError("Distance must be greater than zero.")
    return seconds // kilometres

def pace_list(legs: List[Tuple[int, float]]) -> List[float]:
    if not legs:
        raise ValueError("Legs list cannot be empty.")
    
    for leg in legs:
        if leg[1] <= 0:
            raise ValueError("Distance must be greater than zero.")
        
        seconds, kilometres = leg
        pace = seconds // kilometres
        yield pace
