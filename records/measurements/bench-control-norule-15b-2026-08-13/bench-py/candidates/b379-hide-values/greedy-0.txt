from typing import Dict, Any

def hide_values(store: Dict[str, Any]) -> Dict[str, str]:
    """A store with every value starred out."""
    hidden = {}
    for key, value in store.items():
        if isinstance(value, (str, list)):
            hidden["*" * len(key)] = "*" * len(value)
        else:
            hidden[key] = "*"
    return hidden
