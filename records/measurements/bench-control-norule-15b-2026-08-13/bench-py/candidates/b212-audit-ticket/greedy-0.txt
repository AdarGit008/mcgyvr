def audit_ticket(ticket: str) -> str:
    if not isinstance(ticket, str):
        raise ValueError("Ticket must be a string")
    
    if len(ticket) != 14 or ticket[2] != '-' or ticket[6] != '-':
        return "shape"
    
    letters = sum(ord(c) - ord('A') + 1 for c in ticket[:2])
    digits = sum(int(d) * (i + 1) for i, d in enumerate(ticket[3:7]))
    
    total = letters + digits
    check_digit = total % 10
    
    if ticket[-1] != str(check_digit):
        return "check"
    
    return "ok"
