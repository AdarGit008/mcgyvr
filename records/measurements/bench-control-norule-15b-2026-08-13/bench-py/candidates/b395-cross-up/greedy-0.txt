from typing import List

def cross_up(readings: List[int], level: int) -> int:
    if not readings or level < 0:
        raise ValueError("Invalid input")

    crossings = 0
    for i in range(1, len(readings)):
        if readings[i] > level and (i == 0 or readings[i - 1] <= level):
            crossings += 1

    return crossings
