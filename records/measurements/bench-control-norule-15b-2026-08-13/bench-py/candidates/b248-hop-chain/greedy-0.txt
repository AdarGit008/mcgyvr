from typing import List, Tuple

def hop_chain(hops: List[Tuple[int, int]]) -> bool:
    if len(hops) == 0:
        return True
    
    start = hops[0][0]
    end = hops[0][1]
    
    for i in range(1, len(hops)):
        if hops[i][0] != end + 1 or hops[i][1] != start - 1:
            return False
        start = hops[i][0]
        end = hops[i][1]
    
    return True
