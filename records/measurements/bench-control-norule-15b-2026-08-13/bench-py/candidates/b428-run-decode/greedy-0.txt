def run_decode(coded: str) -> str:
    decoded = ""
    i = 0
    while i < len(coded):
        if coded[i].isdigit():
            count = int(coded[i])
            i += 1
            letter = coded[i]
            decoded += letter * count
        else:
            decoded += coded[i]
        i += 1
    return decoded
