from typing import List


def served_by(queue: List[str], turn: int) -> str:
    if not queue:
        raise ValueError("a queue needs somebody in it")
    return queue[turn % len(queue)]


def queue_round(queue: List[str], turns: int) -> List[str]:
    """Who is served on each turn, the queue coming round again."""
    if not queue:
        raise ValueError("a queue needs somebody in it")
    
    served = []
    for turn in range(min(turns, len(queue))):
        served.append(served_by(queue, turn))
    
    return served
