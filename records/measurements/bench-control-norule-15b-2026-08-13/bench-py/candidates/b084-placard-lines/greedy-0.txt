```python
"""Lay a notice's text into placard lines."""


def placard_lines(text: str, width: int) -> list:
    if not isinstance(text, str) or len(text) == 0:
        raise ValueError("Text must be a non-empty string")
    if not isinstance(width, int) or width <= 0:
        raise ValueError("Width must be a positive integer")
    if text.startswith(" "):
        raise ValueError("Text cannot start with a space")
    if text.endswith(" ") or text.endswith(" "):
        raise ValueError("Text cannot end with two spaces")
    if any(len(word) > width for word in text.split()):
        raise ValueError("Word wider than the placard")
    
    lines = []
    line = ""
    for word in text.split(" "):
        if len(line) + len(word) <= width:
            line = line + " " + word
        else:
            lines.append(line)
            line = word
    lines.append(line)
    return lines
```
