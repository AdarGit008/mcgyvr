def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not isinstance(start, int) or start < 0:
        raise ValueError("start must be a non-negative integer")
    if not isinstance(low, int) or low < 0:
        raise ValueError("low must be a non-negative integer")
    if not isinstance(high, int) or high <= low:
        raise ValueError("high must lie strictly above low")
    if not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")

    heated = False
    coldest = start
    switches = 0

    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("drifts must be an integer")
        if abs(drift) > power:
            continue

        if heated:
            temp += drift + power
        else:
            temp = start + drift

        if temp < coldest:
            coldest = temp

        if temp >= high:
            heated = False
            switches += 1
        elif not heated:
            heated = True
            switches += 1

    return {"temp": temp, "heated": switches, "switches": switches, "coldest": coldest}
