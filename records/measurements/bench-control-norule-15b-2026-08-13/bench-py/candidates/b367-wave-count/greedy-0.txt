from typing import List

def wave_count(readings: List[int]) -> int:
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > readings[i - 1]:
            if i == len(readings) - 1 or readings[i + 1] < readings[i]:
                count += 1
        elif readings[i] < readings[i - 1]:
            if i == 0 or readings[i - 2] > readings[i]:
                count += 1
    return count
