export function turnTail(entries: string[], count: number): string[] {
  if (count === 0) return entries;

  const out: string[] = [];
  const stop = Math.min(count, entries.length);
  for (let i = stop - 1; i >= 0; i -= 1) {
    out.push(entries[i]);
  }
  for (let i = stop; i < entries.length; i += 1) {
    out.push(entries[i]);
  }

  return out;
}
