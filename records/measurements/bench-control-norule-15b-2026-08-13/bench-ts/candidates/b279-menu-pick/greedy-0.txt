```ts
export function menuPick(dishes: { name: string; price: number }[], budget: number): string[] {
  if (!dishes || !budget) throw new Error('Invalid input');

  return dishes.filter(dish => dish.price <= budget)
    .sort((a, b) => a.price === b.price ? a.name.localeCompare(b.name) : a.price - b.price);
}
```
