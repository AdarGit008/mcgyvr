/**
 * Finds the longest streak of games scoring strictly more than the bar.
 * @param scores - Array of integer scores in game order.
 * @param bar - Integer bar value that must be exceeded to count a game.
 * @returns Pair [start, length] of the longest streak, or [-1, 0] if no streak exists.
 */
export function hotStreak(scores: number[], bar: number): number[] {
    let maxLength = 0;
    let maxStart = -1;
    let currentLength = 0;
    let currentStart = -1;

    for (let i = 0; i < scores.length; i++) {
        if (scores[i] > bar) {
            if (currentStart === -1) {
                currentStart = i;
            }
            currentLength++;
        } else {
            if (currentLength > maxLength) {
                maxLength = currentLength;
                maxStart = currentStart;
            }
            currentLength = 0;
            currentStart = -1;
        }
    }

    if (currentLength > maxLength) {
        maxLength = currentLength;
        maxStart = currentStart;
    }

    return [maxStart, maxLength];
}
