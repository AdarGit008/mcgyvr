export function mendShards(copies: Array<Array<number | null>>): number[] {
  if (copies.length === 0) {
    throw new Error("Input list of copies cannot be empty.");
  }
  
  const length = copies[0].length;
  for (const copy of copies) {
    if (copy.length !== length) {
      throw new Error("All copies must have the same length.");
    }
  }

  const mended: number[] = [];
  for (let position = 0; position < length; position++) {
    const counts = new Map<number | null, number>();
    for (const copy of copies) {
      const value = copy[position];
      if (value !== null && typeof value !== "number" || Number.isNaN(value)) {
        throw new Error("Each slot must hold a non-negative integer or the null value.");
      }
      counts.set(value, (counts.get(value) ?? 0) + 1);
    }

    let best: number | null = null;
    let bestCount = 0;
    for (const [value, count] of counts) {
      if (count > bestCount || (count === bestCount && value !== null && best !== null && value < best)) {
        best = value;
        bestCount = count;
      }
    }

    mended.push(best === null ? -1 : best);
  }

  return mended;
}
