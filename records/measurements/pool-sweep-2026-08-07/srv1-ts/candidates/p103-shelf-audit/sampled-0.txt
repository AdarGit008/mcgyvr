/** Fold shelf entries into a final count and a skipped-take tally. */
export function shelfCount(entries: Array<[string, number]>): [number, number] {
  let count = 0;
  let skippedTakes = 0;

  for (const [kind, amount] of entries) {
    if (kind === "add") {
      count += amount;
    } else if (kind === "take") {
      if (typeof amount !== "number" || !Number.isInteger(amount) || amount < 0) {
        throw new Error("Invalid take amount");
      }
      if (amount > count) {
        skippedTakes++;
        continue;
      }
      count -= amount;
    } else if (kind === "fix") {
      if (typeof amount !== "number" || !Number.isInteger(amount) || amount < 0) {
        throw new Error("Invalid fix amount");
      }
      count = amount;
    } else {
      throw new Error(`Invalid kind: ${kind}`);
    }
  }

  return [count, skippedTakes];
}
