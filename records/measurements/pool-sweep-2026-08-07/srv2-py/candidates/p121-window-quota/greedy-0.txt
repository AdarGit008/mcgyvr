def window_quota(limit: int, width: int, calls: list) -> list:
    if limit <= 0 or width <= 0:
        raise ValueError("Limit and width must be positive integers")
    
    quota = {}
    results = []
    
    for time, name in calls:
        if not isinstance(name, str) or not name:
            raise ValueError("Name must be a non-empty string")
        if time < 0:
            raise ValueError("Time must be a non-negative integer")
        
        frame = time // width
        if frame not in quota:
            quota[frame] = {}
        
        if name not in quota[frame]:
            quota[frame][name] = limit
        
        if quota[frame][name] > 0:
            quota[frame][name] -= 1
            results.append("ok")
        else:
            results.append("over")
    
    return results
