def unspool_text(spool: str) -> str:
    if not isinstance(spool, str):
        raise ValueError("Argument must be a string")
    
    output = []
    i = 0
    while i < len(spool):
        char = spool[i]
        if char == '<':
            if i + 1 == len(spool) or spool[i+1] != '<':
                raise ValueError("Single less-than sign without a corresponding greater-than sign")
            output.append('<')
            i += 2
        elif char.isdigit():
            if i + 1 >= len(spool) or not spool[i+1].isdigit() or spool[i] == '0' or spool[i+1] == '0':
                raise ValueError("Reach or haul is not a valid digit sequence")
            
            reach = int(spool[i])
            haul = int(spool[i+1])
            if reach > len(output) or haul > reach:
                raise ValueError("Invalid reach or haul value")
            
            output.extend(output[-reach:][-haul:])
            i += 2
        else:
            output.append(char)
            i += 1
    
    return ''.join(output)
