def wrap_text(text: str, width: int) -> list[str]:
    if not isinstance(text, str) or not isinstance(width, int):
        raise ValueError("Text must be a string and width must be an integer.")
    if width <= 0:
        raise ValueError("Width must be a positive integer.")
    if text.strip() != text:
        raise ValueError("Text cannot have leading, trailing, or doubled spaces.")

    words = text.split()
    lines = []
    current_line = []

    for word in words:
        if len(' '.join(current_line + [word])) <= width:
            current_line.append(word)
        else:
            lines.append(' '.join(current_line))
            current_line = [word]

    lines.append(' '.join(current_line))
    return lines
