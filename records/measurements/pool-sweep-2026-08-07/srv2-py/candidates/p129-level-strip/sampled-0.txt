def level_strip(readings: list[int], low: int, high: int, ramp: str) -> str:
    if len(ramp) == 0:
        raise ValueError("empty ramp")
    if high <= low:
        raise ValueError("span must rise")
    out = []
    for r in readings:
        index = max(0, min(len(ramp) - 1, int((r - low) * (len(ramp) - 1) // (high - low))))
        out.append(ramp[index])
    return "".join(out)
