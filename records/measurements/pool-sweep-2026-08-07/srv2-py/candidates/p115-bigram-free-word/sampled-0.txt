from typing import List

def smallest_free_word(n: int, k: int, banned: List[str]) -> str:
    if n <= 0 or n > 12:
        raise ValueError("n must be a positive integer of at most 12")
    if not (1 <= k <= 6):
        raise ValueError("k must be an integer from 1 to 6")
    if not isinstance(banned, list):
        raise ValueError("banned must be a list")
    if any(not isinstance(item, str) or len(item) != 2 or set(item) - set(ascii_lowercase[:k]) for item in banned):
        raise ValueError("each banned entry must be a two-character string of letters within the first k")

    alphabet = ascii_lowercase[:k]
    banned_combinations = set(banned)

    def is_valid_word(word: str) -> bool:
        return all(word[i:i+2] not in banned_combinations for i in range(len(word) - 1))

    for length in range(1, n + 1):
        for word in product(alphabet, repeat=length):
            if is_valid_word(''.join(word)):
                return ''.join(word)

    raise ValueError("no valid word of length n can be formed")
