def seal_serial(serial: str) -> str:
    if not isinstance(serial, str):
        raise ValueError("Input must be a string")
    if len(serial) != 8 or not serial.isdigit():
        raise ValueError("Input must be exactly eight decimal digits")

    weight_cycle = [3, 7, 1, 3, 7, 1, 3, 7]
    seal_sum = sum(int(digit) * weight for digit, weight in zip(serial, weight_cycle))
    seal_digit = seal_sum % 11
    if seal_digit == 10:
        seal_char = 'K'
    else:
        seal_char = str(seal_digit)

    return serial + seal_char
