def commit_index(log: list[int], matches: list[int], current_term: int) -> dict:
    if not all(isinstance(x, list) for x in [log, matches]):
        raise ValueError("Both log and matches must be lists.")
    
    if not isinstance(current_term, int) or current_term <= 0:
        raise ValueError("Current term must be a whole number of one or more.")
    
    if any(term < 1 or (isinstance(term, float) and term.is_integer() is False) for term in log):
        raise ValueError("All terms in the log must be whole numbers of one or more.")
    
    if any(term < log[i-1] for i, term in enumerate(log)):
        raise ValueError("Terms in the log must not fall below the term before it.")
    
    if any(term > current_term for term in log):
        raise ValueError("No entry may be stamped above the leader's current term.")
    
    max_log_index = len(log) - 1
    quorum_size = (len(matches) + 1) // 2
    
    commit = 0
    safe = 0
    behind = []
    
    for i, match in enumerate(matches):
        if not isinstance(match, int) or match < 0 or match > max_log_index:
            raise ValueError("Copied number must be a whole number between zero and the log's length.")
        
        if match >= commit:
            if match == commit:
                behind.append(i)
            else:
                commit = match
                safe = match
    
    return {"commit": commit, "safe": safe, "behind": behind}
