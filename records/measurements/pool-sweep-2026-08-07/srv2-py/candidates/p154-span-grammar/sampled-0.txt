def expand_span_grammar(pattern: str) -> list:
    def parse_group(group):
        if ".." in group:
            start, end = map(int, group.split(".."))
            width = len(str(start))
            return [f"{i:0{width}d}" for i in range(start, end + 1)]
        else:
            choices = group.split("|")
            for choice in choices:
                if not choice.isalnum():
                    raise ValueError("Invalid choice in group body")
            return choices

    def expand_recursive(pattern):
        if "<" not in pattern and ">" not in pattern:
            return [pattern]
        groups = []
        i = 0
        while i < len(pattern):
            if pattern[i] == "<":
                j = pattern.find(">", i)
                if j == -1 or i + 1 >= j or pattern[i + 1] == ">" or pattern[j - 1] in "<|":
                    raise ValueError("Invalid group syntax")
                groups.append(parse_group(pattern[i + 1 : j]))
                i = j + 1
            else:
                tilde_index = pattern.find("~", i)
                if tilde_index != -1:
                    if tilde_index + 1 >= len(pattern) or pattern[tilde_index + 1] not in "<>||~":
                        raise ValueError("Invalid escape sequence")
                    groups.append([pattern[i : tilde_index + 2]])
                    i = tilde_index + 2
                else:
                    groups.append([pattern[i]])
                    i += 1
        expansions = []
        for group in groups[0]:
            if len(groups) == 1:
                expansions.append(group)
            else:
                sub_expansions = expand_recursive("".join(subgroup) for subgroup in zip(*groups[1:]))
                for sub_expansion in sub_expansions:
                    expansions.append(group + sub_expansion)
        if len(expansions) > 500:
            raise ValueError("Too many expansions")
        return sorted(set(expansions))

    if not pattern:
        raise ValueError("Empty pattern")

    try:
        return expand_recursive(pattern)
    except (ValueError, TypeError) as e:
        raise ValueError(f"Invalid pattern: {e}")
