def count_strip_tilings(width: int) -> int:
    if width < 0 or not isinstance(width, int):
        raise ValueError("Width must be a non-negative integer")
    
    if width == 0:
        return 1
    elif width == 1:
        return 1
    
    # dp[i] will store the number of ways to tile a strip of width i
    dp = [0] * (width + 1)
    dp[0] = 1
    dp[1] = 1
    
    for i in range(2, width + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    
    return dp[width]
