def bracket_seed_order(count: int) -> list:
    if not isinstance(count, int) or count < 2 or count & (count - 1) != 0:
        raise ValueError("count must be a power of two, at least 2")
    
    def generate_bracket(current_count):
        if current_count == 2:
            return [1, 2]
        field = []
        for seed in range(1, current_count // 2 + 1):
            field.extend([seed, current_count - seed])
        next_round = generate_bracket(current_count // 2)
        result = []
        for i in range(len(next_round) // 2):
            result.extend([field[2 * i] + next_round[i], field[2 * i + 1] + next_round[i]])
        return result
    
    return generate_bracket(count)
