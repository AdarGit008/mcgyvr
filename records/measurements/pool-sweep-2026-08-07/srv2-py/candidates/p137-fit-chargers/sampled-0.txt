def fit_chargers(chargers: list, devices: list) -> list:
    handed = [-1] * len(devices)
    for device in devices:
        pick = -1
        for index, charger in enumerate(chargers):
            if charger["plug"] == device["plug"] and charger["low"] <= device["draw"] <= charger["high"]:
                pick = index
                break
        handed[devices.index(device)] = pick
    return handed
