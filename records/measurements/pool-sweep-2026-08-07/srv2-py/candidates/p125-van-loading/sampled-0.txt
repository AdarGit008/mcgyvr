def load_vans(parcels: list[int]) -> list[int]:
    if not parcels or any(weight < 1 for weight in parcels):
        raise ValueError("Parcels must be a non-empty list of positive integers")
    
    first_van = 0
    second_van = 0
    
    for parcel in sorted(parcels, reverse=True):
        if first_van <= second_van:
            first_van += parcel
        else:
            second_van += parcel
    
    return [first_van, second_van]
