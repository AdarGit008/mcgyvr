def transcribe_runes(source: str, table: list[list[str]]) -> str:
    if any(len(pattern) == 0 for pattern, _ in table):
        raise ValueError("Rule patterns cannot be empty")
    
    result = []
    pos = 0
    
    while pos < len(source):
        matched = False
        for pattern, output in table:
            if source.startswith(pattern, pos):
                result.append(output)
                pos += len(pattern)
                matched = True
                break
        if not matched:
            result.append(source[pos])
            pos += 1
    
    return ''.join(result)
