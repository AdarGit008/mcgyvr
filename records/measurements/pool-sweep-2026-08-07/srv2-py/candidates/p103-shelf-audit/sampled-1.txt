def shelf_count(entries: list) -> list:
    """Fold shelf entries into a final count and a skipped-take tally."""
    count = 0
    skipped_takes = 0
    for kind, amount in entries:
        if not isinstance(kind, str) or kind not in ["add", "take", "fix"]:
            raise ValueError("Invalid kind entry")
        if not isinstance(amount, int) or amount < 0:
            raise ValueError("Invalid amount entry")
        if kind == "add":
            count += amount
        elif kind == "take":
            take_amount = min(count, amount)
            count -= take_amount
            skipped_takes += amount - take_amount
        else:
            count = amount
    return [count, skipped_takes]
