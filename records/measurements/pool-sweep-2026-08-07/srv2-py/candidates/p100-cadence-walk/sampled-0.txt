def label_cadence_walk(edges: list[list[str]], cadence: list[str], start: str, goal: str) -> int:
    if not cadence or any(not l for l in cadence):
        raise ValueError("Invalid cadence")
    if not all(isinstance(e, list) and len(e) == 3 and all(isinstance(x, str) and x for x in e) for e in edges):
        raise ValueError("Invalid edge format")
    if start not in (e[0] for e in edges) or goal not in (e[1] for e in edges):
        raise ValueError("Start or goal node not in edges")

    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v, l in edges:
        graph[u].append((v, l))

    queue = deque([(start, 0, 0)])
    visited = set()

    while queue:
        current, steps, cadence_index = queue.popleft()
        if current == goal and steps > 0:
            return steps
        if (current, steps % len(cadence)) in visited:
            continue
        visited.add((current, steps % len(cadence)))
        for neighbor, label in graph[current]:
            if label == cadence[cadence_index % len(cadence)]:
                queue.append((neighbor, steps + 1, (cadence_index + 1) % len(cadence)))

    return -1
