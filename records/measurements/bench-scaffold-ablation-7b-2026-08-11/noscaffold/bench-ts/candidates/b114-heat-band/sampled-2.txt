export function runThermostat(start: number, low: number, high: number, power: number, drifts: number[]): { temp: number; heated: number; switches: number; coldest: number } {
    if (!Number.isInteger(start) || !Number.isInteger(low) || !Number.isInteger(high) || !Number.isInteger(power)) {
        throw new Error('Start, low, high, and power must be integers');
    }
    if (low >= high) {
        throw new Error('Low must be strictly below high');
    }
    if (power <= 0 || !Number.isInteger(power)) {
        throw new Error('Power must be a positive integer');
    }
    if (!Array.isArray(drifts) || drifts.some(d => typeof d !== 'number') || drifts.some(d => !Number.isInteger(d))) {
        throw new Error('Drifts must be an array of integers');
    }

    let temp = start;
    let heated = 0;
    let switches = 0;
    let coldest = start;

    for (const drift of drifts) {
        const heaterOn = temp < low || temp >= high;
        if (heaterOn) {
            temp += power;
        }
        temp += drift;

        if (temp < coldest) {
            coldest = temp;
        }

        if (heated === 0 && heaterOn) {
            switches++;
        } else if (heated > 0 && !heaterOn) {
            switches++;
        }

        heated = heaterOn ? 1 : 0;
    }

    return { temp, heated, switches, coldest };
}
