/**
 * Explodes a kit into its base parts, calculating the total count consumed.
 * @param catalog - A mapping of kit names to their recipes.
 * @param kit - The name of the kit to explode.
 * @param want - The number of units of the kit to produce.
 * @returns A mapping from base part to the total count consumed.
 */
export function explodeKit(catalog: Record<string, { makes: number; parts: [string, number][] }>, kit: string, want: number): Record<string, number> {
    if (want <= 0) throw new Error('Want must be a positive number');

    const result: Record<string, number> = {};

    const explode = (kit: string, need: number): void => {
        const recipe = catalog[kit];
        if (!recipe) throw new Error(`Kit "${kit}" not found in catalog`);

        const runs = Math.ceil(need / recipe.makes);
        for (const [part, count] of recipe.parts) {
            const partNeed = runs * count;
            if (catalog[part]) {
                explode(part, partNeed);
            } else {
                result[part] = (result[part] || 0) + partNeed;
            }
        }
    };

    explode(kit, want);
    return result;
}
