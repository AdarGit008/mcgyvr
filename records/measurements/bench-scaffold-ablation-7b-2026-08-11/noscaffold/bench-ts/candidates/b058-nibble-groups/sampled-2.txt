export function formatBits(value: number, width: number): string {
  if (!Number.isInteger(value) || value < 0) {
    throw new Error("Value must be a non-negative integer");
  }
  if (width % 4 !== 0 || width > 32) {
    throw new Error("Width must be a positive multiple of four and at most 32");
  }
  if (value >= 1 << width) {
    throw new Error("Value does not fit within the specified width");
  }

  const binary = value.toString(2).padStart(width, '0');
  const chunks = [];
  for (let i = 0; i < binary.length; i += 4) {
    chunks.push(binary.substr(i, 4));
  }
  return chunks.join(' ');
}
