export function layConveyor(floor: string[], row: number, col: number, length: number): string[] {
    if (length <= 0) throw new Error("Length must be greater than zero");
    if (row < 0 || row >= floor.length) throw new Error("Row out of bounds");
    if (col < 0 || col >= floor[row].length) throw new Error("Column out of bounds");
    if (col + length - 1 >= floor[row].length) throw new Error("Conveyor run goes off the east edge");

    const result = [...floor];
    for (let i = 0; i < length; i++) {
        if (result[row][col + i] !== ".") throw new Error("Cell not open");
        result[row] = result[row].substring(0, col + i) + "=" + result[row].substring(col + i + 1);
    }
    return result;
}
