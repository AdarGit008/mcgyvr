export function layConveyor(floor: string[], row: number, col: number, length: number): string[] {
  if (length <= 0) throw new Error("Length must be greater than zero");
  if (row < 0 || row >= floor.length) throw new Error("Row out of bounds");
  if (col < 0 || col + length > floor[row].length) throw new Error("Run would leave the plan");

  const newFloor = [...floor];
  for (let i = 0; i < length; i++) {
    if (newFloor[row][col + i] !== ".") throw new Error("Cell not open");
    newFloor[row] = newFloor[row].substring(0, col + i) + "=" + newFloor[row].substring(col + i + 1);
  }

  return newFloor;
}
