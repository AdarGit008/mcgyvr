import re

def parse_option(segment: str) -> list:
    """Parse a key=value segment into [key, value]."""
    match = re.match(r'^(\b\w+\b)=(.*)$', segment)
    if not match:
        raise ValueError("Invalid option format")
    
    key, value = match.groups()
    if value.startswith('"') and value.endswith('"'):
        value = value[1:-1]
    elif value.startswith('"'):
        raise ValueError("Unmatched double quote in value")
    
    return [key, value]

def scan_pairs(text: str) -> list:
    """Split the text into key=value pairs, handling quoted values."""
    if not isinstance(text, str) or not text:
        raise ValueError("Invalid input type or empty string")
    
    result = []
    current_segment = ""
    in_quotes = False
    
    for char in text:
        if char == '"':
            in_quotes = not in_quotes
        elif char == ';' and not in_quotes:
            if current_segment:
                result.append(parse_option(current_segment))
                current_segment = ""
            else:
                raise ValueError("Empty segment")
        else:
            current_segment += char
    
    if current_segment:
        result.append(parse_option(current_segment))
    
    # Check for repeated keys
    seen_keys = set()
    for key, _ in result:
        if key in seen_keys:
            raise ValueError(f"Repeated key: {key}")
        seen_keys.add(key)
    
    return result
