def region_totals(grid: list, queries: list) -> list:
    if not grid or not all(grid):
        raise ValueError("Grid must be non-empty with rows of equal length")
    row_len = len(grid[0])
    if any(len(row) != row_len for row in grid):
        raise ValueError("All rows must have the same number of cells")
    if not all(isinstance(cell, int) for row in grid for cell in row):
        raise ValueError("All cells must be integers")

    prefix_sum = [[0] * (row_len + 1) for _ in range(len(grid) + 1)]
    for i in range(1, len(grid) + 1):
        for j in range(1, row_len + 1):
            prefix_sum[i][j] = grid[i - 1][j - 1] + prefix_sum[i - 1][j] + prefix_sum[i][j - 1] - prefix_sum[i - 1][j - 1]

    results = []
    for query in queries:
        if len(query) != 4:
            raise ValueError("Each query must be a list of exactly four integers")
        top, left, bottom, right = query
        if not (0 <= top < bottom <= len(grid)) or not (0 <= left < right <= row_len):
            raise ValueError("Query indices are out of bounds")
        total = prefix_sum[bottom][right] - prefix_sum[top][right] - prefix_sum[bottom][left] + prefix_sum[top][left]
        results.append(total)

    return results
