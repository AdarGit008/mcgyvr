def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary")
    if start not in links:
        raise ValueError(f"start '{start}' is not a valid post name")
    
    visited = set()
    path = []
    current = start
    
    while True:
        if current in visited:
            raise ValueError(f"handoff loop detected at post '{current}'")
        visited.add(current)
        path.append(current)
        
        next_post = links[current]
        if not next_post:
            break
        if next_post not in links:
            raise ValueError(f"handoff to unknown post '{next_post}'")
        
        current = next_post
    
    return path
