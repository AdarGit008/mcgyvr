def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a mapping")
    if start not in links:
        raise ValueError(f"start '{start}' is not a post named in links")
    
    visited = set()
    path = []
    current = start
    
    while current:
        if current in visited:
            raise ValueError("walk comes back to a previously visited post")
        
        visited.add(current)
        path.append(current)
        
        next_post = links[current]
        if not next_post and len(path) > 1:
            break
        current = next_post
    
    return path
