def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not isinstance(start, int) or not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("start, low, and high must be integers")
    if not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")
    
    if start != int(start) or low != int(low) or high != int(high) or power != int(power):
        raise ValueError("start, low, high, and power must be integers")
    if low >= high:
        raise ValueError("low must be strictly below high")
    
    temp = start
    heated = 0
    switches = 0
    coldest = start
    
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("drift values must be integers")
        
        if temp < low:
            state = True
            switches += 1
        elif temp >= high:
            state = False
            switches += 1
        else:
            state = None
        
        if state is not None and state:
            temp += power + drift
        else:
            temp += drift
        
        coldest = min(coldest, temp)
    
    return {"temp": temp, "heated": heated, "switches": switches, "coldest": coldest}
