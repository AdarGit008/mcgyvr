def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not all(isinstance(x, int) for x in [start, low, high, power]):
        raise ValueError("Start, low, high, and power must be integers.")
    if not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("Low and high must be integers.")
    if not (isinstance(power, int) and power > 0):
        raise ValueError("Power must be a positive integer.")
    if not isinstance(drifts, list):
        raise ValueError("Drifts must be a list.")
    
    temp = start
    heater_on = False
    switches = 0
    coldest = start
    
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("Each element in drifts must be an integer.")
        
        if heater_on:
            temp += power
        
        temp += drift
        
        if temp < low and not heater_on:
            heater_on = True
            switches += 1
        elif temp >= high and heater_on:
            heater_on = False
            switches += 1
        
        coldest = min(coldest, temp)
    
    return {
        "temp": temp,
        "heated": sum(1 for d in drifts if heater_on),
        "switches": switches,
        "coldest": coldest
    }
