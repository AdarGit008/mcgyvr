def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not all(isinstance(x, int) for x in [start, low, high, power]):
        raise ValueError("Start, low, high, and power must be integers")
    if power <= 0:
        raise ValueError("Power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("Drifts must be a list")
    if not all(isinstance(x, int) for x in drifts):
        raise ValueError("All drifts must be integers")
    if low >= high:
        raise ValueError("Low must be strictly below high")

    temp = start
    heated = 0
    switches = 0
    coldest = start

    heater_on = False
    for drift in drifts:
        current_temp = temp + (power if heater_on else 0)
        if current_temp < low and not heater_on:
            heater_on = True
            switches += 1
        elif current_temp >= high and heater_on:
            heater_on = False
            switches += 1

        temp += drift
        if temp < coldest:
            coldest = temp

    final_temp = temp + (power if heater_on else 0)
    if final_temp < coldest:
        coldest = final_temp

    return {
        "temp": final_temp,
        "heated": heated,
        "switches": switches,
        "coldest": coldest
    }
