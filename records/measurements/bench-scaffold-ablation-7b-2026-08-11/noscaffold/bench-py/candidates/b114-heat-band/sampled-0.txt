def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not all(isinstance(x, int) for x in [start, low, high, power]):
        raise ValueError("Start, low, and high must be integers.")
    if not isinstance(power, int) or power <= 0:
        raise ValueError("Power must be a positive integer.")
    if not isinstance(drifts, list):
        raise ValueError("Drifts must be a list.")
    
    if start != int(start) or low != int(low) or high != int(high) or power != int(power):
        raise ValueError("Start, low, and high must be integers.")
    if low >= high:
        raise ValueError("Low must be strictly below high.")
    if not all(isinstance(x, int) for x in drifts):
        raise ValueError("Drifts must contain only integers.")
    
    temp = start
    heated = 0
    switches = 0
    coldest = [temp]
    
    heater_on = False
    
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("Drifts must contain only integers.")
        
        if heater_on:
            temp += power
        
        temp += drift
        
        if temp < low and not heater_on:
            heater_on = True
            switches += 1
        elif temp >= high and heater_on:
            heater_on = False
            switches += 1
        
        coldest.append(temp)
    
    return {
        "temp": temp,
        "heated": heated,
        "switches": switches,
        "coldest": min(coldest),
    }
