"""Tape-measure spans: parse to inches, format canonically, and add."""

INCHES = {"yd": 36, "ft": 12, "in": 1}
ORDER = ("yd", "ft", "in")


def parse_span(text: str) -> int:
    if not isinstance(text, str):
        raise ValueError("text must be a string")
    if not text.strip():
        raise ValueError("text cannot be empty")
    
    parts = text.split()
    prev_unit = None
    total = 0
    
    for part in parts:
        value, unit = part[:-2], part[-2:]
        
        if not value.isdigit() or (value == "0" and len(value) > 1):
            raise ValueError("values must be non-negative integers without leading zeros")
        if unit not in INCHES:
            raise ValueError(f"unknown unit: {unit}")
        if prev_unit is not None and INCHES[prev_unit] < INCHES[unit]:
            raise ValueError(f"units out of order: {prev_unit} before {unit}")
        
        total += int(value) * INCHES[unit]
        prev_unit = unit
    
    return total


def format_span(inches: int) -> str:
    if isinstance(inches, bool) or not isinstance(inches, int) or inches < 0:
        raise ValueError("inches must be a non-negative integer")
    if inches == 0:
        return "0in"
    
    parts = []
    for unit in ORDER:
        value, inches = divmod(inches, INCHES[unit])
        if value > 0:
            parts.append(f"{value}{unit}")
    
    return " ".join(parts)


def add_spans(first: str, second: str) -> str:
    total = parse_span(first) + parse_span(second)
    return format_span(total)
