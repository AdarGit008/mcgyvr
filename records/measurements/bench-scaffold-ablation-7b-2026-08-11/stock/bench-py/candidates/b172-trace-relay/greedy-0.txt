def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict): raise ValueError("links must be a mapping")
    if start not in links: raise ValueError("start is not a post")
    route = []
    post = start
    visited = set()
    
    while True:
        if post in visited:
            raise ValueError("watch comes round again")
        if post == "":
            break
        visited.add(post)
        route.append(post)
        post = links[post]
    
    return route
