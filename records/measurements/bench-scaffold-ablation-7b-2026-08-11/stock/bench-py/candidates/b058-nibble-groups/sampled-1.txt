"""Render a value as fixed-width unsigned binary, nibble-grouped."""


def format_bits(value: int, width: int) -> str:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError("value must be a non-negative integer")
    if width <= 0 or width % 4 != 0 or width > 32:
        raise ValueError("width must be a positive multiple of four up to 32")
    binary = f"{value:0{width}b}"
    return " ".join(binary[i:i+4] for i in range(0, width, 4))
