def region_totals(grid: list, queries: list) -> list:
    if not isinstance(grid, list) or not grid:
        raise ValueError("grid must be a non-empty list of rows")
    width = len(grid[0]) if isinstance(grid[0], list) else 0
    if width == 0:
        raise ValueError("rows must be non-empty lists")
    for row in grid:
        if not isinstance(row, list) or len(row) != width:
            raise ValueError("rows must all share one length")
        for cell in row:
            if isinstance(cell, bool) or not isinstance(cell, int):
                raise ValueError("cells must be integers")

    prefix_sum = [[0] * (width + 1) for _ in range(len(grid) + 1)]
    for i in range(1, len(prefix_sum)):
        for j in range(1, len(prefix_sum[0])):
            prefix_sum[i][j] = grid[i - 1][j - 1] + prefix_sum[i - 1][j] + prefix_sum[i][j - 1] - prefix_sum[i - 1][j - 1]

    results = []
    for query in queries:
        if not isinstance(query, list) or len(query) != 4:
            raise ValueError("each query must be a list of exactly four integers")
        top, left, bottom, right = query
        if not (0 <= top < bottom <= len(grid)) or not (0 <= left < right <= width):
            raise ValueError("query bounds are out of grid dimensions")
        results.append(prefix_sum[bottom][right] - prefix_sum[top][right] - prefix_sum[bottom][left] + prefix_sum[top][left])

    return results
