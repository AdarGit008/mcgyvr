def fill_template(template: str, values: dict) -> str:
    if not isinstance(template, str):
        raise ValueError("fill_template expects a string template")
    
    result = []
    i = 0
    while i < len(template):
        if template[i] == '{':
            j = i + 1
            while j < len(template) and template[j].isalnum():
                j += 1
            if j - i == 1 or not (i > 0 and template[i-1] != '{'):
                raise ValueError("Unterminated placeholder")
            name = template[i+1:j]
            if not name.isidentifier():
                raise ValueError(f"Malformed placeholder: {name}")
            if name not in values:
                raise ValueError(f"Unknown placeholder: {name}")
            result.append(values[name])
            i = j
        else:
            result.append(template[i])
            i += 1
    
    return ''.join(result)
