def fill_template(template: str, values: dict) -> str:
    if not isinstance(template, str):
        raise ValueError("fill_template expects a string template")

    result = []
    i = 0
    while i < len(template):
        if template[i] == '{':
            j = i + 1
            while j < len(template) and template[j].isalnum() or template[j] == '_':
                j += 1
            if j - i > 1:
                placeholder = template[i+1:j-1]
                if not placeholder:
                    raise ValueError("Placeholder cannot be empty")
                if placeholder in values:
                    result.append(values[placeholder])
                else:
                    raise ValueError(f"Unknown placeholder: {placeholder}")
                i = j
            else:
                result.append('{')
                i += 1
        else:
            result.append(template[i])
            i += 1

    return ''.join(result)
