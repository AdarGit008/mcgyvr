"""Lay a straight eastward conveyor run across a factory floor plan."""


def lay_conveyor(floor: list[str], row: int, col: int, length: int) -> list[str]:
    if row < 0 or row >= len(floor):
        raise ValueError("the run starts off the plan")
    if col < 0 or col + length > len(floor[row]):
        raise ValueError("the run leaves the plan")
    cells = list(floor[row])
    for i in range(length):
        if cells[col + i] != ".":
            raise ValueError("a cell the run covers is not open")
        cells[col + i] = "="
    return ["".join(row) if row_index == row else row for row_index, row in enumerate(floor)]
