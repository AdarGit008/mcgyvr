"""Lay a straight eastward conveyor run across a factory floor plan."""


def lay_conveyor(floor: list, row: int, col: int, length: int) -> list:
    if row < 0 or row >= len(floor):
        raise ValueError("the run starts off the plan")
    cells = list(floor[row])
    end_col = col + length
    if end_col > len(cells):
        raise ValueError("the run goes off the plan")
    for c in cells[col:end_col]:
        if c != ".":
            raise ValueError("some of the run's cells are not open")
    cells[col:end_col] = ["="] * (end_col - col)
    return ["".join(row) if i == row else row for i, row in enumerate(floor)]
