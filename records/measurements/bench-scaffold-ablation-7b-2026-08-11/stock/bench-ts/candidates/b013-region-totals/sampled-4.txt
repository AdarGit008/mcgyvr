export function regionTotals(grid: number[][], queries: number[][]): number[] {
  if (!Array.isArray(grid) || grid.length === 0) {
    throw new Error("grid must be a non-empty list of rows");
  }
  const width = Array.isArray(grid[0]) ? grid[0].length : 0;
  if (width === 0) {
    throw new Error("rows must be non-empty lists");
  }
  for (const row of grid) {
    if (!Array.isArray(row) || row.length !== width) {
      throw new Error("rows must all share one length");
    }
    for (const cell of row) {
      if (!Number.isInteger(cell)) {
        throw new Error("cells must be integers");
      }
    }
  }

  const prefixSumTable = Array.from({ length: grid.length + 1 }, () =>
    Array(width + 1).fill(0)
  );

  for (let i = 0; i < grid.length; i++) {
    for (let j = 0; j < width; j++) {
      prefixSumTable[i + 1][j + 1] =
        grid[i][j] +
        prefixSumTable[i][j + 1] +
        prefixSumTable[i + 1][j] -
        prefixSumTable[i][j];
    }
  }

  const results: number[] = [];
  for (const query of queries) {
    if (!Array.isArray(query) || query.length !== 4) {
      throw new Error("query must be a list of exactly four integers");
    }
    const [top, left, bottom, right] = query;
    if (
      top < 0 ||
      left < 0 ||
      bottom > grid.length ||
      right > width ||
      top >= bottom ||
      left >= right
    ) {
      throw new Error("invalid query boundaries");
    }
    results.push(
      prefixSumTable[bottom][right] -
        prefixSumTable[top][right] -
        prefixSumTable[bottom][left] +
        prefixSumTable[top][left]
    );
  }

  return results;
}
