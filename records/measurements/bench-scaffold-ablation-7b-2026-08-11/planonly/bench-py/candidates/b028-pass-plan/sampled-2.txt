def cheapest_pass_plan(trip_days: list[int], passes: list[dict[str, int]]) -> dict[str, int | list[list[int]]]:
    if not all(isinstance(day, int) and day > 0 for day in trip_days):
        raise ValueError("Trip days must be strictly increasing positive integers")
    if not all(isinstance(pass_, dict) and 'span' in pass_ and 'cost' in pass_ for pass_ in passes):
        raise ValueError("Pass list must contain dictionaries with 'span' and 'cost'")
    if not all(isinstance(span, int) and span > 0 for pass_ in passes if 'span' in pass_):
        raise ValueError("Span must be a positive integer")
    if not all(isinstance(cost, int) and cost >= 0 for pass_ in passes if 'cost' in pass_):
        raise ValueError("Cost must be a non-negative integer")

    trip_days = set(trip_days)
    n = len(trip_days)

    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    purchase_plan = [[] for _ in range(n + 1)]

    for i, day in enumerate(trip_days):
        for pass_ in passes:
            span = pass_['span']
            cost = pass_['cost']
            if i - span >= 0 and dp[i - span] + cost < dp[i]:
                dp[i] = dp[i - span] + cost
                purchase_plan[i].extend(purchase_plan[i - span])
                purchase_plan[i].append([day, span])

    if dp[n] == float('inf'):
        raise ValueError("No valid pass plan found")

    return {
        'total': dp[n],
        'purchases': [pair for pair_list in purchase_plan for pair in pair_list]
    }
