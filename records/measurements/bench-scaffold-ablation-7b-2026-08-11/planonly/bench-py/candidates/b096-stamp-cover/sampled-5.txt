def stamp_table(dies: list) -> dict:
    """Return a lookup table for die fragments to their prices."""
    table = {}
    for fragment, price in dies:
        if not isinstance(fragment, str) or not fragment:
            raise ValueError("Fragment must be a non-empty string.")
        if not isinstance(price, int) or price <= 0:
            raise ValueError("Price must be a positive integer.")
        if fragment in table:
            raise ValueError(f"Duplicate fragment: {fragment}")
        table[fragment] = price
    return table

def stamp_cover(label: str, dies: list) -> int:
    """Return the cheapest total price of any sequence of die presses."""
    if not isinstance(label, str) or not label:
        raise ValueError("Label must be a non-empty string.")

    table = stamp_table(dies)
    n = len(label)
    dp = [float('inf')] * (n + 1)
    dp[0] = 0

    for i in range(1, n + 1):
        for fragment, price in table.items():
            if i >= len(fragment) and label[i - len(fragment):i] == fragment:
                dp[i] = min(dp[i], dp[i - len(fragment)] + price)

    if dp[n] == float('inf'):
        raise ValueError("Label cannot be spelled by any sequence of dies.")

    return dp[n]
