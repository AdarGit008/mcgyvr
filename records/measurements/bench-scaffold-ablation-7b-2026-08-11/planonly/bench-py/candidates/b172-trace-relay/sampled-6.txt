def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary")
    if start not in links:
        raise ValueError(f"start post '{start}' is not in links")

    visited = set()
    path = []
    current = start

    while True:
        if current in visited:
            raise ValueError(f"watch comes back to post already visited: {current}")
        visited.add(current)
        path.append(current)

        next_post = links[current]
        if next_post == "":
            break
        if next_post not in links:
            raise ValueError(f"handoff '{next_post}' naming a post that links does not name")
        current = next_post

    return path
