def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict) or not all(isinstance(k, str) and isinstance(v, str) for k, v in links.items()):
        raise ValueError("links must be a dictionary mapping post names to post names")
    if start not in links:
        raise ValueError(f"start post '{start}' is not named in links")

    visited = set()
    path = []

    current_post = start
    while current_post and current_post not in visited:
        visited.add(current_post)
        path.append(current_post)
        current_post = links.get(current_post)

    if current_post in visited:
        raise ValueError("watch comes round again to a post already visited")

    return path
