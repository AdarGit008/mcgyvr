def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a mapping")
    if start not in links:
        raise ValueError(f"start post '{start}' is not a valid post name")
    
    path = []
    visited = set()
    current = start
    
    while True:
        if current in visited:
            raise ValueError("watch comes back to a post already visited")
        if current == "":
            break
        path.append(current)
        visited.add(current)
        current = links[current]
    
    return path
