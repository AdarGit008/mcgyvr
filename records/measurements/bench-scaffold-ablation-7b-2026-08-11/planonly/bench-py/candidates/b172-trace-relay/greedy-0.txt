def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary")
    if start not in links:
        raise ValueError(f"start post '{start}' is not named in links")

    visited = set()
    current_post = start
    path = []

    while True:
        if current_post in visited:
            raise ValueError(f"watch comes back to already visited post: {current_post}")
        visited.add(current_post)
        path.append(current_post)

        next_post = links[current_post]
        if not next_post:
            break

        if next_post not in links:
            raise ValueError(f"handoff naming a post that links does not name: {next_post}")

        current_post = next_post

    return path
