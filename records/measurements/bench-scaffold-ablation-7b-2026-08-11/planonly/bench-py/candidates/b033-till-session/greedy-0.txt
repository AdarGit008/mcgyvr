def run_till_session(events: list, prices: dict) -> dict:
    if not all(isinstance(price, int) and price > 0 for price in prices.values()):
        raise ValueError("All prices must be positive integers")

    state = {
        "items": [],
        "total": 0,
        "paid": 0,
        "change": 0,
        "status": "open",
    }

    cart = {}

    for event in events:
        if not isinstance(event, list) or len(event) != 2:
            raise ValueError("Each event must be a list of two elements")

        action, payload = event

        if state["status"] == "paid" or state["status"] == "cancelled":
            raise ValueError("No further events allowed after the session is paid or cancelled")

        if action == "scan":
            if payload not in prices:
                raise ValueError(f"Item '{payload}' not found in price list")
            cart[payload] = cart.get(payload, 0) + 1
        elif action == "void":
            if payload not in cart or cart[payload] == 0:
                raise ValueError(f"Cannot void item '{payload}' that is not in the cart or already at zero")
            cart[payload] -= 1
            if cart[payload] == 0:
                del cart[payload]
        elif action == "close":
            if not cart:
                raise ValueError("Cannot close an empty cart")
            state["items"] = sorted([[item, quantity] for item, quantity in cart.items()])
            state["total"] = sum(prices[item] * quantity for item, quantity in cart.items())
            state["status"] = "payment"
        elif action == "pay":
            if state["status"] != "payment":
                raise ValueError("Cannot pay during a non-payment session")
            if not isinstance(payload, int) or payload <= 0:
                raise ValueError("Pay amount must be a positive integer")
            state["paid"] += payload
            state["change"] = max(state["paid"] - state["total"], 0)
            state["status"] = "paid"
        elif action == "cancel":
            state["status"] = "cancelled"
        else:
            raise ValueError(f"Unknown action: {action}")

    return state
