def apply_charges(caps: dict, charges: list) -> dict:
    refused = []
    left = {bucket: allowance for bucket, (allowance, _) in caps.items()}

    def _why_refused(charge_id: int, bucket: str, amount: int) -> str:
        if amount <= 0 or not isinstance(amount, int):
            raise ValueError("Amount must be a positive whole number")
        if bucket not in left:
            return "unknown"
        if amount > caps[bucket][1]:
            return "single"
        if amount > left[bucket]:
            return "cap"
        return None

    for charge_id, bucket, amount in charges:
        reason = _why_refused(charge_id, bucket, amount)
        if reason is not None:
            refused.append((charge_id, reason))
        else:
            left[bucket] -= amount

    return {"left": left, "refused": refused}
