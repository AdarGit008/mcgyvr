def parse_amount(text: str) -> int:
    """Turns a decimal amount string into integer cents."""
    if not isinstance(text, str):
        raise ValueError("Input must be a string")
    parts = text.split('.')
    if len(parts) > 2 or (len(parts) == 2 and parts[1].strip() != '00'):
        raise ValueError("Invalid format for amount")
    integer_part = int(parts[0])
    fraction_part = int(parts[1]) if '.' in text else 0
    return integer_part * 100 + fraction_part

def format_amount(cents: int) -> str:
    """Renders non-negative integer cents back into dotted form."""
    if not isinstance(cents, int) or cents < 0:
        raise ValueError("Input must be a non-negative integer")
    return f"{cents // 100}.{cents % 100:02d}"

def allocate_cents(total_cents: int, weights: list) -> list:
    """Splits total_cents across weights proportionally."""
    if not isinstance(weights, list) or not all(isinstance(w, int) and w >= 0 for w in weights):
        raise ValueError("Weights must be a non-empty list of non-negative integers")
    if sum(weights) == 0:
        raise ValueError("Sum of weights must be greater than zero")
    if not isinstance(total_cents, int) or total_cents < 0:
        raise ValueError("Total cents must be a non-negative integer")

    total_weight = sum(weights)
    base_share = total_cents // total_weight
    remainder = total_cents % total_weight
    shares = [base_share] * len(weights)

    for i in range(remainder):
        shares[i] += 1

    return shares
