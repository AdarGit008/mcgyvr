def carve_shift(start: int, end: int, span: int, rest: int, least: int) -> list:
    if not all(isinstance(x, int) for x in [start, end, span, rest, least]):
        raise ValueError("All arguments must be whole numbers")
    if start >= end:
        raise ValueError("Start must precede end")
    if span < 1 or rest < 1:
        raise ValueError("Span and rest must be at least 1")
    if least < 1 or least > span:
        raise ValueError("Least must be at least 1 and at most span")

    stretches = []
    current_start = start

    while current_start + span < end:
        stretch_end = min(current_start + span, end)
        if stretch_end - current_start >= least:
            stretches.append([current_start, stretch_end])
            current_start += span + rest
        else:
            break

    last_stretch_end = min(end, current_start + span)
    if last_stretch_end - current_start >= least:
        stretches.append([current_start, last_stretch_end])

    return stretches
