def lay_conveyor(floor: list, row: int, col: int, length: int) -> list:
    if row < 0 or row >= len(floor):
        raise ValueError("Row is out of bounds")
    if col < 0 or col + length > len(floor[row]):
        raise ValueError("Conveyor run would leave the plan")
    for c in range(length):
        if floor[row][col + c] != ".":
            raise ValueError("Cell is not open")
    new_floor = [row[:] for row in floor]
    new_floor[row][col:col + length] = ["="] * length
    return new_floor
