def file_paths(root: dict) -> list:
    result = []
    
    def walk(node, path):
        if not isinstance(node, dict) or 'name' not in node or not isinstance(node['name'], str) or '/' in node['name']:
            raise ValueError("Invalid node")
        current_path = f"{path}/{node['name']}" if path else node['name']
        result.append(current_path)
        if 'bytes' in node:
            return
        elif 'children' in node and isinstance(node['children'], list):
            for child in node['children']:
                walk(child, current_path)
        else:
            raise ValueError("Invalid node")
    
    walk(root, "")
    return result

def tree_bytes(root: dict) -> int:
    if not isinstance(root, dict) or 'name' not in root or not isinstance(root['name'], str) or '/' in root['name']:
        raise ValueError("Invalid node")
    if 'bytes' in root and 'children' in root:
        raise ValueError("Invalid node")
    if 'bytes' in root:
        return root['bytes']
    elif 'children' in root and isinstance(root['children'], list):
        return sum(tree_bytes(child) for child in root['children'])
    else:
        return 0

def heavy_folders(root: dict, min_bytes: int) -> list:
    if not isinstance(min_bytes, int) or min_bytes < 0:
        raise ValueError("Invalid min_bytes")
    
    result = []
    
    def walk(node, path, total):
        nonlocal min_bytes
        current_path = f"{path}/{node['name']}" if path else node['name']
        total += tree_bytes(node)
        if 'children' in node and isinstance(node['children'], list):
            for child in node['children']:
                result.extend(walk(child, current_path, total))
        elif total >= min_bytes:
            result.append([current_path, total])
        return result
    
    walk(root, "", 0)
    
    # Ensure the root is included if it's a folder
    if 'children' in root and isinstance(root['children'], list):
        result.extend(walk(root, "", 0))
    
    return result
