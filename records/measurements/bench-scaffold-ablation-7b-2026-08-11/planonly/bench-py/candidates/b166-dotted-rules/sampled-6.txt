def match_setting(rules: dict, name: str) -> str | None:
    if not isinstance(name, str) or not name:
        raise ValueError("Name must be a non-empty string")
    
    for selector, value in rules.items():
        if not isinstance(value, str):
            raise ValueError("Rule values must be strings")
        
        if selector == "*":
            continue
        
        if selector.endswith(".*"):
            prefix = selector[:-2]
            if name.startswith(prefix):
                return value
        
        if selector == name:
            return value
    
    for selector, _ in rules.items():
        if selector.endswith(".*") and selector != "*":
            prefix = selector[:-2]
            if name.startswith(prefix) and len(selector) > 1:
                return rules[selector]
    
    return None
