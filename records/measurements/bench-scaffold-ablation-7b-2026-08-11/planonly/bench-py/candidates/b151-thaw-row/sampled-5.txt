def thaw_row(row: str, steps: int) -> str:
    if not isinstance(row, str) or not row:
        raise ValueError("Row must be a non-empty string.")
    if not all(cell in "#." for cell in row):
        raise ValueError("Row can only contain '#' and '.' characters.")
    if not isinstance(steps, int) or steps < 0:
        raise ValueError("Steps must be a non-negative integer.")

    cells = list(row)

    for _ in range(steps):
        new_cells = []
        for i, cell in enumerate(cells):
            if cell == ".":
                new_cells.append(".")
            elif (i == 0 or cells[i - 1] == ".") and (i == len(cells) - 1 or cells[i + 1] == "."):
                new_cells.append(".")
            else:
                new_cells.append("#")
        cells = new_cells

    return "".join(cells)
