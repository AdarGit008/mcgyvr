 ```ts
export function matchSetting(rules: Record<string, string>, name: string): string | null {
  if (typeof name !== 'string' || name === '') {
    throw new Error('Invalid name');
  }

  let bestMatch: [string, string] | null = null;

  for (const selector in rules) {
    const value = rules[selector];
    if (typeof value !== 'string') {
      throw new Error('Invalid rule value');
    }

    if (selector === name) {
      return value;
    } else if (selector.endsWith('.*')) {
      const baseSelector = selector.slice(0, -2);
      if (name.startsWith(baseSelector)) {
        if (!bestMatch || baseSelector.length > bestMatch[0].length) {
          bestMatch = [baseSelector, value];
        }
      }
    } else if (selector === '*') {
      bestMatch = ['*', value];
    }
  }

  return bestMatch ? bestMatch[1] : null;
}
```
