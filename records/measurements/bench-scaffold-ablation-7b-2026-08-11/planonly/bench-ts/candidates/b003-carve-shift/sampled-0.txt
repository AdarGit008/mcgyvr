export function carveShift(start: number, end: number, span: number, rest: number, least: number): number[][] {
  if (!Number.isInteger(start) || !Number.isInteger(end) || !Number.isInteger(span) || !Number.isInteger(rest) || !Number.isInteger(least)) {
    throw new Error("All arguments must be whole numbers.");
  }
  if (start >= end || span < 1 || rest < 1 || least < 1 || least > span) {
    throw new Error("Invalid shift bounds or stretch parameters.");
  }

  const stretches: number[][] = [];
  let currentStart = start;

  while (currentStart + span < end) {
    const to = Math.min(currentStart + span, end);
    if (to - currentStart >= least) {
      stretches.push([currentStart, to]);
      currentStart = to + rest;
    } else {
      break;
    }
  }

  return stretches;
}
