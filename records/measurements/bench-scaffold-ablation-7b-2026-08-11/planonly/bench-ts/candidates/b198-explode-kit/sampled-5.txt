export function explodeKit(catalog: Record<string, { makes: number; parts: [string, number][] }>, kit: string, want: number): Record<string, number> {
  if (want <= 0) throw new Error("Want must be a positive integer");

  const result: Record<string, number> = {};

  function explode(kitName: string, demand: number): void {
    const recipe = catalog[kitName];
    if (!recipe) throw new Error(`Kit ${kitName} not found in the catalog`);

    const runs = Math.ceil(demand / recipe.makes);
    for (const [part, count] of recipe.parts) {
      result[part] = (result[part] || 0) + count * runs;
    }

    for (const [component, componentCount] of recipe.parts) {
      if (catalog[component]) {
        explode(component, componentCount * runs);
      }
    }
  }

  explode(kit, want);
  return result;
}
