/**
 * Render value as unsigned binary in exactly width bits: most significant bit first,
 * left-padded with zeros to the full width, and split into groups of four bits joined by single spaces.
 */
export function formatBits(value: number, width: number): string {
  if (!Number.isInteger(value) || value < 0) {
    throw new Error("Value must be a non-negative integer");
  }
  if (width % 4 !== 0 || width > 32) {
    throw new Error("Width must be a positive multiple of four that does not exceed 32");
  }
  if (value >= 1 << width) {
    throw new Error("Value does not fit in the specified width");
  }

  const binary = value.toString(2).padStart(width, '0');
  const nibbles = [];
  for (let i = 0; i < width; i += 4) {
    nibbles.push(binary.slice(i, i + 4));
  }
  return nibbles.join(' ');
}
