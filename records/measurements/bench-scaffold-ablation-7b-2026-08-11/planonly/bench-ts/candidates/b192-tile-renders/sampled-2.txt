/**
 * @param requests - A list of [tick, tileName] pairs representing requests to render tiles at specific ticks.
 * @param freshFor - The number of ticks a rendered tile remains fresh for.
 * @param size - The maximum number of tiles the cache can hold.
 * @returns An ordered list of [tick, tileName] pairs representing the renders that occurred.
 */
export function tileRenders(requests: [number, string][], freshFor: number, size: number): [number, string][] {
  if (freshFor <= 0 || !Number.isInteger(freshFor)) {
    throw new Error("Invalid freshFor value. It must be a positive whole number.");
  }
  if (!Number.isInteger(size) || size <= 0) {
    throw new Error("Invalid cache size. It must be a positive whole number.");
  }

  const renders: [number, string][] = [];
  const tileCache: { [tileName: string]: { tick: number } } = {};

  for (const request of requests) {
    const [tick, tileName] = request;
    if (tileCache[tileName] && tick >= tileCache[tileName].tick + freshFor) {
      renders.push([tick, tileName]);
      tileCache[tileName].tick = tick;
    } else if (!tileCache[tileName]) {
      while (Object.keys(tileCache).length === size) {
        const oldestTile = Object.entries(tileCache).reduce((a, b) => a[1].tick < b[1].tick ? a : b)[0];
        delete tileCache[oldestTile];
      }
      tileCache[tileName] = { tick };
      renders.push([tick, tileName]);
    }
  }

  return renders;
}
