export function applyOverrides(base: Record<string, string>, overrides: string[]): Record<string, string> {
  if (!Array.isArray(overrides)) {
    throw new Error('overrides must be an array');
  }

  const merged = { ...base };

  for (const override of overrides) {
    const equalsIndex = override.indexOf('=');
    if (equalsIndex === -1 || override[0] === '=') {
      throw new Error(`invalid override: ${override}`);
    }

    const name = override.substring(0, equalsIndex);
    const value = override.substring(equalsIndex + 1);

    if (!name) {
      throw new Error('empty setting name not allowed');
    }

    if (!(name in base)) {
      throw new Error(`unknown setting name: ${name}`);
    }

    merged[name] = value;
  }

  return merged;
}
