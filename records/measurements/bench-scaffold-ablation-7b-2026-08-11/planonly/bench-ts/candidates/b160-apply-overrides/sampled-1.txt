export function applyOverrides(base: Record<string, string>, overrides: string[]): Record<string, string> {
  if (!Array.isArray(overrides)) {
    throw new Error('overrides must be an array');
  }

  const merged = { ...base };

  for (const override of overrides) {
    const [name, value] = override.split(/=([^]*)/, 2);
    if (typeof name !== 'string' || name.trim() === '' || typeof value !== 'string') {
      throw new Error('invalid override format');
    }
    if (!merged.hasOwnProperty(name)) {
      throw new Error(`override for unknown setting: ${name}`);
    }
    merged[name] = value;
  }

  return merged;
}
