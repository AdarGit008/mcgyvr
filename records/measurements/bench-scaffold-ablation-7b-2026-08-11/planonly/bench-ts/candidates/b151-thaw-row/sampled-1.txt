export function thawRow(row: string, steps: number): string {
  if (typeof row !== 'string' || row === '' || !/^[.#]+$/.test(row)) {
    throw new Error('Invalid row format');
  }
  if (!Number.isInteger(steps) || steps < 0) {
    throw new Error('Invalid step count');
  }

  const melt = (r: string): string => {
    let result = r;
    for (let i = 0; i < result.length; i++) {
      if ((i === 0 || result[i - 1] === '.') && result[i] === '#') {
        result = result.slice(0, i) + '.' + result.slice(i + 1);
      } else if ((i === result.length - 1 || result[i + 1] === '.') && result[i] === '#') {
        result = result.slice(0, i) + '.' + result.slice(i + 1);
      }
    }
    return result;
  };

  for (let _ of Array(steps).fill(null)) {
    row = melt(row);
  }

  return row;
}
