export function bumpRelease(version: string, part: string): string {
  if (typeof version !== "string" || version.trim() === "") {
    throw new Error("version must be a non-empty string");
  }
  if (typeof part !== "string" || part.trim() === "") {
    throw new Error("part must be a non-empty string");
  }

  const components: string[] = version.split(".");
  if (components.length !== 3) {
    throw new Error("version must have exactly three dot-separated components");
  }

  for (const comp of components) {
    if (comp.length === 0) {
      throw new Error("version component cannot be empty");
    }
    if (comp[0] === "0" && comp.length > 1) {
      throw new Error("version component cannot have a leading zero");
    }
    if (!/^[0-9]+$/.test(comp)) {
      throw new Error("version component must contain only digits");
    }
  }

  const major = parseInt(components[0], 10);
  const minor = parseInt(components[1], 10);
  const patch = parseInt(components[2], 10);

  if (part === "major") {
    return `${major + 1}.0.0`;
  }
  if (part === "minor") {
    return `${major}.${minor + 1}.0`;
  }
  if (part === "patch") {
    return `${major}.${minor}.${patch + 1}`;
  }
  throw new Error("part must be one of: major, minor, patch");
}
