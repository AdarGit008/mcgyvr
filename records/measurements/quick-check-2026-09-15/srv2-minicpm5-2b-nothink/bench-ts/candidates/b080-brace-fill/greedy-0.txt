export function fillTemplate(template: string, values: Record<string, string>): string {
  if (typeof template !== "string") {
    throw new Error("fillTemplate expects a string template");
  }

  const result: string[] = [];
  let i = 0;
  let braceDepth = 0;

  while (i < template.length) {
    const char = template[i];

    if (char === "{") {
      braceDepth++;
      i++;
      if (braceDepth > 0) {
        const nameStart = i;
        while (i < template.length && template[i] !== "}") {
          i++;
        }
        const nameEnd = i;
        const name = template.slice(nameStart, nameEnd);

        if (name.length === 0) {
          throw new Error("placeholder name is empty");
        }

        const nameRegex = /^[a-zA-Z0-9_]+$/;
        if (!nameRegex.test(name)) {
          throw new Error(`placeholder name contains invalid characters: ${name}`);
        }

        if (i >= template.length) {
          throw new Error("unterminated placeholder");
        }

        const value = values[name];
        if (value === undefined) {
          throw new Error(`placeholder name not found in mapping: ${name}`);
        }

        result.push(value);
        i++;
        braceDepth--;
      } else {
        result.push(char);
      }
    } else if (char === "}") {
      braceDepth--;
      if (braceDepth < 0) {
        throw new Error("unterminated placeholder");
      }
      result.push(char);
      i++;
    } else {
      result.push(char);
      i++;
    }
  }

  if (braceDepth !== 0) {
    throw new Error("unterminated placeholder");
  }

  return result.join("");
}
