def fill_template(template: str, values: dict) -> str:
    """Replace placeholders in a template string with values from a mapping."""
    if not isinstance(template, str):
        raise ValueError("fill_template expects a string template")
    if not isinstance(values, dict):
        raise ValueError("fill_template expects a dict of values")

    result: list[str] = []
    i = 0
    n = len(template)

    while i < n:
        if template[i] == "{":
            j = i + 1
            while j < n and template[j] != "}":
                j += 1
            if j >= n:
                raise ValueError("unterminated placeholder")
            name = template[i + 1:j]
            if not name:
                raise ValueError("empty placeholder name")
            if not all(c.isalnum() or c == "_" for c in name):
                raise ValueError(f"placeholder name '{name}' contains invalid characters")
            if name not in values:
                raise ValueError(f"placeholder name '{name}' not found in values")
            result.append(values[name])
            i = j + 1
        else:
            result.append(template[i])
            i += 1

    return "".join(result)
