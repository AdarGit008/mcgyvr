"""One point-of-sale till session, replayed from its events."""

OPEN = "open"
PAYMENT = "payment"
PAID = "paid"
CANCELLED = "cancelled"


def run_till_session(events: list, prices: dict) -> dict:
    """Replay a point-of-sale till session and return its final state."""
    if not prices:
        raise ValueError("price list must not be empty")

    cart: dict[str, int] = {}
    total: int = 0
    paid: int = 0
    state: str = OPEN

    for event in events:
        if not isinstance(event, list) or len(event) < 1:
            raise ValueError("event must be a non-empty list")
        action, payload = event[0], event[1:]

        if action == "scan":
            if state not in (OPEN, PAYMENT):
                raise ValueError("scan not allowed in non-open session")
            if not isinstance(payload, str):
                raise ValueError("scan payload must be a string")
            if payload not in prices:
                raise ValueError("scan of unknown item")
            if state == OPEN:
                cart[payload] = cart.get(payload, 0) + 1
                total += prices[payload]

        elif action == "void":
            if state not in (OPEN, PAYMENT):
                raise ValueError("void not allowed in non-open session")
            if not isinstance(payload, str):
                raise ValueError("void payload must be a string")
            if payload not in cart or cart[payload] <= 0:
                raise ValueError("void of unknown or zero quantity item")
            cart[payload] -= 1
            if cart[payload] == 0:
                del cart[payload]
            total -= prices[payload]

        elif action == "close":
            if state not in (OPEN, PAYMENT):
                raise ValueError("close not allowed in non-open session")
            if not cart:
                raise ValueError("close with empty cart")
            if state == OPEN:
                state = PAYMENT
            if state == PAYMENT:
                if total <= 0:
                    raise ValueError("cannot close with zero total")
                paid = total
                total = 0
                state = PAID

        elif action == "pay":
            if state not in (PAYMENT, PAID):
                raise ValueError("pay not allowed in non-payment session")
            if not isinstance(payload, int) or payload <= 0:
                raise ValueError("pay amount must be a positive integer")
            paid += payload

        elif action == "cancel":
            if state not in (OPEN, PAYMENT, PAID, CANCELLED):
                raise ValueError("cancel not allowed in non-open/payment/paid session")
            state = CANCELLED

        else:
            raise ValueError(f"unknown action: {action}")

        if state in (PAID, CANCELLED):
            raise ValueError("event after session is paid or cancelled")

    return {
        "items": sorted(cart.items()),
        "total": total,
        "paid": paid,
        "change": paid - total,
    }
