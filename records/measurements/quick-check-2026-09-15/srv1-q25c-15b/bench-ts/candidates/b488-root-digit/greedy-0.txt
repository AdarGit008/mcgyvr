export function rootDigit(value: number): number {
  if (value < 1) {
    throw new Error("Input must be a positive integer.");
  }

  let left = value;
  while (left >= 10) {
    let sum = 0;
    while (left > 0) {
      sum += left % 10;
      left = Math.floor(left / 10);
    }
    left = sum;
  }
  return left;
}
