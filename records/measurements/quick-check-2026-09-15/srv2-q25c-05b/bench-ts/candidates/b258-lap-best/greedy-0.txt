/**
 * The quickest completed lap of a session. 
 * @param laps - An array of lap times.
 * @returns The quickest completed lap of a session, or zero if no completed lap was recorded.
 */
export function lapBest(laps: number[]): number {
  if (laps.length === 0) {
    return 0;
  }
  let best = laps[0];
  for (const lap of laps) {
    if (lap < best) {
      best = lap;
    }
  }
  return best;
}
