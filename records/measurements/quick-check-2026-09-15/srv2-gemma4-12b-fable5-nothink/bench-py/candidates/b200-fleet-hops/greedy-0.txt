def fleet_hops(count: int, lts: list, builds: list, hop: int) -> int:
    """Calculate total hops for all devices to reach the newest release."""
    if hop <= 0:
        raise ValueError("hop must be a positive integer")

    lts_sorted = sorted(lts)
    newest = count - 1
    total_hops = 0

    for build in builds:
        if not isinstance(build, int) or not (0 <= build < count):
            raise ValueError("invalid build version")

        current = build
        while current < newest:
            target = min(current + hop, newest)
            if target > current + 1:
                # Find the first LTS strictly above current
                idx = next((i for i, v in enumerate(lts_sorted) if v > current), None)
                if idx is not None and lts_sorted[idx] < target:
                    current = lts_sorted[idx]
                else:
                    current = target
            else:
                current += 1
            total_hops += 1

    return total_hops
