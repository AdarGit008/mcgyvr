export function vatBack(gross: number, rate: number): number {
  if (typeof gross !== "number" || typeof rate !== "number") {
    throw new Error("Both arguments must be numbers");
  }

  if (!isFinite(gross) || !isFinite(rate)) {
    throw new Error("Both arguments must be finite numbers");
  }

  return Math.floor(gross / (1 + rate / 100));
}
