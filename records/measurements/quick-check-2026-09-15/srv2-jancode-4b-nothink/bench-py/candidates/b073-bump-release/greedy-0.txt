def bump_release(version: str, part: str) -> str:
    if not isinstance(version, str):
        raise ValueError("Version must be a string")
    
    if not version or len(version.split('.')) != 3:
        raise ValueError("Version must have exactly three components separated by dots")
    
    components = version.split('.')
    for component in components:
        if not component or not component.isdigit() or len(component) > 1 and component[0] == '0':
            raise ValueError("Each component must be a non-empty digit string without leading zeros")
    
    part = part.lower()
    if part not in ['major', 'minor', 'patch']:
        raise ValueError("Part must be 'major', 'minor', or 'patch'")
    
    major, minor, patch = map(int, components)
    
    if part == 'major':
        return f"{major + 1}.0.0"
    elif part == 'minor':
        return f"{major}.{minor + 1}.0"
    elif part == 'patch':
        return f"{major}.{minor}.{patch + 1}"
    else:
        raise ValueError("Invalid part name")
