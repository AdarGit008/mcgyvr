/**
 * Scores written out, with a star against those that reach a floor.
 * Entries that do not qualify come back unchanged.
 * The marker is placed against every entry that reaches a floor, written as the entry followed by a star.
 * Entries below the floor are marked instead.
 */
export function markList(scores: number[], floor: number): string[] {
  const out: string[] = [];
  for (const score of scores) {
    if (score < floor) {
      out.push(String(score) + "*");
    } else {
      out.push(String(score));
    }
  }
  return out;
}
