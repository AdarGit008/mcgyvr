def root_digit(value: int) -> int:
    """Fold a count down to a single figure by repeatedly summing its digits."""
    left = value
    if left < 0:
        left = -left
    if left < 10:
        return left
    while left >= 10:
        total = 0
        while left > 0:
            total += left % 10
            left = left // 10
        left = total
    return left
