def bump_release(version: str, part: str) -> str:
    if not isinstance(version, str):
        raise ValueError("Version must be a string")
    if len(version.split('.')) != 3:
        raise ValueError("Version must have exactly three components")
    if any(not c.isdigit() for c in version):
        raise ValueError("Version components must be digits")
    if any(c == '0' for c in version):
        raise ValueError("Version components must not start with zero")
    if part not in ['major', 'minor', 'patch']:
        raise ValueError("Part must be one of 'major', 'minor', or 'patch'")
    
    parts = version.split('.')
    if part == 'major':
        parts[1] = '0'
        parts[2] = '0'
    elif part == 'minor':
        parts[2] = '0'
    elif part == 'patch':
        pass
    
    return '.'.join(parts)
