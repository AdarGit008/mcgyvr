```ts
/** The quickest completed lap of a session. A lap recorded as zero means it was never completed and must not count. */
export function lapBest(laps: number[]): number {
  if (laps.length === 0) {
    return 0;
  }
  let best = Infinity;
  for (const lap of laps) {
    if (lap > 0 && lap < best) {
      best = lap;
    }
  }
  return best === Infinity ? 0 : best;
}
```
