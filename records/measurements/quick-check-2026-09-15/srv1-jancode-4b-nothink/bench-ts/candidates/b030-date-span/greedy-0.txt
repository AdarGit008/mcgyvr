/** How many days lie between two proleptic Gregorian calendar dates. */

const MONTH_LENGTHS = [31, 28, 31, 3-0, 31, 30, 31, 31, 30, 31, 30, 31];

function isLeap(year: number): boolean {
  return year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
}

function monthLength(year: number, month: number): number {
  if (month === 2 && isLeap(year)) {
    return 29;
  }
  return MONTH_LENGTHS[month - 1];
}

function toOrdinal(date: number[]): number {
  const [year, month, day] = date;
  if (year === undefined || month === undefined || day === undefined) {
    throw new Error("Date components must be provided");
  }
  if (!Number.isInteger(year) || !Number.isInteger(month) || !Number.isInteger(day)) {
    throw new Error("Date components must be integers");
  }
  if (month < 1 || month > 12) {
    throw new Error("Month must be between 1 and 12");
  }
  const prior = year - 1;
  let days = prior * 365;
  days += Math.floor(prior / 4);
  days -= Math.floor(prior / 100);
  days += Math.floor(prior / 400);
  for (let m = 1; m < month; m += 1) {
    days += monthLength(year, m);
  }
  return days + day;
}

export function spanDays(start: number[], end: number[]): number {
  if (start.length !== 3 || end.length !== 3) {
    throw new Error("Date must be an array of [year, month, day]");
  }
  if (start[0] < 1) {
    throw new Error("Year must be positive");
  }
  if (start[1] < 1 || start[1] > 12) {
    throw new Error("Month must be between 1 and 12");
  }
  if (start[2] < 1 || start[2] > monthLength(start[0], start[1])) {
    throw new Error("Day must be valid for the given month and year");
  }
  if (end[0] < 1) {
    throw new Error("Year must be positive");
  }
  if (end[1] < 1 || end[1] > 12) {
    throw new Error("Month must be between 1 and 12");
  }
  if (end[2] < 1 || end[2] > monthLength(end[0], end[1])) {
    throw new Error("Day must be valid for the given month and year");
  }
  if (start[0] > end[0] || (start[0] === end[0] && start[1] > end[1]) || (start[0] === end[0] && start[1] === end[1] && start[2] > end[2])) {
    throw new Error("Start date must not be after end date");
  }
  return toOrdinal(end) - toOrdinal(start);
}
