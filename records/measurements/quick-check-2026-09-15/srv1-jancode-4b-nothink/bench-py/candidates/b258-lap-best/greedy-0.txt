def lap_best(laps: list) -> int:
    """The quickest completed lap of a session. A lap recorded as zero means it was never completed and must not count."""
    if not laps:
        return 0
    best = float('inf')
    for lap in laps:
        if lap > 0 and lap < best:
            best = lap
    return best if best != float('inf') else 0
