def trace_relay(links: dict, start: str) -> list:
    if not isinstance(start, str):
        raise ValueError("start must be a string")
    if start == "":
        raise ValueError("start cannot be an empty string")
    if start not in links:
        raise ValueError("unknown starting station")
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary")
    for key in links:
        if not isinstance(key, str):
            raise ValueError("all station names in links must be strings")
        if not isinstance(links[key], str):
            raise ValueError("link values must be strings")

    visited = set()
    path: list[str] = []
    current = start

    while True:
        if current in visited:
            raise ValueError("relay contains a circular chain")
        visited.add(current)
        path.append(current)
        target = links[current]
        if target == "":
            return path
        if target not in links:
            raise ValueError("broken link: target station not in directory")
        current = target
