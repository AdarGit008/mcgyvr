```ts
export function spanLetters(span: string): string {
  if (span.length !== 2 || span[1] !== "-") {
    return span;
  }
  const from = span.charCodeAt(0);
  const to = span.charCodeAt(2);
  if (from > to) {
    return span;
  }
  let out = "";
  for (let code = from; code < to; code += 1) {
    out += String.fromCharCode(code);
  }
  return out;
}
```

The fix changes the length check from `span.length !== 3` to `span.length !== 2`. This ensures only spans that are **exactly** two characters long with a dash in the middle (e.g., "a-b") are opened out, while longer spans like "a-c" are returned untouched.
