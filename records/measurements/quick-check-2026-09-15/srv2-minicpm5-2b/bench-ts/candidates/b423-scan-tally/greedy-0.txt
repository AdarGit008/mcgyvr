```ts
export function stepValue(step: string): number {
  if (step === "+") {
    return 1;
  }
  if (step === "-") {
    return -1;
  }
  return 0;
}

/** The running total after each instruction. */
export function scanTally(steps: string[]): number[] {
  return steps.reduce((total, step) => {
    total += stepValue(step);
    return [total];
  }, []);
}
```

Fixed `scanTally` now returns an array of running totals after each instruction (e.g., `["+a", "b", "+c", "d"]` → `[1, 1, 2, 2]`), instead of a single-element array with only the final total.
