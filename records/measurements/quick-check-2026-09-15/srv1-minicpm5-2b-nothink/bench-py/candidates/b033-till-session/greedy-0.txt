"""One point-of-sale till session, replayed from its events."""

OPEN = "open"
PAYMENT = "payment"
PAID = "paid"
CANCELLED = "cancelled"


def run_till_session(events: list, prices: dict) -> dict:
    """Replay a point-of-sale till session and return its final state."""
    if not prices:
        raise ValueError("price list must not be empty")

    cart: dict[str, int] = {}
    total: int = 0
    paid: int = 0
    state: str = OPEN
    session: dict[str, object] = {
        "items": [],
        "total": 0,
        "paid": 0,
        "change": 0,
        "state": state,
    }

    for event in events:
        if not isinstance(event, list) or len(event) < 1:
            raise ValueError("event must be a non-empty list")
        action, payload = event[0], event[1:]

        if state == OPEN:
            if action == "scan":
                if not isinstance(payload, str):
                    raise ValueError("scan payload must be a string")
                if payload not in prices:
                    raise ValueError(f"item {payload} not in price list")
                if payload not in cart:
                    raise ValueError(f"item {payload} not in cart")
                cart[payload] = cart.get(payload, 0) + 1
                total += prices[payload]
            elif action == "void":
                if not isinstance(payload, str):
                    raise ValueError("void payload must be a string")
                if payload not in cart:
                    raise ValueError(f"item {payload} not in cart")
                cart[payload] = max(0, cart[payload] - 1)
                total -= prices[payload]
            elif action == "close":
                if not cart:
                    raise ValueError("cart must not be empty at close")
                state = PAYMENT
            elif action == "pay":
                if not isinstance(payload, int) or payload <= 0:
                    raise ValueError("pay amount must be a positive integer")
                paid += payload
                if paid >= total:
                    state = PAID
                    change = paid - total
                    total = 0
            elif action == "cancel":
                state = CANCELLED
            else:
                raise ValueError(f"unknown action: {action}")

        elif state == PAYMENT:
            if action == "pay":
                if not isinstance(payload, int) or payload <= 0:
                    raise ValueError("pay amount must be a positive integer")
                paid += payload
                if paid >= total:
                    state = PAID
                    change = paid - total
                    total = 0
            elif action == "cancel":
                state = CANCELLED
            else:
                raise ValueError(f"event after payment/cancel is not allowed: {action}")

        else:
            raise ValueError(f"event after {state} is not allowed: {action}")

    session["items"] = sorted(cart.items(), key=lambda x: x[0])
    return {
        "items": session["items"],
        "total": session["total"],
        "paid": session["paid"],
        "change": session["change"],
    }
