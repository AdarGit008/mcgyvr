def cloud_code(table: dict, code: str) -> str:
    """
    Convert a short weather code into words using the provided table.

    Args:
        table (dict): A dictionary mapping codes to descriptions. Codes may arrive in any case and are matched in upper case.
        code (str): The weather code to convert.

    Returns:
        str: The description of the weather code, or 'unknown' if the code is not found in the table.
    """
    # Convert the input code to uppercase for consistent matching
    code = code.upper()
    
    # Check if the code exists in the table
    if code in table:
        return table[code]
    else:
        return 'unknown'
