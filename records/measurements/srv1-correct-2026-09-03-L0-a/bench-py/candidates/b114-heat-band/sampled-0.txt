def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if isinstance(start, bool) or not isinstance(start, int):
        raise ValueError("start must be an integer")
    if isinstance(low, bool) or not isinstance(low, int):
        raise ValueError("low must be an integer")
    if isinstance(high, bool) or not isinstance(high, int):
        raise ValueError("high must be an integer")
    if low >= high:
        raise ValueError("low must lie strictly below high")
    if isinstance(power, bool) or not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")
    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("drifts must be integers")
    
    temp = start
    heated = 0
    switches = 0
    coldest = start
    
    for drift in drifts:
        if temp < low:
            temp += power
            heated += 1
            switches += 1
        elif temp > high:
            temp -= power
            heated += 1
            switches += 1
        else:
            temp += drift
            heated += 1
            if temp < coldest:
                coldest = temp
    
    return {"temp": temp, "heated": heated, "switches": switches, "coldest": coldest}
