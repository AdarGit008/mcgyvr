def drain_frames(chunks: list, marker: str) -> dict:
    if not isinstance(marker, str) or len(marker) != 1:
        raise ValueError("marker must be a single character")
    buffer = "".join(chunks)
    pending = ""
    frames = []
    i = 0
    while i < len(buffer):
        if buffer[i] == marker:
            if i + 1 < len(buffer) and buffer[i + 1] == marker:
                buffer = buffer[:i] + buffer[i + 2:]
                pending += marker
                continue
            else:
                if pending:
                    frames.append(pending)
                    pending = ""
                frames.append(buffer[i])
                i += 1
        else:
            pending += buffer[i]
            i += 1
    if pending:
        frames.append(pending)
    return {"frames": frames, "pending": ""}
