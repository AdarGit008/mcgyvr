def match_at(grid: list, row: int, column: int, value: int) -> bool:
    if not grid or row < 0 or column < 0 or row >= len(grid) or column >= len(grid[0]):
        raise ValueError("Invalid grid or indices")
    return grid[row][column] == value

def grid_find(grid: list, value: int) -> list:
    for row in range(len(grid)):
        for col in range(len(grid[0])):
            if grid[row][col] == value:
                return [row, col]
    raise ValueError("Value not found in the grid")
