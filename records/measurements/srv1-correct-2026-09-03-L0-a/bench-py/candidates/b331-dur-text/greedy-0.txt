def dur_hours(minutes: int) -> int:
    if minutes < 0:
        raise ValueError("Minutes cannot be negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    hours = minutes // 60
    minutes %= 60
    return f"{hours}h{minutes:02d}m" if hours > 0 else f"{minutes:02d}m"
