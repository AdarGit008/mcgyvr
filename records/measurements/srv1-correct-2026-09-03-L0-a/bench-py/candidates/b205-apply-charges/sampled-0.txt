def _why_refused(cap, left: int, amount: int) -> str:
    if cap is None:
        return "unknown"
    if amount > cap[1]:
        return "single"
    return "cap" if amount > left else ""


def apply_charges(caps: dict, charges: list) -> dict:
    left = {name: cap[0] for name, cap in caps.items()}
    refused = []
    # TODO: put every charge through _why_refused before it is spent, and
    # send the ones that come back with a reason to the refused list
    # instead of taking their amount off the bucket. This placeholder
    # spends every charge blindly.
    for _, bucket, amount in charges:
        if amount <= 0:
            raise ValueError("Amount must be a positive whole number")
        if amount > left[bucket]:
            refused.append((amount, _why_refused(caps[bucket], left[bucket], amount)))
            left[bucket] = 0
        else:
            left[bucket] -= amount
    return {"left": left, "refused": refused}
