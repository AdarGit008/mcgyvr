from typing import List

def step_cost(kind: str) -> int:
    """Returns the cost of a step of the given kind."""
    if kind == "foot":
        return 1
    elif kind == "bike":
        return 2
    elif kind == "car":
        return 3
    else:
        raise ValueError("Unknown step kind")

def path_toll(steps: List[str]) -> int:
    """Takes a list of step kinds and returns the total cost of the path.
    Every third step is free, counting from one."""
    total_cost = 0
    for i, step in enumerate(steps):
        if i % 3 == 2:
            continue
        total_cost += step_cost(step)
    return total_cost
