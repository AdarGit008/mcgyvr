from typing import List

def climb_budget(tolls: List[str]) -> int:
    if not tolls:
        return 0
    
    total_cost = 0
    for i, toll in enumerate(tolls):
        if not toll.isdigit():
            raise ValueError(f"Invalid toll at position {i}: {toll}")
        total_cost += int(toll)
    
    return total_cost
