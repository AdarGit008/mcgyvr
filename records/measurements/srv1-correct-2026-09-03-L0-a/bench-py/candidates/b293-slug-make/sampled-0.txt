```python
def slug_word(word: str) -> str:
    """Lowercase a word, dropping non-alphanumeric characters."""
    return ''.join(c for c in word if c.isalnum())

def slug_make(phrase: str) -> str:
    """Join slugged words of a phrase with hyphens, excluding empty words."""
    slugs = [slug_word(word) for word in phrase.split() if slug_word(word)]
    return '-'.join(slugs)
```
