```python
def flip_one(pair: list) -> list:
    """
    Returns a pair with its two members the other way round.
    
    Args:
    pair (list): A list containing exactly two elements.
    
    Returns:
    list: A new list with the elements of the input pair swapped.
    """
    if len(pair) != 2:
        raise ValueError("Input must be a list with exactly two elements.")
    return [pair[1], pair[0]]

def flip_all(pairs: list) -> list:
    """
    Returns a new list with every pair in the input list flipped.
    
    Args:
    pairs (list): A list of lists, where each inner list contains exactly two elements.
    
    Returns:
    list: A new list with the elements of each pair in the input list swapped.
    """
    return [flip_one(pair) for pair in pairs]
```
