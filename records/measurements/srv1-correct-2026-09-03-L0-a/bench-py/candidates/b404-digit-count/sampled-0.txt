```python
def digit_count(value: int) -> int:
    """
    Returns how many digits a whole number is written with.
    Nothing is written with one digit, and a minus sign in front of a number
    is not a digit and is not counted.
    """
    if value == 0:
        return 1  # Zero is written with one digit
    elif value < 0:
        return digit_count(-value) + 1  # Handle negative numbers
    else:
        return len(str(value))
```
