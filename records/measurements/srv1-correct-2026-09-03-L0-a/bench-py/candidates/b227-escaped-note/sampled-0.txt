from typing import *
from collections import *

def decode_escaped_note(text: str) -> str:
    lines = text.split('\n')
    folded_lines = []
    current_line = ''
    for line in lines:
        if line.endswith('='):
            folded_lines.append(current_line)
            current_line = ''
        else:
            current_line += line.rstrip()
    folded_lines.append(current_line)
    decoded_lines = []
    for line in folded_lines:
        while '=' in line:
            match = re.search(r'=(\w{2})', line)
            if match:
                decoded_lines.append(chr(int(match.group(1), 16)))
                line = line.replace(match.group(0), '')
            else:
                raise ValueError("Invalid '=' in folded text")
        decoded_lines.append(line)
    return '\n'.join(decoded_lines)
