export function replayTesselMatch(rallies: string[]): { winner: string; bands: number[]; points: number[]; serve: string } {
  if (!Array.isArray(rallies)) {
    throw new Error("Invalid input: rallies must be an array");
  }

  const validRallies = ["A", "B"];
  for (const rally of rallies) {
    if (!validRallies.includes(rally)) {
      throw new Error("Invalid input: rallies must only contain 'A' or 'B'");
    }
  }

  let bands: number[] = [0, 0];
  let points: number[] = [0, 0];
  let serve: string = "A";
  let winner: string = "";

  const checkForWinner = () => {
    if (bands[0] >= 3) {
      winner = "A";
    } else if (bands[1] >= 3) {
      winner = "B";
    }
  };

  for (const rally of rallies) {
    if (winner !== "") {
      throw new Error("Invalid input: rallies cannot continue after the match is decided");
    }

    const servingIndex = serve === "A" ? 0 : 1;
    const receivingIndex = 1 - servingIndex;

    if (rally === serve) {
      points[servingIndex]++;
      if (points[servingIndex] >= 7 && points[servingIndex] - points[receivingIndex] >= 2) {
        bands[servingIndex]++;
        points = [0, 0];
        serve = rally;
      }
    } else {
      serve = rally;
    }

    checkForWinner();
  }

  return { winner, bands, points, serve };
}
