/**
 * Replays a day at a lending desk based on given events.
 * @param stock - A record mapping each title to its copy count.
 * @param cap - The per-member loan cap.
 * @param events - An array of events, each a triple [action, member, title].
 * @returns An array of answers for each event, either "ok" or "no:<reason>".
 */
export function replayLoanDesk(stock: Record<string, number>, cap: number, events: Array<[string, string, string]>): string[] {
  if (cap <= 0) {
    throw new Error("Cap must be a positive integer.");
  }

  const loans: Record<string, Set<string>> = {};
  const holds: Record<string, string[]> = {};

  const results: string[] = [];

  for (const event of events) {
    if (!Array.isArray(event) || event.length !== 3) {
      throw new Error("Each event must be a triple [action, member, title].");
    }

    const [action, member, title] = event;

    switch (action) {
      case "borrow":
        results.push(handleBorrow(stock, loans, holds, cap, member, title));
        break;
      case "return":
        results.push(handleReturn(loans, member, title));
        break;
      case "renew":
        results.push(handleRenew(loans, holds, member, title));
        break;
      case "hold":
        results.push(handleHold(holds, loans, stock, member, title));
        break;
      default:
        throw new Error("Invalid action.");
    }
  }

  return results;
}

function handleBorrow(
  stock: Record<string, number>,
  loans: Record<string, Set<string>>,
  holds: Record<string, string[]>,
  cap: number,
  member: string,
  title: string
): string {
  if (!stock[title]) {
    return "no:unknown-title";
  }
  if (loans[member]?.has(title)) {
    return "no:already-out";
  }
  if ((loans[member]?.size ?? 0) >= cap) {
    return "no:member-cap";
  }
  if (stock[title] === 0 && holds[title]?.length > 0) {
    return "no:none-left";
  }
  if (holds[title]?.length > 0 && holds[title][0] !== member) {
    return "no:queued-ahead";
  }

  // Process queue head
  if (holds[title]?.length > 0 && holds[title][0] === member) {
    holds[title].shift();
  }

  // Allocate loan and reduce stock
  if (!loans[member]) {
    loans[member] = new Set();
  }
  loans[member].add(title);
  stock[title]--;

  return "ok";
}

function handleReturn(loans: Record<string, Set<string>>, member: string, title: string): string {
  if (!loans[member]?.has(title)) {
    return "no:not-out";
  }

  // Remove loan and increase stock
  loans[member].delete(title);
  // Stock increase logic is not needed here as per the task description

  return "ok";
}

function handleRenew(loans: Record<string, Set<string>>, holds: Record<string, string[]>, member: string, title: string): string {
  if (!loans[member]?.has(title)) {
    return "no:not-out";
  }
  if (holds[title]?.length > 0) {
    return "no:on-hold";
  }

  // Renewal logic is not needed here as per the task description

  return "ok";
}

function handleHold(holds: Record<string, string[]>, loans: Record<string, Set<string>>, stock: Record<string, number>, member: string, title: string): string {
  if (!stock[title]) {
    return "no:unknown-title";
  }
  if (loans[member]?.has(title)) {
    return "no:own-loan";
  }
  if (holds[title]?.includes(member)) {
    return "no:in-queue";
  }
  if (stock[title] > 0) {
    return "no:take-it";
  }

  // Add member to hold queue
  if (!holds[title]) {
    holds[title] = [];
  }
  holds[title].push(member);

  return "ok";
}
