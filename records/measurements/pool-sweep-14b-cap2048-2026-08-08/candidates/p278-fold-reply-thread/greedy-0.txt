export function foldReplyThread(messages: string[][]): string {
  if (!Array.isArray(messages) || messages.length === 0) {
    throw new Error("Invalid input: messages must be a non-empty array.");
  }

  const messageMap = new Map<string, { id: string; parent: string; text: string; children: string[] }>();
  const rootMessages: string[] = [];

  for (const message of messages) {
    if (!Array.isArray(message) || message.length !== 3) {
      throw new Error("Invalid input: each message must be an array of exactly three strings.");
    }

    const [id, parent, text] = message;

    if (typeof id !== "string" || typeof parent !== "string" || typeof text !== "string") {
      throw new Error("Invalid input: id, parent, and text must all be strings.");
    }

    if (id.trim() === "") {
      throw new Error("Invalid input: id cannot be an empty string.");
    }

    if (text.includes("\n")) {
      throw new Error("Invalid input: text cannot contain a newline character.");
    }

    if (messageMap.has(id)) {
      throw new Error("Invalid input: repeated id found.");
    }

    messageMap.set(id, { id, parent, text, children: [] });

    if (parent === "") {
      rootMessages.push(id);
    } else {
      const parentMessage = messageMap.get(parent);
      if (!parentMessage) {
        throw new Error("Invalid input: parent message not found.");
      }
      parentMessage.children.push(id);
    }
  }

  function detectCycle(messageId: string, visited: Set<string>): boolean {
    if (visited.has(messageId)) {
      return true;
    }
    visited.add(messageId);
    const message = messageMap.get(messageId)!;
    for (const child of message.children) {
      if (detectCycle(child, new Set(visited))) {
        return true;
      }
    }
    return false;
  }

  for (const rootMessage of rootMessages) {
    if (detectCycle(rootMessage, new Set())) {
      throw new Error("Invalid input: parent links form a cycle.");
    }
  }

  function renderMessage(messageId: string, depth: number): string[] {
    const message = messageMap.get(messageId)!;
    const lines: string[] = [];
    const quoteMarker = "> ".repeat(depth);
    lines.push(`${quoteMarker}${message.id} ${message.text}`);
    for (const child of message.children) {
      lines.push(...renderMessage(child, depth + 1));
    }
    return lines;
  }

  const resultLines: string[] = [];
  for (const rootMessage of rootMessages) {
    resultLines.push(...renderMessage(rootMessage, 0));
  }

  return resultLines.join("\n");
}
