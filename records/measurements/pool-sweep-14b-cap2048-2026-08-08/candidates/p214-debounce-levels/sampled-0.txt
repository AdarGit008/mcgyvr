/**
 * Debounces levels based on the given samples and hold value.
 * @param samples - A non-empty list of samples, each being 0 or 1.
 * @param hold - A positive whole number indicating how many samples must disagree with the settled level before changing it.
 * @returns An array representing the settled level at every sample.
 */
export function debounceLevels(samples: unknown, hold: unknown): number[] {
    if (!Array.isArray(samples) || samples.length === 0) {
        throw new Error("Samples must be a non-empty list.");
    }
    if (!samples.every(sample => typeof sample === 'number' && (sample === 0 || sample === 1))) {
        throw new Error("All samples must be 0 or 1.");
    }
    if (typeof hold !== 'number' || hold <= 0 || !Number.isInteger(hold)) {
        throw new Error("Hold must be a positive whole number.");
    }

    const report: number[] = [];
    let settledLevel = samples[0] as number;
    let tally = 0;

    for (const sample of samples) {
        if (sample === settledLevel) {
            tally = 0;
        } else {
            tally++;
            if (tally >= hold) {
                settledLevel = sample;
                tally = 0;
            }
        }
        report.push(settledLevel);
    }

    return report;
}
