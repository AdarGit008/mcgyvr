export function countPhrase(tokens: string[], phrase: string): number {
  if (!phrase.trim()) {
    throw new Error("Phrase cannot be empty or only spaces");
  }

  const phraseTokens = phrase.toLowerCase().split(/\s+/);
  let count = 0;
  let tokenIndex = 0;

  while (tokenIndex <= tokens.length - phraseTokens.length) {
    let match = true;
    for (let i = 0; i < phraseTokens.length; i++) {
      if (tokens[tokenIndex + i].toLowerCase() !== phraseTokens[i]) {
        match = false;
        break;
      }
    }

    if (match) {
      count++;
      tokenIndex += phraseTokens.length;
    } else {
      tokenIndex++;
    }
  }

  return count;
}
