export function holdShelfReplay(slips: string[]): string[] {
  const queue: string[] = [];
  const responses: string[] = [];

  for (const slip of slips) {
    if (typeof slip !== 'string') {
      throw new Error('Invalid slip');
    }

    const parts = slip.split(' ');
    const verb = parts[0];
    const name = parts.slice(1).join(' ');

    switch (verb) {
      case 'join':
        if (!name) {
          throw new Error('Name missing for join');
        }
        if (queue.includes(name)) {
          responses.push('no:again');
        } else {
          queue.push(name);
          const position = queue.length;
          responses.push(`at:${position}`);
        }
        break;

      case 'leave':
        if (!name) {
          throw new Error('Name missing for leave');
        }
        const index = queue.indexOf(name);
        if (index === -1) {
          responses.push('no:absent');
        } else {
          queue.splice(index, 1);
          responses.push('out');
        }
        break;

      case 'serve':
        if (parts.length > 1) {
          throw new Error('Invalid serve slip');
        }
        if (queue.length === 0) {
          responses.push('idle');
        } else {
          const servedName = queue.shift()!;
          responses.push(`take:${servedName}`);
        }
        break;

      default:
        throw new Error('Invalid verb');
    }
  }

  return responses;
}
