/**
 * Calculates the sums of every complete window in the given list of integers.
 * @param values - The list of integers to process.
 * @param size - The size of each window.
 * @param hop - The step between the start of each window.
 * @returns An array containing the sums of each complete window.
 */
export function hopWindowSums(values: number[], size: number, hop: number): number[] {
  if (!Array.isArray(values) || values.some(v => typeof v !== 'number')) {
    throw new Error('Input must be a list of integers.');
  }
  if (typeof size !== 'number' || size <= 0 || !Number.isInteger(size)) {
    throw new Error('Window size must be a positive integer.');
  }
  if (typeof hop !== 'number' || hop <= 0 || !Number.isInteger(hop)) {
    throw new Error('Hop step must be a positive integer.');
  }

  const result: number[] = [];
  for (let i = 0; i + size <= values.length; i += hop) {
    const windowSum = values.slice(i, i + size).reduce((acc, curr) => acc + curr, 0);
    result.push(windowSum);
  }
  return result;
}
