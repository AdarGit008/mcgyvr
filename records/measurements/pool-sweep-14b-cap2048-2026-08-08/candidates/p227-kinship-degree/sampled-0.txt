/**
 * How far apart two people stand in a register of forebears.
 */
export function kinshipDegree(
  register: Record<string, string[]>,
  one: string,
  other: string,
): Record<string, unknown> {
  // Validate the register
  if (typeof register !== "object" || register === null) {
    throw new Error("Invalid register");
  }

  for (const [name, forebears] of Object.entries(register)) {
    if (typeof name !== "string" || name.trim() === "") {
      throw new Error("Invalid key in register");
    }
    if (!Array.isArray(forebears) || forebears.length > 2) {
      throw new Error("Invalid forebear list in register");
    }
    for (const forebear of forebears) {
      if (typeof forebear !== "string" || forebear.trim() === "") {
        throw new Error("Invalid forebear name in register");
      }
      if (!register[forebear]) {
        throw new Error("Forebear not a key in register");
      }
    }
  }

  // Validate the people
  if (typeof one !== "string" || typeof other !== "string") {
    throw new Error("Invalid person name");
  }
  if (!register[one] || !register[other]) {
    throw new Error("Person not a key in register");
  }

  const climb = (start: string): Map<string, number> => {
    const seen = new Map<string, number>();
    let frontier = register[start] ?? [];
    let step = 1;
    while (frontier.length > 0) {
      const next: string[] = [];
      for (const name of frontier) {
        if (!seen.has(name)) {
          seen.set(name, step);
          next.push(...(register[name] ?? []));
        }
      }
      frontier = next;
      step += 1;
    }
    return seen;
  };

  const mine = climb(one);
  const theirs = climb(other);

  // Direct line check
  if (mine.has(other)) {
    return { steps: mine.get(other)! - 0, line: "direct", meet: other };
  }
  if (theirs.has(one)) {
    return { steps: theirs.get(one)! - 0, line: "direct", meet: one };
  }

  // Collateral line check
  for (const [name, up] of mine) {
    const down = theirs.get(name);
    if (down !== undefined) {
      return { steps: up + down, line: "collateral", meet: name };
    }
  }

  // Apart case
  return { steps: 0, line: "apart", meet: "" };
}
