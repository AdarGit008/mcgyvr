export function squashJournal(ops: (string | number)[][]): (string | number)[][] {
  const store: Record<string, number> = {};
  const keyHistory: Record<string, string[]> = {};

  for (let i = 0; i < ops.length; i++) {
    const [op, key1, valueOrKey2] = ops[i];

    if (typeof op !== 'string' || typeof key1 !== 'string' || key1 === '') {
      throw new Error('Invalid operation');
    }

    switch (op) {
      case 'put':
        if (typeof valueOrKey2 !== 'number' || valueOrKey2 <= 0 || !Number.isInteger(valueOrKey2)) {
          throw new Error('Invalid put operation');
        }
        store[key1] = valueOrKey2;
        keyHistory[key1] = [key1];
        break;
      case 'del':
        if (!(key1 in store)) {
          throw new Error('Deleting absent key');
        }
        delete store[key1];
        break;
      case 'ren':
        const oldKey = key1;
        const newKey = valueOrKey2 as string;

        if (typeof newKey !== 'string' || newKey === '') {
          throw new Error('Invalid rename operation');
        }

        if (!(oldKey in store)) {
          throw new Error('Renaming absent key');
        }
        if (newKey in store) {
          throw new Error('Renaming onto existing key');
        }

        store[newKey] = store[oldKey];
        delete store[oldKey];

        keyHistory[newKey] = [...keyHistory[oldKey], newKey];
        delete keyHistory[oldKey];
        break;
      default:
        throw new Error('Unknown operation');
    }
  }

  const result: (string | number)[][] = [];

  for (const [finalKey, value] of Object.entries(store)) {
    const originalKey = keyHistory[finalKey][0];
    result.push(['put', finalKey, value]);
  }

  return result;
}
