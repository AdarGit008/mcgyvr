export function hexRingWalk(center: number[], radius: number): number[][] {
  if (!Array.isArray(center) || center.length !== 2) {
    throw new Error("Center must be a two-element array.");
  }
  if (typeof center[0] !== 'number' || typeof center[1] !== 'number') {
    throw new Error("Center elements must be numbers.");
  }
  if (!Number.isInteger(radius)) {
    throw new Error("Radius must be an integer.");
  }
  if (radius < 0) {
    throw new Error("Radius cannot be negative.");
  }

  const directions = [
    [1, 0],       // east
    [1, -1],      // northeast
    [0, -1],      // northwest
    [-1, 0],      // west
    [-1, 1],     // southwest
    [0, 1]        // southeast
  ];

  const path: number[][] = [];
  let current = [...center];

  if (radius === 0) {
    return [current];
  }

  // Start at the cell reached by taking radius southwest steps away from the centre
  current[0] -= radius;
  current[1] += radius;

  for (let directionIndex = 0; directionIndex < directions.length; directionIndex++) {
    const [dq, dr] = directions[directionIndex];
    for (let step = 0; step < radius; step++) {
      path.push([...current]);
      current[0] += dq;
      current[1] += dr;
    }
  }

  return path;
}
