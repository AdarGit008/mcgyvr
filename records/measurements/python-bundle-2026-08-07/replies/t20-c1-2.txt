```python
def read_config(text: str) -> dict[str, str]:
    config = {}
    lines = text.splitlines()
    
    for line in lines:
        line = line.strip()
        
        if not line or line.startswith('#'):
            continue
        
        parts = line.split('=', 1)
        
        if len(parts) != 2:
            raise ValueError(f"Invalid line format: {line}")
        
        key, value = parts
        config[key] = value
    
    return config
```
