def parse_semver(version: str) -> tuple[int, int, int, str | None]:
    parts = version.split('-')
    
    if len(parts) > 2:
        raise ValueError("too many parts")
    
    major, minor, patch = map(int, parts[0].split('.'))
    
    prerelease = None
    if len(parts) == 2:
        prerelease = parts[1]
        
        # Validate prerelease part
        if not prerelease or any(not c.isalnum() and c not in ('-', '.') for c in prerelease):
            raise ValueError("invalid prerelease part")
    
    return (major, minor, patch, prerelease)
