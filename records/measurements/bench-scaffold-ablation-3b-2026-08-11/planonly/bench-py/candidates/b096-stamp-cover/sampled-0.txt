def stamp_table(dies: list) -> dict:
    """Builds a lookup table from dies where each fragment is mapped to its price."""
    table = {}
    for fragment, price in dies:
        if not isinstance(fragment, str) or not isinstance(price, int):
            raise ValueError("Invalid fragment or price.")
        if fragment in table:
            raise ValueError(f"Fragment '{fragment}' appears twice.")
        table[fragment] = price
    return table

def stamp_cover(label: str, dies: list) -> int:
    """Calculates the cheapest total price of die presses to spell a label."""
    if not isinstance(label, str) or not label:
        raise ValueError("Label must be a non-empty string.")
    
    table = stamp_table(dies)
    n = len(label)
    
    # Initialize DP array where dp[i] is the cheapest cost to spell the first i characters of the label
    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    
    for i in range(1, n + 1):
        for fragment, price in table.items():
            if i >= len(fragment) and label[i - len(fragment):i] == fragment:
                dp[i] = min(dp[i], dp[i - len(fragment)] + price)
    
    return dp[-1]
