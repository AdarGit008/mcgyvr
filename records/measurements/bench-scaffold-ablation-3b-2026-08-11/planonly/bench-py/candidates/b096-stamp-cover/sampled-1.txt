def stamp_cover(label: str, dies: list) -> int:
    """Calculate the cheapest total price of any sequence of die presses."""
    if not label or not isinstance(label, str):
        raise ValueError("Label must be a non-empty string.")

    table = stamp_table(dies)
    
    n = len(label)

    # dp[i] is the minimum cost to cover the first i characters of the label
    dp = [float('inf')] * (n + 1)
    dp[0] = 0

    for i in range(1, n + 1):
        for j in range(i):
            if label[j:i] in table:
                dp[i] = min(dp[i], dp[j] + table[label[j:i]])

    if dp[n] == float('inf'):
        raise ValueError("No sequence of dies can spell the label.")

    return dp[n]


def stamp_table(dies: list) -> dict:
    """Build a lookup of fragment to price."""
    table = {}

    for die in dies:
        if not (isinstance(die, list) and len(die) == 2):
            raise ValueError("Each entry in the dies argument must be [fragment, price].")
        
        fragment, price = die

        if not isinstance(fragment, str) or not isinstance(price, int) or not (price > 0):
            raise ValueError("Fragment must be a non-empty string and price must be a positive integer.")

        if fragment in table:
            raise ValueError(f"Duplicate fragment '{fragment}' found in dies.")
        
        table[fragment] = price

    return table
