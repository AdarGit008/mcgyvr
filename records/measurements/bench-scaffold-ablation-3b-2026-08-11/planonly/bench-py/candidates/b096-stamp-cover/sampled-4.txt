def stamp_table(dies: list) -> dict:
    """Builds a lookup table for the cheapest price per fragment from dies."""
    table = {}
    
    for die in dies:
        fragment, price = die
        
        if not isinstance(fragment, str) or not fragment.strip():
            raise ValueError("Fragment must be a non-empty string.")
        
        if not isinstance(price, int) or price <= 0:
            raise ValueError("Price must be a positive integer.")
        
        if fragment in table:
            raise ValueError(f"Duplicate fragment '{fragment}' found.")
        
        table[fragment] = price
    
    return table

def stamp_cover(label: str, dies: list) -> int:
    """Finds the cheapest total price of any sequence of die presses to spell the label exactly."""
    
    if not isinstance(label, str) or not label.strip():
        raise ValueError("Label must be a non-empty string.")
    
    table = stamp_table(dies)
    
    n = len(label)
    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    
    for i in range(1, n + 1):
        for fragment, price in table.items():
            if label[i - 1:i] == fragment:
                dp[i] = min(dp[i], dp[i - len(fragment)] + price)
    
    return dp[-1]
