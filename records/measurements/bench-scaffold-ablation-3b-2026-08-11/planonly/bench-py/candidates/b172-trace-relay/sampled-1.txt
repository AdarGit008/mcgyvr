def trace_relay(links: dict, start: str) -> list:
    """
    Walks the handoffs to the empty name, refusing an unknown post and a watch that comes round again.
    
    :param links: A mapping where each key is a post's name and the value is the next post's name.
    :param start: The starting post's name for the watch.
    :return: A list of post names in order from start to the end post, with no duplicates.
    """
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary.")
    
    if start not in links:
        raise ValueError(f"start '{start}' is not a valid post name in links.")
    
    visited = set()
    path = [start]
    current_post = start
    
    while current_post != "":
        if current_post in visited:
            raise ValueError("Watch comes back to an already visited post.")
        
        visited.add(current_post)
        next_post = links.get(current_post)
        if next_post is None:
            raise ValueError(f"Handoff from '{current_post}' does not exist.")
            
        path.append(next_post)
        current_post = next_post
    
    return path
