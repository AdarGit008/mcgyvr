from typing import List, Dict, Tuple

def cheapest_pass_plan(trip_days: List[int], passes: List[Dict[str, int]]) -> Dict[str, Union[int, List[Tuple[int, int]]]]:
    if not trip_days or not passes:
        return {'total': 0, 'purchases': []}
    
    # Validate input
    for day in trip_days:
        if day < 1:
            raise ValueError("Trip days must be strictly increasing positive integers.")
        
    for pass_kind in passes:
        if 'span' not in pass_kind or 'cost' not in pass_kind:
            raise ValueError("Pass kind must contain 'span' and 'cost'.")
        
        span = pass_kind['span']
        cost = pass_kind['cost']
        
        if span <= 0 or cost < 0:
            raise ValueError("Span must be a positive integer, cost must be a non-negative integer.")
    
    # Sort trips by day
    trip_days.sort()
    
    total = 0
    purchases: List[Tuple[int, int]] = []
    
    i = 0
    for start_day in range(1, len(trip_days) + 1):
        if i == len(purchases):
            # Find the cheapest pass that starts on or after the current trip day
            best_cost = float('inf')
            best_span = 0
            for pass_kind in passes:
                if pass_kind['span'] >= start_day:
                    if pass_kind['cost'] < best_cost:
                        best_cost = pass_kind['cost']
                        best_span = pass_kind['span']
            
            if best_cost == float('inf'):
                raise ValueError("Cannot cover all trip days with available passes.")
            
            purchases.append((start_day, best_span))
        
        total += best_cost
        i += 1
    
    return {'total': total, 'purchases': purchases}
