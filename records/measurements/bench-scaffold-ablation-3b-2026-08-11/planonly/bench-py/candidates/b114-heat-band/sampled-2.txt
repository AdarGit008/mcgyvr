def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not isinstance(start, int):
        raise ValueError("start must be an integer")
    if not isinstance(low, int):
        raise ValueError("low must be an integer")
    if not isinstance(high, int):
        raise ValueError("high must be an integer")
    if low >= high:
        raise ValueError("low must be strictly below high")
    if not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not all(isinstance(drift, int) for drift in drifts):
        raise ValueError("drifts must be a list of integers")

    temp = start
    heated = 0
    switches = 0
    coldest = start

    for drift in drifts:
        if temp < low:
            heated += 1
            temp += power + drift
            switches += 1
            coldest = min(coldest, temp)
        elif temp >= high:
            heated += 1
            temp += power + drift
            switches += 1
            coldest = min(coldest, temp)
        else:
            temp += drift

    return {
        "temp": temp,
        "heated": heated,
        "switches": switches,
        "coldest": coldest
    }
