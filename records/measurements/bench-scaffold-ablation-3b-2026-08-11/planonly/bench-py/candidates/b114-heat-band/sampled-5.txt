from typing import *
from collections import *

def run_thermostat(start: int, low: int, high: int, power: int, drifts: List[int]) -> dict:
    if not all(isinstance(x, int) for x in [start, low, high, power]):
        raise ValueError("all parameters must be integers")
    if low >= high:
        raise ValueError("low must be strictly below high")
    if power <= 0:
        raise ValueError("power must be a positive integer")
    if not all(isinstance(x, int) for x in drifts):
        raise ValueError("drifts must be a list of integers")

    temp = start
    heated = 0
    switches = 0
    coldest = start

    for drift in drifts:
        if temp < low:
            heated += 1
            switches += 1
            temp += drift + power
        elif temp > high:
            heated -= 1
            switches += 1
            temp += drift - power
        else:
            temp += drift

        coldest = min(coldest, temp)

    return {
        'temp': temp,
        'heated': heated,
        'switches': switches,
        'coldest': coldest
    }
