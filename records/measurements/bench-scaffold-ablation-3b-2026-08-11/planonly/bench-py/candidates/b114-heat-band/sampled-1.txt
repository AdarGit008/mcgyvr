def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not isinstance(start, int):
        raise ValueError("Start temperature must be an integer.")
    if not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("Low and high values must be integers.")
    if low >= high:
        raise ValueError("Low value must be strictly below high.")
    if not isinstance(power, int) or power <= 0:
        raise ValueError("Power must be a positive integer.")
    if not isinstance(drifts, list):
        raise ValueError("Drifts must be a list.")
    for d in drifts:
        if not isinstance(d, int):
            raise ValueError("Each drift value must be an integer.")

    temp = start
    heated = 0
    switches = 0
    coldest = start

    for d in drifts:
        if temp < low:
            temp += d + power
            heated += 1
            switches += 1
        elif temp >= high:
            temp += d - power
            heated += 1
            switches += 1
        else:
            temp += d

        coldest = min(coldest, temp)

    return {
        "temp": temp,
        "heated": heated,
        "switches": switches,
        "coldest": coldest
    }
