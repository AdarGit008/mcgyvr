export function carveShift(start: number, end: number, span: number, rest: number, least: number): number[][] {
  if (!Number.isInteger(start) || !Number.isInteger(end) || !Number.isInteger(span) || !Number.isInteger(rest) || !Number.isInteger(least)) {
    throw new Error("Arguments must be whole numbers.");
  }
  if (start >= end) {
    throw new Error("Start must precede end.");
  }
  if (span < 1 || rest < 1 || least > span) {
    throw new Error("Span and rest must be at least 1, and least must not exceed span.");
  }

  const stretches: number[][] = [];
  let current = start;

  while (current < end) {
    const nextEnd = Math.min(current + span, end);
    if (nextEnd - current >= least) {
      stretches.push([current, nextEnd]);
    }
    current += rest;
  }

  return stretches;
}
