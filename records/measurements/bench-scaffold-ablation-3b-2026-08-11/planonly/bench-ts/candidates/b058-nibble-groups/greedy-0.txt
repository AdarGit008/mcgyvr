export function formatBits(value: number, width: number): string {
  if (!Number.isInteger(value) || value < 0) {
    throw new Error("Value must be a non-negative integer.");
  }
  if (width <= 0 || width % 4 !== 0 || width > 32) {
    throw new Error("Width must be a positive multiple of four and no more than 32 bits.");
  }

  const binaryString = value.toString(2);
  const paddedBinaryString = binaryString.padStart(width, '0');

  let result = '';
  for (let i = 0; i < width; i += 4) {
    const nibble = paddedBinaryString.substring(i, i + 4);
    if (result !== '') {
      result += ' ';
    }
    result += nibble;
  }

  return result;
}
