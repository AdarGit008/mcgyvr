export function maskFromPrefix(prefix: number): string {
  if (!Number.isInteger(prefix) || prefix < 0 || prefix > 32) {
    throw new Error("Invalid prefix: must be an integer from 0 to 32.");
  }

  const octets = Array.from({ length: 4 }, (_, i) => Math.floor(prefix / (Math.pow(2, 8 * (3 - i))) % 256));
  return octets.join(".");
}

export function prefixFromMask(mask: string): number {
  if (typeof mask !== "string") {
    throw new Error("Invalid mask: must be a string.");
  }

  const parts = mask.split(".");
  if (parts.length !== 4) {
    throw new Error("Invalid mask: must have exactly four dot-separated fields.");
  }

  let prefix = 0;
  for (let i = 0; i < 4; i++) {
    const octet = parseInt(parts[i], 10);
    if (isNaN(octet) || octet < 0 || octet > 255) {
      throw new Error("Invalid mask: each field must be a decimal digit from 0 to 255.");
    }
    prefix += octet * Math.pow(2, 8 * (3 - i));
  }

  return prefix;
}
