export function traceRelay(links: Record<string, string>, start: string): string[] {
  if (typeof links !== 'object' || !links) {
    throw new Error('Links must be a non-empty object.');
  }
  if (!Object.keys(links).includes(start)) {
    throw new Error(`Start post '${start}' is not in the links.`);
  }

  const visited = new Set<string>();
  const path: string[] = [start];

  while (path[path.length - 1] !== '') {
    const currentPost = path[path.length - 1];
    if (!visited.has(currentPost)) {
      visited.add(currentPost);
      const nextPost = links[currentPost];
      if (!nextPost || typeof nextPost !== 'string') {
        throw new Error(`Invalid handoff from '${currentPost}'.`);
      }
      path.push(nextPost);
    } else {
      throw new Error('Watch comes back to a post already visited.');
    }
  }

  return path;
}
