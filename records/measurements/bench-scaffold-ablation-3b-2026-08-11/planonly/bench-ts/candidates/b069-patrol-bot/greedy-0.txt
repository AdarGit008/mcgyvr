export function runPatrol(width: number, start: number, moves: number[]): Record<string, number> {
  if (!Number.isInteger(width) || width <= 0) {
    throw new Error("Width must be a positive integer.");
  }
  if (!Number.isInteger(start) || start < 0 || start >= width) {
    throw new Error("Start position must be an integer within the corridor.");
  }
  if (!Array.isArray(moves)) {
    throw new Error("Moves argument must be a list.");
  }

  let position = start;
  let bumps = 0;
  const visited: Set<number> = new Set([start]);

  for (const delta of moves) {
    if (!Number.isInteger(delta) || delta === 0) {
      throw new Error("Move delta must be a non-zero integer.");
    }

    const [newPosition, isWallCut] = clampMove(position, delta, width);
    position = newPosition;
    bumps += isWallCut ? 1 : 0;

    visited.add(position);
  }

  return { position, bumps, visited: visited.size };
}
