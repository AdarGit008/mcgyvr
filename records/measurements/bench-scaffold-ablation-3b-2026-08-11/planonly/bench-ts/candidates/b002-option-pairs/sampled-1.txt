export function scanPairs(input: string): string[][] {
  if (typeof input !== 'string' || input === '') {
    throw new Error('Input must be a non-empty string');
  }

  const result: string[][] = [];
  let currentSegment = '';
  let inQuotes = false;

  for (let i = 0; i < input.length; i++) {
    if (input[i] === '"') {
      inQuotes = !inQuotes;
      continue;
    }

    if (!inQuotes && input[i] === ';') {
      if (currentSegment.trim() !== '') {
        result.push(parseOption(currentSegment));
        currentSegment = '';
      }
      continue;
    }

    currentSegment += input[i];
  }

  // Process the last segment
  if (currentSegment.trim() !== '') {
    result.push(parseOption(currentSegment));
  }

  return result;
}

export function parseOption(segment: string): string[] {
  const [key, value] = segment.split('=');

  if (!/^[a-zA-Z_][a-zA-Z0-9_]*$/.test(key)) {
    throw new Error('Invalid key identifier');
  }

  let parsedValue = value;
  if (parsedValue.startsWith('"') && parsedValue.endsWith('"')) {
    parsedValue = parsedValue.substring(1, parsedValue.length - 1);
    // Remove escaped quotes
    parsedValue = parsedValue.replace(/\\\\"/g, '"');
  } else if (parsedValue === '') {
    throw new Error('Empty quoted value');
  }

  return [key, parsedValue];
}

The `parseOption` function checks for valid keys and parses quoted values while handling escaped quotes.
