export function scanPairs(input: string): string[][] {
  if (typeof input !== 'string' || input === '') {
    throw new Error('Input must be a non-empty string');
  }

  const pairs = [];
  let currentPair: [string, string] | null = null;

  for (let i = 0; i < input.length; i++) {
    if (currentPair === null) {
      // Start of a new pair
      currentPair = parseOption(input.substring(i));
      if (!currentPair) {
        throw new Error('Invalid option segment');
      }
      pairs.push(currentPair);
      i += currentPair[0].length + 1; // Move past the parsed key and '='
    } else {
      // Continue parsing the current pair
      const [key, value] = parseOption(input.substring(i));
      if (!key || !value) {
        throw new Error('Invalid option segment');
      }
      pairs.push([currentPair[0], value]);
      i += key.length + 1; // Move past the parsed key and '='
      currentPair = null;
    }
  }

  return pairs;
}

function parseOption(segment: string): [string, string] | null {
  const equalsIndex = segment.indexOf('=');
  if (equalsIndex === -1) {
    // No equals sign, treat as a bare identifier with no value
    if (!/^[a-zA-Z_][a-zA-Z0-9_]*$/.test(segment)) {
      return null;
    }
    return [segment, ''];
  }

  const key = segment.substring(0, equalsIndex);
  let value = segment.substring(equalsIndex + 1);

  // Strip double quotes from the value if present
  if (value.startsWith('"') && value.endsWith('"')) {
    value = value.slice(1, -1).replace(/\\"/g, '"');
  }

  return [key, value];
}
