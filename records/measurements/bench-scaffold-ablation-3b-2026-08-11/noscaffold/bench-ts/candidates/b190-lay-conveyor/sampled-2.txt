export function layConveyor(floor: string[], row: number, col: number, length: number): string[] {
    if (row < 0 || row >= floor.length) {
        throw new Error("Row out of bounds");
    }
    if (col < 0 || col + length > floor[row].length) {
        throw new Error("Run would leave the plan");
    }

    const result = [...floor];
    for (let i = 0; i < length; i++) {
        const cell = floor[row][col + i];
        if (cell !== '.') {
            throw new Error("Cell is occupied by a machine");
        }
        result[row][col + i] = '=';
    }

    return result;
}
