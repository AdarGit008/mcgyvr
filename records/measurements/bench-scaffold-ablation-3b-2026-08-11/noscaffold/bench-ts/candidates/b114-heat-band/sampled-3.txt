export function runThermostat(start: number, low: number, high: number, power: number, drifts: number[]): { temp: number; heated: number; switches: number; coldest: number } {
  if (typeof start !== 'number' || !Number.isInteger(start)) throw new Error('start must be an integer');
  if (typeof low !== 'number' || !Number.isInteger(low) || low >= high) throw new Error('low must be an integer strictly below high');
  if (typeof high !== 'number' || !Number.isInteger(high)) throw new Error('high must be an integer');
  if (typeof power !== 'number' || !Number.isInteger(power) || power <= 0) throw new Error('power must be a positive integer');
  if (!Array.isArray(drifts) || !drifts.every(Number.isInteger)) throw new Error('drifts must be a list of integers');

  let temp = start;
  let heated = 0;
  let switches = 0;
  let coldest = start;

  for (const drift of drifts) {
    if (temp < low) {
      heated++;
      switches++;
      temp += power + drift;
      coldest = Math.min(coldest, temp);
    } else if (temp >= high) {
      switches++;
      heated--;
      temp += drift;
      coldest = Math.min(coldest, temp);
    } else {
      temp += drift;
      coldest = Math.min(coldest, temp);
    }
  }

  return { temp, heated, switches, coldest };
}
