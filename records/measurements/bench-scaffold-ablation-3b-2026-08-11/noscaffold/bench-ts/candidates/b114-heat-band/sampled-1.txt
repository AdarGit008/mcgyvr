export function runThermostat(start: number, low: number, high: number, power: number, drifts: number[]): { temp: number; heated: number; switches: number; coldest: number } {
  if (!Number.isInteger(start) || !Number.isInteger(low) || !Number.isInteger(high) || !Number.isInteger(power)) {
    throw new Error("All inputs must be integers.");
  }
  if (low >= high) {
    throw new Error("Low temperature must strictly be below high.");
  }
  if (!Array.isArray(drifts) || !drifts.every(Number.isInteger)) {
    throw new Error("Drifts must be an array of integers.");
  }

  let temp = start;
  let heated = 0;
  let switches = 0;
  let coldest = start;

  for (const drift of drifts) {
    if (!Number.isInteger(drift)) {
      throw new Error("Drift must be an integer.");
    }

    if (temp < low) {
      heated++;
      temp += power + drift;
      switches++;
      if (temp < coldest) {
        coldest = temp;
      }
    } else if (temp >= high) {
      heated++;
      temp -= drift;
      switches++;
      if (temp < coldest) {
        coldest = temp;
      }
    }

    // Check for overflow or underflow
    if (isNaN(temp)) {
      throw new Error("Temperature calculation overflowed.");
    }
  }

  return { temp, heated, switches, coldest };
}
