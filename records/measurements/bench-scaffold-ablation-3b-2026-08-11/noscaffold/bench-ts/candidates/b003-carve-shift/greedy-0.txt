export function carveShift(start: number, end: number, span: number, rest: number, least: number): number[][] {
  if (!Number.isInteger(start) || !Number.isInteger(end) || !Number.isInteger(span) || !Number.isInteger(rest) || !Number.isInteger(least)) {
    throw new Error("All arguments must be whole numbers.");
  }
  if (start >= end) {
    throw new Error("Start must precede end.");
  }
  if (span < 1 || rest < 1 || least < 1 || least > span) {
    throw new Error("Span, rest, and least must be at least 1 and least cannot exceed span.");
  }

  const stretches: number[][] = [];
  let currentStart = start;
  let remainingTime = end - start;

  while (remainingTime >= least) {
    const stretchLength = Math.min(span, remainingTime);
    stretches.push([currentStart, currentStart + stretchLength]);
    currentStart += stretchLength + rest;
    remainingTime -= stretchLength + rest;
  }

  return stretches;
}
