export function runPatrol(width: number, start: number, moves: number[]): Record<string, number> {
  if (!Number.isInteger(width) || width <= 0) {
    throw new Error("Width must be a positive integer.");
  }
  if (!Number.isInteger(start) || start < 0 || start >= width) {
    throw new Error("Start position must be an integer within the corridor.");
  }
  if (!Array.isArray(moves)) {
    throw new Error("Moves argument must be a list (array).");
  }

  let position = start;
  let bumps = 0;
  const visited: Set<number> = new Set([position]);

  for (const delta of moves) {
    const [newPos, cutShort] = clampMove(position, delta, width);
    position = newPos;

    if (cutShort) {
      bumps++;
    }

    if (!visited.has(position)) {
      visited.add(position);
    }
  }

  return { position, bumps, visited: visited.size };
}

function clampMove(position: number, delta: number, width: number): [number, boolean] {
  let newPos = position + delta;
  if (newPos < 0) {
    newPos = 0;
    return [newPos, true];
  } else if (newPos >= width) {
    newPos = width - 1;
    return [newPos, true];
  }
  return [newPos, false];
}
