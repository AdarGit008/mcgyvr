def parse_amount(text: str) -> int:
    """Converts a decimal amount string to integer cents."""
    if not isinstance(text, str):
        raise ValueError("Input must be a string.")
    
    parts = text.split('.')
    if len(parts) != 2 or len(parts[1]) != 2:
        raise ValueError("Exactly two fraction digits are required.")
    
    integer_part = int(parts[0])
    fractional_part = int(parts[1])
    
    total_cents = integer_part * 100 + fractional_part
    
    if total_cents < 0 or total_cents > 9999:
        raise ValueError("Total cents must be between 0 and 9999.")
    
    return total_cents

def format_amount(cents: int) -> str:
    """Converts integer cents back to a dotted amount string."""
    if not isinstance(cents, int):
        raise ValueError("Input must be an integer.")
    
    if cents < 0 or cents > 9999:
        raise ValueError("Cents must be between 0 and 9999.")
    
    integer_part = cents // 100
    fractional_part = cents % 100
    
    return f"{integer_part}.{fractional_part:02d}"

def allocate_cents(total_cents: int, weights: list) -> list:
    """Allocates total_cents across a list of integer weights proportionally."""
    if not isinstance(weights, list):
        raise ValueError("Weights must be a list.")
    if len(weights) == 0:
        raise ValueError("Weight list cannot be empty.")
    
    for weight in weights:
        if not isinstance(weight, int) or weight < 0:
            raise ValueError("All weights must be non-negative integers.")
    
    total_weights = sum(weights)
    if total_weights == 0:
        raise ValueError("Total weights cannot be zero.")
    
    if total_cents < 0 or total_cents > total_weights * 100:
        raise ValueError("Total cents must not exceed the maximum possible allocation given the weights.")
    
    shares = [weight * total_cents // total_weights for weight in weights]
    remainder = total_cents % total_weights
    
    i = 0
    while remainder > 0:
        if weights[i] != 0:
            shares[i] += 1
            remainder -= 1
        i += 1
    
    return shares
