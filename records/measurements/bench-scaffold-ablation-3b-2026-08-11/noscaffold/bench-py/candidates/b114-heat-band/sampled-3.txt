def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not all(isinstance(x, int) for x in [start, low, high, power]):
        raise ValueError("All start, low, high, and power must be integers.")
    if low >= high:
        raise ValueError("Low must be strictly below high.")
    if power <= 0:
        raise ValueError("Power must be a positive integer.")
    if not all(isinstance(x, int) for x in drifts):
        raise ValueError("All elements of the drifts list must be integers.")

    heated = switches = coldest = temp = start

    for drift in drifts:
        if temp < low:
            heated += 1
            temp += power + drift
            switches += 1
        elif temp >= high:
            heated -= 1
            temp += drift
            switches += 1
        else:
            temp += drift

        coldest = min(coldest, temp)

    return {
        "temp": temp,
        "heated": heated,
        "switches": switches,
        "coldest": coldest
    }
