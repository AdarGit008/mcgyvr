def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if not (isinstance(start, int) and isinstance(low, int) and isinstance(high, int)
            and isinstance(power, int) and isinstance(drifts, list)):
        raise ValueError("All inputs must be integers")
    
    if low >= high or power <= 0:
        raise ValueError("Low must be strictly below high and power must be positive")

    temp = start
    heated = 0
    switches = 0
    coldest = start

    for drift in drifts:
        if not isinstance(drift, int):
            raise ValueError("Drifts must be integers")
        
        if temp < low:
            heated += 1
            switches += 1
            temp += power + drift
        elif temp > high:
            heated -= 1
            switches += 1
            temp += drift
        else:
            temp += drift

        coldest = min(coldest, temp)

    return {
        "temp": temp,
        "heated": heated,
        "switches": switches,
        "coldest": coldest
    }
