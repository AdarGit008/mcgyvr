def lay_conveyor(floor: list, row: int, col: int, length: int) -> list:
    if len(floor) == 0 or len(floor[0]) == 0:
        raise ValueError("The floor plan is empty.")
    
    if row < 0 or row >= len(floor):
        raise ValueError("Invalid row index.")
    
    if col < 0 or col + length > len(floor[row]):
        raise ValueError("Run would leave the plan.")
    
    for i in range(length):
        if floor[row][col + i] != ".":
            raise ValueError(f"Cell at [{row}, {col + i}] is not open.")
    
    new_floor = [list(row) for row in floor]
    for i in range(length):
        new_floor[row][col + i] = "="
    
    return new_floor
