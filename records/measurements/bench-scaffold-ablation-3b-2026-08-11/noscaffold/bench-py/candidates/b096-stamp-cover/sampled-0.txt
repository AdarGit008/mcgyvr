def stamp_table(dies: list) -> dict:
    """Converts a list of [fragment, price] entries into a lookup of fragment to price."""
    table = {}
    for frag, price in dies:
        if not isinstance(frag, str) or not isinstance(price, int):
            raise ValueError("Die entry must be a pair of non-empty string and positive integer.")
        if frag in table:
            raise ValueError(f"Fragment '{frag}' listed twice.")
        table[frag] = price
    return table

def stamp_cover(label: str, dies: list) -> int:
    """Returns the cheapest total price of any sequence of die presses to spell a label."""
    if not isinstance(label, str):
        raise ValueError("Label must be a non-empty string.")

    # Create a lookup table for fragment prices
    fragment_table = stamp_table(dies)

    # Initialize the minimum cost array with infinity
    min_cost = [float('inf')] * (len(label) + 1)
    min_cost[0] = 0

    # Fill in the min_cost array
    for i, char in enumerate(label):
        for j in range(i + 1):
            current_fragment = label[j:i+1]
            if current_fragment in fragment_table:
                min_cost[i+1] = min(min_cost[i+1], min_cost[j] + fragment_table[current_fragment])

    # Return the minimum cost to spell the entire label
    return min_cost[-1]
