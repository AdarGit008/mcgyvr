def stamp_table(dies: list) -> dict:
    """
    Converts a list of [fragment, price] entries into a lookup table for fragment to price,
    rejecting invalid fragments and prices.
    """
    result = {}
    for frag, price in dies:
        if not isinstance(frag, str) or not frag.strip():
            raise ValueError("Invalid fragment: must be a non-empty string.")
        if not isinstance(price, int) or price <= 0:
            raise ValueError("Invalid price: must be a positive integer.")
        if frag in result:
            raise ValueError(f"Fragment '{frag}' is listed twice.")
        result[frag] = price
    return result

def stamp_cover(label: str, dies: list) -> int:
    """
    Returns the cheapest total price of any sequence of die presses whose fragments,
    laid end to end, spell the label exactly.
    """
    if not isinstance(label, str) or not label.strip():
        raise ValueError("Invalid label: must be a non-empty string.")
    
    table = stamp_table(dies)
    
    n = len(label)
    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    
    for i in range(1, n + 1):
        for j in range(i):
            if label[j:i] in table:
                dp[i] = min(dp[i], dp[j] + table[label[j:i]])
    
    return dp[n]
