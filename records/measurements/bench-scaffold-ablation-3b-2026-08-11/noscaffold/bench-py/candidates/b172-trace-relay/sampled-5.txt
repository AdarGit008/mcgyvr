def trace_relay(links: dict, start: str) -> list:
    """
    Trace the night watch handoffs from the start post to the end post.

    :param links: A dictionary mapping each post name to the next post name.
    :param start: The name of the starting post.
    :return: A list of post names in order, including the start and end posts.
    """
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary.")
    
    if start not in links:
        raise ValueError(f"start '{start}' is not a valid post name.")

    visited = set()
    path = [start]

    while True:
        current_post = path[-1]
        
        if current_post == "":
            if len(path) > 1:
                return path
            else:
                raise ValueError("Run of handoffs comes back to start post.")
        
        next_post = links.get(current_post)
        
        if next_post is None or next_post in visited:
            raise ValueError(f"Invalid handoff for '{current_post}'.")
        
        if next_post == "":
            return path
        
        visited.add(current_post)
        path.append(next_post)

    # This line should never be reached due to the while loop's condition.
    raise ValueError("Unreachable code.")
