def trace_relay(links: dict, start: str) -> list:
    """
    Trace the path of watch handoffs from start to the empty name.

    :param links: A dictionary mapping each post's name to the next post's name.
    :param start: The starting post's name.
    :return: A list of post names visited in order, ending with an empty string.
    """
    if not isinstance(links, dict):
        raise ValueError("links must be a dictionary.")
    
    if start not in links:
        raise ValueError(f"start '{start}' is not a valid post name.")
    
    path = [start]
    current_post = start
    
    while True:
        next_post = links.get(current_post)
        
        if not isinstance(next_post, str):
            raise ValueError("links must map to strings.")
        
        if next_post == "":
            break
        
        if next_post in path:
            raise ValueError(f"Detected a cycle: {path}.")
        
        path.append(next_post)
        current_post = next_post
    
    return path
