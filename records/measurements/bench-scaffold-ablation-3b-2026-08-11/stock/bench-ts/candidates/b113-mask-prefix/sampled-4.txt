export function maskFromPrefix(prefix: number): string {
  if (!Number.isInteger(prefix) || prefix < 0 || prefix > 32) {
    throw new Error("prefix must be an integer from 0 to 32");
  }
  const octets: number[] = [];
  for (let slot = 0; slot < 4; slot += 1) {
    const ones = Math.min(8, Math.max(0, prefix - slot * 8));
    octets.push(256 - 2 ** (8 - ones));
  }
  return octets.join(".");
}

export function prefixFromMask(mask: string): number {
  if (typeof mask !== "string") {
    throw new Error("mask must be a string");
  }

  const parts = mask.split(".").map((part, index) => {
    if (!/\d/.test(part)) {
      throw new Error(`Invalid octet at position ${index + 1}: '${part}'`);
    }
    const octet = parseInt(part, 10);
    if (octet < 0 || octet > 255) {
      throw new Error(`Octet out of range at position ${index + 1}: '${octet}'`);
    }
    return octet;
  });

  if (parts.length !== 4) {
    throw new Error("Mask must have exactly four dot-separated fields");
  }

  let count = 0;
  for (let i = 3; i >= 0; i--) {
    const octet = parts[i];
    if (octet === 255) {
      count += 8;
    } else if (octet & (octet + 1)) {
      throw new Error(`Mask has broken ones: '${mask}'`);
    } else {
      break;
    }
  }

  return count;
}
