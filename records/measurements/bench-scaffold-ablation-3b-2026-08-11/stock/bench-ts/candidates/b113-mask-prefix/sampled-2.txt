export function maskFromPrefix(prefix: number): string {
  if (!Number.isInteger(prefix) || prefix < 0 || prefix > 32) {
    throw new Error("prefix must be an integer from 0 to 32");
  }
  const octets: number[] = [];
  for (let slot = 0; slot < 4; slot += 1) {
    const ones = Math.min(8, Math.max(0, prefix - slot * 8));
    octets.push(256 - 2 ** (8 - ones));
  }
  return octets.join(".");
}

export function prefixFromMask(mask: string): number {
  if (typeof mask !== "string") {
    throw new Error("mask must be a string");
  }
  const octets = mask.split(".").map((octet, index) => {
    if (!/^\d+$/.test(octet)) {
      throw new Error(`invalid octet at position ${index}: "${octet}"`);
    }
    const num = parseInt(octet, 10);
    if (num < 0 || num > 255) {
      throw new Error(`invalid octet at position ${index}: "${octet}"`);
    }
    return num;
  });
  if (octets.length !== 4) {
    throw new Error("mask must contain exactly four dot-separated fields");
  }

  let leadingOnes = 0;
  for (let i = 0; i < octets.length; i++) {
    const octet = octets[i];
    for (let j = 7; j >= 0; j--) {
      if ((octet >> j) & 1) {
        leadingOnes += 1;
      } else {
        break;
      }
    }
    if (leadingOnes > 32 - octets.length + i) {
      throw new Error("one bits are broken by a zero");
    }
  }

  return leadingOnes;
}
