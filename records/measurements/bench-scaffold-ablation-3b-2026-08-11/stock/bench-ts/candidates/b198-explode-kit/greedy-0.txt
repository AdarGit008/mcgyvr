/** Total the base parts an order of a kit consumes. */
export function explodeKit(catalog: Record<string, { makes: number; parts: [string, number][] }>, kit: string, want: number): Record<string, number> {
  if (!Object.prototype.hasOwnProperty.call(catalog, kit)) {
    throw new Error(`the catalog does not define ${kit}`);
  }
  const totals: Record<string, number> = {};
  function processKit(kitName: string, units: number): void {
    if (units <= 0) return;
    const recipe = catalog[kitName];
    const { makes, parts } = recipe;
    const runs = Math.ceil(units / makes);
    totals[kitName] += runs * makes;
    for (const [component, count] of parts) {
      processKit(component, runs * count);
    }
  }
  processKit(kit, want);
  return totals;
}
