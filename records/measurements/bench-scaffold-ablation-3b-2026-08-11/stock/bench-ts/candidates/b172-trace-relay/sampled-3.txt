export function traceRelay(links: Record<string, string>, start: string): string[] {
  if (links === null || typeof links !== "object" || Array.isArray(links)) throw new Error("links must be a mapping");
  if (!(start in links)) throw new Error("start is not a post");

  const route: string[] = [start];
  let post = start;

  while (post !== "") {
    const nextPost = links[post];
    if (!nextPost) throw new Error("unknown post");
    if (route.includes(nextPost)) throw new Error("watch comes around again");
    route.push(nextPost);
    post = nextPost;
  }

  return route;
}
