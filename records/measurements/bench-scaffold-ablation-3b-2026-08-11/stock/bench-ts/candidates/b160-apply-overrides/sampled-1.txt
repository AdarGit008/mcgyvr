export function applyOverrides(
  base: Record<string, string>,
  overrides: string[],
): Record<string, string> {
  if (!Array.isArray(overrides)) {
    throw new Error("Overrides must be a list.");
  }

  const merged = { ...base };

  for (const override of overrides) {
    const [name, value] = override.split("=").map((part) => part.trim());

    if (typeof name !== "string" || typeof value !== "string") {
      throw new Error("Each override must be a string with an equals sign.");
    }

    if (!name || !value) {
      throw new Error("Override names and values cannot be empty strings.");
    }

    merged[name] = value;
  }

  return merged;
}
