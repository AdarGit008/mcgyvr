"""Cut a fabric roll for the greatest takings the price board allows."""


def cut_roll(length: int, board: list[tuple[int, int]]) -> dict:
    if isinstance(length, bool) or not isinstance(length, int) or length < 0:
        raise ValueError("length must be a non-negative whole number")
    
    # Sort boards by descending price and ascending length
    order = sorted(board, key=lambda entry: (-entry[1], entry[0]))
    
    # Initialize arrays to keep track of the best takings and corresponding pieces
    best = [0] * (length + 1)
    taken = [-1] * (length + 1)
    
    for metres in range(1, length + 1):
        best[metres] = -1
        for piece, price in order:
            if piece <= metres and price + best[metres - piece] > best[metres]:
                best[metres] = price + best[metres - piece]
                taken[metres] = piece
        
        # If not taking the last full board piece, check if we can take less
        if best[metres - 1] > best[metres]:
            best[metres] = best[metres - 1]
            taken[metres] = 0
    
    # Walk back from the full roll to collect the pieces cut at each step
    pieces: list[int] = []
    current_metres = length
    while current_metres > 0:
        if taken[current_metres] == 0:
            pieces.append(current_metres)
        current_metres -= taken[current_metres]
    
    # Reverse the pieces to get them in the order they were cut
    pieces.reverse()
    
    return {"takings": best[length], "pieces": pieces}
