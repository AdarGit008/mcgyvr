def stamp_table(dies: list) -> dict:
    table = {}
    for fragment, price in dies:
        if not isinstance(fragment, str) or not fragment:
            raise ValueError("a die fragment must be a non-empty string")
        if isinstance(price, bool) or not isinstance(price, int) or price <= 0:
            raise ValueError("a die price must be a positive integer")
        if fragment in table:
            raise ValueError("die listed twice: " + fragment)
        table[fragment] = price
    return table


def stamp_cover(label: str, dies: list) -> int:
    # Validate the label
    if not isinstance(label, str) or not label:
        raise ValueError("label must be a non-empty string")
    
    # Build the stamp table
    try:
        stamp_table_ = stamp_table(dies)
    except ValueError as e:
        return f"Error building stamp table: {str(e)}"
    
    # Initialize memoization dictionary for dp
    memo = {}
    
    def min_cost(index):
        if index == len(label):
            return 0
        if index in memo:
            return memo[index]
        
        # Try all possible ways to start a new sequence
        min_cost_val = float('inf')
        for fragment, price in stamp_table_.items():
            if label.startswith(fragment, index):
                cost = price + min_cost(index + len(fragment))
                min_cost_val = min(min_cost_val, cost)
        
        memo[index] = min_cost_val
        return min_cost_val
    
    # Find the minimum cost to cover the entire label
    result = min_cost(0)
    
    # If the result is infinity, it means no valid sequence was found
    if result == float('inf'):
        raise ValueError("no sequence can spell the label")
    
    return result
