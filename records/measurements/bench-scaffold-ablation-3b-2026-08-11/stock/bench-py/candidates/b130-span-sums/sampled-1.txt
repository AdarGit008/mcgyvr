```python
"""Tape-measure spans: parse to inches, format canonically, and add."""

INCHES = {"yd": 36, "ft": 12, "in": 1}
ORDER = ("yd", "ft", "in")


def parse_span(text: str) -> int:
    if not isinstance(text, str):
        raise ValueError("input must be a string")
    
    parts = text.split()
    if len(parts) != 3:
        raise ValueError("span must contain exactly three space-separated parts")
    
    total = 0
    for part in parts:
        value, unit = part[:-2], part[-2:]
        
        if not value.isdigit() or int(value) < 0:
            raise ValueError("value must be a non-negative integer without leading zero")
        
        if unit not in INCHES or unit != ORDER[ORDER.index(unit)]:
            raise ValueError(f"unit {unit} is not valid or out of order")
        
        total += int(value) * INCHES[unit]
    
    return total


def format_span(inches: int) -> str:
    if not isinstance(inches, int) or inches < 0:
        raise ValueError("inches must be a non-negative integer")
    
    if inches == 0:
        return "0in"
    
    parts = []
    yards, rest = divmod(inches, 36)
    feet, rest = divmod(rest, 12)
    
    if yards:
        parts.append(f"{yards}yd")
    if feet:
        parts.append(f"{feet}ft")
    if rest:
        parts.append(f"{rest}in")
    
    return " ".join(parts)


def add_spans(first: str, second: str) -> str:
    total = parse_span(first) + parse_span(second)
    return format_span(total)
```
