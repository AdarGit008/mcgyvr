import re


def parse_option(segment: str) -> list:
    key, eq, raw = segment.partition("=")
    if not eq or not re.fullmatch(r"[A-Za-z][A-Za-z0-9_]*", key):
        raise ValueError("malformed option segment: " + segment)
    if len(raw) >= 2 and raw.startswith('"') and raw.endswith('"'):
        raw = raw[1:-1]
    return [key, raw]


def scan_pairs(text: str) -> list:
    if not isinstance(text, str):
        raise ValueError("input must be a string")
    if text == "":
        raise ValueError("empty input")

    pairs = []
    current_key = None
    in_quotes = False

    for char in text:
        if char == '"':
            if in_quotes:
                in_quotes = False
            else:
                in_quotes = True
        elif char == ';':
            if not in_quotes:
                if current_key is not None:
                    pairs.append(parse_option(f"{current_key}={raw}"))
                    current_key = None
                    raw = ""
        elif in_quotes:
            raw += char
        else:
            if current_key is None:
                current_key = char
            else:
                current_key += char

    # Handle the last segment
    if current_key is not None:
        pairs.append(parse_option(f"{current_key}={raw}"))

    return pairs
