def line_tag(body):
    if not isinstance(body, str):
        raise ValueError("line_tag expects a string")
    return format(sum(ord(ch) for ch in body) % 256, "02x")


def check_line(line: str) -> str:
    if not isinstance(line, str) or len(line) < 3:
        raise ValueError("check_line expects a non-empty string of at least three characters")
    
    tilde_index = line.rfind('~')
    if tilde_index == -1 or tilde_index == len(line) - 2:
        raise ValueError("check_line requires a line with exactly one '~' and two characters after it")
    
    body = line[:tilde_index]
    tag = line[tilde_index + 1:]
    
    expected_tag = line_tag(body)
    if tag != expected_tag:
        raise ValueError(f"check_line: tag mismatch: '{tag}' does not match '{expected_tag}' for body '{body}'")
    
    return body
