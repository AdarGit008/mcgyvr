"""Prune a file listing with ordered drop and keep rules."""


def prune_listing(listing: list, rules: list) -> list:
    def segment_fits(pattern: str, text: str) -> bool:
        p = t = mark = 0
        star = -1
        while t < len(text):
            if p < len(pattern) and pattern[p] == "*":
                star = p
                mark = t
                p += 1
            elif p < len(pattern) and pattern[p] == text[t]:
                p += 1
                t += 1
            elif star >= 0:
                p = star + 1
                mark += 1
                t = mark
            else:
                return False
        while p < len(pattern) and pattern[p] == "*":
            p += 1
        return p == len(pattern)

    def split_strict(text: str) -> list:
        if not isinstance(text, str) or text == "":
            raise ValueError(f"{text} must be a non-empty string")
        segments = text.split("/")
        if any(segment == "" for segment in segments):
            raise ValueError(f"{text} has an empty segment: {text}")
        return segments

    # Validate every rule and path with split_strict, then keep each
    # path the last matching rule allows; a path no rule matches survives.
    def match_path(path: str) -> bool:
        for rule in rules:
            if rule.startswith("!"):
                # Keep rule
                if segment_fits(rule[1:], path):
                    return True
            else:
                # Drop rule
                if not segment_fits(rule, path):
                    return False
        return True

    matched_paths = [path for path in listing if match_path(path)]
    return matched_paths
