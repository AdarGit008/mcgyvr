"""Prune a file listing with ordered drop and keep rules."""


def prune_listing(listing: list, rules: list) -> list:
    def segment_fits(pattern: str, text: str) -> bool:
        p = t = mark = 0
        star = -1
        while t < len(text):
            if p < len(pattern) and pattern[p] == "*":
                star = p
                mark = t
                p += 1
            elif p < len(pattern) and pattern[p] == text[t]:
                p += 1
                t += 1
            elif star >= 0:
                p = star + 1
                mark += 1
                t = mark
            else:
                return False
        while p < len(pattern) and pattern[p] == "*":
            p += 1
        return p == len(pattern)

    def split_strict(text: str, what: str) -> list:
        if not isinstance(text, str) or text == "":
            raise ValueError(f"{what} must be a non-empty string")
        segments = text.split("/")
        if any(segment == "" for segment in segments):
            raise ValueError(f"{what} has an empty segment: {text}")
        return [segment for segment in segments if segment]

    # Validate every rule and path with split_strict, then keep each
    # path the last matching rule allows; a path no rule matches survives.
    filtered_listing = []
    rule_index = 0

    for path in listing:
        path_segments = split_strict(path, "path")
        if not path_segments:
            continue

        while rule_index < len(rules):
            rule = rules[rule_index]
            if rule.strip() == "":
                raise ValueError("Rule cannot be just whitespace")

            keep = rule.startswith("!")
            if keep or segment_fits(rule, path):
                filtered_listing.append(path)
                break
            rule_index += 1

    return filtered_listing
