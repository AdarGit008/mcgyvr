"""Lay a straight eastward conveyor run across a factory floor plan."""


def lay_conveyor(floor: list, row: int, col: int, length: int) -> list:
    if row < 0 or row >= len(floor):
        raise ValueError("the run starts off the plan")
    
    # Ensure the run does not pass the last column
    if col + length > len(floor[0]):
        raise ValueError("run would pass the last column")
    
    # Check that all cells in the run are open
    for i in range(length):
        if floor[row][col + i] != ".":
            raise ValueError("cell is not open")
    
    # Create a copy of the original row
    cells = list(floor[row])
    
    # Write the conveyor run
    for i in range(length):
        cells[col + i] = "="
    
    # Return the modified floor with only the changed row
    return [cells if r == row else floor[r] for r in range(len(floor))]
