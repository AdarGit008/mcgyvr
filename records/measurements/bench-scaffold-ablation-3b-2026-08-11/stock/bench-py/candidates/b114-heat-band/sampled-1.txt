def run_thermostat(start: int, low: int, high: int, power: int, drifts: list) -> dict:
    if isinstance(start, bool) or not isinstance(start, int):
        raise ValueError("start must be an integer")
    if isinstance(low, bool) or not isinstance(low, int) or low >= high:
        raise ValueError("low must lie strictly below high")
    if isinstance(high, bool) or not isinstance(high, int) or low >= high:
        raise ValueError("high must lie strictly below low")
    if isinstance(power, bool) or not isinstance(power, int) or power <= 0:
        raise ValueError("power must be a positive integer")
    if not isinstance(drifts, list):
        raise ValueError("drifts must be a list")

    temp = start
    heated = 0
    switches = 0
    coldest = start
    heater_on = False

    for drift in drifts:
        if isinstance(drift, bool) or not isinstance(drift, int):
            raise ValueError("drifts must be a list of integers")

        if temp < low and not heater_on:
            heated += 1
            heater_on = True
            coldest = min(coldest, start)

        if temp >= high and heater_on:
            heated += 1
            heater_on = False

        temp += drift + (power if heater_on else 0)
        coldest = min(coldest, temp)

    return {"temp": temp, "heated": heated, "switches": switches, "coldest": coldest}
