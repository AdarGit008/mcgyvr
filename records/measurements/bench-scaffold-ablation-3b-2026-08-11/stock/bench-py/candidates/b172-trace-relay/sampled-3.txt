```python
from typing import *
from collections import *

def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict): raise ValueError("links must be a mapping")
    if start not in links:
        raise ValueError("start is not a post")

    route = [start]
    visited = set(route)
    post = start

    while post != '':
        next_post = links.get(post)
        
        if next_post is None or next_post == '' or next_post in visited:
            raise ValueError("Invalid handoff sequence or watch comes back to a previously visited post")
        
        route.append(next_post)
        visited.add(next_post)
        post = next_post

    return route
```
