def split_at(entries: list, marker: str) -> list:
    """
    Break a list in two at the first entry equal to a marker,
    returning what stood before it and what stood after,
    with the marker itself in neither.
    If the marker is not found, returns the entire list and an empty list.
    """
    for i, entry in enumerate(entries):
        if entry == marker:
            return entries[:i], entries[i + 1:]
    return entries, []
