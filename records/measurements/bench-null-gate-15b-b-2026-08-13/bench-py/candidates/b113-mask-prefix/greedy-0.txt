def mask_from_prefix(prefix):
    if isinstance(prefix, bool) or not isinstance(prefix, int) or not 0 <= prefix <= 32:
        raise ValueError("prefix must be an integer from 0 to 32")
    octets = []
    for slot in range(4):
        ones = min(8, max(0, prefix - slot * 8))
        octets.append(str(256 - 2 ** (8 - ones)))
    return ".".join(octets)


def prefix_from_mask(mask):
    if not isinstance(mask, str):
        raise ValueError("mask must be a string")
    parts = mask.split('.')
    if len(parts) != 4:
        raise ValueError("mask must have exactly four dot-separated fields")
    for part in parts:
        if not part.isdigit() or (part.startswith('0') and part[1:] != ''):
            raise ValueError("field must hold decimal digits without leading zeros except for 0 itself")
        octet = int(part)
        if octet > 255:
            raise ValueError("octet must be between 0 and 255")
    # Count the run of leading ones
    count = 0
    for part in parts:
        if part == '0':
            break
        count += len(part)
    return count
