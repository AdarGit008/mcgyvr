from typing import List, Union

def field_mean(records: List[dict], field: str) -> int:
    total = 0
    count = 0
    for record in records:
        if field in record and isinstance(record[field], (int, float)):
            total += record[field]
            count += 1
    if count == 0:
        return 0
    return total // count
