def kettle_hold(readings: list, target: int) -> int:
    """
    Count the number of readings at the very end of the run that sat at or above the target
    without interruption. A run that dips below the target stops the count there.

    :param readings: List of temperature readings in the order they were taken.
    :param target: The target temperature.
    :return: The count of readings that sat at or above the target.
    """
    if not readings:
        return 0

    count = 0
    for reading in readings[::-1]:
        if reading >= target:
            count += 1
        else:
            break

    return count
