def hold_queue(callers: list, limit: int) -> list:
    if limit <= 0:
        raise ValueError("Limit must be a positive integer")
    
    caller_queue = []
    
    for caller in callers:
        caller_queue.append(caller)
        if len(caller_queue) > limit:
            caller_queue.pop(0)
    
    return caller_queue
