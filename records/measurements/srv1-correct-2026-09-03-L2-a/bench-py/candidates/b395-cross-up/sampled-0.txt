def cross_up(readings: list, level: int) -> int:
    """Return how many times a run of readings crosses a level going upward."""
    if not readings or level is None:
        raise ValueError("Input readings or level cannot be None.")
    
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > level > readings[i-1]:
            count += 1
    return count
