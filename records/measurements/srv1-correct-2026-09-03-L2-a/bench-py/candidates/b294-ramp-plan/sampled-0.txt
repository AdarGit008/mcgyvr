def ramp_step(current: int, target: int, step: int) -> int:
    if step == 0:
        raise ValueError("Step cannot be zero.")
    if current == target:
        return target
    if current < target:
        return current + min(step, target - current)
    else:
        return current - min(step, current - target)

def ramp_plan(start: int, target: int, step: int) -> list:
    if step == 0:
        return [start]
    if start == target:
        return [target]
    if start < target:
        return range(start, target + 1, step)
    else:
        return range(start, target - 1, -step)
