def link_find(line: str, opens: str, closes: str) -> list:
    """
    Find all substrings between opening and closing markers.

    Args:
    line (str): The line of text to search within.
    opens (str): The opening marker.
    closes (str): The closing marker.

    Returns:
    list: A list of substrings found between matching markers.
    """
    result = []
    start = -1
    for i, char in enumerate(line):
        if char == opens:
            start = i
        elif char == closes:
            if start != -1:
                result.append(line[start + 1:i])
                start = -1
    return result
