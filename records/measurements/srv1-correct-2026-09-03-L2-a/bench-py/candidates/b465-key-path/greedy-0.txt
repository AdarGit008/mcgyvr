def key_path(store: dict, head: str) -> list:
    """
    Given a store whose names are joined by full stops, return the names standing directly under a stated head,
    without the head and its stop. Names under a deeper head are left out, in the order the store holds them.
    
    :param store: A dictionary where keys are store names and values are lists of items.
    :param head: The head store name to search under.
    :return: A list of names standing directly under the head.
    """
    if head not in store:
        raise ValueError("Head store not found in the store.")
    
    return store[head]
