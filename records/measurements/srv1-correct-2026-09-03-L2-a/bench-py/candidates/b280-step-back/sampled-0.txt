def step_back(moves: str) -> int:
    """
    A walker starts at zero and follows a line of moves, F stepping one forward and B stepping one back.
    Return the furthest forward position reached at any point during the walk. A walk that never goes forward reaches zero,
    and any letter that is neither F nor B is ignored.
    """
    max_forward = 0
    current_position = 0
    for move in moves:
        if move == 'F':
            current_position += 1
        elif move == 'B':
            current_position -= 1
        max_forward = max(max_forward, current_position)
    return max_forward
