def row_key(name: str) -> str:
    """Return the first letter of a name in upper case."""
    if not name:
        raise ValueError("Name cannot be empty")
    return name[0].upper()

def index_rows(names: list) -> dict:
    """Return a dictionary with row keys as keys and lists of names as values."""
    if not names:
        return {}
    index = {}
    for name in names:
        row_key = row_key(name)
        if row_key not in index:
            index[row_key] = []
        index[row_key].append(name)
    return index
