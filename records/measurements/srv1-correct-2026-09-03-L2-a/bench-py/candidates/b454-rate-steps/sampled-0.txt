def step_rate(amount: int, fixed: int, percent: int) -> int:
    """
    Returns the charge for one step: a fixed sum plus a whole-percent share of the amount, rounded down.
    
    Parameters:
    - amount: int, the total amount to be charged.
    - fixed: int, the fixed part of the charge.
    - percent: int, the percentage share of the amount.
    
    Returns:
    - int, the total charge for one step.
    """
    return fixed + (amount * percent) // 100

def rate_steps(amounts: list, fixed: int, percent: int) -> list:
    """
    Returns the charge for each of several amounts, in order.
    
    Parameters:
    - amounts: list of int, the amounts for which to calculate the charge.
    - fixed: int, the fixed part of the charge.
    - percent: int, the percentage share of the amount.
    
    Returns:
    - list of int, the charges for each amount.
    """
    return [fixed + (amount * percent) // 100 for amount in amounts]
