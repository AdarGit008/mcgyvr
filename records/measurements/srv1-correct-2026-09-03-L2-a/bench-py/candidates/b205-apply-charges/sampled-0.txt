def _why_refused(cap, left: int, amount: int) -> str:
    if cap is None:
        return "unknown"
    if amount > cap[1]:
        return "single"
    return "cap" if amount > left else ""


def apply_charges(caps: dict, charges: list) -> dict:
    left = {name: cap[0] for name, cap in caps.items()}
    refused = []
    for _, bucket, amount in charges:
        if amount <= 0 or not isinstance(amount, int):
            raise ValueError("Invalid amount")
        if bucket not in caps:
            refused.append((amount, "unknown"))
            continue
        cap = caps[bucket]
        reason = _why_refused(cap, left[bucket], amount)
        if reason:
            refused.append((amount, reason))
        else:
            left[bucket] -= amount
    return {"left": left, "refused": refused}
