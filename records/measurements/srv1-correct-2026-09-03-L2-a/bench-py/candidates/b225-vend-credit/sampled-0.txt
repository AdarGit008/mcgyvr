```python
def vend_credit(coins: list, price: int) -> int:
    if not coins:
        raise ValueError("No coins were inserted.")
    
    if not isinstance(price, int) or price < 0 or price % 5 != 0:
        raise ValueError("Invalid price. It must be a whole number that is a multiple of 5.")
    
    credit = 0
    items_dropped = 0
    
    for coin in coins:
        if coin not in [5, 10, 25, 100]:
            raise ValueError("Invalid coin inserted. Only 5, 10, 25, and 100 cents are accepted.")
        credit += coin
        while credit >= price:
            items_dropped += 1
            credit -= price
    
    return credit
```
