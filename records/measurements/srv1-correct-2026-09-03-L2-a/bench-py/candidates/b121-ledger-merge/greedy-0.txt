def merge_ledgers(base: list, ours: list, theirs: list) -> list:
    if not isinstance(base, list) or not isinstance(ours, list) or not isinstance(theirs, list):
        raise ValueError("All arguments must be lists.")
    
    merged = []
    for account, cents in base:
        if account in ours and account in theirs:
            merged.append([account, cents + ours[account] + theirs[account]])
        elif account in ours:
            merged.append([account, cents + ours[account]])
        elif account in theirs:
            merged.append([account, cents + theirs[account]])
        else:
            merged.append([account, cents])
    
    for account, cents in ours:
        if account not in base and account not in theirs:
            merged.append([account, cents])
    
    for account, cents in theirs:
        if account not in base and account not in ours:
            merged.append([account, cents])
    
    merged.sort(key=lambda x: x[0])
    return merged
