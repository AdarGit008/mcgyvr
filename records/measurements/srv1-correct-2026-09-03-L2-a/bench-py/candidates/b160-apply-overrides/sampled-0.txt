def apply_overrides(base: dict, overrides: list) -> dict:
    if not isinstance(overrides, list):
        raise ValueError("overrides must be a list")
    for override in overrides:
        if not isinstance(override, str):
            raise ValueError("override must be a string")
        if '=' not in override:
            raise ValueError("override must have exactly one equals sign")
        name, value = override.split('=', maxsplit=1)
        if not name:
            raise ValueError("setting name cannot be empty")
        if name not in base:
            raise ValueError("setting name must be in the base")
        base[name] = value
    return base
