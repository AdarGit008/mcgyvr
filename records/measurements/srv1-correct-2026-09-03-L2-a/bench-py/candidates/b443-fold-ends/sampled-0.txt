def fold_ends(values: list) -> list:
    """
    Fold a list onto itself by adding the first entry to the last, the second to the second-last,
    and so on, returning the totals in that order. A list of an odd length leaves its middle entry as its own total.
    
    :param values: List of integers to be folded
    :return: List of totals from the folded list
    """
    if not values:
        return []
    
    total = 0
    result = []
    
    for i in range(len(values) // 2):
        total += values[i] + values[-i - 1]
        result.append(total)
    
    # If the list has an odd number of elements, add the middle element as its own total
    if len(values) % 2 != 0:
        result.append(values[len(values) // 2])
    
    return result
