def trace_relay(links: dict, start: str) -> list:
    if not isinstance(links, dict):
        raise ValueError("links must be a mapping")
    if start not in links:
        raise ValueError("start is not a post")
    route = []
    post = start
    seen = set()
    while post != "" and post not in seen:
        seen.add(post)
        route.append(post)
        post = links.get(post, "")
    if post in seen:
        raise ValueError("run of handoffs that comes back to a post already visited")
    return route
