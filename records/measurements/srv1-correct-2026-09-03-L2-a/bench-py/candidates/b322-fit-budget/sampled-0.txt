```python
def fit_budget(prices: list, budget: int) -> int:
    if budget < 0:
        raise ValueError("Budget cannot be negative")
    
    prices.sort()
    total_items = 0
    current_budget = budget
    
    for price in prices:
        if current_budget >= price:
            current_budget -= price
            total_items += 1
        else:
            break
    
    return total_items
```
