export function fitBudget(prices: number[], budget: number): number {
  if (budget < 0) {
    throw new Error("Amount cannot be negative");
  }

  let totalSpent = 0;
  let count = 0;

  for (const price of prices) {
    if (totalSpent + price > budget) break;
    totalSpent += price;
    count++;
  }

  return count;
}
