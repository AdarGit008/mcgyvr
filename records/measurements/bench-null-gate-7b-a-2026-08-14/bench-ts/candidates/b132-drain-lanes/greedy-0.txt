/** Drain labelled items across planned lanes, round by round. */
export function drainLanes(
  plan: [string, number][],
  items: [string, string][],
): { order: string[]; rounds: number } {
  if (plan.length === 0) {
    throw new Error("Plan cannot be empty");
  }

  const laneSet = new Set<string>();
  for (const [lane] of plan) {
    if (typeof lane !== "string" || lane.trim() === "" || !Number.isInteger(lane[1]) || lane[1] <= 0) {
      throw new Error("Plan entries must be a non-empty string lane and a positive integer quota");
    }
    if (laneSet.has(lane)) {
      throw new Error("Lane declared twice in the plan");
    }
    laneSet.add(lane);
  }

  const queues = new Map<string, string[]>();
  for (const [lane] of plan) {
    queues.set(lane, []);
  }

  for (const [label, lane] of items) {
    if (typeof label !== "string" || typeof lane !== "string" || label.trim() === "" || lane.trim() === "") {
      throw new Error("Items must be a pair of two strings");
    }
    const queue = queues.get(lane);
    if (queue !== undefined) {
      queue.push(label);
    }
  }

  const order: string[] = [];
  let rounds = 0;
  while (true) {
    rounds += 1;
    for (const [lane, quota] of plan) {
      const queue = queues.get(lane) as string[];
      let taken = 0;
      while (taken < quota && queue.length > 0) {
        order.push(queue.shift() as string);
        taken += 1;
      }
      if (queue.length > 0) {
        return { order, rounds };
      }
    }
  }
}
