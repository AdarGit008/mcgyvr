def merge_settings(base: dict, ours: dict, theirs: dict) -> dict:
    if not all(isinstance(k, str) and isinstance(v, str) for k, v in base.items()):
        raise ValueError("Base must be a flat mapping of string keys to string values")
    if not all(isinstance(k, str) and isinstance(v, str) for k, v in ours.items()):
        raise ValueError("Ours must be a flat mapping of string keys to string values")
    if not all(isinstance(k, str) and isinstance(v, str) for k, v in theirs.items()):
        raise ValueError("Theirs must be a flat mapping of string keys to string values")

    result = {}
    for key in set(base).union(ours, theirs):
        base_value = base.get(key)
        ours_value = ours.get(key)
        theirs_value = theirs.get(key)

        if base_value == ours_value == theirs_value:
            result[key] = base_value
        elif (base_value is None and ours_value is not None) or (base_value is not None and ours_value is None):
            result[key] = ours_value
        elif (base_value is None and theirs_value is not None) or (base_value is not None and theirs_value is None):
            result[key] = theirs_value
        else:
            raise ValueError(f"Conflict at key {key}: cannot merge different edits")

    return result
