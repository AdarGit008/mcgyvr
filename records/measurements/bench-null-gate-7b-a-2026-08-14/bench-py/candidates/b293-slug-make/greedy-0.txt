```python
import re

def slug_word(word: str) -> str:
    """Lowercase a word and remove non-alphanumeric characters."""
    return ''.join(filter(str.isalnum, word.lower()))

def slug_make(phrase: str) -> str:
    """Join slugged words of a phrase with single hyphens, excluding empty slugs."""
    slugged_words = [slug_word(word) for word in phrase.split()]
    return '-'.join(word for word in slugged_words if word)
```
