def plural_of(noun: str) -> str:
    """Return the plural of an English noun."""
    if noun.endswith('s') or noun.endswith('x') or noun.endswith('ch') or noun.endswith('sh'):
        return noun + 'es'
    elif noun.endswith('y') and noun[-2] not in 'aeiou':
        return noun[:-1] + 'ies'
    else:
        return noun + 's'
