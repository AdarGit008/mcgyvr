def dur_hours(minutes: int) -> int:
    """Return the whole hours in a count of minutes."""
    if minutes < 0:
        raise ValueError("Minutes cannot be negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    """Write the count as hours and minutes joined, like 1h30m."""
    if minutes < 0:
        raise ValueError("Minutes cannot be negative")
    hours = minutes // 60
    mins = minutes % 60
    if hours == 0:
        return f"{mins}m"
    elif mins == 0:
        return f"{hours}h"
    else:
        return f"{hours}h{mins:02d}m"
