def step_cost(kind: str) -> int:
    """Return the cost of one step of a given kind."""
    if kind == "walk":
        return 1
    elif kind == "run":
        return 2
    else:
        raise ValueError("Invalid step kind")

def path_toll(steps: list[str]) -> int:
    """Total the toll for a path, with every third step free."""
    total_cost = 0
    for i, step in enumerate(steps):
        if (i + 1) % 3 != 0:
            total_cost += step_cost(step)
    return total_cost
