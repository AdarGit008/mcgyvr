"""Render a value as fixed-width unsigned binary, nibble-grouped."""


def format_bits(value: int, width: int) -> str:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError("value must be a non-negative integer")
    
    if width <= 0 or width % 4 != 0 or width > 32:
        raise ValueError("width must be a positive multiple of four and less than or equal to 32")
    
    binary_str = bin(value)[2:]  # Convert value to binary string, remove '0b' prefix
    if len(binary_str) > width:
        raise ValueError(f"value {value} does not fit in {width} bits")
    
    # Pad the binary string with leading zeros to reach the desired width
    padded_binary = binary_str.zfill(width)
    
    # Split the padded binary string into groups of four bits and join them with spaces
    nibble_groups = [padded_binary[i:i+4] for i in range(0, len(padded_binary), 4)]
    
    return ' '.join(nibble_groups)
