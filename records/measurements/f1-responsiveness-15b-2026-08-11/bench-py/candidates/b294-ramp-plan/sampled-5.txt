def ramp_step(current: int, target: int, step: int) -> int:
    if step == 0:
        raise ValueError("Step cannot be zero.")
    return min(target, max(current + step, target - step))

def ramp_plan(start: int, target: int, step: int) -> list:
    result = []
    while start < target or start > target:
        result.append(start)
        start += step
    return result
