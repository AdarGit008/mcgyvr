def hex_split(colour: str) -> list:
    if len(colour) != 7 or colour[0] != '#':
        raise ValueError("Invalid input. The colour must be a hash followed by six hexadecimal digits.")
    
    return [colour[i:i+2].lower() for i in range(1, 7, 2)]

def hex_join(parts: list) -> str:
    if len(parts) != 3:
        raise ValueError("Invalid input. Must provide exactly three parts.")
    
    return '#' + ''.join(parts)
