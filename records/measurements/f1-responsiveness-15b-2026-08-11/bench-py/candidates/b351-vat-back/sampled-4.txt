def vat_back(gross: int, rate: int) -> int:
    if rate < 0:
        raise ValueError("Rate must be non-negative.")
    
    # Calculate the amount of tax by multiplying gross by the rate
    tax = (rate / 100) * gross
    
    # Subtract the total tax from the gross to find the price before tax
    pre_tax_price = gross - tax
    
    # Return the result, rounding down to the nearest integer
    return int(pre_tax_price)
