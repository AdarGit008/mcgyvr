def vat_back(gross: int, rate: int) -> int:
    if rate < 0:
        raise ValueError("Rate must be non-negative")
    
    # Calculate the pre-tax price by subtracting the tax from the gross
    pre_tax_price = gross / (1 + rate / 100)
    
    # Return the pre-tax price rounded down to the nearest integer
    return int(pre_tax_price)
