def vat_back(gross: int, rate: int) -> int:
    if rate < 0:
        raise ValueError("Rate must be non-negative")
    
    # Calculate the amount of tax paid
    tax = gross * (rate / 100)
    
    # Subtract the tax from the gross price to get the original price before VAT was added
    original_price = gross - tax
    
    # Round down the original price to the nearest integer
    return int(original_price)
