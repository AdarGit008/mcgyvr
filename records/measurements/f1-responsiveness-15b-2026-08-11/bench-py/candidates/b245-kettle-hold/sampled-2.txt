def kettle_hold(readings: list, target: int) -> int:
    if not readings or target <= 0:
        return 0

    holding_count = 1
    for i in range(1, len(readings)):
        if readings[i] >= target and readings[i-1] < target:
            holding_count += 1
        elif readings[i] < target:
            break

    return holding_count
