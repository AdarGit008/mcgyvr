```python
from typing import List

def kettle_hold(readings: List[int], target: int) -> int:
    if not readings or target < 0:
        raise ValueError("Invalid input")

    counter = 0
    for i in range(len(readings) - 1, -1, -1):
        if readings[i] >= target:
            counter += 1
        else:
            break

    return counter
```
