from typing import *

def hide_values(store: dict) -> dict:
    """A store with every value starred out."""
    hidden = {}
    for key, value in store.items():
        if isinstance(value, str):
            hidden[value] = "*" * len(value)
        elif isinstance(value, list):
            hidden[key] = [f"*{item}*" if isinstance(item, str) else item for item in value]
        else:
            hidden[key] = "*" * len(str(value))
    return hidden
