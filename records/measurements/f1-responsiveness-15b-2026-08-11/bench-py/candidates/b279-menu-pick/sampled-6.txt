```python
from typing import List

def menu_pick(dishes: List[tuple[str, int]], budget: int) -> List[str]:
    # Filter dishes that cost no more than the budget
    affordable_dishes = [dish for dish in dishes if dish[1] <= budget]
    
    # Sort dishes by price and then by name (ascending order)
    affordable_dishes.sort(key=lambda x: (x[1], x[0]))
    
    return [dish[0] for dish in affordable_dishes]
```
