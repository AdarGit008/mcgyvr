def gross_price(net: int, rate: int) -> int:
    """A net price with a whole-percent tax added, in whole pence."""
    if rate == 0:
        return net
    elif rate > 100:
        raise ValueError("Rate cannot be greater than 100%")
    
    tax_pence = (rate / 100) * net
    tax_pence = int(tax_pence)
    return net + tax_pence
