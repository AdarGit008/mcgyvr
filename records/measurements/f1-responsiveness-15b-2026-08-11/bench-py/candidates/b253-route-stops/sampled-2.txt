```python
from typing import List

def route_stops(route: str) -> List[str]:
    return [stop.strip() for stop in route.split('>') if stop.strip()]

def route_hops(route: str) -> int:
    stops = route_stops(route)
    if not stops:
        return 0
    return len(stops) - 1 if len(stops) > 1 else 1
```
