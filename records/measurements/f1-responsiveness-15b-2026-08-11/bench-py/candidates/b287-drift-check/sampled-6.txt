from typing import *
from collections import *

def drift_check(readings: list, step: int) -> int:
    if not readings:
        return -1

    for i in range(1, len(readings)):
        current_reading = readings[i]
        previous_reading = readings[i - 1]

        if abs(current_reading - previous_reading) > step:
            return i

    return -1
