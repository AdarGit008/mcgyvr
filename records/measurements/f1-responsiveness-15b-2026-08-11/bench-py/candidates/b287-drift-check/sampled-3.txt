```python
class Solution:
    def drift_check(self, readings: list, step: int) -> int:
        if len(readings) <= 1:
            return -1
        
        prev_value = readings[0]
        
        for i in range(1, len(readings)):
            current_value = readings[i]
            
            diff = abs(current_value - prev_value)
            
            if diff > step:
                return i
            
            prev_value = current_value
        
        return -1
```
