def run_decode(coded: str) -> str:
    decoded = ""
    i = 0
    while i < len(coded):
        letter = coded[i]
        count = ""
        if i + 1 < len(coded) and coded[i + 1].isdigit():
            j = i + 1
            while j < len(coded) and coded[j].isdigit():
                count += coded[j]
                j += 1
            i = j - 1
        decoded += letter * int(count or 1)
    return decoded
