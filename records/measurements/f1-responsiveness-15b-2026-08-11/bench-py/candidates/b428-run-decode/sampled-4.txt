def run_decode(coded: str) -> str:
    decoded = []
    i = 0
    while i < len(coded):
        letter = coded[i]
        if letter.isdigit():
            multiplier = int(letter)
            i += 1
            while i < len(coded) and coded[i].isdigit():
                multiplier = multiplier * 10 + int(coded[i])
                i += 1
            decoded.extend([letter] * multiplier)
        else:
            decoded.append(letter)
            i += 1
    return ''.join(decoded)
