def tide_marks(levels: list) -> list:
    if not levels:
        return []

    peaks_positions = []
    n = len(levels)

    # Check for peaks from the second element to the second last element
    for i in range(1, n - 1):
        if levels[i] > levels[i - 1] and levels[i] > levels[i + 1]:
            peaks_positions.append(i)

    return peaks_positions
