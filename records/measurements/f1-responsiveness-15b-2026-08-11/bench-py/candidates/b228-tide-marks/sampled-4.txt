```python
from typing import *
from collections import *

def tide_marks(levels: List[int]) -> List[int]:
    n = len(levels)
    if n < 3:
        return []
    
    peaks = []
    
    for i in range(1, n - 1):
        if levels[i] > levels[i-1] and levels[i] > levels[i+1]:
            peaks.append(i)
    
    return peaks
```
