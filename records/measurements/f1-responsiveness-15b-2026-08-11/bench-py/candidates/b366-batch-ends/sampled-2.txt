def batch_ends(count: int, size: int) -> list:
    """The position of the last entry of each full batch."""
    if count <= 0 or size <= 0:
        raise ValueError("Invalid input: count and size must be positive integers.")
    
    ends = []
    start = 0
    while start + size < count:
        ends.append(start)
        start += size
    # No need to append the last batch if it's empty
    return ends[:-1] if len(ends) > 0 else ends
