```python
def strip_tags(line: str) -> str:
    """Remove every marker from a line of text, a marker being an opening angle bracket, everything up to the next closing bracket."""
    
    result = ""
    in_tag = False
    
    for char in line:
        if char == '<':
            in_tag = True
        elif char == '>':
            in_tag = False
            break
        
        if not in_tag:
            result += char
    
    return result
```
