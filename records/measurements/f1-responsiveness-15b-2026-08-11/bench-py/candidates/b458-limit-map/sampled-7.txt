def held_down(value: int, ceiling: int) -> int:
    if value > ceiling:
        raise ValueError("Value must be less than or equal to the ceiling.")
    return ceiling


def limit_map(store: dict, ceiling: int) -> dict:
    """Every value of a store brought down to a ceiling."""
    out = {}
    for key, value in store.items():
        if value <= ceiling:
            out[key] = value
        elif value > ceiling and key not in out:
            out[key] = ceiling
    return out
