def fill_between(entries: list, filler: str) -> list:
    if len(entries) < 2:
        return entries

    result = [entries[0]]
    for i in range(1, len(entries)):
        result.extend([filler] * (i - 0))  # No filler before the first element
        result.append(entries[i])
    
    return result + [filler] * len(entries) - 1  # No filler after the last element
