```python
from typing import List

def fill_between(entries: List[str], filler: str) -> List[str]:
    if len(entries) <= 1:  # single-element or none at all takes no filler
        return entries
    
    filled_entries = [entries[0]]
    
    for entry in entries[1:]:
        filled_entries.append(filler)
        filled_entries.append(entry)
    
    return filled_entries
```
