def slug_word(word: str) -> str:
    """Lowercase a word and drop all non-alphanumeric characters."""
    return ''.join(filter(str.isalnum, word.lower()))

def slug_make(phrase: str) -> str:
    """
    Join slugged words of a phrase with single hyphens,
    leaving out any word that slugs away to nothing.
    """
    words = [slug_word(word) for word in phrase.split()]
    return '-'.join([word for word in words if word])
