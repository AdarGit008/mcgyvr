```python
from typing import *
from collections import *

def slug_word(word: str) -> str:
    """Lowercase a word and remove non-alphanumeric characters."""
    return ''.join(filter(str.isalnum, word.lower()))

def slug_make(phrase: str) -> str:
    """Join slugged words of a phrase with a hyphen, omitting empty ones."""
    slugs = [slug_word(word) for word in phrase.split()]
    return '-'.join(slug for slug in slugs if slug)
```
