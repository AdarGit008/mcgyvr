def slug_word(word: str) -> str:
    """Lowercase a word and drop all characters that are not letters or digits."""
    return ''.join(c.lower() if c.isalnum() else '' for c in word)

def slug_make(phrase: str) -> str:
    """Join the slugged words of a phrase with single hyphens, leaving out any word that slugs away to nothing."""
    slugs = [slug_word(word) for word in phrase.split()]
    return '-'.join(slug for slug in slugs if slug)
