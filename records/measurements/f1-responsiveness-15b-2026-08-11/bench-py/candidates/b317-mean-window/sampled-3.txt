```python
def mean_window(readings: list, size: int) -> list:
    if size == 0:
        return []
    
    if len(readings) < size:
        return []
    
    means = []
    for i in range(len(readings) - size + 1):
        window_readings = readings[i:i+size]
        mean = sum(window_readings) // size
        means.append(mean)
    
    return means
```
