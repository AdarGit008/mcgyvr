def dur_hours(minutes: int) -> int:
    if minutes < 0:
        raise ValueError("Duration cannot be negative")
    return minutes // 60

def dur_text(minutes: int) -> str:
    hours = minutes // 60
    minutes %= 60
    if hours == 0:
        return f"{minutes}m"
    elif minutes == 1:
        return f"{hours}h{minutes}m"
    else:
        return f"{hours}h{minutes}m"
