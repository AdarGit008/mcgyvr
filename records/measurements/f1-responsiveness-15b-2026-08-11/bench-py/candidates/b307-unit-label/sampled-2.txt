from typing import *
from collections import *

def short_unit(unit: str) -> str:
    return unit[:3]


def label_of(amount: int, unit: str) -> str:
    """An amount written with its unit."""
    if amount < 0:
        raise ValueError("amount cannot be negative")
    if amount == 1 or unit.endswith('s'):
        return f"{amount} {unit}"
    else:
        return f"{amount} {unit}s"
