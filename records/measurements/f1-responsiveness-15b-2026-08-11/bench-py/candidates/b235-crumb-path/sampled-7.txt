```python
from typing import List

def crumb_split(trail: str) -> List[str]:
    """
    Split a breadcrumb trail on forward slashes into segments, discarding empty ones.

    Args:
        trail (str): The breadcrumb trail to split.

    Returns:
        list: A list of non-empty segments.
    """
    return [part for part in trail.split('/') if part]

def crumb_join(parts: List[str]) -> str:
    """
    Join a list of segments into a single breadcrumb trail, combining them with a single slash between them.

    Args:
        parts (list): The list of segments to join.

    Returns:
        str: The joined breadcrumb trail.
    """
    return '/'.join(part for part in parts if part)
```
