```python
def count_by(records: list, field: str) -> dict:
    if not field:
        raise ValueError("Field name cannot be empty.")

    result = {}
    for record in records:
        value = record.get(field)
        if value is None:
            continue
        if value not in result:
            result[value] = 1
        else:
            result[value] += 1

    return result
```
