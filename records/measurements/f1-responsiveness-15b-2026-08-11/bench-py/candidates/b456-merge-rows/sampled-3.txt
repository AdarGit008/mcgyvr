def row_keys(row: dict) -> list:
    return list(row.keys())


def merge_rows(under: dict, over: dict) -> dict:
    """One row laid over another, the second row's values win where both hold a name."""
    merged = dict(over)
    for key in row_keys(under):
        if key in merged and merged[key] != under[key]:
            del merged[key]
    return merged
