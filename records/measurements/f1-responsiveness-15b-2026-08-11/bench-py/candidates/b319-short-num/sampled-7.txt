def short_num(value: int) -> str:
    """Convert a number to its shortened form."""
    
    if value < 100:
        return str(value)
    
    if value >= 1000 and value < 1000000:
        return f"{value // 100}k"
    
    if value >= 1000000 and value < 1_000_000_000:
        return f"{value // 1_000_000}m"
    
    raise ValueError("Value must be less than a million")
