def size_unit(bytes_count: int) -> str:
    if bytes_count < 1024:
        return "B"
    elif bytes_count < 1024 * 1024:
        return "KB"
    else:
        return "MB"


def size_text(bytes_count: int) -> str:
    """A byte count written in whole units."""
    if bytes_count < 0:
        raise ValueError("a byte count cannot be negative")
    unit = size_unit(bytes_count)
    if unit == "B":
        return f"{bytes_count} B"
    elif unit == "KB":
        return f"{bytes_count // 1024:.2f} KB"
    else:
        return f"{bytes_count // (1024 * 1000):.2f} MB"
