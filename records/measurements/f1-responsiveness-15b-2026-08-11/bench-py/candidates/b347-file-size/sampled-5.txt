def size_unit(bytes_count: int) -> str:
    if bytes_count < 1024:
        return "B"
    if bytes_count < 1024 * 1024:
        return "KB"
    return "MB"


def size_text(bytes_count: int) -> str:
    """A byte count written in whole units."""
    if bytes_count < 0:
        raise ValueError("a byte count cannot be negative")
    unit = size_unit(bytes_count)
    if unit == "B":
        return str(bytes_count) + " B"
    if unit == "KB":
        return str(bytes_count // 1000) + " KB"
    # If bytes_count is between 1MB and less than 1GB, it should be rounded to the nearest MB
    elif 1024 * 1024 <= bytes_count < 1024 ** 3:
        return str(bytes_count // (1024 * 1024)) + " MB"
    else:
        # For counts of 1GB or more, it should be rounded to the nearest GB
        return str(bytes_count // (1024 ** 3)) + " GB"
