```python
from typing import List

def step_value(step: str) -> int:
    if step == "+":
        return 1
    if step == "-":
        return -1
    return 0


def scan_tally(steps: List[str]) -> List[int]:
    """The running total after each instruction."""
    total = 0
    for step in steps:
        total += step_value(step)
    return [total[-1]]
```
