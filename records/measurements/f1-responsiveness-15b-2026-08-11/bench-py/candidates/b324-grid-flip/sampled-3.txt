```python
from typing import *
from collections.abc import *

def grid_flip(grid: list) -> List[List[int]]:
    if not grid:
        return []

    flipped_grid = [list(row) for row in zip(*grid)]
    return flipped_grid
```

Tests (run with `pytest -s solution.py`):
```python
from pytest import *
import solution

class TestGridFlip:

    def test_empty_grid(self):
        assert solution.grid_flip([]) == []

    def test_single_element_grid(self):
        assert solution.grid_flip([1]) == [[1]]

    def test_multiple_elements_grid(self):
        assert solution.grid_flip([[1, 2], [3, 4]]) == [[1, 3], [2, 4]]

    def test_zero_rows(self):
        assert solution.grid_flip([]) == []

    def test_single_row(self):
        assert solution.grid_flip([[5]]) == [[5]]

    def test_duplicate_values(self):
        assert solution.grid_flip([[1, 1, 1], [2, 2, 2]]) == [[1, 2], [1, 2], [1, 2]]
```
