from typing import List


def wave_count(readings: List) -> int:
    count = 0

    for i in range(1, len(readings)):
        if readings[i - 1] < readings[i]:
            # Rise followed by fall
            count += 1
        elif readings[i - 1] > readings[i]:
            # Fall followed by rise
            count += 1

    return count
