def wave_count(readings: list) -> int:
    changes = 0
    for i in range(1, len(readings)):
        if (readings[i - 1] < readings[i] and readings[i] > readings[i + 1]) or \
           (readings[i - 1] > readings[i] and readings[i] < readings[i + 1]):
            changes += 1
    return changes
