def wave_count(readings):
    if not readings:
        return 0

    count = 0
    for i in range(len(readings) - 1):
        if (readings[i] > readings[i + 1]) and (readings[i + 1] < readings[i + 2]):
            count += 1
        elif (readings[i] < readings[i + 1]) and (readings[i + 1] > readings[i + 2]):
            count += 1

    return count
