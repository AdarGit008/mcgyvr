from typing import List

def wave_count(readings: List[int]) -> int:
    if not readings or len(readings) < 2:
        return 0
    
    count = 0
    for i in range(1, len(readings)):
        if (readings[i] > readings[i-1] and readings[i-1] <= 0) or \
           (readings[i] < readings[i-1] and readings[i-1] >= 0):
            count += 1
    
    return count
