def swap_pair(first: str, second: str) -> str:
    return second + first

def swap_all(text: str) -> str:
    result = ''
    i = 0
    # Swap pairs while there are at least two characters left
    while i < len(text) - 1:
        result += text[i+1] + text[i]
        i += 2
    # Add the last character if it's not part of a pair
    if i == len(text) - 1:
        result += text[-1]
    return result
