def swap_pair(first: str, second: str) -> str:
    """Return two characters from the input strings swapped."""
    return second + first

def swap_all(text: str) -> str:
    """Swap each pair of adjacent characters in the text."""
    if not text:
        return ""
    
    result = ""
    for i in range(0, len(text), 2):
        if i < len(text) - 1:
            result += text[i+1] + text[i]
        else:
            result += text[i]
    
    return result
