from typing import List

def poll_gaps(minutes: List[int]) -> List[int]:
    if len(minutes) < 2:
        return []
    
    first_run = minutes[0]
    last_run = minutes[-1]
    gaps = []
    
    for i in range(1, len(minutes)):
        if minutes[i] != minutes[i-1]:  # Check for duplicate runs
            gaps.append(minutes[i] - minutes[i-1])
    
    return sorted(gaps)
