from typing import List


def poll_gaps(minutes: List[int]) -> List[int]:
    if len(minutes) < 2:
        return []

    min_time = minutes[0]
    max_time = minutes[-1]

    gaps = []
    for i in range(1, len(minutes)):
        gap = minutes[i] - minutes[i - 1]
        if gap > 1:
            gaps.append(gap)

    return sorted(gaps)
