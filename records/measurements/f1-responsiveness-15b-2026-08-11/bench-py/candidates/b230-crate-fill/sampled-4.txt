```python
from typing import List

def crate_fill(items: int, crates: int) -> List[int]:
    if items == 0:
        return []
    
    # Calculate base size and remainder of items to distribute
    base_size = items // crates
    remainder = items % crates
    
    # Generate the list of sizes for each crate
    sizes = [base_size] * crates
    for i in range(remainder):
        sizes[i] += 1
    
    return sizes

def crate_total(sizes: List[int]) -> int:
    return sum(sizes)
```
