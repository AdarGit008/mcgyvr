```python
from typing import List

def crate_fill(items: int, crates: int) -> List[int]:
    if items <= 0:
        return []

    item_size = items // crates
    remainder = items % crates

    sizes = [item_size] * crates
    for i in range(remainder):
        sizes[i] += 1

    return sizes

def crate_total(sizes: List[int]) -> int:
    return sum(sizes)
```
