def payout(stake: int, odds: int) -> int:
    if stake <= 0:
        raise ValueError("Stake must be positive")
    return stake * odds

def settle_all(bets: list) -> int:
    total = 0
    for bet in bets:
        if bet.stake > 0 and bet.odds > 0:
            total += payout(bet.stake, bet.odds)
    return total
