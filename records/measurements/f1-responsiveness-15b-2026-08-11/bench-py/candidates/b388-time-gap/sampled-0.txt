from typing import Tuple

def time_gap(from_hour: int, from_minute: int, to_hour: int, to_minute: int) -> int:
    # Convert hours and minutes to total minutes since midnight
    from_minutes = (from_hour * 60) + from_minute
    to_minutes = (to_hour * 60) + to_minute
    
    # If to_time is less than or equal to from_time, consider it as the following day
    if to_minutes <= from_minutes:
        return (24 * 60) - from_minutes + to_minutes
    else:
        return to_minutes - from_minutes
