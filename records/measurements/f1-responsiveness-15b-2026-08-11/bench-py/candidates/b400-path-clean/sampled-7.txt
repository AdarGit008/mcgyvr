```python
from typing import *
from collections import *

def path_clean(path: str) -> str:
    """Remove every segment of two dots by dropping it together with the segment standing before it, and return what remains joined by slashes. Two dots with nothing before them are dropped alone."""
    cleaned_path = []
    for i in range(len(path)):
        if path[i] == '.' and (i == 0 or path[i - 1] != '/'):
            continue
        cleaned_path.append(path[i])
    return ''.join(cleaned_path)
```
