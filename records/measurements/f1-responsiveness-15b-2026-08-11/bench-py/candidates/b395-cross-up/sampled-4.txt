from typing import List

def cross_up(readings: List[int], level: int) -> int:
    """
    Count how many times the readings cross a level going upward.

    Args:
        readings (List[int]): A list of integers representing sensor readings.
        level (int): The level to cross.

    Returns:
        int: The number of times the readings cross the level going upward.
    """
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > level and readings[i-1] <= level:
            count += 1
    return count
