from typing import List

def cross_up(readings: List[int], level: int) -> int:
    """
    Counts the number of times a reading crosses a level going upward.

    Args:
        readings (List[int]): The list of readings to analyze.
        level (int): The target level for crossing.

    Returns:
        int: The count of readings that cross the specified level going upward.
    """
    count = 0
    for i in range(1, len(readings)):
        if readings[i] > level and readings[i - 1] <= level:
            count += 1
    return count
