from typing import List

def cross_up(readings: List[int], level: int) -> int:
    if len(readings) == 0:
        return 0
    
    crossings = 0
    for i in range(1, len(readings)):
        if readings[i] > level and readings[i - 1] <= level:
            crossings += 1
    
    return crossings
