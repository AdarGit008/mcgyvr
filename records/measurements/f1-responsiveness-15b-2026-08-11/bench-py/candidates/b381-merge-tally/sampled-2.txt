def add_one(tally: dict, name: str, amount: int) -> dict:
    if name in tally:
        tally[name] += amount
    else:
        tally[name] = amount
    return tally


def merge_tally(left: dict, right: dict) -> dict:
    for key in left:
        if key not in right:
            right[key] = left[key]
        else:
            right[key] += left[key]
    for key in right:
        if key not in left:
            left[key] = right[key]
        else:
            left[key] += right[key]
    return left
