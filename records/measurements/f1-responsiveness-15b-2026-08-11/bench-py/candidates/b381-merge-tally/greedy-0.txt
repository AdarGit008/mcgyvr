def add_one(tally: dict, name: str, amount: int) -> dict:
    if name in tally:
        tally[name] += amount
    else:
        tally[name] = amount
    return tally

def merge_tally(left: dict, right: dict) -> dict:
    merged = left.copy()
    for name, count in right.items():
        if name in merged:
            merged[name] += count
        else:
            merged[name] = count
    return merged
