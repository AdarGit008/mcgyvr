def add_one(tally: dict, name: str, amount: int) -> dict:
    if name in tally:
        tally[name] += amount
    else:
        tally[name] = amount
    return tally

def merge_tally(left: dict, right: dict) -> dict:
    for key in left:
        if key in right:
            left[key] += right[key]
    for key in right:
        if key not in left:
            left[key] = right[key]
    return left
