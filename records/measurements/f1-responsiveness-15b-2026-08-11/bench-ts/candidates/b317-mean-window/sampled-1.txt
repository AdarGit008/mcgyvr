function meanWindow(readings: number[], size: number): number[] {
  if (size <= 0) {
    throw new Error('Window size must be positive');
  }

  const means = [];
  for (let i = 0; i < readings.length - size + 1; ++i) {
    const windowReadings = readings.slice(i, i + size);
    const sum = windowReadings.reduce((acc, val) => acc + val, 0);
    const mean = Math.floor(sum / windowReadings.length);
    means.push(mean);
  }

  return means;
}
