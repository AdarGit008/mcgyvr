export function takeBack(values: number[], target: number): number {
  if (values.length === 0 || target <= 0) {
    return -1;
  }

  let totalSum = 0;

  for (let i = values.length - 1; i >= 0; i--) {
    totalSum += values[i];
    if (totalSum >= target) {
      return values.length - i - 1;
    }
  }

  return -1;
}
