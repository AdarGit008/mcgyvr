function fieldOf(record: Record<string, number>, field: string): number {
    if (!record.hasOwnProperty(field)) {
        throw new Error(`Record does not have field '${field}'.`);
    }
    return record[field];
}

function sortPairs(records: Record<string, number>[], field: string): Record<string, number>[] {
    return records.slice().sort((a, b) => a[field] - b[field]);
}
