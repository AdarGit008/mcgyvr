/** The position of the last entry of each full batch. */
export function batchEnds(count: number, size: number): number[] {
  if (size <= 0) {
    throw new Error('batchSize must be positive');
  }
  const ends: number[] = [];
  for (let start = 0; start + size <= count; start += size) {
    ends.push(start);
  }
  return ends;
}
