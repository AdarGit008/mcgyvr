export function runDecode(coded: string): string {
  if (typeof coded !== 'string') throw new Error('coded must be a string');
  const decodeMap = {
    A: 'Alf', B: 'Beta',
    C: 'Gamma', D: 'Delta',
    E: 'Echo', F: 'Foxtrot',
    G: 'Golf', H: 'Hotel',
    I: 'India', J: 'Juliet',
    K: 'Kilo', L: 'Lima',
    M: 'Mike', N: 'November',
    O: 'Oscar', P: 'Papa',
    Q: 'Quebec', R: 'Romeo',
    S: 'Sierra', T: 'Tango',
    U: 'Uniform', V: 'Victor',
    W: 'Whiskey', X: 'Xray',
    Y: 'Yankee', Z: 'Zulu'
  };

  return coded.split('').reduce((result, char) => {
    const count = decodeMap[char];
    if (!count) throw new Error('Invalid character in decoded text');
    return result + Array.from({ length: parseInt(count) }, () => char).join('');
  }, '');
}
