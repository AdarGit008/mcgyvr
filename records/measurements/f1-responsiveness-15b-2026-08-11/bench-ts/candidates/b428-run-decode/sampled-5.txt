/**
 * Decodes encoded run-length encoded text.
 *
 * @param coded - The run-length encoded text to decode.
 * @returns The decoded string of letters.
 */
export function runDecode(coded: string): string {
  if (typeof coded !== 'string') {
    throw new Error("Input must be a string.");
  }

  let decoded = '';
  for (let i = 0; i < coded.length; i += 2) {
    const letter = coded[i];
    const countStr = coded[i + 1];

    if (!countStr || isNaN(countStr)) {
      throw new Error("Invalid count or non-numeric count.");
    }

    const count = parseInt(countStr, 10);
    decoded += letter.repeat(count);
  }

  return decoded;
}
