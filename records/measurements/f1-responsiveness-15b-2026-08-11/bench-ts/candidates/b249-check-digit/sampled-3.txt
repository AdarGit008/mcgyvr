export function checkDigit(code: string): number {
  if (typeof code !== 'string' || !code) return 0;

  const digits = code.split('').filter(char => char.match(/\d/));

  if (digits.length === 0) return 0;

  let total = digits.reduce((acc, digit) => acc + parseInt(digit), 0);

  while (total % 10 !== 0) {
    total += 1;
  }

  return total % 10;
}
