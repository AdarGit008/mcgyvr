```ts
/** A mapping's value for a key, or a fallback when it is absent. */
export function lookUp(
  store: Record<string, string>,
  key: string,
  fallback: string,
): string {
  if (key === "" || !store[key]) {
    return fallback;
  }
  const value = store[key];
  return value ? value : fallback;
}
```
