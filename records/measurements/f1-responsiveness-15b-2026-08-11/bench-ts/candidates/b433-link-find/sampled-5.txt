function linkFind(line: string, opens: string, closes: string): string[] {
  const result: string[] = [];
  let stack: string[] = [];

  for (let i = 0; i < line.length; i++) {
    if (line[i] === opens) {
      stack.push(opens);
      continue;
    }

    if (stack.includes(closes)) {
      result.push(line.substring(i));
      break;
    }
  }

  return result;
}
