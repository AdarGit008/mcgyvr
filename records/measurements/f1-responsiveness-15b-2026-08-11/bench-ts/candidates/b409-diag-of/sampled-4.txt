export function cellAt(grid: number[][], row: number, column: number): number | undefined {
  if (!Array.isArray(grid) || !Array.isArray(grid[0]) || row < 0 || column < 0 || row >= grid.length || column >= grid[0].length) return undefined;
  return grid[row][column];
}

export function diagOf(grid: number[][]): number[] {
  const result: number[] = [];
  for (let i = 0; i < grid.length; i++) {
    if (grid[i] !== undefined && grid[i][i] !== undefined) result.push(grid[i][i]);
  }
  return result;
}
