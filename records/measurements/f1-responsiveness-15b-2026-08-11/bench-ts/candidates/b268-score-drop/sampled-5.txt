export function scoreDrop(scores: number[]): number {
  if (scores.length <= 1) return 0;

  const sortedScores = scores.slice().sort((a, b) => a - b);
  let totalScore = 0;

  for (let i = 1; i < sortedScores.length - 1; i++) {
    totalScore += sortedScores[i];
  }

  return totalScore;
}
