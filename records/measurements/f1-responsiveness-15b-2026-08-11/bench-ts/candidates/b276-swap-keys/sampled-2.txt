export function swapKeys(names: Record<string, string>): Record<string, string> {
  const result: Record<string, string> = {};

  for (const code of Object.keys(names).sort()) {
    if (code === '') continue;

    let name;
    for (name in names) {
      if (names[name] === code) break;
    }

    if (name) {
      result[code] = name;
    }
  }

  return result;
}
