export function capAfter(passage: string): string {
    const sentences = passage.split('. ');
    let result = '';

    for (let i = 0; i < sentences.length; i++) {
        if (i === 0) {
            // Start of the passage
            result += sentences[i].charAt(0).toUpperCase() + sentences[i].slice(1);
        } else if (sentences[i] !== '') {
            // Sentence starting at index 0 or after a full stop
            result += '. ' + sentences[i].charAt(0).toUpperCase() + sentences[i].slice(1);
        }
    }

    return result;
}
