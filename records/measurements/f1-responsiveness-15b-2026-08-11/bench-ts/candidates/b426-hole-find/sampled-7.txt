function stepsOn(first: number, count: number): number[] {
  if (!Number.isInteger(first) || !Number.isInteger(count) || count <= 0 || first > first + count - 1) {
    throw new Error('Invalid input');
  }
  return Array.from({ length: count }, (_, i) => first + i);
}

function holeFind(seen: number[], first: number, count: number): number | undefined {
  const sequence = stepsOn(first, count);
  for (let i = 0; i < sequence.length; i++) {
    if (!seen.includes(sequence[i])) {
      return sequence[i];
    }
  }
  return undefined;
}
