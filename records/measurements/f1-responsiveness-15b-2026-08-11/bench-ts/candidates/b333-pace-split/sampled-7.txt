function paceOf(seconds: number, kilometres: number): number {
  if (kilometres === 0) {
    throw new Error('Leg distance cannot be zero');
  }

  return Math.floor(seconds / kilometres);
}

function paceList(legs: { seconds: number; kilometres: number }[]): number[] {
  return legs.filter(leg => leg.kilometres > 0).map(leg => paceOf(leg.seconds, leg.kilometres));
}
