export function paceOf(seconds: number, kilometres: number): number {
  if (kilometres <= 0) throw new Error('Distance must be positive');
  const pace = seconds / kilometres;
  return Math.floor(pace);
}

export function paceList(legs: { seconds: number; kilometres: number }[]): number[] {
  return legs.filter(leg => leg.kilometres > 0).map(leg => paceOf(leg.seconds, leg.kilometres));
}
