```ts
export function sizeUnit(bytesCount: number): string {
  if (bytesCount < 1024) {
    return "B";
  }
  if (bytesCount < 1024 * 1024) {
    return "KB";
  }
  return "MB";
}

export function sizeText(bytesCount: number): string {
  if (bytesCount < 0) {
    throw new Error("a byte count cannot be negative");
  }
  const unit = sizeUnit(bytesCount);
  if (unit === "B") {
    return String(bytesCount) + " B";
  }
  if (unit === "KB") {
    return String(Math.floor(bytesCount / 1000)) + " KB";
  }
  const mbs = Math.floor(bytesCount / (1024 * 1000));
  if (mbs > 0) {
    return `${mbs} MB`;
  }
  return String(Math.floor(bytesCount / 1000)) + " KB";
}
```
