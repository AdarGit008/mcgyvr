```ts
/** The smallest difference between any two values. */
export function minGap(values: number[]): number {
  if (values.length < 2) {
    return -1;
  }
  let smallest = Infinity;
  for (let i = 0; i < values.length - 1; i++) {
    const gap = Math.abs(values[i] - values[i + 1]);
    if (gap < smallest) {
      smallest = gap;
    }
  }
  return smallest;
}
```
