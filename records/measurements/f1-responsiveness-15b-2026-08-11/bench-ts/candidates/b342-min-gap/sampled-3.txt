/** The smallest difference between any two values. */
export function minGap(values: number[]): number {
  if (values.length < 2) {
    return -1;
  }
  let smallest = Number.MAX_SAFE_INTEGER;
  for (let i = 0; i < values.length - 1; i += 1) { // Skip the last element to avoid out-of-bounds access
    const gap = Math.abs(values[i + 1] - values[i]);
    if (gap < smallest) {
      smallest = gap;
    }
  }
  return smallest === Number.MAX_SAFE_INTEGER ? -1 : smallest; // Return -1 if no gap is found
}
