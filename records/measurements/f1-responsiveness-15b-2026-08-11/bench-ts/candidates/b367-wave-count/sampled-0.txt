export function waveCount(readings: number[]): number {
    let count = 0;
    for (let i = 1; i < readings.length; i++) {
        if ((readings[i] > readings[i - 1]) && (readings[i] < readings[i + 1])) {
            count++;
        } else if ((readings[i] < readings[i - 1]) && (readings[i] > readings[i + 1])) {
            count++;
        }
    }
    return count;
}
