export function routeStops(route: string): string[] {
  if (route === "") return [];
  // Split the route by '>' characters, remove any leading/trailing spaces
  const stops = route.split('>').map(stop => stop.trim());
  // Remove any empty strings that might have resulted from splitting or trimming
  stops.splice(0, stops.indexOf("") + 1);
  stops.splice(stops.lastIndexOf(""), 1);
  return stops;
}

export function routeHops(route: string): number {
  if (route === "") return 0;
  const stops = routeStops(route);
  // For each stop in the route, calculate the number of journeys to the next stop
  let count = stops.length - 1; // Subtract 1 for the final journey
  return count < 0 ? 0 : count; // Return 0 if there are no stops
}
