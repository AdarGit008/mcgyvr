/** How many of each coin to hand back, largest coin first. */
export function tillFloat(amount: number, coins: number[]): number[] {
  const counts: number[] = [];
  let left = amount;
  for (const coin of coins.sort((a, b) => b - a)) { // Sort coins in descending order
    if (left >= coin) {
      counts.push(Math.floor(left / coin)); // Use Math.floor to count the exact number of coins
      left -= coin * counts[counts.length - 1];
    }
  }
  return counts;
}
