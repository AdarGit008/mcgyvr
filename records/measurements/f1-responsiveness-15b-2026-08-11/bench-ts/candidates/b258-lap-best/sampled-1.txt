/** The quickest completed lap of a session. */
export function lapBest(laps: number[]): number {
  if (laps.length === 0) {
    return 0;
  }
  let best = Infinity; // Initialize with infinity to handle zero correctly
  for (const lap of laps) {
    if (lap < best && lap !== 0) { // Ensure the lap is not zero before updating best
      best = lap;
    }
  }
  return best === Infinity ? 0 : best; // Return zero if no completed lap was found
}
