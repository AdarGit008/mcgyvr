function wordReverse(line: string): string {
  if (line === null || line === undefined) throw new Error("Input cannot be null or undefined");

  return line.split(' ').map(word => {
    let reversed = '';
    for (let i = word.length - 1; i >= 0; i--) reversed += word[i];
    return reversed;
  }).join(' ');
}
