/**
 * Group items into slots of a fixed capacity, in order. If the capacity is zero,
 * return the original array with no modifications.
 */
export function slotPack(items: string[], capacity: number): string[][] {
  if (capacity === 0) return items;

  const slots: string[][] = [];
  for (let i = 0; i < items.length; i += capacity) {
    slots.push(items.slice(i, i + capacity));
  }
  return slots;
}
