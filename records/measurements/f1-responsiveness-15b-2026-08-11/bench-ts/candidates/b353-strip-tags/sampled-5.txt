export function stripTags(line: string): string {
  let result = '';
  let inTag = false;
  for (let i = 0; i < line.length; i++) {
    const char = line[i];
    if (char === '<') {
      inTag = true;
      continue;
    }
    if (inTag && char === '>') {
      inTag = false;
      break;
    }
    if (!inTag) {
      result += char;
    }
  }
  return result;
}
