/** How many of a store's values reach a floor. */
export function overCount(
  store: Record<string, number>,
  floor: number,
): number {
  let found = 0;
  for (const key in store) {
    if (!(key in store)) continue; // Skip non-existing keys
    const value = Number(store[key]);
    if (!Number.isFinite(value)) throw new Error('Invalid value');
    if (value < floor) {
      found += 1;
    }
  }
  return found;
}
