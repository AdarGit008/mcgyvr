import { strict as assert } from 'assert';

export function gridFlip(grid: number[][]): number[][] {
    if (grid.length === 0) return [];
    
    const transposedGrid = grid[0].map((_, colIndex) => grid.map(row => row[colIndex]));
    return transposedGrid;
}

// Check function to verify the correctness of the solution
function checkGridFlip() {
    assert.deepEqual(gridFlip([]), [], 'Empty grid');
    assert.deepEqual(gridFlip([[1, 2], [3, 4]]), [[1, 3], [2, 4]], 'Single-row grid');
    assert.deepEqual(gridFlip([[5, 6], [7, 8], [9, 10]]), [[5, 7, 9], [6, 8, 10]], 'Multi-row grid');
}

// Run the check function
checkGridFlip();
