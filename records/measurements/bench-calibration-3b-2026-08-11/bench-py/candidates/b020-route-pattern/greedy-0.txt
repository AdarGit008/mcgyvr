```python
"""Match a slash-separated route pattern against a request path."""


def match_route(pattern: str, path: str) -> dict | None:
    """Return captured segments as a dict, or None when there is no match."""
    def validate_pattern(p):
        if not p:
            raise ValueError("Pattern cannot be empty.")
        for segment in p.split("/"):
            if segment == "":
                raise ValueError("Empty segments are not allowed in the pattern.")
            if segment.startswith(":") and ":" not in segment[1:]:
                raise ValueError("Capture segments must have a name.")

    validate_pattern(pattern)

    pats = pattern.split("/") if pattern else []
    parts = path.split("/") if path else []

    def match_from(pi: int, si: int, caps: dict) -> dict | None:
        if pi == len(pats):
            return caps if si == len(parts) else None
        seg = pats[pi]
        if seg == "**":
            for take in range(si + 1, len(parts) + 1):
                found = match_from(pi + 1, take, caps)
                if found is not None:
                    return found
            return None
        if si == len(parts):
            return None
        if seg.startswith(":"):
            merged = dict(caps)
            merged[seg[1:]] = parts[si]
            return match_from(pi + 1, si + 1, merged)
        if seg == parts[si]:
            return match_from(pi + 1, si + 1, caps)
        return None

    return match_from(0, 0, {})
```
