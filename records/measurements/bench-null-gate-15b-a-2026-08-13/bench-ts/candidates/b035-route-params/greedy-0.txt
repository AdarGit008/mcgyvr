function splitSegments(path: string): string[] {
  if (typeof path !== 'string' || path === '' || !path.startsWith('/')) {
    throw new Error('Invalid path');
  }
  return path.slice(1).split('/');
}

function firstRoute(patterns: string[], path: string): number {
  for (let i = 0; i < patterns.length; i++) {
    if (matchRoute(patterns[i], path) !== null) {
      return i;
    }
  }
  return -1;
}

export function matchRoute(pattern: string, path: string): Record<string, string> | null {
  const segments = splitSegments(path);
  const patternSegments = splitSegments(pattern);

  if (segments.length !== patternSegments.length) {
    throw new Error('Pattern and path do not have the same number of segments');
  }

  const captureNames: { [key: string]: string } = {};

  for (let i = 0; i < patternSegments.length; i++) {
    const segment = patternSegments[i];
    if (segment.startsWith(':')) {
      const name = segment.slice(1);
      if (! /^[a-zA-Z_][a-zA-Z0-9_]*/.test(name)) {
        throw new Error('Invalid capture name');
      }
      if (captureNames[name]) {
        throw new Error('Capture name repeated within the pattern');
      }
      captureNames[name] = segments[i];
    } else if (segment === '*') {
      continue;
    } else if (segment === '**') {
      const nextPatternSegment = patternSegments[i + 1];
      if (!nextPatternSegment.startsWith(':')) {
        throw new Error('Invalid "**" in pattern');
      }
      i++;
    } else if (segment !== segments[i]) {
      return null;
    }
  }

  return captureNames;
}
