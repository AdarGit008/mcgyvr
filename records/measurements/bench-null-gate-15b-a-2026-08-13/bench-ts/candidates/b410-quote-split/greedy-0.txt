export function inQuote(ch: string): boolean {
  return ch === '"' || ch === "'";
}

export function quoteSplit(line: string): string[] {
  const result: string[] = [];
  let currentQuotationMark = null;
  let currentString = '';

  for (let i = 0; i < line.length; i++) {
    const char = line[i];
    if (char === '"') {
      if (currentQuotationMark !== null) {
        // Close the previous quotation mark
        result.push(currentString);
        currentString = '';
      }
      currentQuotationMark = char;
    } else if (char === ',') {
      if (currentQuotationMark === null) {
        // Add a comma to the current string without closing it
        currentString += char;
      } else {
        // Close the quotation mark and add the comma to the result
        result.push(currentString);
        currentString = '';
        currentQuotationMark = null;
      }
    } else {
      if (currentQuotationMark === null) {
        // Add the character to the current string without closing it
        currentString += char;
      } else {
        // Continue building the current string inside a quotation mark
        currentString += char;
      }
    }
  }

  // Add the last part of the line if there's no comma after the last quote
  if (currentQuotationMark !== null) {
    result.push(currentString);
  }

  return result;
}
