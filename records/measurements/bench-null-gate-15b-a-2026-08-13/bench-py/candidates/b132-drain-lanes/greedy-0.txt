"""Drain labelled items across planned lanes, round by round."""


def drain_lanes(plan: list, items: list) -> dict:
    if not plan or not all(isinstance(item, tuple) and len(item) == 2 for item in items):
        raise ValueError("Invalid plan or items")
    
    queues = {}
    for lane, _quota in plan:
        if lane in queues:
            queues[lane] = []
        else:
            queues[lane] = [item[0] for item in items if item[1] == lane]
    
    order = []
    rounds = 0
    while True:
        rounds += 1
        for lane, quota in plan:
            queue = queues[lane]
            taken = 0
            while taken < quota and queue:
                order.append(queue.pop(0))
                taken += 1
            if taken < quota:
                return {"order": order, "rounds": rounds}
