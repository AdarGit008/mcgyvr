def wave_count(readings: list) -> int:
    if not readings:
        return 0

    count = 0
    for i in range(1, len(readings)):
        if (readings[i] > readings[i - 1]) and (readings[i] < readings[i + 1]):
            count += 1
        elif (readings[i] < readings[i - 1]) and (readings[i] > readings[i + 1]):
            count += 1

    return count
