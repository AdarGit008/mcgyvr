def split_at(entries: list, marker: str) -> list:
    """
    Break a list in two at the first entry equal to a marker, returning what stood before it and what stood after, with the marker itself in neither.

    :param entries: List of elements.
    :param marker: The element to split the list at.
    :return: A tuple containing two lists: (elements before marker, elements after marker).
    """
    if not entries:
        return [], []

    for i, entry in enumerate(entries):
        if entry == marker:
            return entries[:i], entries[i+1:]
    return entries, []
