def trace_parcel(events: list) -> list:
    if not isinstance(events, list):
        raise ValueError("Input must be a list")
    
    states = ["created"]
    for event in events:
        if event not in ["created", "packed", "shipped", "delivered", "returned"]:
            raise ValueError(f"Invalid event: {event}")
        
        if event == "created":
            continue
        elif event == "packed":
            states.append("packed")
        elif event == "shipped":
            states.append("shipped")
        elif event == "delivered":
            states.append("delivered")
        elif event == "returned":
            states.append("returned")
    
    return states
