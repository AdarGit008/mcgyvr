```python
"""Melt a row of ice one simultaneous step at a time."""


def thaw_row(row: str, steps: int) -> str:
    if not isinstance(row, str) or not row:
        raise ValueError("thaw_row expects a non-empty string row")
    if not isinstance(steps, int) or steps < 0:
        raise ValueError("thaw_row expects a non-negative integer step count")

    def melt_cell(cell):
        return "." if cell in ("#", ".") else None

    def is_meltwater(cell):
        return cell == "#"

    def melt_row(row):
        new_row = []
        for i, cell in enumerate(row):
            if i == 0 or i == len(row) - 1:
                new_row.append(melt_cell(cell))
            elif is_meltwater(cell) and (i > 0 and row[i-1] != "#" or i < len(row)-1 and row[i+1] != "#"):
                new_row.append(".")
            else:
                new_row.append(cell)
        return "".join(new_row)

    for _ in range(steps):
        row = melt_row(row)

    return row
```
